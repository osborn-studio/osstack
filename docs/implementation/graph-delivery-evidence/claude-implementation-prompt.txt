        new GraphDeliveryEpochStore(this.options.db).activate(this.fence(), next, deliveryEpoch(this.compiled, next).cost, this.feedbackRef);
        resume = { kind: 'interrupt', value: { ...decision.route, feedbackRef: this.feedbackRef } };
        this.epoch = next;
      }
    } catch (error) {
      const current = this.current();
      const held = this.options.db.prepare('SELECT 1 FROM graph_action_admissions WHERE graph_run_id=? AND released_at IS NULL').get(current.graphRunId);
      const unresolved = this.options.db.prepare("SELECT 1 FROM graph_actions WHERE graph_run_id=? AND state IN ('intent','running','unknown','cancel_pending')").get(current.graphRunId);
      const reconcile = !!held || !!unresolved || !lastStopped?.actionsSettled || lastStopped.worker.kind !== 'exited';
      try {
        this.check();
        if (!['ready','failed','cancelled','cancel_pending'].includes(current.status) && current.leaseExpiresAt! > Date.now()) {
          this.runs.project({ ...this.fence(current), status: reconcile ? 'reconciliation_required' : 'failed' });
        }
      } catch { /* Revoked/stale hosts cannot project replacement or cancellation state. */ }
      return { route: { kind: reconcile ? 'reconcile' : 'stop', epoch: this.epoch, reason: String(error).slice(0, 4096) } };
    } finally { this.closed = true; await this.runtime.close(); if (this.connection) this.broker.unregister(this.connection); }
  }

export interface ExpectedVerificationBranch { nodeId: string; effectSlot: string; logicalAttempt: number }
export interface VerificationJoinInput {
  binding: VerificationBinding;
  /** Current ledger attempts; zero means no invocation was admitted. */
  expected: { review: ExpectedVerificationBranch; tests: ExpectedVerificationBranch };
  branches: VerificationBranch[];
}
export interface VerificationJoin {
  schemaVersion: 1;
  binding: VerificationBinding;
  expected: VerificationJoinInput['expected'];
  status: 'complete' | 'blocked';
  branches: VerificationBranch[];
  missing: Array<'review' | 'tests'>;
  evidenceDigest: string;

function parseExpected(value: unknown, binding: VerificationBinding): ExpectedVerificationBranch {
  const row = verificationObject(value, ['nodeId', 'effectSlot', 'logicalAttempt']);
  if (!Number.isSafeInteger(row.logicalAttempt) || (row.logicalAttempt as number) < 0) throw Error('OUTPUT_CONTRACT_INVALID: expected attempt');
  const effect = GraphEffectIdentity.parse({ schemaVersion: 1, scope: binding.scope, graphRunId: binding.graphRunId, specHash: binding.specHash,
    epoch: binding.epoch, nodeId: row.nodeId, effectSlot: row.effectSlot, logicalAttempt: row.logicalAttempt === 0 ? 1 : row.logicalAttempt });
  return { nodeId: effect.nodeId, effectSlot: effect.effectSlot, logicalAttempt: row.logicalAttempt as number };
}
/** Pure assembly over a trusted ledger snapshot. verifyEvidence must compare each
 * envelope to host-verified durable evidence, not just accept its reported digest.
 * This function grants no authority, schedules no tasks, and has no second FSM.
 * Recomputing this deterministic bundle after recovery is not a new external effect. */
export function joinVerificationBranches(value: unknown, verifyEvidence: (branch: VerificationBranch) => void): VerificationJoin {
  if (typeof verifyEvidence !== 'function') throw Error('trusted evidence verifier required');
  const input = verificationObject(value, ['binding', 'expected', 'branches']);
  const binding = parseVerificationBinding(input.binding);
  const expectedInput = verificationObject(input.expected, ['review', 'tests']);
  const expected = { review: parseExpected(expectedInput.review, binding), tests: parseExpected(expectedInput.tests, binding) };
  if (expected.review.nodeId === expected.tests.nodeId) throw Error('JOIN_REVISION_MISMATCH: independent predecessor nodes required');
  if (!Array.isArray(input.branches) || input.branches.length > 2) throw Error('OUTPUT_CONTRACT_INVALID: predecessor count');
  const byKind = new Map<'review' | 'tests', VerificationBranch>();
  for (const raw of input.branches) {
    const branch = validateVerificationBranch(raw, binding); const target = expected[branch.kind];
    if (byKind.has(branch.kind)) throw Error('JOIN_REVISION_MISMATCH: duplicate predecessor');
    if (branch.nodeId !== target.nodeId || branch.effectSlot !== target.effectSlot || (branch.logicalAttempt ?? 0) !== target.logicalAttempt) throw Error('JOIN_REVISION_MISMATCH: stale predecessor');
    if (target.logicalAttempt > 0) {
      const key = computeGraphEffectKey({ schemaVersion: 1, scope: binding.scope, graphRunId: binding.graphRunId, specHash: binding.specHash,
        epoch: binding.epoch, nodeId: target.nodeId, effectSlot: target.effectSlot, logicalAttempt: target.logicalAttempt });
      if (branch.effectKey !== key) throw Error('JOIN_REVISION_MISMATCH: missing or incorrect effect key');
    }
    byKind.set(branch.kind, freezeWorkerData(branch));
  }
  const branches: VerificationBranch[] = []; const missing: Array<'review' | 'tests'> = [];
  for (const kind of ['review', 'tests'] as const) {
    const branch = byKind.get(kind);
    if (!branch) { missing.push(kind); continue; }
    const verified: unknown = verifyEvidence(branch);
    if (verified !== undefined) { void Promise.resolve(verified).catch(() => {}); throw Error('evidence verification must be synchronous'); }
    branches.push(branch);
  }
  const body = { schemaVersion: 1 as const, binding, expected, status: missing.length === 0 && branches.every(b => b.state === 'confirmed') ? 'complete' as const : 'blocked' as const, branches, missing };
  return freezeWorkerData({ ...body, evidenceDigest: createHash('sha256').update(`osstack.graph.join.v1\n${verificationCanonical(body)}`).digest('hex') });
}
/** Recheck a serialized join's structure and digest. This is not evidence authentication. */
export function parseVerificationJoin(value: unknown): VerificationJoin {
  const input = verificationObject(value, ['schemaVersion', 'binding', 'expected', 'status', 'branches', 'missing', 'evidenceDigest']);
  const rebuilt = joinVerificationBranches({ binding: input.binding, expected: input.expected, branches: input.branches }, () => {});
  if (verificationCanonical(rebuilt) !== verificationCanonical(value)) throw Error('OUTPUT_CONTRACT_INVALID: join bundle');
  return rebuilt;
}

/** Deterministic recommendation over a host-verified join and original ledger
 * snapshot. It never allocates attempts, advances durable epochs, or authorizes
 * execution. The host must recheck the current fence/grant and atomically reserve
 * through the existing ledger before acting on retry_required/repair_required.
 * now is supplied by that host; the replayable orchestration does not read clocks. */
export function routeVerification(joinValue: unknown, stateValue: unknown): VerificationRoute {
  const joined = parseVerificationJoin(joinValue); const state = parseState(stateValue);
  if (verificationCanonical(joined.binding) !== verificationCanonical(state.binding)) throw Error('JOIN_REVISION_MISMATCH: routing snapshot');
  const epoch = state.binding.epoch; const limits = state.spec.limits;
  for (const kind of ['review', 'tests'] as const) {
    if (state.branchAttemptsSpent[kind] < joined.expected[kind].logicalAttempt) throw Error('INPUT_CONTRACT_INVALID: branch accounting');
  }
  const stop = (reason: string): VerificationRoute => ({ kind: 'stop', epoch, reason });
  const reconcile = (reason: string): VerificationRoute => ({ kind: 'reconcile', epoch, reason });
  if (['ready', 'failed', 'cancelled'].includes(state.runStatus)) return stop('terminal_run');
  if (state.runStatus === 'cancel_pending' || state.runStatus === 'reconciliation_required' || state.unresolvedActions > 0
    || joined.branches.some(b => b.state === 'unknown' || b.state === 'cancel_pending')) return reconcile('unresolved_execution');
  if (state.pendingAdmissions > 0) return reconcile('resource_release_pending');
  if (state.runStatus !== 'running') return stop('run_not_running');
  if (joined.branches.some(b => b.state === 'cancelled')) return stop('cancelled_branch');
  if (epoch >= limits.maxEpochs) return stop('epoch_budget');
  if (state.now - state.runCreatedAt >= limits.deadlineSeconds * 1000) return stop('deadline_budget');
  const spentTokens = BigInt(state.reservedTokens) + BigInt(state.chargedTokens);
  if (state.accountingBlocked || spentTokens > BigInt(limits.tokenReservationLimit) || state.actionStarts > limits.maxActionStarts) return stop('accounting_budget');
  if (joined.missing.length) return stop('missing_predecessor');
  const capacity = (costs: VerificationActionCost[]): boolean => {
    const actions = costs.reduce((sum, c) => sum + BigInt(c.actions), BigInt(state.actionStarts));
    const totalTokens = costs.reduce((sum, c) => sum + BigInt(c.tokens), spentTokens);
    return actions <= BigInt(limits.maxActionStarts) && totalTokens <= BigInt(limits.tokenReservationLimit);
  };
  const failed = joined.branches.filter(b => b.state === 'failed');
  if (failed.length) {
    if (failed.some(b => !b.error?.retryable || !b.logicalAttempt || !['OUTPUT_CONTRACT_INVALID', 'INTERNAL_ERROR'].includes(b.error.code))) return stop('branch_failure');
    if (failed.some(b => state.branchAttemptsSpent[b.kind] >= limits.maxBranchAttempts)) return stop('branch_budget');
    if (!capacity(failed.map(b => state.nextCost[b.kind]))) return stop('action_or_token_budget');
    return freezeWorkerData({ kind: 'retry_required', epoch, branches: failed.map(b => b.kind) });
  }
  const review = joined.branches.find(b => b.kind === 'review')?.result;
  const tests = joined.branches.find(b => b.kind === 'tests')?.result;
  if (joined.status !== 'complete' || review?.kind !== 'review' || tests?.kind !== 'tests') return stop('invalid_predecessor');
  if (review.verdict === 'accept' && tests.passed) return freezeWorkerData({ kind: 'ready', epoch, evidenceDigest: joined.evidenceDigest });
  if (epoch >= 2 || epoch + 1 >= limits.maxEpochs) return stop('epoch_budget');
  if (!capacity([state.nextCost.repair])) return stop('action_or_token_budget');
  return freezeWorkerData({ kind: 'repair_required', epoch, nextEpoch: epoch + 1, evidenceDigest: joined.evidenceDigest });
}
