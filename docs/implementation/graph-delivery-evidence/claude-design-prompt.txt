const specInput = (field: string, kind: GraphFieldKind = 'ref'): GraphInputMapping => ({ from: 'spec', field, kind });
const output = (nodeId: string, field: string, kind: GraphFieldKind = 'ref'): GraphInputMapping => ({ from: 'node', nodeId, field, kind });

function fixedPlan(maxEpochs: number, maxBranchAttempts: number): GraphPlan {
  const nodes: GraphPlanNode[] = [];
  for (let epoch = 0; epoch < maxEpochs; epoch++) {
    const prefix = `e${epoch}`;
    const prior = `e${epoch - 1}`;
    const work = epoch === 0 ? 'implement' : 'repair';
    const add = (name: string, operation: GraphOperation, inputs: Record<string, GraphInputMapping>, maxLedgerAttempts = 1) => {
      nodes.push({ id: `${prefix}.${name}`, taskName: `code-delivery-v1.${prefix}.${name}`, epoch, operation,
        effectSlot: 'main', inputs, maxLedgerAttempts });
    };
    add('workspace', 'workspace_create', { repositoryRef: specInput('repositoryRef'), baseRevision: epoch === 0
      ? specInput('baseRevision', 'commit') : output(`${prior}.freeze`, 'candidateRevision', 'commit') });
    add(work, work, { inputArtifactRef: { from: 'start', field: 'inputArtifactRef', kind: 'ref' },
      workspaceLeaseRef: output(`${prefix}.workspace`, 'workspaceLeaseRef'), ...(epoch === 0 ? {} : {
        candidateRevision: output(`${prior}.freeze`, 'candidateRevision', 'commit'),
        feedbackRef: output(`${prior}.join`, 'feedbackRef'),
      }) });
    add('freeze', 'freeze', { workspaceLeaseRef: output(`${prefix}.${work}`, 'workspaceLeaseRef'), receiptRef: output(`${prefix}.${work}`, 'receiptRef') });
    add('review', 'review', { candidateRevision: output(`${prefix}.freeze`, 'candidateRevision', 'commit'),
      acceptanceRef: specInput('acceptanceRef'), workspaceLeaseRef: output(`${prefix}.workspace`, 'workspaceLeaseRef') }, maxBranchAttempts);
    // Host resolves acceptanceRef to an authorized command specification. Never shell text.
    add('tests', 'tests', { candidateRevision: output(`${prefix}.freeze`, 'candidateRevision', 'commit'),
      commandSpecRef: specInput('acceptanceRef'), workspaceLeaseRef: output(`${prefix}.workspace`, 'workspaceLeaseRef') }, maxBranchAttempts);
    add('join', 'join', { candidate: output(`${prefix}.freeze`, '$', 'candidate'), review: output(`${prefix}.review`, '$', 'review'),
      tests: output(`${prefix}.tests`, '$', 'tests') });
  }
  return { template: 'code-delivery-v1', maxEpochs, maxBranchAttempts, nodes };
}

/** Validates mappings AND exact fixed topology; this is not a user graph builder. */
export function validateGraphPlan(value: unknown, registryValue: unknown = GRAPH_REGISTRY): GraphPlan {
  const registry = validateGraphRegistry(registryValue);
  try {
    const plan = GraphJsonValue.parse(value) as unknown as GraphPlan;
    if (!Number.isInteger(plan.maxEpochs) || plan.maxEpochs < 1 || plan.maxEpochs > 3
      || !Number.isInteger(plan.maxBranchAttempts) || plan.maxBranchAttempts < 1 || plan.maxBranchAttempts > 32) throw new Error('bounds');
    const seen = new Map<string, GraphPlanNode>();
    for (const node of plan.nodes) {
      const contract = registry.nodes[node.operation];
      if (!contract || seen.has(node.id)) throw new Error('unregistered node');
      if (Object.keys(node.inputs).sort().join() !== Object.keys(contract.inputs).sort().join()) throw new Error('required inputs');
      for (const [field, mapping] of Object.entries(node.inputs)) {
        let actual: GraphFieldKind | undefined;
        if (mapping.from === 'spec') actual = specFields[mapping.field];
        else if (mapping.from === 'start') actual = mapping.field === 'inputArtifactRef' ? 'ref' : undefined;
        else if (mapping.from === 'node') {
          const predecessor = seen.get(mapping.nodeId);
          if (!predecessor) throw new Error('unreachable input');
          const producer = registry.nodes[predecessor.operation];
          if (mapping.field === '$') actual = producer.outputKind === 'candidate_ready' ? 'candidate'
            : producer.outputKind === 'review' ? 'review' : producer.outputKind === 'tests' ? 'tests' : undefined;
          else if (mapping.field === 'evidenceRefs.0' && producer.outputs.evidenceRefs === 'refs') actual = 'ref';
          else actual = producer.outputs[mapping.field];
        }
        if (!actual || actual !== mapping.kind || actual !== contract.inputs[field]) throw new Error('input kind/output existence');
      }
      seen.set(node.id, node);
    }
    if (workerCanonical(plan) !== workerCanonical(fixedPlan(plan.maxEpochs, plan.maxBranchAttempts))) throw new Error('fixed topology');
    return freezeWorkerData(plan);
  } catch {
    throw new Error('PLAN_INVALID: fixed topology, reachable typed outputs and required inputs only');
  }
}

/** Pure compilation: no filesystem, executor, clock, random source, or plugin lookup.
 * Emitted source is a versioned first-epoch execution scaffold, not the G21 template.
 * All possible epochs/host-authorized branch retries reserve capacity up front. */
export function compileGraphSpec(value: unknown, registryValue: unknown = GRAPH_REGISTRY): CompiledGraphSpec {
  const spec = freezeWorkerData(GraphSpec.parse(value));
  const registry = validateGraphRegistry(registryValue);
  const plan = validateGraphPlan(fixedPlan(spec.limits.maxEpochs, spec.limits.maxBranchAttempts), registry);
  const budget = { maxActionStarts: 0, tokenReservationLimit: 0 };
  for (const node of plan.nodes) {
    const contract = registry.nodes[node.operation];
    if (contract.tool !== 'pure') budget.maxActionStarts += node.maxLedgerAttempts;
    budget.tokenReservationLimit += node.maxLedgerAttempts * contract.reservedTokens;
  }
  // Each review/tests attempt may require a fresh isolated restore, including
  // retries. Reserve these host lifecycle starts even though G21 will register
  // and dispatch them; the G18 scaffold is not the complete delivery template.
  budget.maxActionStarts += 2 * spec.limits.maxEpochs * spec.limits.maxBranchAttempts;
  if (budget.maxActionStarts > spec.limits.maxActionStarts || budget.tokenReservationLimit > spec.limits.tokenReservationLimit) {
    throw new Error('BUDGET_EXHAUSTED: limits cannot reserve the expanded fixed plan');
  }
  const specHash = sha256(`osstack.graph.spec.v1\n${workerCanonical(spec)}`);
  const identity = { compilerVersion: GRAPH_COMPILER_VERSION, registryVersion: registry.registryVersion, schemaVersion: registry.schemaVersion };
  const source = emitSource({ ...identity, spec, specHash, plan, registry, budget });
  return freezeWorkerData({ ...identity, spec, specHash, plan, budget, source, compiledProgramDigest: sha256(source) });
}

        checkpoint: (key, request) => this.options.checkpoints.execute(key, request),
      });
    } catch (error) { retire(); throw error; }

      if (!record.exit) record.exit = freezeWorkerData(structuredClone(exit));
      retire();
    });
    // The bounded stop result stays immutable; a later actual OS event is a new
    // observation available to the trusted recovery verifier, never a timeout proof.
    void handle.finalExit.then(exit => { record.exit = freezeWorkerData(structuredClone(exit)); retire(); });
    return Object.freeze({ ...handle, cancel: (reason?: string) => { retire(); handle.cancel(reason); },
      stop: (reason?: string) => { retire(); return this.stop(handle.connection, reason); } });
  }
  observedExit(connection: object): GraphWorkerExitEvidence | undefined {
    const receipt = this.connections.get(connection)?.exit;
    return receipt === undefined ? undefined : structuredClone(receipt);
  }
  async stop(connection: object, reason = 'GRAPH_STOPPED'): Promise<GraphRuntimeStop> {
    const record = this.connections.get(connection); if (!record) throw new Error('GRAPH_CONNECTION_UNKNOWN');
    record.retired = true;
    this.options.broker.unregister(connection); this.options.checkpoints.unregister(connection);
    // cancel synchronously aborts the same signal passed to each actual broker call.
    const worker = record.handle.stop(reason);
    let timer: ReturnType<typeof setTimeout> | undefined;
    const drained = Promise.allSettled([...record.pending]).then(() => true);
    const bounded = Promise.race([drained, new Promise<boolean>(resolve => { timer = setTimeout(() => resolve(false), this.drainTimeoutMs); })]);
    const [exit, actionsSettled] = await Promise.all([worker, bounded]);
    if (timer) clearTimeout(timer);
    return { worker: exit, actionsSettled };
  }
  async close(): Promise<void> {
    this.closed = true;
    await Promise.all([...this.connections.keys()].map(connection => this.stop(connection, 'GRAPH_RUNTIME_CLOSED')));
  }
}

    return { scope: binding.scope, graphRunId: binding.graphRunId, identity: binding.identity,
      expectedGeneration: binding.generation, expectedOwner: binding.owner, expectedRevision: run.revision, now };
  }

  handle(connection: object, value: unknown, options: { signal?: AbortSignal } = {}): Promise<WorkerActionResponse> {
    return this.executeRequest(connection, value, options);
  }

  /** Trusted host command, never part of worker transport. Only this call owns the
   * freshly ledger-allocated retry intent; ordinary replays cannot claim it. */
  async retryAuthorized(connection: object, effectKey: string, authorizeRetry: () => void,
    options: { signal?: AbortSignal } = {}): Promise<WorkerActionResponse> {
    const binding = this.connections.get(connection);
    if (!binding) throw new Error("UNAUTHENTICATED");
    this.authorize(binding);
    const permitted: unknown = authorizeRetry();
    if (permitted !== undefined) {
      void Promise.resolve(permitted).catch(() => {});
      throw new Error("invalid synchronous retry authority");
    }
    if (options.signal?.aborted) throw new Error("retry cancelled");
    const prior = this.actions.get(this.fence(binding), effectKey).action;
    const registered = binding.actions.get(slot(prior.effect.nodeId, prior.effect.effectSlot));
    if (!registered || prior.effect.epoch !== binding.epoch || canonical(registered.action) !== canonical(prior.action)) throw new Error("unregistered retry slot");
    const next = this.actions.retry(this.fence(binding), effectKey).action;
    return this.executeRequest(connection, { schemaVersion: 1, kind: "worker_action", requestId: `retry:${randomUUID()}`,
      scope: binding.scope, graphRunId: binding.graphRunId, generation: binding.generation, specHash: binding.identity.specHash,
      epoch: binding.epoch, nodeId: prior.effect.nodeId, effectSlot: prior.effect.effectSlot, action: registered.action }, options, { action: next, binding }, authorizeRetry);
  }

  private async executeRequest(connection: object, value: unknown, options: { signal?: AbortSignal }, retry?: { action: GraphAction; binding: Binding }, authorizeRetry?: () => void): Promise<WorkerActionResponse> {
    let request: WorkerActionRequest;
    try { request = snapshot(WorkerActionRequest.parse(value)); }
    catch { return { schemaVersion: 1, kind: "worker_action_rejected", requestId: "invalid-request", error: graphError("INPUT_CONTRACT_INVALID", "invalid worker action request") }; }
    // Own the allocated retry before rechecking grants. If admission is denied,
    // settle this caller's unstarted intent while its original fence is still live.
    // Expired/replaced owners must instead use G15, never a refreshed generation.
    const binding = this.connections.get(connection) ?? retry?.binding;
    const retryEffectKey = retry?.action.effectKey;
    const signal = options.signal ?? new AbortController().signal;

    let lease: ExecutionLease | undefined; let owned: GraphAction | undefined = retry?.action; let invoked = false; let uncertain = false; let observedTokens = 0; let persistedEvidenceRef: string | undefined;

      if (request.graphRunId !== binding.graphRunId) throw new Denied("RUN_MISMATCH", "run does not match registered connection");
      if (request.generation !== binding.generation) throw new Denied("STALE_GENERATION", "generation does not match registered connection");
      if (request.specHash !== binding.identity.specHash) throw new Denied("SPEC_MISMATCH", "spec does not match registered program");
      if (request.epoch !== binding.epoch) throw new Denied("EPOCH_MISMATCH", "epoch does not match registered program");

      if (!action) throw new Denied("POLICY_DENIED", "unregistered compiled action slot");

      if (canonical(action.action) !== canonical(request.action)) throw new Denied("IDEMPOTENCY_CONFLICT", "action differs from immutable compiled registration");

        if (authorizeRetry) {
          const permitted: unknown = authorizeRetry();
          if (permitted !== undefined) { void Promise.resolve(permitted).catch(() => {}); throw new Denied("POLICY_DENIED", "retry authority must be synchronous"); }
        }
      };
      check();
      const intent = this.actions.reserve(this.fence(binding), { epoch: binding.epoch, nodeId: action.nodeId, effectSlot: action.effectSlot,
        reservationTokens: action.tokenReservation, action: action.action });

      if (retryEffectKey && (intent.action.state !== "intent" || intent.action.generation !== binding.generation)) {
        return this.response(request, intent.action);

      // handler call, after G04's async authorization and availability windows.

          // non-execution, so retain admission and require reconciliation.

        && !evidence.command.cancellation && !evidence.command.failure_code && !signal.aborted;

          : action.action.kind.startsWith("workspace_") ? "workspace" : action.action.kind;

        if ("workspaceLeaseRef" in result && "workspaceLeaseRef" in action.action && result.workspaceLeaseRef !== action.action.workspaceLeaseRef) throw new Error("persisted workspace mismatch");
        if ("candidateRevision" in result && "candidateRevision" in action.action && result.candidateRevision !== action.action.candidateRevision) throw new Error("persisted revision mismatch");
      }

      return this.response(request, settled.action);
    } catch (error) {

          if (lease) this.retained.set(owned.effectKey, { lease, context: { effectKey: owned.effectKey,
            scope: binding.scope, graphRunId: binding.graphRunId, generation: binding.generation,
            owner: binding.owner, workspacePath: executionWorkspace?.workspace_path ?? binding.workspace.workspace_path } });
          try { const unknown = this.actions.markUncertain(this.fence(binding), owned.effectKey, { state: "unknown", evidenceRef: persistedEvidenceRef ?? `unknown:${randomUUID()}`, observedTokens: Math.max(owned.reservationTokens, observedTokens) });

          return { schemaVersion: 1, kind: "worker_action_response", requestId: request.requestId, generation: request.generation,
            effect: owned.effect, outcome: { state: "unknown", error: graphError("ACTION_OUTCOME_UNKNOWN", String(error)), reconciliationRef: owned.effectKey } };
        }
        // No physical invocation: resolve this caller's own intent without calling
        // it unknown. Only a future explicit host retry can use this proof;
        // cancellations remain sealed and worker replay never allocates attempts.
        try { this.actions.recordOutcome(this.fence(binding), owned.effectKey, { state: "failed", evidenceRef: `not-started:${randomUUID()}`,
          actualTokens: 0, retrySafe: !signal.aborted, error: { message: "admission refused before invocation" } }); } catch { /* A stale owner cannot mutate a newer generation. */ }
      }
      return reject(error);
    } finally {
      if (lease && !uncertain) await this.releaseAdmission(lease, binding, owned, executionWorkspace);
    }
  }

  private async releaseAdmission(lease: ExecutionLease, binding: Binding | undefined, owned: GraphAction | undefined, workspace?: RegisteredCandidateWorkspace): Promise<void> {
    let physicallyReleased = false;
    try {
      await lease.release(); physicallyReleased = true;
      if (owned) this.admissions.released(owned.effectKey, this.options.now?.() ?? Date.now());
    }
    catch (error) {
      // Release failure deliberately stops the chain without replacing durable
      // effect truth. Keep its exact admission reachable for release-only recovery.
      if (!physicallyReleased) lease.retain(String(error));
      if (binding && owned) this.retained.set(owned.effectKey, { lease, context: {
        effectKey: owned.effectKey, scope: binding.scope, graphRunId: binding.graphRunId,
        generation: binding.generation, owner: binding.owner, workspacePath: workspace?.workspace_path ?? binding.workspace.workspace_path,
      } });
      throw error;
    }
  }

  private response(request: WorkerActionRequest, action: GraphAction): WorkerActionResponse {
    const base = { schemaVersion: 1 as const, kind: "worker_action_response" as const, requestId: request.requestId,
      generation: request.generation, effect: action.effect };
    if (action.state === "confirmed") return WorkerActionResponse.parse({ ...base, outcome: { state: "confirmed", result: action.outcome?.state === "confirmed" ? action.outcome.result : undefined } });
    if (action.state === "failed") {
      const error = GraphErrorSchema.safeParse(action.outcome?.state === "failed" ? action.outcome.error : undefined);
      return WorkerActionResponse.parse({ ...base, outcome: { state: "failed", error: error.success ? error.data : graphError("POLICY_DENIED", "prior invocation failed; no automatic retry") } });
    }
    return { ...base, outcome: { state: "unknown", error: graphError("ACTION_OUTCOME_UNKNOWN", "existing action requires reconciliation; invocation was not repeated"), reconciliationRef: action.effectKey } };
  }

  private async execute(binding: Binding, action: BoundAction, signal: AbortSignal, beforeEffect: (executionRunId: string) => void, observeTokens: (tokens: number) => void, lease?: ExecutionLease): Promise<GraphActionEvidence> {

              outputLimit: 16_000, ...(invocationSignal ? { signal: invocationSignal } : {}) });

              created_at: Date.now() / 1000, terminal: true, payload: { status: command.exitCode === 0 ? "completed" : "failed", last_seq: 0 } };
          } catch (error) { threw = true; throw error; }
        })();

    };

      adapters: [wrapper], signal, cancellationGraceMs: 50,
      authorize: async () => { this.authorize(binding); this.fence(binding); return { allow: true }; } });
    if (!invocationRunId) throw new Denied("POLICY_DENIED", candidate.reason ?? "candidate preflight refused");
    if (threw || candidate.status === "execution_unconfirmed" || candidate.execution?.failure_code) throw new Error("adapter result or execution unconfirmed");
    let stopTimer: ReturnType<typeof setTimeout> | undefined;
    const stopped = await Promise.race([wrapper.confirmStopped!(invocationRunId),
      new Promise<{ stopped: false }>(resolve => { stopTimer = setTimeout(() => resolve({ stopped: false }), 1000); })])
      .finally(() => { if (stopTimer) clearTimeout(stopTimer); });
    if (stopped.stopped !== true || !stopped.evidence?.trim()) throw new Error("external stop unconfirmed");
    const usage = candidate.execution;
    if ([usage?.tokens_in, usage?.tokens_out].some(value => value !== undefined && (!Number.isSafeInteger(value) || value < 0))) throw new Error("invalid adapter usage");
    return { status: candidate.status === "implementation_complete" ? "completed" : "failed", stopped: true, candidate, ...(command ? { command } : {}) };
  }

            ...(kind === 'review' ? { verdict: envelope.report!.verdict } : { passed: envelope.acceptance.passExitCodes.includes(envelope.evidence.command!.exitCode) }) });
          if (verificationCanonical(outcome.result) !== verificationCanonical(result)) throw Error('JOIN_RESULT_MISMATCH');
          branches.push({ ...base, state: 'confirmed', result });
        } else {
          const failure = envelope.verificationFailure;
          // A cancelled settlement can retain a valid receipt yet have no G19
          // classification. Preserve the ledger failure instead of promoting the
          // receipt's code verdict, and leave host retry policy to G21.
          const error = failure ? GraphError.parse(outcome.error) : GraphError.parse({ schemaVersion: 1, code: 'INTERNAL_ERROR',
            message: 'Stopped invocation has no retryable verification classification', retryable: false, nextAction: 'stop' });
          if (failure && (error.code !== failure.code || error.message !== failure.message || error.retryable !== failure.retrySafe || outcome.retrySafe !== failure.retrySafe)) throw Error('JOIN_FAILURE_MISMATCH');
          branches.push({ ...base, state: 'failed', error });
        }
      } else {
        const cancelling = action.state === 'cancel_pending';
        branches.push({ ...base, state: cancelling ? 'cancel_pending' : 'unknown', error: { schemaVersion: 1, code: cancelling ? 'CANCEL_UNCONFIRMED' : 'ACTION_OUTCOME_UNKNOWN',
          message: 'Ledger execution requires reconciliation', retryable: false, nextAction: 'reconcile' } });
      }
    }
    // Reject races after asynchronous evidence reads without refreshing away the
    // premise that was verified. Admission releases may change without revision.
    this.check(); const time = now(); const current = store.snapshot({ ...fence, now: time });
    if (current.digest !== snapshot.digest) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
    const joined = joinVerificationBranches({ binding, expected, branches }, branch => {
      if (!branches.some(original => verificationCanonical(branch) === verificationCanonical(original))) throw Error('JOIN_BRANCH_MISMATCH');
    });
    const spent = (kind: Kind) => selected[kind] ? 1 + snapshot.actions.filter(a => a.effect.epoch === binding.epoch
      && a.effect.nodeId === slots[kind].nodeId && a.effect.logicalAttempt > 1).length : 0;
    const route = routeVerification(joined, { binding, spec: snapshot.run.spec, runStatus: snapshot.run.status, runCreatedAt: snapshot.run.createdAt, now: time,
      ...snapshot.budget, pendingAdmissions: snapshot.pendingAdmissions, unresolvedActions: snapshot.unresolvedActions,
      branchAttemptsSpent: { review: spent('review'), tests: spent('tests') }, nextCost: this.options.nextCost });
    if (joined.status !== 'complete' || snapshot.unresolvedActions || snapshot.pendingAdmissions || snapshot.run.status !== 'running') return freezeWorkerData({ joined, route, snapshot });
    attested = { digest: snapshot.digest, joined };
    const committed = store.commit(snapshot, joined, time);
    return freezeWorkerData({ joined, route, snapshot: store.snapshot({ ...fence, expectedRevision: committed.run.revision, now: time }),
      publication: { record: committed.record, created: committed.created } });
  }
}

const hash = (value: unknown) => createHash('sha256').update(workerCanonical(value)).digest('hex');
function fence(run: GraphRun, now: number): GraphRunFence {

    expectedOwner: run.leaseOwner, expectedRevision: run.revision, now };

/** A durable immutable evidence publication, not an epoch scheduler or node FSM.
 * The trusted host verifier must bind the bundle to current ledger attempts and
 * independent artifact/executor evidence. No worker JSON port calls this store. */
export class GraphVerificationStore {
  constructor(private readonly db: Database.Database,
    private readonly verify: (snapshot: GraphVerificationSnapshot, bundle: GraphJsonValue) => void) {}
  private capture(run: GraphRun): GraphVerificationSnapshot {
    const state = readGraphActionState(this.db, run.graphRunId);
    const admissions = this.db.prepare('SELECT effect_key,released_at FROM graph_action_admissions WHERE graph_run_id=? ORDER BY effect_key').all(run.graphRunId) as Array<{ effect_key: string; released_at: number | null }>;
    const body = { run, ...state, pendingAdmissions: admissions.filter(a => a.released_at === null).length,
      unresolvedActions: state.actions.filter(a => ['intent','running','unknown','cancel_pending'].includes(a.state)).length };
    return freezeWorkerData({ ...body, digest: hash({ body, admissions }) });
  }
  snapshot(input: GraphRunFence): GraphVerificationSnapshot {
    return fencedGraphTransaction(this.db, input, run => ({ changed: false, value: this.capture(run) })).value;
  }
  commit(snapshotValue: GraphVerificationSnapshot, bundleValue: unknown, now: number): { record: GraphVerificationRecord; created: boolean; run: GraphRun } {
    const snapshot = freezeWorkerData(structuredClone(snapshotValue)); const bundle = freezeWorkerData(GraphJsonValue.parse(bundleValue));
    const result = fencedGraphTransaction(this.db, fence(snapshot.run, now), run => {
      const current = this.capture(run);
      if (current.digest !== snapshot.digest || workerCanonical(current) !== workerCanonical(snapshot)) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
      if (current.unresolvedActions) throw Error('UNRESOLVED_ACTION');
      if (current.pendingAdmissions) throw Error('ADMISSION_RELEASE_PENDING');
      if (!bundle || typeof bundle !== 'object' || Array.isArray(bundle)) throw Error('INVALID_JOIN');
      const object = bundle as Record<string, GraphJsonValue>; const binding = object.binding;
      if (!binding || typeof binding !== 'object' || Array.isArray(binding)) throw Error('INVALID_JOIN_BINDING');
      const b = binding as Record<string, GraphJsonValue>;
      const candidate = GraphResult.parse({ schemaVersion: 1, kind: 'candidate_ready', candidateRevision: b.candidateRevision, artifactDigest: b.artifactDigest, artifactRefs: ['join-validation'] });
      if (candidate.kind !== 'candidate_ready' || object.schemaVersion !== 1 || object.status !== 'complete'
        || b.graphRunId !== run.graphRunId || workerCanonical(b.scope) !== workerCanonical(run.scope) || b.specHash !== run.identity.specHash
        || !Number.isSafeInteger(b.epoch) || (b.epoch as number) < 0 || (b.epoch as number) >= (run.spec?.limits.maxEpochs ?? 0)
        || b.acceptanceRef !== run.spec?.input.acceptanceRef || typeof object.evidenceDigest !== 'string' || !/^[a-f0-9]{64}$/.test(object.evidenceDigest)) throw Error('INVALID_JOIN_BINDING');
      if (typeof this.verify !== 'function') throw Error('trusted verifier required');
      const verified: unknown = this.verify(current, bundle);
      if (verified !== undefined) { void Promise.resolve(verified).catch(() => {}); throw Error('join verifier must be synchronous'); }
      const receiptRef = `graph-join:${hash(binding)}`;
      const saved = this.db.prepare('SELECT bundle_json,generation,owner,created_at,receipt_ref FROM graph_verification_joins WHERE graph_run_id=? AND epoch=? AND candidate_revision=?')
        .get(run.graphRunId, b.epoch, candidate.candidateRevision) as { bundle_json: string; generation: number; owner: string; created_at: number; receipt_ref: string } | undefined;
      if (saved && (saved.bundle_json !== workerCanonical(bundle) || saved.receipt_ref !== receiptRef)) throw Error('JOIN_PUBLICATION_CONFLICT');
      if (!saved) this.db.prepare('INSERT INTO graph_verification_joins(receipt_ref,graph_run_id,epoch,candidate_revision,generation,owner,bundle_json,created_at) VALUES(?,?,?,?,?,?,?,?)')
        .run(receiptRef, run.graphRunId, b.epoch, candidate.candidateRevision, run.generation, run.leaseOwner, workerCanonical(bundle), now);
      const record: GraphVerificationRecord = { receiptRef, graphRunId: run.graphRunId, epoch: b.epoch as number, candidateRevision: candidate.candidateRevision,
        generation: saved?.generation ?? run.generation, owner: saved?.owner ?? run.leaseOwner!, bundle, createdAt: saved?.created_at ?? now };
      return { changed: !saved, value: { record, created: !saved } };
    });
    return freezeWorkerData({ ...result.value, run: result.run });
  }
}

const count = z.number().int().min(0).max(Number.MAX_SAFE_INTEGER);

  z.object({ state: z.literal("confirmed"), evidenceRef: ref, actualTokens: count.nullable(), result: z.unknown() }).strict(),

]);

export type GraphActionOutcomeInput =

function canonical(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  const record = value as Record<string, unknown>;

}

/** Internal core snapshot helper; callers own their fenced SQL transaction. */
export function readGraphActionState(db: Database.Database, graphRunId: string): { actions: GraphAction[]; budget: GraphActionBudget } {
  return { actions: (db.prepare("SELECT * FROM graph_actions WHERE graph_run_id=? ORDER BY epoch,node_id,effect_slot,logical_attempt").all(graphRunId) as ActionRow[]).map(decode), budget: graphActionBudget(db, graphRunId) };
}
function graphActionBudget(db: Database.Database, graphRunId: string): GraphActionBudget {
    let reserved = 0n; let charged = 0n; let actionStarts = 0; let accountingBlocked = false;
    // Ledger rows ARE the authoritative budget entries. Derived in the same
    // transaction, so there is no mutable budget mirror to reset on resume.
    const rows = db.prepare(`SELECT state, reservation_tokens, actual_tokens, observed_tokens, accounting_blocked
      FROM graph_actions WHERE graph_run_id = ?`).iterate(graphRunId) as Iterable<Pick<ActionRow,
        "state" | "reservation_tokens" | "actual_tokens" | "observed_tokens" | "accounting_blocked">>;
    for (const row of rows) {
      actionStarts++;
      accountingBlocked ||= row.accounting_blocked === 1;
      if ((row.state === "confirmed" || row.state === "failed") && row.actual_tokens !== null) {
        // Failed attempts consume their full reservation even if less usage was
        // reported. Unknown usage never releases capacity.
        charged += BigInt(row.state === "failed" ? Math.max(row.reservation_tokens, row.actual_tokens) : row.actual_tokens);
      } else reserved += BigInt(Math.max(row.reservation_tokens, row.observed_tokens));
    }
    return { actionStarts, reservedTokens: reserved.toString(), chargedTokens: charged.toString(), accountingBlocked };
}

export class GraphActionStore {
  constructor(private readonly db: Database.Database) {}

  private write(fence: GraphRunFence, fn: (run: GraphRun) => Mutation): GraphActionResult {
    const result = fencedGraphTransaction(this.db, fence, (run) => {
      if (!run.spec) fail("ORIGINAL_BUDGET_MISSING");
      const value = fn(run);
      return { changed: value.changed, value: { ...value, budget: this.budget(run) } };
    });
    return { ...result.value, run: result.run };
  }
  private find(run: GraphRun, key: string): GraphAction {
    const row = this.db.prepare("SELECT * FROM graph_actions WHERE graph_run_id = ? AND effect_key = ?")
      .get(run.graphRunId, key) as ActionRow | undefined;
    return row ? decode(row) : fail("ACTION_NOT_FOUND");
  }
  private current(run: GraphRun, slot: Pick<GraphActionAdmission, "epoch" | "nodeId" | "effectSlot">): GraphAction | null {
    const row = this.db.prepare(`SELECT * FROM graph_actions WHERE graph_run_id = ? AND epoch = ? AND node_id = ? AND effect_slot = ?
      ORDER BY logical_attempt DESC LIMIT 1`).get(run.graphRunId, slot.epoch, slot.nodeId, slot.effectSlot) as ActionRow | undefined;
    return row ? decode(row) : null;
  }
  private budget(run: GraphRun): GraphActionBudget { return graphActionBudget(this.db, run.graphRunId); }
  private admit(run: GraphRun, slot: GraphActionAdmission, retry: boolean, now: number): void {
    const limits = run.spec!.limits;
    if (run.status === "cancel_pending" || run.status === "reconciliation_required") fail("RUN_STOPPING");
    if (this.db.prepare(`SELECT 1 FROM graph_actions WHERE graph_run_id = ? AND
      (state IN ('unknown','cancel_pending') OR (state IN ('intent','running') AND generation <> ?)) LIMIT 1`)
      .get(run.graphRunId, run.generation)) fail("UNRESOLVED_ACTION");
    const budget = this.budget(run);
    if (budget.accountingBlocked) fail("ACCOUNTING_BLOCKED");
    if (slot.epoch >= limits.maxEpochs) fail("EPOCH_BUDGET_EXHAUSTED");
    if (now - run.createdAt >= limits.deadlineSeconds * 1000) fail("WALL_CLOCK_BUDGET_EXHAUSTED");
    if (budget.actionStarts >= limits.maxActionStarts) fail("ACTION_BUDGET_EXHAUSTED");
    if (retry) {
      const row = this.db.prepare(`SELECT count(*) AS retries FROM graph_actions
        WHERE graph_run_id = ? AND epoch = ? AND node_id = ? AND logical_attempt > 1`)
        .get(run.graphRunId, slot.epoch, slot.nodeId) as { retries: number };
      if (1 + row.retries >= limits.maxBranchAttempts) fail("BRANCH_BUDGET_EXHAUSTED");
    }
    if (BigInt(budget.reservedTokens) + BigInt(budget.chargedTokens) + BigInt(slot.reservationTokens) > BigInt(limits.tokenReservationLimit)) {
      fail("TOKEN_BUDGET_EXHAUSTED");
    }
  }
  private insert(run: GraphRun, input: GraphActionAdmission, businessDigest: string, attempt: number, now: number): Mutation {
    const effect = GraphEffectIdentity.parse({ schemaVersion: 1, scope: run.scope, graphRunId: run.graphRunId,
      specHash: run.identity.specHash, epoch: input.epoch, nodeId: input.nodeId, effectSlot: input.effectSlot, logicalAttempt: attempt });
    const key = computeGraphEffectKey(effect);
    this.db.prepare(`INSERT INTO graph_actions (effect_key, graph_run_id, epoch, node_id, effect_slot, logical_attempt,
      effect_json, action_json, business_digest, generation, reservation_tokens, state, created_at, updated_at)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'intent', ?, ?)`).run(key, run.graphRunId, effect.epoch, effect.nodeId,
        effect.effectSlot, attempt, canonical(effect), canonical(input.action), businessDigest, run.generation, input.reservationTokens, now, now);
    return { action: this.find(run, key), changed: true, created: true };
  }

  /** Persist intent/reservation before invocation. Existing slots NEVER allocate
   * an attempt. Host must markRunning and invoke only when that call changed=true;
   * a lost response is pending, not permission to invoke again. */
  reserve(fence: GraphRunFence, value: GraphActionAdmission): GraphActionResult {
    const parsed = admissionSchema.parse(GraphJsonValue.parse(value));
    return this.write(fence, (run) => {
      const action = WorkerActionRequest.parse({ schemaVersion: 1, kind: "worker_action", requestId: "storage",
        generation: run.generation, scope: run.scope, graphRunId: run.graphRunId, specHash: run.identity.specHash,
        epoch: parsed.epoch, nodeId: parsed.nodeId, effectSlot: parsed.effectSlot, action: parsed.action }).action;
      const input = { ...parsed, action };
      const digest = createHash("sha256").update(`osstack.graph.action.business.v1\n${canonical(action)}`).digest("hex");
      const prior = this.current(run, input);
      if (prior) {
        if (prior.businessDigest !== digest || prior.reservationTokens !== input.reservationTokens) fail("ACTION_INPUT_CONFLICT");
        return { action: prior, changed: false, created: false };
      }
      this.admit(run, input, false, fence.now);
      return this.insert(run, input, digest, 1, fence.now);
    });
  }

  /** Fenced evidence lookup, including historical attempts. No raw public cache. */
  get(fence: GraphRunFence, effectKey: string): GraphActionResult {
    keySchema.parse(effectKey);
    return this.write(fence, (run) => ({ action: this.find(run, effectKey), changed: false, created: false }));
  }

  /** Explicit trusted retry command, never a worker flag. Proof of retry safety
   * must already be verified by G12 and recorded on a failed outcome. The old
   * action supplies ALL key/payload/cap inputs. Unknown is not retryable. */
  retry(fence: GraphRunFence, effectKey: string): GraphActionResult {
    keySchema.parse(effectKey);
    return this.write(fence, (run) => {
      const prior = this.find(run, effectKey);
      if (this.current(run, prior.effect)!.effectKey !== effectKey) fail("NOT_CURRENT_ATTEMPT");
      if (prior.state !== "failed" || !prior.retrySafe) fail("RETRY_NOT_SAFE");
      if (this.db.prepare("SELECT 1 FROM graph_action_admissions WHERE effect_key=? AND released_at IS NULL").get(effectKey)) fail("ADMISSION_RELEASE_PENDING");
      const input = { epoch: prior.effect.epoch, nodeId: prior.effect.nodeId, effectSlot: prior.effect.effectSlot,
        action: prior.action, reservationTokens: prior.reservationTokens };
      this.admit(run, input, true, fence.now);
      return this.insert(run, input, prior.businessDigest, prior.effect.logicalAttempt + 1, fence.now);
    });
  }

  markRunning(fence: GraphRunFence, effectKey: string, executionRef: string): GraphActionResult {
    keySchema.parse(effectKey); ref.parse(executionRef);
    return this.write(fence, (run) => {
      const prior = this.find(run, effectKey);
      if (prior.generation !== run.generation) fail("ACTION_GENERATION_MISMATCH");
      if (prior.state === "running") {
        if (prior.executionRef !== executionRef) fail("EXECUTION_CONFLICT");
        return { action: prior, changed: false, created: false };
      }
      if (prior.state !== "intent") fail("INVALID_ACTION_STATE");
      // Recheck stopping, unknown and deadline immediately before the host can
      // invoke. Capacity was already reserved by intent, so do not reserve twice.
      if (run.status === "cancel_pending" || run.status === "reconciliation_required") fail("RUN_STOPPING");
      if (this.db.prepare("SELECT 1 FROM graph_actions WHERE graph_run_id = ? AND state IN ('unknown','cancel_pending') LIMIT 1")
        .get(run.graphRunId)) fail("UNRESOLVED_ACTION");
      if (this.budget(run).accountingBlocked) fail("ACCOUNTING_BLOCKED");
      if (fence.now - run.createdAt >= run.spec!.limits.deadlineSeconds * 1000) fail("WALL_CLOCK_BUDGET_EXHAUSTED");
      this.db.prepare("UPDATE graph_actions SET state = 'running', execution_ref = ?, updated_at = ? WHERE effect_key = ?")
        .run(executionRef, fence.now, effectKey);
      return { action: this.find(run, effectKey), changed: true, created: false };
    });
  }

  /** Adapter/supervisor uncertainty only. G11 has no release/reconciliation API:
   * worker receipts or later 'not applied' claims cannot clear unknown capacity. */
  markUncertain(fence: GraphRunFence, effectKey: string, value: GraphActionUncertainty): GraphActionResult {
    keySchema.parse(effectKey);
    const input = uncertainSchema.parse(GraphJsonValue.parse(value));
    return this.write(fence, (run) => {
      const prior = this.find(run, effectKey);
      if (prior.generation !== run.generation) fail("ACTION_GENERATION_MISMATCH");
      if (prior.state === "confirmed" || prior.state === "failed") fail("INVALID_ACTION_STATE");
      if (prior.uncertainty && canonical(prior.uncertainty) === canonical(input)) return { action: prior, changed: false, created: false };
      this.db.prepare(`UPDATE graph_actions SET state = ?, uncertainty_json = ?, observed_tokens = ?,
        accounting_blocked = ?, updated_at = ? WHERE effect_key = ?`).run(input.state, canonical(input),
          Math.max(prior.observedTokens, input.observedTokens), Number(prior.accountingBlocked || input.observedTokens > prior.reservationTokens), fence.now, effectKey);
      return { action: this.find(run, effectKey), changed: true, created: false };
    });
  }

  /** Only trusted adapter outcomes, never worker-provided evidence. Invalid usage
   * does not discard a real result: persist null usage, retain capacity and stop
   * new actions. Over-cap actual usage is charged exactly and also stops actions.
   * Settlement after the wall-clock budget elapsed remains allowed with a live
   * fence; effects already started must not disappear because their budget ended. */
  recordOutcome(fence: GraphRunFence, effectKey: string, value: GraphActionOutcomeInput): GraphActionResult {
    keySchema.parse(effectKey);
    const usage = count.safeParse(value.actualTokens);
    const input = outcomeSchema.parse(GraphJsonValue.parse({ ...value, actualTokens: usage.success ? usage.data : null }));
    const outcome: GraphActionOutcome = input.state === "confirmed"
      ? { ...input, result: GraphResult.parse(input.result) }
      : { ...input, error: GraphJsonValue.parse(input.error) };
    return this.write(fence, (run) => {
      const prior = this.find(run, effectKey);
      if (prior.outcome) {
        if (canonical(prior.outcome) !== canonical(outcome)) fail("OUTCOME_CONFLICT");
        return { action: prior, changed: false, created: false };
      }
      if (prior.generation !== run.generation) fail("ACTION_GENERATION_MISMATCH");
      if (prior.state === "unknown" || prior.state === "cancel_pending") fail("UNRESOLVED_ACTION");
      if (prior.state !== "running" && !(prior.state === "intent" && outcome.state === "failed")) fail("INVALID_ACTION_STATE");
      const blocked = !usage.success || usage.data > prior.reservationTokens;
      this.db.prepare(`UPDATE graph_actions SET state = ?, outcome_json = ?, actual_tokens = ?, retry_safe = ?,
        accounting_blocked = ?, updated_at = ? WHERE effect_key = ?`).run(outcome.state, canonical(outcome), outcome.actualTokens,
          Number(outcome.state === "failed" && outcome.retrySafe), Number(blocked), fence.now, effectKey);
      return { action: this.find(run, effectKey), changed: true, created: false };
    });
  }
}

// Internal synchronous primitive shared by action and checkpoint storage. Never
// export a callback-taking transaction API from core's public entry point.
import type Database from "better-sqlite3";
import { GraphJsonValue } from "@osstack/protocol";
import { preflightDiskSpace } from "./db.js";
import { GraphRunStore, checkGraphRunFence, graphRunFenceSchema, type GraphRun, type GraphRunFence } from "./graph-runs.js";

export interface GraphStorageMutation<T> { value: T; changed: boolean }

/** Acquires BEGIN IMMEDIATE before any fence/cache read. Callback is internal,
 * synchronous SQL only; no adapter invocation, async authorization, or nested
 * transactions. Every mutation advances run revision; a read/cache hit does not.
 * Host must serialize requests and refresh its fence from the returned run. */
export function fencedGraphTransaction<T>(
  db: Database.Database,
  value: GraphRunFence,
  operation: (run: GraphRun) => GraphStorageMutation<T>,
): { value: T; run: GraphRun } {
  const input = graphRunFenceSchema.parse(GraphJsonValue.parse(value));
  if (db.inTransaction) throw new Error("Graph storage requires its own BEGIN IMMEDIATE transaction");
  preflightDiskSpace(db.name);
  return db.transaction(() => {
    const store = new GraphRunStore(db);
    const run = checkGraphRunFence(store.get(input.scope, input.graphRunId), input, true);
    const result = operation(run);
    // Runtime guard supplements the non-Promise callback return type. Callers are
    // trusted internal code; this cannot sandbox an intentionally async callback.
    if (!result || typeof result.changed !== "boolean" || "then" in result) throw new Error("Graph storage callback must be synchronous");
    if (result.changed) {
      db.prepare("UPDATE graph_runs SET revision = ?, updated_at = ? WHERE graph_run_id = ?")
        .run(run.revision + 1, input.now, run.graphRunId);
    }
    return { value: result.value, run: result.changed ? store.get(input.scope, input.graphRunId)! : run };
  }).immediate();
}

FILE packages/graph/src/route.ts
import { createHash } from 'node:crypto';
import { GraphRunState, GraphSpec, freezeWorkerData, workerCanonical } from '@osstack/protocol';
import { parseVerificationJoin } from './join.js';
import { parseVerificationBinding, verificationCanonical, verificationObject, type VerificationBinding } from './nodes/branch.js';
export interface VerificationActionCost { actions: number; tokens: number }
export interface VerificationRoutingState {
  binding: VerificationBinding; spec: GraphSpec; runStatus: GraphRunState;
  runCreatedAt: number; now: number; actionStarts: number;
  reservedTokens: string; chargedTokens: string; accountingBlocked: boolean;
  pendingAdmissions: number; unresolvedActions: number;
  branchAttemptsSpent: { review: number; tests: number };
  nextCost: { repair: VerificationActionCost; review: VerificationActionCost; tests: VerificationActionCost };
}
export type VerificationRoute =
  | { kind: 'ready'; epoch: number; evidenceDigest: string }
  | { kind: 'repair_required'; epoch: number; nextEpoch: number; evidenceDigest: string }
  | { kind: 'retry_required'; epoch: number; branches: Array<'review' | 'tests'> }
  | { kind: 'reconcile'; epoch: number; reason: string }
  | { kind: 'stop'; epoch: number; reason: string };
function count(value: unknown): number {
  if (typeof value !== 'number' || !Number.isSafeInteger(value) || value < 0) throw Error('INPUT_CONTRACT_INVALID: counter');
  return value;
}
function tokens(value: unknown): string {
  if (typeof value !== 'string' || !/^(?:0|[1-9][0-9]{0,127})$/.test(value)) throw Error('INPUT_CONTRACT_INVALID: token counter');
  return value;
}
function cost(value: unknown): VerificationActionCost {
  const object = verificationObject(value, ['actions', 'tokens']);
  const actions = count(object.actions); if (!actions) throw Error('INPUT_CONTRACT_INVALID: action cost');
  return { actions, tokens: count(object.tokens) };
}
function parseState(value: unknown): VerificationRoutingState {
  const s = verificationObject(value, ['binding', 'spec', 'runStatus', 'runCreatedAt', 'now', 'actionStarts', 'reservedTokens', 'chargedTokens', 'accountingBlocked', 'pendingAdmissions', 'unresolvedActions', 'branchAttemptsSpent', 'nextCost']);
  const binding = parseVerificationBinding(s.binding); const spec = GraphSpec.parse(s.spec);
  if (createHash('sha256').update(`osstack.graph.spec.v1\n${workerCanonical(spec)}`).digest('hex') !== binding.specHash
    || spec.input.acceptanceRef !== binding.acceptanceRef) throw Error('SPEC_MISMATCH: original routing limits');
  const attempts = verificationObject(s.branchAttemptsSpent, ['review', 'tests']);
  const costs = verificationObject(s.nextCost, ['repair', 'review', 'tests']);
  const runCreatedAt = count(s.runCreatedAt); const now = count(s.now);
  if (now < runCreatedAt || typeof s.accountingBlocked !== 'boolean') throw Error('INPUT_CONTRACT_INVALID: time/accounting');
  return { binding, spec, runStatus: GraphRunState.parse(s.runStatus), runCreatedAt, now, actionStarts: count(s.actionStarts),
    reservedTokens: tokens(s.reservedTokens), chargedTokens: tokens(s.chargedTokens), accountingBlocked: s.accountingBlocked,
    pendingAdmissions: count(s.pendingAdmissions), unresolvedActions: count(s.unresolvedActions),
    branchAttemptsSpent: { review: count(attempts.review), tests: count(attempts.tests) },
    nextCost: { repair: cost(costs.repair), review: cost(costs.review), tests: cost(costs.tests) } };
}
/** Deterministic recommendation over a host-verified join and original ledger
 * snapshot. It never allocates attempts, advances durable epochs, or authorizes
 * execution. The host must recheck the current fence/grant and atomically reserve
 * through the existing ledger before acting on retry_required/repair_required.
 * now is supplied by that host; the replayable orchestration does not read clocks. */
export function routeVerification(joinValue: unknown, stateValue: unknown): VerificationRoute {
  const joined = parseVerificationJoin(joinValue); const state = parseState(stateValue);
  if (verificationCanonical(joined.binding) !== verificationCanonical(state.binding)) throw Error('JOIN_REVISION_MISMATCH: routing snapshot');
  const epoch = state.binding.epoch; const limits = state.spec.limits;
  for (const kind of ['review', 'tests'] as const) {
    if (state.branchAttemptsSpent[kind] < joined.expected[kind].logicalAttempt) throw Error('INPUT_CONTRACT_INVALID: branch accounting');
  }
  const stop = (reason: string): VerificationRoute => ({ kind: 'stop', epoch, reason });
  const reconcile = (reason: string): VerificationRoute => ({ kind: 'reconcile', epoch, reason });
  if (['ready', 'failed', 'cancelled'].includes(state.runStatus)) return stop('terminal_run');
  if (state.runStatus === 'cancel_pending' || state.runStatus === 'reconciliation_required' || state.unresolvedActions > 0
    || joined.branches.some(b => b.state === 'unknown' || b.state === 'cancel_pending')) return reconcile('unresolved_execution');
  if (state.pendingAdmissions > 0) return reconcile('resource_release_pending');
  if (state.runStatus !== 'running') return stop('run_not_running');
  if (joined.branches.some(b => b.state === 'cancelled')) return stop('cancelled_branch');
  if (epoch >= limits.maxEpochs) return stop('epoch_budget');
  if (state.now - state.runCreatedAt >= limits.deadlineSeconds * 1000) return stop('deadline_budget');
  const spentTokens = BigInt(state.reservedTokens) + BigInt(state.chargedTokens);
  if (state.accountingBlocked || spentTokens > BigInt(limits.tokenReservationLimit) || state.actionStarts > limits.maxActionStarts) return stop('accounting_budget');
  if (joined.missing.length) return stop('missing_predecessor');
  const capacity = (costs: VerificationActionCost[]): boolean => {
    const actions = costs.reduce((sum, c) => sum + BigInt(c.actions), BigInt(state.actionStarts));
    const totalTokens = costs.reduce((sum, c) => sum + BigInt(c.tokens), spentTokens);
    return actions <= BigInt(limits.maxActionStarts) && totalTokens <= BigInt(limits.tokenReservationLimit);
  };
  const failed = joined.branches.filter(b => b.state === 'failed');
  if (failed.length) {
    if (failed.some(b => !b.error?.retryable || !b.logicalAttempt || !['OUTPUT_CONTRACT_INVALID', 'INTERNAL_ERROR'].includes(b.error.code))) return stop('branch_failure');
    if (failed.some(b => state.branchAttemptsSpent[b.kind] >= limits.maxBranchAttempts)) return stop('branch_budget');
    if (!capacity(failed.map(b => state.nextCost[b.kind]))) return stop('action_or_token_budget');
    return freezeWorkerData({ kind: 'retry_required', epoch, branches: failed.map(b => b.kind) });
  }
  const review = joined.branches.find(b => b.kind === 'review')?.result;
  const tests = joined.branches.find(b => b.kind === 'tests')?.result;
  if (joined.status !== 'complete' || review?.kind !== 'review' || tests?.kind !== 'tests') return stop('invalid_predecessor');
  if (review.verdict === 'accept' && tests.passed) return freezeWorkerData({ kind: 'ready', epoch, evidenceDigest: joined.evidenceDigest });
  if (epoch >= 2 || epoch + 1 >= limits.maxEpochs) return stop('epoch_budget');
  if (!capacity([state.nextCost.repair])) return stop('action_or_token_budget');
  return freezeWorkerData({ kind: 'repair_required', epoch, nextEpoch: epoch + 1, evidenceDigest: joined.evidenceDigest });
}
