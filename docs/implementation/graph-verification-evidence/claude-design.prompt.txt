  /** Trusted recovery service only. Commit the decision before verifying/releasing;
   * this method deliberately cannot mutate outcomes or allocate another attempt. */
  async releaseReconciled(effectKey: string, verify: (context: GraphRetainedAction) => Promise<void>): Promise<boolean> {
    const entry = this.retained.get(effectKey);
    if (!entry) return false;
    if (!entry.lease.releaseAfterReconciliation) throw new Error("lease reconciliation unsupported");
    const verification: unknown = await verify(snapshot(entry.context));
    if (verification !== undefined) throw new Error("invalid reconciliation authority");
    await entry.lease.releaseAfterReconciliation(() => verify(snapshot(entry.context)));
    this.admissions.released(effectKey, this.options.now?.() ?? Date.now());
    this.retained.delete(effectKey);
    return true;
  }

  private authorize(binding: Binding): void {
    const result: unknown = binding.authorize();
    if (result !== undefined) {
      // TypeScript allows async functions in void positions. Never silently
      // treat a pending/rejected promise as synchronous permission.
      void Promise.resolve(result).catch(() => {});
      throw new Denied("POLICY_DENIED", "revocation check must be synchronous");
    }
  }

  private fence(binding: Binding): GraphRunFence {
    const run = this.runs.get(binding.scope, binding.graphRunId);
    if (!run) throw new Denied("RUN_MISMATCH", "registered run not found");
    if (run.generation !== binding.generation || run.leaseOwner !== binding.owner) throw new Denied("STALE_GENERATION", "registered worker lease changed");
    if (canonical(run.identity) !== canonical(binding.identity)) throw new Denied("CHECKPOINT_INCOMPATIBLE", "registered program identity changed");
    const now = this.options.now?.() ?? Date.now();
    if (!run.leaseExpiresAt || run.leaseExpiresAt <= now) throw new Denied("STALE_GENERATION", "registered worker lease expired");
    return { scope: binding.scope, graphRunId: binding.graphRunId, identity: binding.identity,
      expectedGeneration: binding.generation, expectedOwner: binding.owner, expectedRevision: run.revision, now };
  }

  handle(connection: object, value: unknown, options: { signal?: AbortSignal } = {}): Promise<WorkerActionResponse> {
    return this.executeRequest(connection, value, options);
  }

  /** Trusted host command, never part of worker transport. Only this call owns the
   * freshly ledger-allocated retry intent; ordinary replays cannot claim it. */
  async retryAuthorized(connection: object, effectKey: string, authorizeRetry: () => void,
    options: { signal?: AbortSignal } = {}): Promise<WorkerActionResponse> {
    const binding = this.connections.get(connection);
    if (!binding) throw new Error("UNAUTHENTICATED");
    this.authorize(binding);
    const permitted: unknown = authorizeRetry();
    if (permitted !== undefined) {
      void Promise.resolve(permitted).catch(() => {});
      throw new Error("invalid synchronous retry authority");
    }
    if (options.signal?.aborted) throw new Error("retry cancelled");
    const prior = this.actions.get(this.fence(binding), effectKey).action;
    const registered = binding.actions.get(slot(prior.effect.nodeId, prior.effect.effectSlot));
    if (!registered || prior.effect.epoch !== binding.epoch || canonical(registered.action) !== canonical(prior.action)) throw new Error("unregistered retry slot");
    const next = this.actions.retry(this.fence(binding), effectKey).action;
    return this.executeRequest(connection, { schemaVersion: 1, kind: "worker_action", requestId: `retry:${randomUUID()}`,
      scope: binding.scope, graphRunId: binding.graphRunId, generation: binding.generation, specHash: binding.identity.specHash,
      epoch: binding.epoch, nodeId: prior.effect.nodeId, effectSlot: prior.effect.effectSlot, action: registered.action }, options, next.effectKey, authorizeRetry);
  }

        // it unknown. Only a future explicit host retry can use this proof;
        // cancellations remain sealed and worker replay never allocates attempts.

    }

  private async releaseAdmission(lease: ExecutionLease, binding: Binding | undefined, owned: GraphAction | undefined): Promise<void> {
    let physicallyReleased = false;
    try {
      await lease.release(); physicallyReleased = true;
      if (owned) this.admissions.released(owned.effectKey, this.options.now?.() ?? Date.now());
    }
    catch (error) {
      // Release failure deliberately stops the chain without replacing durable
      // effect truth. Keep its exact admission reachable for release-only recovery.
      if (!physicallyReleased) lease.retain(String(error));
      if (binding && owned) this.retained.set(owned.effectKey, { lease, context: {
        effectKey: owned.effectKey, scope: binding.scope, graphRunId: binding.graphRunId,
        generation: binding.generation, owner: binding.owner, workspacePath: binding.workspace.workspace_path,
      } });
      throw error;
    }
  }

  private response(request: WorkerActionRequest, action: GraphAction): WorkerActionResponse {
    const base = { schemaVersion: 1 as const, kind: "worker_action_response" as const, requestId: request.requestId,
      generation: request.generation, effect: action.effect };
    if (action.state === "confirmed") return WorkerActionResponse.parse({ ...base, outcome: { state: "confirmed", result: action.outcome?.state === "confirmed" ? action.outcome.result : undefined } });
    if (action.state === "failed") return WorkerActionResponse.parse({ ...base, outcome: { state: "failed", error: graphError("POLICY_DENIED", "prior invocation failed; no automatic retry") } });
    return { ...base, outcome: { state: "unknown", error: graphError("ACTION_OUTCOME_UNKNOWN", "existing action requires reconciliation; invocation was not repeated"), reconciliationRef: action.effectKey } };
  }

  private async execute(binding: Binding, action: BoundAction, signal: AbortSignal, beforeEffect: (executionRunId: string) => void, observeTokens: (tokens: number) => void, lease?: ExecutionLease): Promise<GraphActionEvidence> {
    const execution = action.execution;
    if (execution.kind === "freeze") {
      if (!lease?.assertActive) throw new Error("freeze requires canonical lease verification");
      const check = (): void => { if (signal.aborted) throw new Error("freeze cancelled"); this.authorize(binding); this.fence(binding); };
      await this.options.authorizePolicy?.(binding.workspace.workspace_path); check();
      await lease.assertActive(); check(); beforeEffect(`freeze:${randomUUID()}`);
      const frozenCandidate = await execution.store.freeze(execution.input, { ...execution.authority, lease,
        verifyLease: async (input, actualLease) => {
          if (actualLease !== lease) throw new Error("freeze lease identity mismatch");
          check(); await lease.assertActive!(); check();
          const decision: unknown = await execution.authority.verifyLease(input, actualLease);
          if (decision !== undefined) throw new Error("invalid freeze authority");
          check();
        } });
      return { status: "completed", frozenCandidate };
    }
    if (execution.kind === "workspace") {
      await this.options.authorizePolicy?.(binding.workspace.workspace_path);
      beforeEffect(`workspace:${randomUUID()}`);
      const result = await execution.handler({ workspace: binding.workspace, signal });
      if (result?.stopped !== true || !["completed", "failed"].includes(result.status)) throw new Error("workspace outcome unconfirmed");
      return { status: result.status, workspace: result.evidence };
    }
    let threw = false; let command: CommandResult | undefined; let invocationRunId: string | undefined;
    const original = execution.kind === "adapter" ? execution.adapter : undefined;
    const capabilities = execution.kind === "adapter" ? execution.adapter.capabilities().filter(cap => cap.id === execution.capabilityId) : [CapabilitySchema.parse({ id: "registered-command.write", kind: "code.write", agent_id: "registered-command", description: "Registered command" })];
    const policy = this.options.authorizePolicy;
    const cumulativeUsage: Record<string, number> = { tokens_in: 0, tokens_out: 0 };
    const wrapper: AgentAdapter = {
      id: original?.id ?? "registered-command", display_name: "Governed graph action", cost_tier: "low",
      capabilities: () => capabilities, isAvailable: original?.isAvailable ?? (async () => ({ available: true, last_checked_at: 0 })),
      invoke: (task, invocationSignal) => {
        return (async function* () {
          await policy?.(binding.workspace.workspace_path);
          if (invocationSignal.aborted) throw new Denied("POLICY_DENIED", "execution cancelled before invocation");
          beforeEffect(task.run_id); invocationRunId = task.run_id;
          try {
            if (original) {
              for await (const event of original.invoke(task, invocationSignal)) {
                const payload: unknown = event.type === "usage.updated" || event.type === "usage.final" ? event.payload
                  : event.type === "done" && event.payload && typeof event.payload === "object" ? (event.payload as { usage?: unknown }).usage : undefined;
                if (payload !== undefined) {
                  if (!payload || typeof payload !== "object" || Array.isArray(payload)) throw new Error("invalid adapter usage payload");
                  const usage = payload as Record<string, unknown>;
                  let tokens = 0;
                  for (const field of ["tokens_in", "tokens_out"]) {
                    const amount = usage[field];
                    if (amount !== undefined) {
                      if (typeof amount !== "number" || !Number.isSafeInteger(amount) || amount < 0) throw new Error("invalid adapter token usage");
                      cumulativeUsage[field] = Math.max(cumulativeUsage[field]!, amount);
                    }
                  }
                  // Providers may send input/output counters in separate events.
                  // Retain each cumulative maximum instead of only max(event sum).
                  tokens = cumulativeUsage["tokens_in"]! + cumulativeUsage["tokens_out"]!;
                  if (!Number.isSafeInteger(tokens)) throw new Error("invalid adapter total usage");
                  observeTokens(tokens);
                  const cost = usage["cost_usd"];
                  if (cost !== undefined && (typeof cost !== "number" || !Number.isFinite(cost) || cost < 0)) throw new Error("invalid adapter cost usage");
                }
                yield event;
              }
              return;
            }
            if (execution.kind !== "command") throw new Error("invalid execution registration");
            // runShellCommand starts synchronously before its returned promise; no
            // untrusted await separates the fence above from the actual spawn.
            command = await runShellCommand(execution.command, { cwd: binding.workspace.workspace_path, timeoutMs: 120_000,
              outputLimit: 16_000, ...(invocationSignal ? { signal: invocationSignal } : {}) });
            yield { event_id: randomUUID(), run_id: task.run_id, session_id: task.session_id, seq: 1, type: "done" as const,
              created_at: Date.now() / 1000, terminal: true, payload: { status: command.exitCode === 0 ? "completed" : "failed", last_seq: 0 } };
          } catch (error) { threw = true; throw error; }
        })();
      },
      confirmStopped: async runId => original?.confirmStopped ? original.confirmStopped(runId)
        : { stopped: command?.termination?.stop_confirmed === true, evidence: command?.termination?.evidence ?? "no trusted stop receipt" },
    };
    const candidate = await executeCandidateImplementation({ plan: { ...execution.plan, agent_id: wrapper.id }, workspace: binding.workspace,
      adapters: [wrapper], signal, cancellationGraceMs: 50,
      authorize: async () => { this.authorize(binding); this.fence(binding); return { allow: true }; } });
    if (!invocationRunId) throw new Denied("POLICY_DENIED", candidate.reason ?? "candidate preflight refused");
    if (threw || candidate.status === "execution_unconfirmed" || candidate.execution?.failure_code) throw new Error("adapter result or execution unconfirmed");
    let stopTimer: ReturnType<typeof setTimeout> | undefined;
    const stopped = await Promise.race([wrapper.confirmStopped!(invocationRunId),
      new Promise<{ stopped: false }>(resolve => { stopTimer = setTimeout(() => resolve({ stopped: false }), 1000); })])
      .finally(() => { if (stopTimer) clearTimeout(stopTimer); });
    if (stopped.stopped !== true || !stopped.evidence?.trim()) throw new Error("external stop unconfirmed");
    const usage = candidate.execution;
    if ([usage?.tokens_in, usage?.tokens_out].some(value => value !== undefined && (!Number.isSafeInteger(value) || value < 0))) throw new Error("invalid adapter usage");
    return { status: candidate.status === "implementation_complete" ? "completed" : "failed", candidate, ...(command ? { command } : {}) };
  }

import { acquireRepoWriteLock, FifoSemaphore, SemaphoreAcquireAbortedError, type SemaphoreLease, type RepoWriteLock } from '@osstack/autonomy';
import { resolveRepositoryScope } from '@osstack/core';

export interface ExecutionLease {
  release(): Promise<void>;
  /** Keep ownership until a trusted caller has confirmed external exit. */
  retain(reason: string): void;
  /** Trusted host only; never accept a worker-provided proof callback. */
  releaseAfterReconciliation?(verify: () => Promise<void>): Promise<void>;
  /** Assert this exact canonical writer lease is still live. */
  assertActive?(): Promise<void>;
}
export interface AdmissionInput {
  kind: 'read' | 'write';
  runId: string;
  /** Supplied by the daemon registration, never by an adapter context. */
  workspacePath?: string;
  repoId?: string;
  signal?: AbortSignal;
  authorize?: () => Promise<void>;
}
export class GovernedExecution {
  private readonly activeRuns = new Set<string>();
  private readonly quarantinedRepositories = new Map<string, Set<string>>();
  private quarantine(repoId: string, runId: string): void {
    const owners = this.quarantinedRepositories.get(repoId) ?? new Set<string>();
    owners.add(runId); this.quarantinedRepositories.set(repoId, owners);
  }
  constructor(private readonly options: {
    reads: FifoSemaphore;
    writes: FifoSemaphore;
    repositories: Map<string, FifoSemaphore>;
    checkHealth: () => Promise<void>;
  }) {}

  get hasWriters(): boolean { return this.options.writes.in_use > 0; }

  async acquire(input: AdmissionInput): Promise<ExecutionLease> {
    const check = async (): Promise<void> => {
      if (input.signal?.aborted) throw new SemaphoreAcquireAbortedError();
      if (input.kind === 'write') await this.options.checkHealth();
      await input.authorize?.();
    };
    await check();
    const scope = input.workspacePath === undefined ? undefined
      : await resolveRepositoryScope(input.workspacePath, input.repoId === undefined ? {} : { repoId: input.repoId });
    if (input.kind === 'write' && !scope) throw new Error('trusted repository scope required for writes');
    if (input.kind === "write" && scope && this.quarantinedRepositories.has(scope.repoId)) throw new Error("repository external stop unconfirmed; reconciliation required");
    if (this.activeRuns.has(input.runId)) throw new Error("run already admitted or queued");
    this.activeRuns.add(input.runId);
    let local: SemaphoreLease | undefined;
    let global: SemaphoreLease | undefined;
    let lock: RepoWriteLock | undefined;
    let queue: FifoSemaphore | undefined;
    const releasePermits = (): void => {
      global?.release(); local?.release();
      this.activeRuns.delete(input.runId);
      if (scope && queue?.in_use === 0 && queue.pending === 0) this.options.repositories.delete(scope.repoId);
    };
    try {
      if (input.kind === 'write' && scope) {
        queue = this.options.repositories.get(scope.repoId) ?? new FifoSemaphore(1);
        this.options.repositories.set(scope.repoId, queue);
        local = await queue.acquire(input.signal);
      }
      global = await (input.kind === 'read' ? this.options.reads : this.options.writes).acquire(input.signal);
      if (input.kind === 'write' && scope) {
        lock = await acquireRepoWriteLock({ root_dir: join(scope.commonDir, "osstack", "writer-locks"), recover_stale: false, repo_id: scope.repoId, autonomy_run_id: input.runId });
      }
      if (input.kind === "write" && scope && this.quarantinedRepositories.has(scope.repoId)) throw new Error("repository external stop unconfirmed; reconciliation required");
      await check();
    } catch (error) {
      try { await lock?.release(); } finally { releasePermits(); }
      throw error;
    }
    let released = false;
    let retained = false;
    // A renewal failure quarantines this repository. Its unknown producer still
    // consumes write capacity, while unrelated reads remain available.
    let renewing: Promise<void> = Promise.resolve();
    const timer = lock === undefined ? undefined : setInterval(() => {
      renewing = renewing.then(async () => {
        if (!released) await lock!.renew();
      }).catch((error: unknown) => {
        retained = true;
        if (scope) this.quarantine(scope.repoId, input.runId);
        process.stderr.write(`repository lease renewal failed; retained ${input.runId}: ${String(error)}\n`);
        if (timer) clearInterval(timer);
      });
    }, 20_000);
    timer?.unref();
    let reconciled: Promise<void> | undefined;
    return {
      assertActive: async () => {
        if (released || retained) throw new Error("execution lease unavailable");
        await lock?.assertActive();
        if (released || retained) throw new Error("execution lease unavailable");
      },
      releaseAfterReconciliation: (verify) => {
        if (released) return Promise.resolve();
        if (reconciled) return reconciled;
        if (typeof verify !== 'function') return Promise.reject(new Error('trusted reconciliation verifier required'));
        reconciled = (async () => {
          // Stop renewal first. No authority is freed until the verifier succeeds
          // and the canonical token is released under its filesystem guard.
          if (timer) clearInterval(timer);
          await renewing;
          const result: unknown = await verify();
          if (result !== undefined) throw new Error('invalid reconciliation authority');
          await lock?.release();
          released = true; retained = false;
          if (scope) {
            const owners = this.quarantinedRepositories.get(scope.repoId);
            owners?.delete(input.runId);
            if (owners?.size === 0) this.quarantinedRepositories.delete(scope.repoId);
          }
          releasePermits();
        })().catch((error: unknown) => {
          retained = true;
          if (scope) this.quarantine(scope.repoId, input.runId);
          throw error;
        }).finally(() => { reconciled = undefined; });
        return reconciled;
      },
      retain: (reason) => { retained = true; if (timer) clearInterval(timer); if (scope) this.quarantine(scope.repoId, input.runId); process.stderr.write(`osstack: retaining write lease ${input.runId}: ${reason}\n`); },
      release: async () => {
        if (released) return;
        if (retained) throw new Error("external stop unconfirmed; reconciliation required before release");
        released = true;
        if (timer) clearInterval(timer);
        await renewing;
        if (retained) { released = false; throw new Error("external stop unconfirmed; reconciliation required before release"); }
        // Ownership failure must not free the in-process writer authority.
        try { await lock?.release(); }
        catch (error) {
          released = false; retained = true;
          if (scope) this.quarantine(scope.repoId, input.runId);
          process.stderr.write(`repository lock release failed; retained ${input.runId}: ${String(error)}\n`);
          throw error;
        }
        releasePermits();
      },
    };
  }
}

/** Architecture isolation only: no DB/home/connection is supplied to this process.
 * This is NOT an OS sandbox against trusted code running as the same OS user. */
export function runGraphWorker(): void {
  const socket = new Socket({ fd: 3, readable: true, writable: true });
  const abort = new AbortController();
  let binding: GraphWorkerBinding | undefined;
  let client: GraphWorkerClient | undefined;
  let heartbeat: ReturnType<typeof setInterval> | undefined;
  let completed = false;
  const fail = (error: Error) => {
    abort.abort(); client?.close(error); if (heartbeat) clearInterval(heartbeat);
    // Only the manager's OS exit event can attest that this process has stopped.
    process.exitCode = 1; socket.destroy();
  };

    const message = GraphWorkerMessage.parse(value);
    if (binding) {
      if (workerCanonical(binding) !== workerCanonical(message.binding)) throw new Error('WORKER_IDENTITY');
      if (message.kind === 'cancel') { abort.abort(); client?.close(new Error('WORKER_CANCELLED')); return; }
      client!.receive(message); return;

    binding = freezeWorkerData(message.binding);
    client = new GraphWorkerClient(binding, transport, abort.signal, message.maxInFlight, message.requestTimeoutMs);
    heartbeat = setInterval(() => { try { transport.send({ kind: 'heartbeat', binding }); } catch (error) { fail(new Error(String(error))); } }, message.heartbeatMs);
    void execute(message, client).catch(error => finish(false, undefined, error));
  }, fail);
  function finish(ok: boolean, value?: GraphJsonValue, error?: unknown): void {
    if (completed) return; completed = true;
    try { transport.send({ kind: 'done', binding, ok, ...(ok ? { value } : { error: String(error).slice(0, 4096) }) }); }
    catch (sendError) { fail(new Error(String(sendError))); }
    // Remain alive until host stop. done/invoke rejection deliberately is not exit.
  }
  async function execute(init: Extract<GraphWorkerMessage, { kind: 'init' }>, contextClient: GraphWorkerClient): Promise<void> {
    const identity = init.binding.identity;
    if (identity.engineVersion !== GRAPH_KERNEL.version || identity.checkpointVersion !== GRAPH_KERNEL.checkpointVersion
      || createHash('sha256').update(init.source).digest('hex') !== identity.compiledProgramDigest) throw new Error('PROGRAM_IDENTITY');
    if (computeGraphInputDigest(init.input) !== identity.inputDigest || workerCanonical(init.input.scope) !== workerCanonical(init.binding.scope)
      || createHash('sha256').update(`osstack.graph.spec.v1\n${workerCanonical(init.input.spec)}`).digest('hex') !== identity.specHash) throw new Error('INPUT_IDENTITY');
    // Import the verified snapshot, not a path that could change between hash and import.
    const program = await import(`data:text/javascript;base64,${Buffer.from(init.source).toString('base64')}`) as GraphWorkerProgram;
    if (typeof program.run !== 'function') throw new Error('WORKER_PROGRAM_INVALID');

    const resume = init.resume ? freezeWorkerData(init.resume) : undefined;
    const context = Object.freeze({ binding: init.binding, input, ...(resume ? { resume } : {}),
      action: contextClient.action, checkpoint: contextClient.checkpoint, signal: abort.signal, task });
    const workflow = entrypoint({ name: init.programId, checkpointer: new RemoteCheckpointSaver(init.binding.graphRunId, contextClient.checkpoint) },
      async (_input: StartGraphRequest) => GraphJsonValue.parse(await program.run(context)));
    if (abort.signal.aborted) throw new Error('WORKER_CANCELLED');
    transport.send({ kind: 'ready', binding });
    const invocation = resume ? resume.kind === 'continue' ? null : new Command({ resume: resume.value }) : input;
    const result = await workflow.invoke(invocation, { configurable: { thread_id: init.binding.graphRunId }, durability: 'sync', signal: abort.signal });
    finish(true, GraphJsonValue.parse(result));
  }
}

/** Decoded JSON limits; transport MUST also cap raw bytes before JSON.parse. */
export const GRAPH_JSON_LIMITS = Object.freeze({ maxBytes: 65_536, maxDepth: 16, maxValues: 4_096, maxStringLength: 16_384 });

function isBoundedJson(value: unknown): boolean {
  let values = 0;
  let bytes = 0;
  const ancestors = new Set<object>();
  function visit(item: unknown, depth: number): boolean {
    if (++values > GRAPH_JSON_LIMITS.maxValues || depth > GRAPH_JSON_LIMITS.maxDepth) return false;
    if (item === null || typeof item === "boolean" || (typeof item === "number" && Number.isFinite(item))) {
      bytes += Buffer.byteLength(JSON.stringify(item));
    } else if (typeof item === "string") {
      if (item.length > GRAPH_JSON_LIMITS.maxStringLength) return false;
      bytes += Buffer.byteLength(JSON.stringify(item));
    } else if (typeof item === "object") {
      if (ancestors.has(item)) return false;
      const array = Array.isArray(item);
      const proto = Object.getPrototypeOf(item);
      if (array ? proto !== Array.prototype : proto !== Object.prototype && proto !== null) return false;
      const keys = Reflect.ownKeys(item);
      if (keys.length > GRAPH_JSON_LIMITS.maxValues) return false;
      if (array && (item.length > GRAPH_JSON_LIMITS.maxValues || keys.length !== item.length + 1)) return false;
      ancestors.add(item);
      bytes += 2;
      let index = 0;
      for (const key of keys) {
        if (array && key === "length") continue;
        if (typeof key !== "string" || key.length > 256 || ["__proto__", "constructor", "prototype"].includes(key)) return false;
        if (array && key !== String(index++)) return false;
        const descriptor = Object.getOwnPropertyDescriptor(item, key);
        if (!descriptor?.enumerable || !("value" in descriptor)) return false;
        bytes += (array ? 0 : Buffer.byteLength(JSON.stringify(key)) + 1) + 1;
        if (bytes > GRAPH_JSON_LIMITS.maxBytes || !visit(descriptor.value, depth + 1)) return false;
      }
      ancestors.delete(item);
    } else return false;
    return bytes <= GRAPH_JSON_LIMITS.maxBytes;
  }
  // IPC must supply decoded JSON, never executable objects. Accessors are not read.
  // A hostile in-process Proxy is not a supported transport input.
  try { return visit(value, 0); } catch { return false; }
}

    if (!isBoundedJson(value)) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, message: "Expected bounded JSON-only graph data" });
      return z.NEVER;
    }
    return value;
  }, schema);
}

const WorkerResponse = z.object({
  schemaVersion: Version, kind: z.literal("worker_action_response"), requestId: Id, generation: Count.min(1), effect: Effect,
  outcome: z.discriminatedUnion("state", [
    z.object({ state: z.literal("confirmed"), result: Result }).strict(),
    z.object({ state: z.literal("intent") }).strict(),
    z.object({ state: z.literal("running"), executionRef: Id }).strict(),
    z.object({ state: z.literal("failed"), error: z.union([ErrorObject.options[0], ErrorObject.options[2]]) }).strict(),
    z.object({ state: z.literal("unknown"), error: ErrorObject, reconciliationRef: Id }).strict(),
    z.object({ state: z.literal("cancel_pending"), error: ErrorObject, executionRef: Id }).strict(),
    z.object({ state: z.literal("reconciliation_required"), error: ErrorObject, reconciliationRef: Id }).strict(),
    z.object({ state: z.literal("cancelled"), receiptRef: Id }).strict(),
  ]),
  metadata: Metadata.optional(),
}).strict();
export const WorkerActionResponse = wire(z.discriminatedUnion("kind", [
  WorkerResponse,
  // No scope, generation or effect is asserted when authentication/admission fails.
  z.object({ schemaVersion: Version, kind: z.literal("worker_action_rejected"), requestId: Id, error: ErrorObject }).strict(),
]).superRefine((response, ctx) => {
  if (response.kind !== "worker_action_response") return;
  const outcome = response.outcome;
  if ("error" in outcome) {
    if ((outcome.error.nodeId !== undefined && outcome.error.nodeId !== response.effect.nodeId) ||
        (outcome.error.epoch !== undefined && outcome.error.epoch !== response.effect.epoch)) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, path: ["outcome", "error"], message: "Error location must match effect" });
    }
    const expectedCodes = outcome.state === "cancel_pending" ? ["CANCEL_UNCONFIRMED"]
      : outcome.state === "unknown" || outcome.state === "reconciliation_required" ? ["ACTION_OUTCOME_UNKNOWN", "CANCEL_UNCONFIRMED"] : undefined;
    if (expectedCodes && !expectedCodes.includes(outcome.error.code)) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, path: ["outcome", "error", "code"], message: "Error code must describe uncertain execution" });
    }
  }
  if ((outcome.state === "unknown" || outcome.state === "cancel_pending" || outcome.state === "reconciliation_required") &&
      outcome.error.retryable) {
    ctx.addIssue({ code: z.ZodIssueCode.custom, path: ["outcome", "error"], message: "Uncertain execution cannot be retried" });
  }
}));
export type WorkerActionResponse = z.infer<typeof WorkerActionResponse>;

const TransportContext = wire(z.object({
  schemaVersion: Version, scope: Scope,
  worker: z.object({ graphRunId: Id, generation: Count.min(1), specHash: Digest, epoch: Epoch }).strict().optional(),
}).strict());
/** Trusted integration input, deliberately not part of the wire/schema manifest. */
export type AuthenticatedGraphTransportContext = z.infer<typeof TransportContext>;
/**
 * Trusted daemon code must resolve an authenticated connection and its current
 * lease independently of the request/checkpoint. This is an integration seam,
 * NOT authentication implemented by this package. Never derive this callback's
 * result from worker scope, caller flags or a cast. G10/G12 own real verification,
 * authorization, revocation and atomic fencing. Resolve async registry/lease data
 * before calling this synchronous boundary, in the protected admission section. Remote identities fail closed
 * until that integration can verify them; there is no local-owner fallback.
 */
export type ResolveAuthenticatedGraphTransport = () => AuthenticatedGraphTransportContext | null;

export class GraphContractError extends Error {
  constructor(public readonly code: GraphErrorCode) {
    super(code);
    this.name = "GraphContractError";
  }
}

function authenticate(resolve: ResolveAuthenticatedGraphTransport): AuthenticatedGraphTransportContext {
  if (typeof resolve !== "function") throw new GraphContractError("UNAUTHENTICATED");
  try {
    const result = TransportContext.safeParse(resolve());
    if (result.success) return result.data;
  } catch { /* Fail closed on unavailable/failed authentication. */ }
  throw new GraphContractError("UNAUTHENTICATED");
}

function canonical(value: GraphJsonValue): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  return `{${Object.keys(value).sort().map((key) => `${JSON.stringify(key)}:${canonical(value[key]!)}`).join(",")}}`;
}
function sameScope(left: GraphScope, right: GraphScope): void {
  if (canonical(left) !== canonical(right)) throw new GraphContractError("SCOPE_MISMATCH");
}

function parseContract<T extends z.ZodTypeAny>(schema: T, value: unknown, code: GraphErrorCode = "INPUT_CONTRACT_INVALID"): z.output<T> {
  const parsed = schema.safeParse(value);
  if (!parsed.success) throw new GraphContractError(code === "INPUT_CONTRACT_INVALID" && parsed.error.issues.some((issue) => issue.path[0] === "spec") ? "SPEC_INVALID" : code);
  return parsed.data;
}

/** Control-plane mutations require a principal connection, never a worker lease. */
export function parseInboundGraphRequest(value: unknown, resolve: ResolveAuthenticatedGraphTransport): GraphRequest {
  const context = authenticate(resolve);
  if (context.worker) throw new GraphContractError("UNAUTHENTICATED");
  const request = parseContract(GraphRequest, value);
  sameScope(request.scope, context.scope);
  return request;
}

export function parseInboundWorkerActionRequest(value: unknown, resolve: ResolveAuthenticatedGraphTransport): WorkerActionRequest {
  const context = authenticate(resolve);
  if (!context.worker) throw new GraphContractError("UNAUTHENTICATED");
  const request = parseContract(WorkerActionRequest, value);
  sameScope(request.scope, context.scope);
  if (request.graphRunId !== context.worker.graphRunId) throw new GraphContractError("RUN_MISMATCH");
  if (request.generation !== context.worker.generation) throw new GraphContractError("STALE_GENERATION");
  if (request.specHash !== context.worker.specHash) throw new GraphContractError("SPEC_MISMATCH");
  if (request.epoch !== context.worker.epoch) throw new GraphContractError("EPOCH_MISMATCH");
  return request;
}

/** Pair data received on an authenticated daemon channel. This does not verify artifact truth. */
export function validateWorkerActionResponse(requestValue: unknown, responseValue: unknown): WorkerActionResponse {
  const request = parseContract(WorkerActionRequest, requestValue);
  const response = parseContract(WorkerActionResponse, responseValue, "OUTPUT_CONTRACT_INVALID");
  const mismatch = (): never => { throw new GraphContractError("RESPONSE_MISMATCH"); };
  if (response.requestId !== request.requestId) mismatch();
  if (response.kind === "worker_action_rejected") return response;
  if (response.generation !== request.generation) mismatch();
  sameScope(request.scope, response.effect.scope);
  for (const key of ["graphRunId", "specHash", "epoch", "nodeId", "effectSlot"] as const) {
    if (request[key] !== response.effect[key]) mismatch();
  }
  if (response.outcome.state === "confirmed") {
    const kinds: Record<WorkerActionRequest["action"]["kind"], GraphResult["kind"]> = {
      implement: "implementation_complete", repair: "implementation_complete", freeze: "candidate_ready",
      review: "review", tests: "tests", workspace_create: "workspace", workspace_release: "workspace",
    };
    const result = response.outcome.result;
    if (result.kind !== kinds[request.action.kind]) mismatch();
    if ("candidateRevision" in request.action && "candidateRevision" in result && request.action.candidateRevision !== result.candidateRevision) mismatch();
    if ("workspaceLeaseRef" in request.action && "workspaceLeaseRef" in result && request.action.workspaceLeaseRef !== result.workspaceLeaseRef) mismatch();
  }
  return response;
}

function hash(domain: string, value: GraphJsonValue): string {
  return createHash("sha256").update(`${domain}\n${canonical(value)}`, "utf8").digest("hex");
}
/**
 * Digest v1 = SHA-256 of domain + LF + canonical strictly parsed business JSON.
 * Only root metadata (trace/requestedAt) and root idempotencyKey are omitted.
 * Kind, scope, source, expectedRevision and all nested data remain significant.
 * Object keys sort by JS UTF-16 order; arrays retain order; JSON number encoding.
 */
export function computeGraphInputDigest(value: unknown): string {
  const { metadata: _metadata, idempotencyKey: _key, ...business } = GraphRequest.parse(value);
  return hash("osstack.graph.input.v1", GraphJsonValue.parse(business));
}

export function computeGraphEffectKey(value: unknown): string {
  return hash("osstack.graph.effect.v1", GraphEffectIdentity.parse(value));
}

export const GraphIdempotencyRecord = wire(z.object({
  schemaVersion: Version, scope: Scope, idempotencyKey: Id, inputDigest: Digest,
}).strict());
export type GraphIdempotencyRecord = z.infer<typeof GraphIdempotencyRecord>;
const MutationContext = wire(z.object({ run: z.object({ scope: Scope, graphRunId: Id, currentRevision: Count }).strict().optional(), prior: GraphIdempotencyRecord.optional() }).strict());

/**
 * Pure decision over a trusted snapshot, NOT durable dedup/CAS or authorization.
 * After authentication/fencing, look up prior by full scope + key (all kinds).
 * Same key/digest replays even if revision has since advanced; changed digest
 * conflicts. A fresh resume/cancel must match currentRevision exactly. Start has
 * no revision. G08 must bind decision, unique key and mutation in one transaction.
 */
export function validateGraphMutation(value: unknown, snapshot: unknown): "apply" | "replay" {
  const request = parseContract(GraphRequest, value);
  const { prior, run } = parseContract(MutationContext, snapshot);
  if (request.kind === "start" && run) throw new GraphContractError("INPUT_CONTRACT_INVALID");
  if (request.kind !== "start" && run) {
    sameScope(request.scope, run.scope);
    if (request.graphRunId !== run.graphRunId) throw new GraphContractError("RUN_MISMATCH");
  }
  if (prior) {
    sameScope(prior.scope, request.scope);
    if (prior.idempotencyKey !== request.idempotencyKey || prior.inputDigest !== computeGraphInputDigest(request)) {
      throw new GraphContractError("IDEMPOTENCY_CONFLICT");
    }
    return "replay";
  }
  if (request.kind !== "start" && request.expectedRevision !== run?.currentRevision) throw new GraphContractError("REVISION_CONFLICT");
  return "apply";
}

/** Same-generation transport retransmission: keep requestId and every business field. */
export function validateWorkerTransportRetry(previous: unknown, retry: unknown): "replay" {
  const { metadata: _previousMetadata, ...first } = WorkerActionRequest.parse(previous);
  const { metadata: _retryMetadata, ...second } = WorkerActionRequest.parse(retry);
  if (canonical(first) !== canonical(second)) throw new GraphContractError("IDEMPOTENCY_CONFLICT");
  return "replay";
}
