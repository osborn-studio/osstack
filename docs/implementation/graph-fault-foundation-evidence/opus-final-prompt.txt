## P0 — `spawn_failed` is not proof of "no child", but is consumed as such

**`packages/cli/src/graph-worker-manager.ts` `spawnFailed` → `graph-process-journal.ts` `GraphProcessJournal.observe` → `packages/core/src/state/graph-recovery.ts` (`w.kind === "not_started"` block).**

`child.on('error', spawnFailed)` is not spawn-specific. Node emits `'error'` on a `ChildProcess` when the process could not be spawned **and also when the process could not be killed, and when sending a message fails**. `phase` is then inferred from `registered` (a registration-progress flag), not from whether a pid was ever assigned:

```ts
phase: registered ? 'spawn' as const : 'registration' as const
```

`observe()` concludes `leader: { state: 'absent', reason: 'authenticated_no_child' }` whenever there is no `launched` event and any `spawn_failed` event exists. `GraphRecoveryStore` then accepts a `not_started` proof for a consumed generation on exactly `events.length === 1 && events[0].kind === 'spawn_failed'`.

Reproduction (low probability, high severity):
1. `consume()` commits (`consumed = true`), broker/checkpoint registration succeeds.
2. Child spawns. The `'spawn'` handler runs `assertActive()` **first** (newly added in this diff) and throws because a concurrent `stop`/`cancel` deactivated the record — so `lifecycle.launched` never runs and **no `launched` row exists** for a process that is alive.
3. `cancel()` kills; the kill fails (ESRCH/EPERM/permission-changed child) → `'error'` → `spawnFailed` → `observe()` appends `spawn_failed` (`phase: 'spawn'`).
4. Journal now holds exactly one event: `spawn_failed`. `observeWorkerProcess` reports `leader.state === 'absent'`, and recovery admits a `not_started` takeover **while the worker is still running**.

The invariant "a `spawn_failed` row implies no child ever existed" is claimed by both the journal comment and the recovery comment but is enforced nowhere. Minimum fix: gate the no-child classification on `child?.pid === undefined` (or a `spawned` boolean set in the `'spawn'` handler *before* `assertActive()`), and record post-spawn `'error'` events under a distinct kind that recovery does not accept.

## P0 — Journal write failures at exit are silently discarded

**`packages/cli/src/graph-worker-manager.ts`, `observe()`.**

function deferred<T>() {
  let resolve!: (value: T) => void; let reject!: (error: Error) => void;
  const promise = new Promise<T>((res, rej) => { resolve = res; reject = rej; });
  // A caller may wait for exit first. Internal observation prevents early lifecycle
  // failures from creating unhandled rejections without changing the public promise.
  void promise.catch(() => {});
  return { promise, resolve, reject };
}
function boundedOption(value: number | undefined, fallback: number, max: number): number {
  const result = value ?? fallback;

  return result;
}

function shuffle<T>(items: T[]): T[] {
  let state = seed || 1;
  for (let i = items.length - 1; i > 0; i--) {
    state ^= state << 13; state ^= state >>> 17; state ^= state << 5;
    const index = (state >>> 0) % (i + 1);

import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';

export type GraphProcessIdentity =
  | { kind: 'linux'; pid: number; pgid: number; bootId: string; startTicks: string }
  | { kind: 'unverifiable'; pid: number; pgid: number | null; platform: string; detail: string };
export type GraphProcessObservation = {
  state: 'running' | 'absent' | 'unknown';
  reason: string;
};
function validPid(pid: number): void {
  if (!Number.isSafeInteger(pid) || pid < 1) throw new Error('GRAPH_PROCESS_PID');
}
function linuxIdentity(pid: number): GraphProcessIdentity {
  const bootId = readFileSync('/proc/sys/kernel/random/boot_id', 'utf8').trim();
  const stat = readFileSync(`/proc/${pid}/stat`, 'utf8');
  // comm can contain spaces and parentheses. Fields after its final ')' start at state (3).
  const fields = stat.slice(stat.lastIndexOf(')') + 2).trim().split(/\s+/);
  const pgid = Number(fields[2]); const startTicks = fields[19];
  if (!bootId || !startTicks || !/^\d+$/.test(startTicks) || !Number.isSafeInteger(pgid) || pgid < 1
    || Number(stat.slice(0, stat.indexOf(' '))) !== pid) throw new Error('GRAPH_PROCESS_STAT');
  return { kind: 'linux', pid, pgid, bootId, startTicks };
}
/** Called for the direct child before init. A coarse start timestamp is diagnostic only. */
export function captureGraphProcessIdentity(pid: number): GraphProcessIdentity {
  validPid(pid);
  try {
    if (process.platform === 'linux') return linuxIdentity(pid);
    if (process.platform === 'darwin') {
      const detail = execFileSync('/bin/ps', ['-p', String(pid), '-o', 'pgid=', '-o', 'lstart='],
        { encoding: 'utf8', timeout: 1000, stdio: ['ignore', 'pipe', 'pipe'], env: { LC_ALL: 'C' } }).trim();
      const pgid = Number(detail.split(/\s+/)[0]);
      return { kind: 'unverifiable', pid, pgid: Number.isSafeInteger(pgid) && pgid > 0 ? pgid : null,
        platform: process.platform, detail: `coarse_lstart:${detail}` };
    }
  } catch { /* Missing or denied OS identity must not become a stop proof. */ }
  return { kind: 'unverifiable', pid, pgid: null, platform: process.platform, detail: 'os_identity_unavailable' };
}
export function compareGraphProcessIdentity(saved: GraphProcessIdentity, current: GraphProcessIdentity): GraphProcessObservation {
  if (saved.kind !== 'linux' || current.kind !== 'linux') return { state: 'unknown', reason: 'identity_unverifiable' };
  if (saved.pid !== current.pid || saved.pgid !== current.pgid || saved.bootId !== current.bootId || saved.startTicks !== current.startTicks) {
    return { state: 'unknown', reason: 'identity_mismatch' };
  }
  return { state: 'running', reason: 'same_linux_process' };
}
/** Read-only leader observation. Never proves descendants or opaque adapter effects stopped. */
export function observeGraphProcessIdentity(saved: GraphProcessIdentity): GraphProcessObservation {
  try { validPid(saved.pid); } catch { return { state: 'unknown', reason: 'identity_invalid' }; }
  if (saved.kind !== 'linux' || process.platform !== 'linux') return { state: 'unknown', reason: 'identity_unverifiable' };
  try {
    if (readFileSync('/proc/sys/kernel/random/boot_id', 'utf8').trim() !== saved.bootId) return { state: 'unknown', reason: 'boot_identity_mismatch' };

import type Database from 'better-sqlite3';
import { GraphRunStore } from '@osstack/core';
import { GraphWorkerBinding, workerCanonical } from '@osstack/protocol';
import { observeGraphProcessIdentity, type GraphProcessObservation } from './graph-process-identity.js';
import type { GraphWorkerLaunch, GraphWorkerExitEvidence } from './graph-worker-manager.js';

type ProcessEvent = { kind: 'launched'; launch: GraphWorkerLaunch } | GraphWorkerExitEvidence;
export interface GraphWorkerProcessReport {
  binding: GraphWorkerBinding;
  intentRecorded: boolean;
  launch: GraphWorkerLaunch | null;
  observations: GraphWorkerExitEvidence[];
  leader: GraphProcessObservation;
  executors: { state: 'unknown'; reason: 'opaque_adapters_and_descendants_unproven' };
  recovery: 'held';
}
/** Append-only host observations, not recovery attestations. */
export class GraphProcessJournal {
  constructor(private readonly db: Database.Database) {}
  private binding(value: GraphWorkerBinding): GraphWorkerBinding {
    const binding = GraphWorkerBinding.parse(value);
    const run = new GraphRunStore(this.db).get(binding.scope, binding.graphRunId);
    if (!run || workerCanonical(run.identity) !== workerCanonical(binding.identity)) throw new Error('GRAPH_PROCESS_BINDING');
    return binding;
  }
  private intent(binding: GraphWorkerBinding): boolean {
    const row = this.db.prepare('SELECT owner, identity_json FROM graph_worker_launches WHERE graph_run_id = ? AND generation = ?')
      .get(binding.graphRunId, binding.generation) as { owner: string; identity_json: string } | undefined;
    if (!row) return false;
    if (row.owner !== binding.owner || row.identity_json !== workerCanonical(binding.identity)) throw new Error('GRAPH_PROCESS_BINDING');
    return true;
  }
  consume(value: GraphWorkerBinding): void {
    const binding = this.binding(value);
    if (this.intent(binding)) throw new Error('GRAPH_GENERATION_ALREADY_LAUNCHED');
    this.db.prepare('INSERT INTO graph_worker_launches(graph_run_id,generation,owner,identity_json,launched_at) VALUES (?,?,?,?,?)')
      .run(binding.graphRunId, binding.generation, binding.owner, workerCanonical(binding.identity), Date.now());
  }
  private append(value: GraphWorkerBinding, event: ProcessEvent): void {
    const binding = this.binding(value);
    if (!this.intent(binding)) throw new Error('GRAPH_PROCESS_INTENT_REQUIRED');
    this.db.prepare('INSERT INTO graph_worker_process_events(graph_run_id,generation,kind,event_json,observed_at) VALUES (?,?,?,?,?)')
      .run(binding.graphRunId, binding.generation, event.kind, workerCanonical(event), Date.now());
  }
  launched(binding: GraphWorkerBinding, launch: GraphWorkerLaunch): void {
    if (!Number.isSafeInteger(launch.pid) || launch.pid < 1 || launch.processIdentity.pid !== launch.pid
      || (launch.pgid !== null && (!Number.isSafeInteger(launch.pgid) || launch.pgid < 1))
      || (launch.processIdentity.pgid !== null && launch.processIdentity.pgid !== launch.pgid)) throw new Error('GRAPH_PROCESS_LAUNCH_IDENTITY');
    this.append(binding, { kind: 'launched', launch });
  }
  observed(binding: GraphWorkerBinding, evidence: GraphWorkerExitEvidence): void {
    if (evidence.kind !== 'exit_unconfirmed' && (evidence.graphRunId !== binding.graphRunId
      || evidence.generation !== binding.generation || evidence.owner !== binding.owner)) throw new Error('GRAPH_PROCESS_BINDING');
    this.append(binding, evidence);
  }
  observe(value: GraphWorkerBinding): GraphWorkerProcessReport {
    const binding = this.binding(value); const intentRecorded = this.intent(binding);
    const rows = this.db.prepare('SELECT event_json FROM graph_worker_process_events WHERE graph_run_id = ? AND generation = ? ORDER BY sequence')
      .all(binding.graphRunId, binding.generation) as { event_json: string }[];
    const events = rows.map(row => JSON.parse(row.event_json) as ProcessEvent);
    const launch = events.find((event): event is Extract<ProcessEvent, { kind: 'launched' }> => event.kind === 'launched')?.launch ?? null;
    const observations = events.filter((event): event is GraphWorkerExitEvidence => event.kind !== 'launched');
    const leader: GraphProcessObservation = launch ? observeGraphProcessIdentity(launch.processIdentity)
      : observations.some(event => event.kind === 'spawn_failed') ? { state: 'absent', reason: 'authenticated_no_child' }
      : { state: 'unknown', reason: intentRecorded ? 'launch_window_unobserved' : 'no_launch_intent' };
    return { binding, intentRecorded, launch, observations, leader,
      executors: { state: 'unknown', reason: 'opaque_adapters_and_descendants_unproven' }, recovery: 'held' };
  }
}
