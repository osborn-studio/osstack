115:     const prior = this.store.history(request.fence).find((r) => r.decision.receiptId === request.receiptId);
116:     if (prior && prior.snapshot.action?.effectKey !== snapshot.action?.effectKey) fail("DECISION_CONFLICT");
117:     if (prior && prior.snapshot.run.generation !== snapshot.run.generation) {
118:       // A lost takeover ACK is a read of immutable history, never permission to
119:       // stop the replacement worker. Release of an old effect is a separate,
120:       // explicit operation; do not infer authority over a newer generation.
121:       if (request.kind === "takeover" && prior.decision.kind === "takeover") {
122:         if (request.owner !== prior.decision.owner || request.leaseExpiresAt !== prior.decision.leaseExpiresAt) fail("DECISION_CONFLICT");
123:         if (snapshot.run.generation === prior.snapshot.run.generation + 1 && snapshot.run.leaseOwner === prior.decision.owner) {
124:           return { exit: "takeover", result: { run: snapshot.run, action: snapshot.action, record: prior, changed: false } };
125:         }
126:       }
127:       fail("RECOVERY_GENERATION_CHANGED");
128:     }
129:     const proof = await this.proof(connection, registration, snapshot);
130:     let decision: GraphRecoveryDecision;
131:     if (request.kind === "effect") {

133:       // Match storage normalization on replay: invalid usage means unknown usage,
134:       // while the physical outcome/result remains durable and immutable.
135:       const verdict = received && received.outcome !== "unknown" &&
136:         (typeof received.actualTokens !== "number" || !Number.isSafeInteger(received.actualTokens) || received.actualTokens < 0)
137:         ? { ...received, actualTokens: null } : received;
138:       this.check(connection, registration, snapshot);
139:       decision = { kind: "effect", receiptId: request.receiptId, verdict };
140:     } else if (request.kind === "project") decision = { kind: "project", receiptId: request.receiptId, status: request.status };
141:     else decision = { kind: "takeover", receiptId: request.receiptId, owner: request.owner, leaseExpiresAt: request.leaseExpiresAt };
142:     // Proof time includes outcome verification, but no snapshot revision is refreshed.
143:     proof.verifiedAt = this.now();
144:     let result: GraphRecoveryResult;
145:     if (prior) {
146:       if (canonical(prior.decision) !== canonical(decision)) fail("DECISION_CONFLICT");
147:       result = this.verified(connection, registration, snapshot, proof, decision, () => this.store.verifyDecision(snapshot, request.receiptId, proof, this.now()));
148:     } else result = this.verified(connection, registration, snapshot, proof, decision, () => this.store.commit(snapshot, decision, proof, this.now()));
149:     const exit = decision.kind === "effect" ? decision.verdict.outcome : decision.kind === "project" ? "projected" : "takeover";
150:     if (exit === "unknown" || decision.kind !== "effect") return { exit, result };
151:     // Decision is already durable. Release failure never rolls back physical truth.
152:     try { await this.release(connection, registration, result); }
153:     catch (error) { return { exit: "release_pending", result, releaseError: error instanceof Error ? error.message : "release failed" }; }
154:     return { exit, result };
155:   }
