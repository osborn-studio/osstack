export function deliveryCanonical(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(deliveryCanonical).join(",")}]`;
  if (value !== null && typeof value === "object")
    return `{${Object.keys(value)
      .sort()
      .map((k) => `${JSON.stringify(k)}:${deliveryCanonical((value as Record<string, unknown>)[k])}`)
      .join(",")}}`;
  const encoded = JSON.stringify(value);
  if (encoded === undefined) throw new Error("invalid_delivery_value");
  return encoded;
}
const hash = (value: unknown) => createHash("sha256").update(deliveryCanonical(value)).digest("hex");
const bodyHash = (body: string) => createHash("sha256").update(body).digest("hex");

export function serializeFollowUpDelivery(value: unknown): string {
  return JSON.stringify(value).replace(/[\u007f-\u009f\u2028\u2029\p{Cf}]/gu, (c) =>
    c
      .split("")
      .map((u) => `\\u${u.charCodeAt(0).toString(16).padStart(4, "0")}`)
      .join(""),
  );
}
function bounded<T>(value: T): T {
  if (Buffer.byteLength(serializeFollowUpDelivery(value)) + 1 > 65536)
    throw new FollowUpDeliveryError("delivery_response_too_large");
  return value;
}
function parsed<T extends z.ZodTypeAny>(schema: T, value: unknown): z.infer<T> {
  const result = schema.safeParse(value);
  if (!result.success) throw new FollowUpDeliveryError("invalid_delivery_request");
  return result.data;
}
function repoPath(path: string): string {
  try {
    if (!isAbsolute(path) || /[\p{C}]/u.test(path)) throw Error();
    const repo = realpathSync(path);
    if (!statSync(repo).isDirectory() || Buffer.byteLength(repo) > 4096) throw Error();
    accessSync(repo, constants.R_OK | constants.X_OK);
    return repo;
  } catch {
    throw new FollowUpDeliveryError("delivery_repository_unavailable");
  }
}
function storedRepoPath(path: string): string {
  if (!isAbsolute(path) || resolve(path) !== path || /[\p{C}]/u.test(path) || Buffer.byteLength(path) > 4096)
    throw new FollowUpDeliveryError("delivery_repository_unavailable");
  return path;
}
function frozen<T>(value: T): T {
  if (value !== null && typeof value === "object") {
    Object.freeze(value);
    for (const child of Object.values(value)) frozen(child);
  }
  return value;
}

  async handle(method: string, input: unknown): Promise<FollowUpDeliveryResult | FollowUpDeliveryListResult> {
    switch (method) {
      case "follow-up-delivery.prepare":
        return this.prepare(input);
      case "follow-up-delivery.list":
        return this.list(input);
      case "follow-up-delivery.inspect":
        return this.inspect(input);
      case "follow-up-delivery.authorize":
        return this.authorize(input);
      case "follow-up-delivery.revoke":
        return this.revoke(input);
      case "follow-up-delivery.dispatch":
        return this.dispatch(input);
      case "follow-up-delivery.reconcile":
        return this.reconcile(input);
      default:
        throw new FollowUpDeliveryError("delivery_unsupported_method");
    }
  }
  prepare(input: unknown): FollowUpDeliveryResult {
    const p = parsed(FollowUpDeliveryPrepareParams, input),
      repo = repoPath(p.repo_path),
      createDigest = hash({ ...p, repo_path: repo });
    return this.safe(() =>
      this.durableWrite(() => {
        const existing = this.db
          .prepare(
            "SELECT * FROM follow_up_delivery_attempts WHERE repo_path=? AND tenant='default' AND idempotency_key=?",
          )
          .get(repo, p.idempotency_key) as Row | undefined;
        if (existing) {
          if (existing.create_digest !== createDigest)
            throw new FollowUpDeliveryError("delivery_idempotency_conflict");
          return this.result(existing, "unchanged", true);
        }
        this.obligation(repo, p.obligation_id, p.expected_obligation_revision);
        if (
          this.db
            .prepare(
              "SELECT 1 FROM follow_up_delivery_attempts WHERE repo_path=? AND tenant='default' AND obligation_id=? AND state IN ('prepared','started','delivery_unknown')",
            )
            .get(repo, p.obligation_id)
        )
          throw new FollowUpDeliveryError("delivery_unresolved_attempt");
        const target = this.target(repo, p.destination_ref),
          now = this.clock(),
          attemptId = randomUUID();
        const payload = {
          repo_path: repo,
          default_tenant: true as const,
          attempt_id: attemptId,
          obligation_id: p.obligation_id,
          obligation_revision: p.expected_obligation_revision,
          destination: target,
          destination_digest: hash(target),
          body: p.body,
          body_sha256: bodyHash(p.body),
        };
        const envelope = FollowUpDeliveryEnvelope.parse({ ...payload, envelope_digest: hash(payload) });
        if (Buffer.byteLength(JSON.stringify(envelope)) > 49152)
          throw new FollowUpDeliveryError("delivery_response_too_large");
        this.db
          .prepare(
            "INSERT INTO follow_up_delivery_attempts(attempt_id,repo_path,tenant,obligation_id,obligation_revision,idempotency_key,create_digest,envelope_json,state,revision,created_at_ms,updated_at_ms,qualification) VALUES(?,?,'default',?,?,?,?,?,'prepared',1,?,?,'qualification_pending')",
          )
          .run(
            attemptId,
            repo,
            p.obligation_id,
            p.expected_obligation_revision,
            p.idempotency_key,
            createDigest,
            JSON.stringify(envelope),
            now,
            now,
          );
        const row = this.row(repo, attemptId);
        this.reserveReceiptSpace(row);
        this.event(attemptId, 1, "prepared", { envelope_digest: envelope.envelope_digest }, now);
        return this.result(row, "prepared");
      }),
    );
  }
  inspect(input: unknown): FollowUpDeliveryResult {
    const p = parsed(FollowUpDeliveryInspectParams, input),
      repo = storedRepoPath(p.repo_path);
    return this.safe(() => this.result(this.row(repo, p.attempt_id), "unchanged"));
  }
  list(input: unknown): FollowUpDeliveryListResult {
    const p = parsed(FollowUpDeliveryListParams, input),
      repo = storedRepoPath(p.repo_path);
    return this.safe(() => {
      const rows = this.db
        .prepare(
          "SELECT * FROM follow_up_delivery_attempts WHERE repo_path=? AND tenant='default' AND attempt_id>? ORDER BY attempt_id LIMIT ?",
        )
        .all(repo, p.after_id ?? "", p.limit + 1) as Row[];
      const items: z.infer<typeof FollowUpDeliveryListItem>[] = [];
      let after: string | null = null;
      for (let i = 0; i < Math.min(rows.length, p.limit); i++) {
        const row = rows[i]!,
          view = this.view(row),
          { envelope, authorization_subject: _subject, receipt: _receipt, ...rest } = view;
        const item = FollowUpDeliveryListItem.parse({
          ...rest,
          destination_ref: envelope.destination.destination_ref,
          body_sha256: envelope.body_sha256,
        });
        const more = i + 1 < rows.length,
          candidate = {
            schema_version: 1 as const,
            scope: { repo_path: repo, default_tenant: true as const },
            attempts: [...items, item],
            next_after_id: more ? row.attempt_id : null,
          };
        if (Buffer.byteLength(serializeFollowUpDelivery(candidate)) + 1 > 65536) {
          if (!items.length) throw new FollowUpDeliveryError("delivery_response_too_large");
          after = items[items.length - 1]!.attempt_id;
          break;
        }
        items.push(item);
        after = candidate.next_after_id;
      }
      return bounded(
        FollowUpDeliveryListResult.parse({
          schema_version: 1,
          scope: { repo_path: repo, default_tenant: true },
          attempts: items,
          next_after_id: after,
        }),
      );
    });
  }
  authorize(input: unknown): FollowUpDeliveryResult {
    const p = parsed(FollowUpDeliveryAuthorizeParams, input),
      repo = repoPath(p.repo_path);
    return this.safe(() =>
      this.durableWrite(() => {
        const row = this.row(repo, p.attempt_id),
          existing = this.grant(row.attempt_id);
        if (existing && deliveryCanonical(JSON.parse(existing.signed_json)) === deliveryCanonical(p.grant))
          return this.result(row, "unchanged", true);
        this.revision(row, p.expected_revision);
        if (row.state !== "prepared") throw new FollowUpDeliveryError("delivery_terminal");
        if (existing) throw new FollowUpDeliveryError("delivery_invalid_authorization");
        const now = this.clock(row.updated_at_ms);
        this.verifyGrant(row, p.grant, now);
        if (
          this.db
            .prepare("SELECT 1 FROM follow_up_delivery_grants WHERE grant_id=?")
            .get(p.grant.body.grant_id)
        )
          throw new FollowUpDeliveryError("delivery_invalid_authorization");
        this.db
          .prepare(
            "INSERT INTO follow_up_delivery_grants(grant_id,attempt_id,key_id,signed_json,state,created_at_ms,updated_at_ms) VALUES(?,?,?,?,'active',?,?)",
          )
          .run(p.grant.body.grant_id, row.attempt_id, p.grant.key_id, JSON.stringify(p.grant), now, now);
        return this.result(
          this.bump(row, "prepared", "authorized", now, {
            grant_id: p.grant.body.grant_id,
            key_id: p.grant.key_id,
          }),
          "authorized",
        );
      }),
    );
  }
  revoke(input: unknown): FollowUpDeliveryResult {
    const p = parsed(FollowUpDeliveryRevokeParams, input),
      repo = storedRepoPath(p.repo_path);
    return this.safe(() =>
      this.durableWrite(() => {
        const row = this.row(repo, p.attempt_id);
        this.revision(row, p.expected_revision);
        if (["delivered", "definitely_not_sent"].includes(row.state))
          return this.result(row, "unchanged", true);
        const now = this.evidenceClock(row.updated_at_ms);
        this.db
          .prepare(
            "UPDATE follow_up_delivery_grants SET state='revoked',updated_at_ms=? WHERE attempt_id=? AND state='active'",
          )
          .run(now, row.attempt_id);
        return this.result(
          this.bump(
            row,
            row.state === "prepared" ? "definitely_not_sent" : "delivery_unknown",
            row.state === "prepared" ? "revoked_before_send" : "revoked_after_start",
            now,
            { reason: p.reason, timestamp_basis: "monotone_revocation_record" },
          ),
          "revoked",
        );
      }),
    );
  }
  async dispatch(input: unknown): Promise<FollowUpDeliveryResult> {
    const p = parsed(FollowUpDeliveryActionParams, input),
      repo = repoPath(p.repo_path);
    const claim = this.safe(() =>
      this.durableWrite(() => {
        const row = this.row(repo, p.attempt_id);
        this.revision(row, p.expected_revision);
        if (row.state !== "prepared")
          return {
            result: this.result(
              row,
              row.state === "started"
                ? "in_flight"
                : row.state === "delivery_unknown"
                  ? "delivery_unknown"
                  : "unchanged",
              true,
            ),
          };
        const envelope = this.envelope(row),
          transport = this.transport(envelope);
        if (!transport) return { result: this.result(row, "qualification_pending") };
        const grant = this.grant(row.attempt_id);
        if (!grant) throw new FollowUpDeliveryError("delivery_authorization_required");
        if (grant.state !== "active") throw new FollowUpDeliveryError("delivery_authorization_revoked");
        const now = this.clock(row.updated_at_ms);
        // Verify current key, exact target, immutable content and time INSIDE the consuming transaction.
        this.verifyGrant(row, SignedFollowUpDeliveryGrant.parse(JSON.parse(grant.signed_json)), now);
        this.obligation(repo, row.obligation_id, row.obligation_revision);
        const consumed = this.db
          .prepare(
            "UPDATE follow_up_delivery_grants SET state='consumed',updated_at_ms=? WHERE attempt_id=? AND state='active'",
          )
          .run(now, row.attempt_id);
        if (consumed.changes !== 1) throw new FollowUpDeliveryError("delivery_authorization_revoked");
        const started = this.bump(
          row,
          "started",
          "send_started",
          now,
          { grant_id: grant.grant_id },
          transport.qualification,
        );
        return { row: started, envelope: frozen(envelope), transport };
      }),
    );
    if ("result" in claim) return claim.result;
    // No retry or fallback exists here. A throw after this call cannot prove non-delivery.
    const outcome = await this.boundedCall((signal) => claim.transport.sendOnce(claim.envelope, signal));
    return this.safe(() =>
      this.durableWrite(() => {
        const current = this.row(repo, p.attempt_id);
        if (current.state !== "started" || current.revision !== claim.row.revision)
          return this.result(current, "unchanged", true);
        const now = this.evidenceClock(current.updated_at_ms),
          receipt = outcome.ok ? this.receipt(claim.envelope, outcome.value) : null;
        if (!receipt)
          return this.result(
            this.bump(
              current,
              "delivery_unknown",
              outcome.ok ? "receipt_unconfirmed" : "send_outcome_unknown",
              now,
              {},
            ),
            "delivery_unknown",
          );
        return this.result(
          this.bump(
            current,
            "delivered",
            "provider_receipt",
            now,
            { receipt_sha256: hash(receipt) },
            current.qualification,
            receipt,
          ),
          "delivered",
        );
      }),
    );
  }
  async reconcile(input: unknown): Promise<FollowUpDeliveryResult> {
    const p = parsed(FollowUpDeliveryActionParams, input),
      repo = storedRepoPath(p.repo_path);
    const claim = this.safe(() =>
      this.durableWrite(() => {
        let row = this.row(repo, p.attempt_id);
        this.revision(row, p.expected_revision);
        if (row.state === "prepared") throw new FollowUpDeliveryError("delivery_authorization_required");
        if (["delivered", "definitely_not_sent"].includes(row.state))
          return { result: this.result(row, "unchanged", true) };
        const envelope = this.envelope(row),
          transport = this.transport(envelope);
        if (!transport?.lookup_receipt || transport.qualification !== row.qualification)
          return { result: this.result(row, "qualification_pending") };
        const now = this.evidenceClock(row.updated_at_ms);
        if (row.state === "started") row = this.bump(row, "delivery_unknown", "started_unconfirmed", now, {});
        return { row, envelope: frozen(envelope), transport };
      }),
    );
    if ("result" in claim) return claim.result;
    const outcome = await this.boundedCall((signal) => claim.transport.lookup(claim.envelope, signal));
    return this.safe(() =>
      this.durableWrite(() => {
        const row = this.row(repo, p.attempt_id);
        if (row.state !== "delivery_unknown" || row.revision !== claim.row.revision)
          return this.result(row, "unchanged", true);
        const now = this.evidenceClock(row.updated_at_ms),
          receipt = outcome.ok ? this.receipt(claim.envelope, outcome.value) : null;
        if (!receipt)
          return this.result(
            this.bump(row, "delivery_unknown", "lookup_unconfirmed", now, {}),
            "delivery_unknown",
          );
        return this.result(
          this.bump(
            row,
            "delivered",
            "provider_lookup_receipt",
            now,
            { receipt_sha256: hash(receipt) },
            row.qualification,
            receipt,
          ),
          "delivered",
        );
      }),
    );
  }
  private async boundedCall(
    call: (signal: AbortSignal) => Promise<unknown>,
  ): Promise<{ ok: true; value: unknown } | { ok: false }> {
    const controller = new AbortController();
    let timer: NodeJS.Timeout | undefined;
    try {
      return await Promise.race([
        Promise.resolve()
          .then(() => call(controller.signal))
          .then(
            (value) => ({ ok: true as const, value }),
            () => ({ ok: false as const }),
          ),
        new Promise<{ ok: false }>((resolve) => {
          timer = setTimeout(
            () => {
              controller.abort();
              resolve({ ok: false });
            },
            Math.max(1, Math.min(30000, this.options.timeoutMs ?? 5000)),
          );
        }),
      ]);
    } finally {
      if (timer) clearTimeout(timer);
      controller.abort();
    }
  }
  private target(repo: string, ref: string): FollowUpDeliveryTarget {
    const result = FollowUpDeliveryTarget.safeParse(this.options.targets?.get(ref));
    if (!result.success || result.data.destination_ref !== ref || result.data.repo_path !== repo)
      throw new FollowUpDeliveryError("delivery_destination_unavailable");
    return result.data;
  }
  private transport(envelope: FollowUpDeliveryEnvelope): FollowUpTransport | undefined {
    const current = this.target(envelope.repo_path, envelope.destination.destination_ref);
    if (hash(current) !== envelope.destination_digest)
      throw new FollowUpDeliveryError("delivery_destination_unavailable");
    const t = this.options.transports?.get(current.provider_id);
    if (
      !t ||
      t.automatic_retries ||
      !(["test_only", "sandbox_qualified", "production_qualified"] as string[]).includes(t.qualification) ||
      (t.qualification === "test_only" && !this.options.allowTestTransports)
    )
      return undefined;
    return t;
  }
  private subject(envelope: FollowUpDeliveryEnvelope) {
    return FollowUpDeliverySubject.parse({
      domain: "osstack.follow-up-delivery.v1",
      repo_path: envelope.repo_path,
      default_tenant: true,
      attempt_id: envelope.attempt_id,
      provider_id: envelope.destination.provider_id,
      account_id: envelope.destination.account_id,
      destination_digest: envelope.destination_digest,
      body_sha256: envelope.body_sha256,
      envelope_digest: envelope.envelope_digest,
    });
  }
  private verifyGrant(row: Row, grant: SignedFollowUpDeliveryGrant, now: number): void {
    const envelope = this.envelope(row),
      body = grant.body,
      { grant_id: _id, issued_at_ms, expires_at_ms, max_calls: _calls, ...subject } = body,
      key = this.options.trustedKeys?.get(grant.key_id);
    if (
      key?.purpose !== "follow_up_delivery" ||
      key.publicKey.type !== "public" ||
      key.publicKey.asymmetricKeyType !== "ed25519" ||
      key.repo_path !== row.repo_path ||
      key.provider_id !== envelope.destination.provider_id ||
      !key.destination_refs.includes(envelope.destination.destination_ref) ||
      !Number.isSafeInteger(key.max_validity_ms) ||
      key.max_validity_ms < 1 ||
      expires_at_ms - issued_at_ms > key.max_validity_ms ||
      deliveryCanonical(subject) !== deliveryCanonical(this.subject(envelope))
    )
      throw new FollowUpDeliveryError("delivery_invalid_authorization");
    if (now < issued_at_ms || now >= expires_at_ms)
      throw new FollowUpDeliveryError("delivery_authorization_expired");
    if (
      hash(this.target(row.repo_path, envelope.destination.destination_ref)) !== envelope.destination_digest
    )
      throw new FollowUpDeliveryError("delivery_destination_unavailable");
    if (
      !verify(
        null,
        Buffer.from(deliveryCanonical(body)),
        key.publicKey,
        Buffer.from(grant.signature, "base64"),
      )
    )
      throw new FollowUpDeliveryError("delivery_invalid_authorization");
  }
  private receipt(envelope: FollowUpDeliveryEnvelope, raw: unknown): FollowUpDeliveryReceipt | null {
    const result = FollowUpDeliveryReceipt.safeParse(raw);
    if (!result.success) return null;
    const r = result.data;
    return r.provider_id === envelope.destination.provider_id &&
      r.account_id === envelope.destination.account_id &&
      r.target_node_id === envelope.destination.target_node_id &&
      r.author_node_id === envelope.destination.actor_node_id &&
      r.destination_digest === envelope.destination_digest &&
      r.body_sha256 === envelope.body_sha256 &&
      r.attempt_id === envelope.attempt_id
      ? r
      : null;
  }
  private envelope(row: Row): FollowUpDeliveryEnvelope {
    const e = FollowUpDeliveryEnvelope.parse(JSON.parse(row.envelope_json)),
      { envelope_digest, ...payload } = e;
    if (
      e.attempt_id !== row.attempt_id ||
      e.repo_path !== row.repo_path ||
      e.obligation_id !== row.obligation_id ||
      e.obligation_revision !== row.obligation_revision ||
      e.body_sha256 !== bodyHash(e.body) ||
      e.destination_digest !== hash(e.destination) ||
      envelope_digest !== hash(payload)
    )
      throw new FollowUpDeliveryError("delivery_database_unavailable");
    return e;
  }
  private grant(id: string): GrantRow | undefined {
    return this.db.prepare("SELECT * FROM follow_up_delivery_grants WHERE attempt_id=?").get(id) as
      | GrantRow
      | undefined;
  }
  private row(repo: string, id: string): Row {
    const r = this.db
      .prepare(
        "SELECT * FROM follow_up_delivery_attempts WHERE repo_path=? AND tenant='default' AND attempt_id=?",
      )
      .get(repo, id) as Row | undefined;
    if (!r) throw new FollowUpDeliveryError("delivery_not_found");
    return r;
  }
  private revision(row: Row, expected: number) {
    if (row.revision !== expected) throw new FollowUpDeliveryError("delivery_revision_conflict");
  }
  private obligation(repo: string, id: string, revision: number) {
    const row = this.db
      .prepare(
        "SELECT revision FROM follow_up_obligations WHERE repo_path=? AND tenant='default' AND obligation_id=?",
      )
      .get(repo, id) as { revision: number } | undefined;
    if (!row || row.revision !== revision) throw new FollowUpDeliveryError("delivery_obligation_changed");
  }
  private readClock(): number {
    const now = (this.options.nowMs ?? Date.now)();
    if (Number.isSafeInteger(now) && now >= 0 && now <= 8_640_000_000_000_000)
      this.observedClock = Math.max(this.observedClock, now);
    return now;
  }
  private highWater(): number {
    return (
      this.db.prepare("SELECT observed_at_ms FROM follow_up_delivery_clock WHERE singleton=1").get() as {
        observed_at_ms: number;
      }
    ).observed_at_ms;
  }
  private clock(minimum = 0): number {
    const now = this.readClock();
    if (
      !Number.isSafeInteger(now) ||
      now < Math.max(minimum, this.highWater()) ||
      now > 8_640_000_000_000_000
    )
      throw new FollowUpDeliveryError("delivery_invalid_clock");
    return now;
  }
  private evidenceClock(minimum: number): number {
    this.readClock();
    return Math.max(minimum, this.observedClock, this.highWater());
  }
  /** FULL commit is required before an irreversible call. No await permits connection sharing here. */
  private durableWrite<T>(call: () => T): T {
    if (this.db.inTransaction || this.db.pragma("journal_mode", { simple: true }) !== "wal")
      throw new FollowUpDeliveryError("delivery_durability_unavailable");
    const observedBefore = this.observedClock;
    const previous = this.db.pragma("synchronous", { simple: true }) as number;
    const previousFullFsync = this.db.pragma("fullfsync", { simple: true }) as number;
    const persistClock = () => {
      this.db
        .prepare("UPDATE follow_up_delivery_clock SET observed_at_ms=MAX(observed_at_ms,?) WHERE singleton=1")
        .run(this.observedClock);
    };
    try {
      this.db.pragma("synchronous=FULL");
      if (process.platform === "darwin") this.db.pragma("fullfsync=ON");
      if (
        this.db.pragma("synchronous", { simple: true }) !== 2 ||
        (process.platform === "darwin" && this.db.pragma("fullfsync", { simple: true }) !== 1)
      )
        throw new FollowUpDeliveryError("delivery_durability_unavailable");
      try {
        preflightDiskSpace(this.dbPath);
        return this.db
          .transaction(() => {
            const result = call();
            persistClock();
            return result;
          })
          .immediate();
      } catch (error) {
        // Failed expiry checks also advance the observed clock: a later rollback cannot revive the grant.
        if (this.observedClock > observedBefore) {
          preflightDiskSpace(this.dbPath);
          this.db.transaction(persistClock).immediate();
        }
        throw error;
      }
    } finally {
      this.db.pragma(`synchronous=${previous}`);
      if (process.platform === "darwin") this.db.pragma(`fullfsync=${previousFullFsync}`);
    }
  }
  private safe<T>(call: () => T): T {
    try {
      return call();
    } catch (error) {
      if (error instanceof FollowUpDeliveryError) throw error;
      throw new FollowUpDeliveryError("delivery_database_unavailable");
    }
  }
  private bump(
    row: Row,
    state: Row["state"],
    reason: string,
    now: number,
    evidence: unknown,
    qualification = row.qualification,
    receipt: FollowUpDeliveryReceipt | null = null,
  ): Row {
    if (row.revision >= Number.MAX_SAFE_INTEGER - 1)
      throw new FollowUpDeliveryError("delivery_revision_conflict");
    const result = this.db
      .prepare(
        "UPDATE follow_up_delivery_attempts SET state=?,revision=revision+1,updated_at_ms=?,qualification=?,receipt_json=?,receipt_sha256=?,reason=? WHERE attempt_id=? AND revision=?",
      )
      .run(
        state,
        now,
        qualification,
        receipt ? JSON.stringify(receipt) : null,
        receipt ? hash(receipt) : null,
        reason,
        row.attempt_id,
        row.revision,
      );
    if (result.changes !== 1) throw new FollowUpDeliveryError("delivery_revision_conflict");
    this.event(row.attempt_id, row.revision + 1, reason, evidence, now);
    return this.row(row.repo_path, row.attempt_id);
  }
  private event(id: string, revision: number, transition: string, evidence: unknown, now: number) {
    this.db
      .prepare(
        "INSERT INTO follow_up_delivery_events(attempt_id,revision,transition,evidence_json,created_at_ms) VALUES(?,?,?,?,?)",
      )
      .run(id, revision, transition, JSON.stringify(evidence), now);
  }
  private view(row: Row): FollowUpDeliveryAttempt {
    const envelope = this.envelope(row);
    return FollowUpDeliveryAttempt.parse({
      attempt_id: row.attempt_id,
      repo_path: row.repo_path,
      default_tenant: true,
      obligation_id: row.obligation_id,
      obligation_revision: row.obligation_revision,
      state: row.state,
      revision: row.revision,
      created_at_ms: row.created_at_ms,
      updated_at_ms: row.updated_at_ms,
      envelope,
      authorization_subject: this.subject(envelope),
      authorization_status: this.grant(row.attempt_id)?.state ?? "missing",
      qualification: row.qualification,
      receipt: row.receipt_json === null ? null : JSON.parse(row.receipt_json),
      receipt_sha256: row.receipt_sha256,
      reason: row.reason,
      delivery_uncertain: row.state === "started" || row.state === "delivery_unknown",
      promise_fulfillment: "independent_local_obligation",
    });
  }
  /** Reserve the largest valid receipt before an irreversible transport call is possible. */
  private reserveReceiptSpace(row: Row): void {
    const view = this.view(row),
      e = view.envelope;
    const receipt: FollowUpDeliveryReceipt = {
      provider_id: e.destination.provider_id,
      account_id: e.destination.account_id,
      target_node_id: e.destination.target_node_id,
      author_node_id: e.destination.actor_node_id,
      destination_digest: e.destination_digest,
      body_sha256: e.body_sha256,
      attempt_id: e.attempt_id,
      provider_message_id: "x" + "\0".repeat(255),
      provider_request_id: "x" + "\0".repeat(255),
    };
    bounded({
      schema_version: 1,
      outcome: "delivery_unknown",
      attempt: {
        ...view,
        state: "delivery_unknown",
        revision: Number.MAX_SAFE_INTEGER - 1,
        updated_at_ms: 8_640_000_000_000_000,
        qualification: "production_qualified",
        authorization_status: "consumed",
        receipt,
        receipt_sha256: "f".repeat(64),
        reason: "x".repeat(128),
        delivery_uncertain: true,
      },
      duplicate: false,
    });
  }
  private result(
    row: Row,
    outcome: FollowUpDeliveryResult["outcome"],
    duplicate = false,
  ): FollowUpDeliveryResult {
    return bounded(
      FollowUpDeliveryResult.parse({ schema_version: 1, outcome, attempt: this.view(row), duplicate }),
    );
  }

// SQLite connection management.
//
// Decisions locked in plan v0.6:
//   - better-sqlite3 (synchronous, single connection per daemon process)
//   - WAL mode for concurrent reader/writer safety
//   - synchronous=NORMAL (WAL-safe, faster than FULL)
//   - 64 MiB page cache
//   - foreign_keys ON (so ON DELETE CASCADE works for runs/events on task delete)
//
// CRITICAL GAP G1 (plan v0.6 §8.3): disk-full must never cause a silent task
// write loss. Every write goes through preflightDiskSpace(), which throws
// DiskFullError before any INSERT/UPDATE can corrupt or stall a transaction.

import Database from "better-sqlite3";
import { statfsSync } from "node:fs";
import { dirname } from "node:path";
import { CURRENT_SCHEMA_VERSION, MIGRATIONS } from "./migrations.js";

/** Default safety threshold for disk-full preflight (100 MiB). */
export const DEFAULT_MIN_FREE_BYTES = 100 * 1024 * 1024;

/** Cache size in -kibibyte units. negative = KiB (per SQLite docs). 64 MiB. */
const PAGE_CACHE_KIB = -65536;

export class DiskFullError extends Error {
  readonly path: string;
  readonly freeBytes: number;
  readonly minFreeBytes: number;
  constructor(path: string, freeBytes: number, minFreeBytes: number) {
    super(
      `Disk full preflight failed for ${path}: ${freeBytes} bytes free, ` +
      `need ≥${minFreeBytes}. Refusing write so Hermes can be notified.`,
    );
    this.name = "DiskFullError";
    this.path = path;
    this.freeBytes = freeBytes;
    this.minFreeBytes = minFreeBytes;
  }
}

export interface OpenDbOptions {
  /** Absolute path to the SQLite file (parent dir must exist). */
  path: string;
  /** When false, opens read-only. */
  readwrite?: boolean;
  /** Override the disk-full safety threshold (bytes). */
  minFreeBytes?: number;
}

/**
 * Run a free-space check against the filesystem holding `path`. Throws
 * DiskFullError if below threshold; otherwise returns free byte count.
 * Skips the check (returns Infinity) if statfsSync is unsupported on the
 * platform — better to allow writes than to crash a daemon on a kernel quirk.
 */
export function preflightDiskSpace(
  path: string,
  minFreeBytes: number = DEFAULT_MIN_FREE_BYTES,
): number {
  let stats: { bsize: number; bavail: number };
  try {
    stats = statfsSync(dirname(path));
  } catch {
    return Number.POSITIVE_INFINITY;
  }
  const free = Number(stats.bsize) * Number(stats.bavail);
  if (free < minFreeBytes) throw new DiskFullError(path, free, minFreeBytes);
  return free;
}

/**
 * Open the osstack state database, apply pragmas, and run pending migrations.
 * Caller owns the lifetime — pair every openDb() with a db.close() in shutdown.
 */
export function openDb(opts: OpenDbOptions): Database.Database {
  const { path, readwrite = true, minFreeBytes = DEFAULT_MIN_FREE_BYTES } = opts;
  if (readwrite) preflightDiskSpace(path, minFreeBytes);

  const db = new Database(path, { readonly: !readwrite });

  try {
    // Refuse an unsupported schema before changing journal mode or running DDL,
    // including read-only inspection. A failed open never escapes with a live handle.
    assertSupportedVersion(db.pragma("user_version", { simple: true }) as number);
    assertMigrationLineage(db);
    // Pragmas are applied per-connection in SQLite. Apply on every open.
    if (readwrite) db.pragma("journal_mode = WAL");
    db.pragma("synchronous = NORMAL");
    db.pragma(`cache_size = ${PAGE_CACHE_KIB}`);
    db.pragma("foreign_keys = ON");
    db.pragma("busy_timeout = 5000");

    if (readwrite) runMigrations(db);
    return db;
  } catch (error) {
    db.close();
    throw error;
  }
}

// This schema is table-additive. Derive expected table introductions once;
// a future DROP/RENAME migration must also update this lineage contract.
const tableIntroductions = MIGRATIONS.flatMap(migration =>
  [...migration.sql.matchAll(/^\s*CREATE\s+TABLE(?:\s+IF\s+NOT\s+EXISTS)?\s+([a-zA-Z_]\w*)\s*\(/gm)]
    .map(match => ({ name: match[1]!, version: migration.id })));

/** Reject pure or mixed unreleased Graph schemas before application pragmas or
 * DDL. SQLite may itself checkpoint an abandoned WAL on close; logical ledger
 * content and user_version must remain intact for the original binary. */
function assertMigrationLineage(db: Database.Database): void {
  const current = db.pragma("user_version", { simple: true }) as number;
  const actual = new Set((db.prepare("SELECT name FROM sqlite_master WHERE type='table'").all() as { name: string }[]).map(row => row.name));
  // Experimental Graph or Attention tables must never be migrated as released main.
  const experimentalTable = (name: string) => name.startsWith("graph_") || name.startsWith("attention_") || name === "daemon_owner";
  if (current < 25 && ![...actual].some(experimentalTable)) return;
  const expected = new Set(tableIntroductions.filter(table => table.version <= current).map(table => table.name));
  if ([...expected].some(name => !actual.has(name))
    || [...actual].some(name => experimentalTable(name) && !expected.has(name))) {
    throw new Error("GRAPH_EXPERIMENTAL_SCHEMA_CONFLICT: preserve this home and reconcile with its original build; do not reset user_version or reuse it for the integrated runtime");
  }
}

function assertSupportedVersion(current: number): void {
  if (current > CURRENT_SCHEMA_VERSION) {
    throw new Error(`Migration state inconsistent: user_version=${current}, expected=${CURRENT_SCHEMA_VERSION}`);
  }
}

/**
 * Apply pending migrations idempotently. `user_version` PRAGMA tracks progress.
 * The version pragma guarantees forward-only application. Many migrations
 * contain non-idempotent CREATE/ALTER statements; never reset user_version.
 */
export function runMigrations(db: Database.Database): void {
  const currentRow = db.pragma("user_version", { simple: true }) as number;
  let current = Number(currentRow);
  assertSupportedVersion(current);
  assertMigrationLineage(db);
  for (const m of MIGRATIONS) {
    if (m.id <= current) continue;
    db.exec("BEGIN");
    try {
      db.exec(m.sql);
      db.pragma(`user_version = ${m.id}`);
      db.exec("COMMIT");
      current = m.id;
    } catch (err) {
      db.exec("ROLLBACK");
      throw err;
    }
  }
  if (current !== CURRENT_SCHEMA_VERSION) {
    throw new Error(
      `Migration state inconsistent: user_version=${current}, expected=${CURRENT_SCHEMA_VERSION}`,
    );
  }
}

/**
 * Run a function inside a transaction with preflight. Use for any write path
 * called from the daemon hot loop. Wraps better-sqlite3's transaction() so
 * disk-space errors fire BEFORE BEGIN, not mid-transaction.
 */
export function withWrite<T>(
  db: Database.Database,
  path: string,
  fn: (db: Database.Database) => T,
  minFreeBytes: number = DEFAULT_MIN_FREE_BYTES,
): T {
  preflightDiskSpace(path, minFreeBytes);
  const tx = db.transaction(fn);
  return tx(db);
}

export async function handleFollowUpDeliveryCommand(
  args: string[],
  deps: FollowUpDeliveryCommandDependencies,
): Promise<number> {
  const out =
      deps.stdout ??
      ((text) => {
        process.stdout.write(text);
      }),
    err =
      deps.stderr ??
      ((text) => {
        process.stderr.write(text);
      });
  if (!args.length || (args.length === 1 && ["help", "--help"].includes(args[0]!))) {
    out(FOLLOW_UP_DELIVERY_HELP);
    return 0;
  }
  const action = args[0];
  let params: unknown;
  try {
    if (
      args.length > 20 ||
      args.some((a) => Buffer.byteLength(a) > 4096) ||
      !["prepare", "list", "inspect", "authorize", "revoke", "dispatch", "reconcile"].includes(action!)
    )
      throw Error();
    const identity = action !== "prepare" && action !== "list",
      id = identity ? args[1] : undefined;
    const allowed = [
      "--repo",
      "--default-tenant",
      "--json",
      ...(action === "prepare"
        ? ["--input"]
        : action === "list"
          ? ["--limit", "--after"]
          : action === "inspect"
            ? []
            : action === "authorize"
              ? ["--revision", "--grant"]
              : action === "revoke"
                ? ["--revision", "--reason"]
                : ["--revision"]),
    ];
    const flags = new Map<string, string | true>();
    for (let i = identity ? 2 : 1; i < args.length; i++) {
      const key = args[i]!;
      if (!allowed.includes(key) || flags.has(key)) throw Error();
      if (key === "--default-tenant" || key === "--json") flags.set(key, true);
      else {
        const value = args[++i];
        if (!value || value.startsWith("--")) throw Error();
        flags.set(key, value);
      }
    }
    const scope = { repo_path: flags.get("--repo"), default_tenant: flags.get("--default-tenant") };
    const number = (key: string) => {
      const value = flags.get(key);
      if (typeof value !== "string" || !/^[1-9][0-9]*$/.test(value)) throw Error();
      return Number(value);
    };
    const file = async (key: string) => {
      const path = flags.get(key);
      if (typeof path !== "string") throw Error();
      return readInput(path);
    };
    if (action === "prepare") {
      const body = FollowUpDeliveryPrepareParams.omit({ repo_path: true, default_tenant: true }).parse(
        await file("--input"),
      );
      params = FollowUpDeliveryPrepareParams.parse({ ...body, ...scope });
    } else if (action === "list")
      params = FollowUpDeliveryListParams.parse({
        ...scope,
        ...(flags.has("--limit") ? { limit: number("--limit") } : {}),
        ...(flags.has("--after") ? { after_id: flags.get("--after") } : {}),
      });
    else if (action === "inspect") params = FollowUpDeliveryInspectParams.parse({ ...scope, attempt_id: id });
    else {
      const base = { ...scope, attempt_id: id, expected_revision: number("--revision") };
      params =
        action === "authorize"
          ? FollowUpDeliveryAuthorizeParams.parse({
              ...base,
              grant: SignedFollowUpDeliveryGrant.parse(await file("--grant")),
            })
          : action === "revoke"
            ? FollowUpDeliveryRevokeParams.parse({ ...base, reason: flags.get("--reason") })
            : FollowUpDeliveryActionParams.parse(base);
    }
  } catch {
    err("invalid_delivery_request\n");
    return 1;
  }
  let raw: unknown;
  try {
    raw = await deps.rpc(`follow-up-delivery.${action}`, params);
  } catch (error) {
    const code = error instanceof Error ? error.message : "";
    err(
      `${(FOLLOW_UP_DELIVERY_ERRORS as readonly string[]).includes(code) ? code : "delivery_request_failed"}\n`,
    );
    return 1;
  }
  try {
    if (Buffer.byteLength(serializeFollowUpDelivery(raw)) + 1 > 65536) throw Error();

      action === "list" ? FollowUpDeliveryListResult.parse(raw) : FollowUpDeliveryResult.parse(raw);
    const text = `${serializeFollowUpDelivery(result)}\n`;
    if (Buffer.byteLength(text) > 65536) throw Error();

    return action === "dispatch" || action === "reconcile"
      ? "attempt" in result &&
        result.attempt.state === "delivered" &&
        result.attempt.qualification === "production_qualified"
        ? 0
        : 1
      : 0;
  } catch {
    err("invalid_delivery_response\n");
    return 1;
  }
}
async function readInput(path: string): Promise<unknown> {
  const file = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
  try {
    const stat = await file.stat();
    if (!stat.isFile() || stat.size > 65536) throw Error();
    const bytes = Buffer.alloc(65537);
    let length = 0;
    while (length < bytes.length) {
      const read = await file.read(bytes, length, bytes.length - length, null);
      if (!read.bytesRead) break;
      length += read.bytesRead;
    }
    if (length > 65536) throw Error();
    return JSON.parse(new TextDecoder("utf-8", { fatal: true }).decode(bytes.subarray(0, length)));
  } finally {
    await file.close();
  }
}

# Public follow-up delivery ledger (O2a)

This increment implements local preparation, signed authorization, single-attempt dispatch, revocation and explicit reconciliation. No production transport, external target registry or authorization issuer is configured by default. Real public delivery remains **qualification_pending**. The built-in code never generates a signing key or treats a caller's actor name, approval boolean, receipt JSON, or task success as permission to send.

## Review and authority

`follow-up-delivery prepare --repo /canonical/repo --default-tenant --input prepare.json` accepts a file containing `obligation_id`, `expected_obligation_revision`, `destination_ref`, `body`, and `idempotency_key`. The body is exact UTF-8 data, not instructions. The destination reference must resolve in a host-owned registry to a fixed GitHub account, repository, issue number and immutable target node ID and expected actor node ID. Arbitrary URL/recipient/provider fields in the request are rejected. Output contains the complete frozen envelope and digests for review.

Preparation is not authorization. An external trusted authority must review and sign `authorization_subject` plus a unique `grant_id`, `issued_at_ms`, `expires_at_ms`, and `max_calls: 1` using Ed25519 over `deliveryCanonical(body)`. Host trust registration restricts the public key's purpose, canonical repository, provider, destination references and maximum validity. No issuer or private-key handling is included in this increment. A signature made with a caller-generated unregistered key has no authority.

`follow-up-delivery authorize ID --repo /canonical/repo --default-tenant --revision N --grant grant.json` imports `{ body, key_id, signature }`. Grant fields bind the exact attempt, scope, provider/account, target digest, body digest and envelope digest. Dispatch rechecks signature, key registration, exact target, expiry and O1 obligation revision **inside the transaction that consumes authorization and records started**, immediately before the transport call. Host key removal or scoped revoke before this transaction prevents sending. Revocation after it cannot undo a call already in progress.

The CLI uses the existing daemon's authenticated local RPC. Root wiring must inject host registries; requests never select a transport or public key. Constructor `FollowUpDeliveryService(db, dbPath, options)` performs no I/O/migration. Methods are `follow-up-delivery.prepare|list|inspect|authorize|revoke|dispatch|reconcile`; all require an explicit canonical repository/default tenant. Writes use the existing DB lease. Same-user database/host-registry compromise is outside this boundary.

## Results and recovery

`dispatch ID ... --revision N` performs at most one `sendOnce` call for that attempt. Missing transport, test transports without the host's explicit test flag, and adapters declaring automatic SDK retries yield `qualification_pending` without consuming the grant or creating started. Transport qualification labels are host configuration, not self-proving attestations; deployment must establish qualification separately.

A durable started row is written before the call. Disconnect, timeout, invalid receipt, process death or lost response leave started/`delivery_unknown` with consumed authority. Neither duplicate dispatch nor a new idempotency key for the same unresolved obligation can resend. A timeout's late response never silently overwrites the returned outcome. Abort is best effort and is not proof that the provider stopped.

`reconcile ID ... --revision N` calls only the registered provider's receipt lookup. The CLI cannot submit delivered=true or receipt data. A valid receipt must match provider, account, immutable target, pinned author node ID, body digest and attempt correlation; it stores bounded provider message/request IDs and its digest. Lookup failure or absence is still unknown. A future real provider must verify author, target and content from its authoritative response; this generic service cannot independently authenticate arbitrary adapter claims.

`revoke ID ... --revision N --reason TEXT` makes an unsent prepared attempt definitely_not_sent. If sending already started, it remains unknown. `inspect ID` and `list [--limit 1..50] [--after ID]` read scoped state. List uses UUID keyset order with explicit continuation, not a snapshot; concurrent inserts before the marker require a new scan. History is append-only. O1 `fulfilled_local` and external sending remain independent: delivery never resolves the promise.

Output is JSON, bounded to 64 KiB with terminal controls escaped and exact body preserved on parse. Prepare reserves room for the largest valid receipt before any send. Files must be regular, bounded, valid UTF-8, and not symlinks. `dispatch`/`reconcile` exit 0 only with `delivered` and `production_qualified`; test/sandbox evidence and uncertain/pending results exit 1, with the actual evidence still printed.

## Qualification and rollback

Local tests exercise actual SQLite transactions and signatures, two connections, CLI-to-service invocation, a SIGKILL child immediately after committed consumption, response loss, timeouts and late completion, target/author/body mismatch, key removal, revocation, scope, stale revisions and bounded outputs. Fake send counts prove local invocation behavior only; no real message is sent and no real provider delivery guarantee is claimed.

O2b must separately supply a reviewed provider adapter, explicit human target/body approval issuer, pinned credentials/account, disabled SDK retries, real sandbox receipt lookups and controlled disconnect/crash qualification. Until then retain empty production registries. To roll back, disable the command/provider registration and retain migration 45's ledger. Do not drop unknown attempts, reset grants or down-migrate a serving DB to enable retries. Outstanding uncertainty requires reconciliation or operator investigation outside automatic sending.

Revocation records a logical nondecreasing ledger time when the host clock regresses; it remains available to remove authority even then. Grant validation and dispatch still use strict wall-clock checks. Once consumed, grant status stays consumed permanently; a later revocation is a separate immutable transition event, not a rewrite of consumption. If revocation or reconciliation races an in-flight send, the stale sender does not overwrite the newer uncertainty state. An explicit authoritative lookup can subsequently prove delivery.

A completed delivery does not authorize another message. A later update for the same obligation needs a new prepared envelope, a new attempt ID and a newly reviewed signature. Unresolved attempts block this path. An expired or removed-key grant on a prepared attempt is safely unsent but continues to reserve its obligation; inspect and explicitly revoke it before preparing a replacement. There is no automatic expiry cleanup or resend.

## Daemon shutdown contract

The host must stop accepting new calls, track every `service.handle(...)` promise and await their settlement before closing the database. Transport and lookup waits have a logical maximum of 30,000 ms each (default 5,000 ms); the final ledger transaction must also finish before the handle settles. Do not close the DB merely because a shorter shutdown deadline elapsed. Uncooperative asynchronous transports can remain alive after abort, but their late fulfillment/rejection has no database continuation: the already-settled race discards it. A blocking synchronous adapter or synchronous DB lock cannot be preempted by a JavaScript timer; host adapters must avoid blocking the event loop. No background timer initiates another dispatch or reconciliation.

## Durability and clock boundary

O2 writes temporarily require `journal_mode=WAL` and verified `synchronous=FULL`; on macOS they additionally enable and verify `fullfsync`. The synchronous transaction commits before the provider call, then restores the connection's prior settings without an intervening await. Other OSStack DB behavior is unchanged. This follows SQLite's documented [WAL durability settings](https://www.sqlite.org/pragma.html#pragma_synchronous) and [macOS fullfsync flag](https://www.sqlite.org/pragma.html#pragma_fullfsync). SIGKILL tests prove process-crash recovery, not physical power-cut or storage-device qualification; the filesystem/VFS/device must honor synchronization, which remains part of deployment qualification.

Migration 45 also retains the maximum valid clock observed by O2 writes. A rejected expiry check persists its observation before returning, so a later clock rollback cannot revive that known-expired grant, even after service restart. Admission rejects a clock below this durable record. Post-send evidence and revocation use logical nondecreasing ledger timestamps, so a regressed clock does not prevent recording an irreversible result or removing authority. This is not an independent trusted time service: time never observed by the ledger cannot be reconstructed after downtime, and host-clock integrity remains a deployment requirement.

Read/list/revoke/reconcile match the original stored canonical path string without requiring its directory to remain present; they cannot rebind the record to a moved directory or follow a new symlink. Prepare/authorize/dispatch still require the live canonical directory. Thus a moved/deleted repository can be inspected and its old authority removed using its exact historical scope. Body controls are permitted as exact data but escaped in review JSON; a future provider may reject them and must not silently normalize approved text. Pagination UUID case is normalized before comparison.

## Remaining availability limits before O2b qualification

A trusted-clock excursion far into the future can conservatively quarantine future admissions until that observed time is reached. Inspection/revocation still work. There is no automatic high-water reset: silently lowering it could revive already expired grants. Reviewed clock-recovery tooling is follow-on work.

Reconciliation currently requires the original destination registration and the same qualified transport to remain registered. Retain that historical registration until uncertain attempts are resolved. A transport without receipt lookup can currently send, but cannot automatically recover uncertainty; it must not be production-qualified under this contract. Inspect/list may combine adjacent committed attempt/grant observations under concurrent writes; mutation CAS remains authoritative. These are recorded reliability limitations, not real-provider qualification. Daemon/CLI integration qualification and its final audit are still required.
