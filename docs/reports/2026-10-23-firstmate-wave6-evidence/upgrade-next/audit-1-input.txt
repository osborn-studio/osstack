    phase = 'installing'; await append('started', null); quiescent(); installStarted = true;
    const installed = InstallSchema.parse(await call(signal => port.install(plan, group, signal))); quiescent();
    if (installed.plan_digest !== plan.digest || installed.installation_id !== group.installation_id || installed.version !== plan.plan.target.version || installed.build_digest !== plan.plan.target.build_digest) throw Error('upgrade_install_mismatch');
    phase = 'installed'; await append('confirmed', null, installed, installed.receipt_ref); clockCheck(); phase = 'restarting';
    for (const home of group.homes) {
      await append('started', home); clockCheck(); const receipt = ReceiptSchema.parse(await call(signal => port.restart(home, signal)));
      if (receipt.home_id !== home.home_id || receipt.source_generation !== home.identity?.generation) throw Error('upgrade_restart_binding');
      await append('confirmed', home, receipt, receipt.receipt_ref); clockCheck();
    }
    phase = 'verifying';
    for (const home of group.homes) {
      await append('started', home); clockCheck(); const observed = await call(signal => port.observe(home, signal)); const identity = DeploymentIdentitySchema.parse(observed.identity);
      if (observed.healthy !== true || identity.home_id !== home.home_id || identity.generation === home.identity?.generation || identity.deployment_kind !== 'npm'
        || identity.version !== plan.plan.target.version || identity.build_digest !== plan.plan.target.build_digest || identity.build_evidence !== 'host_manifest'
        || identity.protocol_version !== plan.plan.target.protocol_version || identity.database_schema !== plan.plan.target.database_schema) throw Error('upgrade_version_mismatch');
      ref.parse(observed.receipt_ref); await append('confirmed', home, identity, observed.receipt_ref); clockCheck();
    }
    phase = 'verified'; clockCheck(); await append('confirmed', null); clockCheck(); await writer.releaseVerified(); closed = true; return result('verified');
  } catch (error) {
    const message = error instanceof Error && /^upgrade_[a-z_]+$/.test(error.message) ? error.message : 'upgrade_port_failure';
    if (!writer && message !== 'upgrade_recovery_required') return result('qualification_pending', message);
    const state: UpgradePhase = message === 'upgrade_recovery_required' || sideEffectStarted && !installStarted ? 'recovery_required'
      : !installStarted ? 'aborted_before_install' : message === 'upgrade_version_mismatch' ? 'version_mismatch' : phase === 'restarting' && message !== 'upgrade_operation_unknown' ? 'restart_failed' : 'recovery_required';
    phase = state; closed = true;
    // A failed journal write leaves its previous started record. Never undo an external fact or steal the claim.
    try { await append('failed', null, undefined, undefined, message); } catch { /* Caller must inspect the retained claim and previous record. */ }
    return result(state, message);
  }
}

=== docs/guides/multi-home-upgrade-plan.md ===
# Multi-home upgrade planning and journal

U2a provides plans and a private phase journal. Production CLI integration exposes only plan/status. It does not register an apply operation, install packages, open home databases, start daemons or restart actual homes. Every ordinary plan remains `qualification_pending`; a home inventory is a snapshot, never proof that all consumers of a shared binary are known.

## Plan contract

`createUpgradePlan(request, inventory.homes, nowMs)` reuses `DeploymentIdentitySchema` from the real home inventory. The request contains:

```json
{
  "schema_version": 1,
  "expires_at_ms": 1792800000000,
  "target": {
    "version": "0.22.0",
    "build_digest": "aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa",
    "protocol_version": 2,
    "database_schema": 45,
    "readable_schema_min": 44,
    "readable_schema_max": 45
  },
  "budget": { "max_total_ms": 300000, "max_step_ms": 60000, "max_steps": 32 },
  "groups": [{
    "group_id": "npm-primary",
    "installation_id": "registered-shared-installation",
    "deployment_kind": "npm",
    "home_ids": ["home_00000000000000000000000000000001", "home_00000000000000000000000000000002"]
  }]
}
```

Replace the example expiry, target evidence and home IDs with actual registered values. Tags and version ranges are rejected. A plan binds the exact target, inventory identity/generation/build/schema, sockets, group membership, group order, expiration and budgets into its digest. The target's schema readability range is a requested constraint; only a future qualified updater can attest compatibility. A caller-supplied range cannot grant installation permission.

Limits: 32 total homes; 32 groups; expiry within 24 hours; per-step limit at most 5 minutes; total limit at most 1 hour; at most 512 port calls. `budget_scope: "per_group"` is explicit: the coordinator handles one group per attempt, not a cross-group global maintenance window. Deadlines bound logical waiting and request cancellation; they do not prove physical stop. Stale, missing or incompatible observations remain recorded as unqualified reasons. Checkout deployments cannot use the npm path.

## Persistence and status

The caller creates an explicitly selected, existing private directory (0700, same UID, canonical path). `new UpgradeJournal(root).create(frozenPlan)` saves an immutable digest-checked plan. `status(planDigest, groupId)` only reads the plan and the bounded hash-linked phase records. It returns `phase`, `completed`, and `resume_allowed: false`; reading does not retry, steal a lock or invoke an updater.

Records are published create-only after file sync, followed by directory sync. File ownership, permissions, type, hard links, byte bounds and paths are checked. Claims are keyed by installation ID across plan digests and acquired create-only. No PID heuristic, elapsed timeout or maintenance window can steal a claim. A completed verified run releases its claim; an interrupted or failed run retains it for reconciliation. Even a dead process is not proof that an installation had no effect. Claim cleanup/reconciliation is deliberately not automated in this slice.

The journal hash chain detects corruption and binds local records. It is neither a signature nor cross-host authentication, and it does not protect against a malicious process with the same host account. Preserve the directory across disabling or upgrading this feature.

## Qualified host boundary and future native adapter

`coordinateUpgrade({journal, plan, groupId, port?})` is available for injected host adapters and fake fault tests. With no port, it only persists the plan and returns `qualification_pending`. No production implementation of `NativeUpgradePort` is supplied.

A future host `resolveAuthority` must independently prove every consumer of the canonical installation, return the exact plan/group/installation/consumer binding, a current explicit authorization and qualification reference, expiration, and synchronous `assertCurrent`. Installation IDs must be canonical host registration identities: letting callers assign aliases for one installation would bypass shared-installation serialization. The registry manifest and its claimed completeness cannot satisfy this proof. These values come from the trusted host callback, never user JSON or model text.

The same port supplies `observe`, `drain`, `snapshot`, `install` and `restart`. Each drain must return exact home/generation evidence, confirmed stop, zero durable Graph holds, zero unknown actions and a synchronous live maintenance-fence check. Every shared consumer must stop before any snapshot or installation. Fences are rechecked before and after snapshots and installation. The future adapter must enforce these requirements at the actual native effect boundary; receipt shapes alone cannot establish them.

The existing `upgrade.ts` provides deployed-home resolution and single-deployment primitives, but its current command has a fixed launchd label, process-global environment and failure paths that restore old database files. Do not call that entire command as the new install port. U2b must separate and qualify suitable native primitives first. No snapshot in U2a authorizes restoring a database containing newer external facts: the plan fixes `rollback: "no_database_restore"`, and the port exposes no database rollback operation.

## Failure behavior

The sequence is preflight identity checks, all-consumer drain, quiescent, consistent snapshots, one installation, per-home restart, then exact healthy target identity verification. Each effect records `started` before calling the port and records evidence after confirmation. The verified daemon must have the pinned home ID, a new generation, target version/build/protocol/schema, and healthy status.

An install timeout, missing receipt, revoked fence, wrong identity or restart failure never triggers another install or automatic database rollback. Late receipts cannot advance the journal. Reopening a partial journal returns `recovery_required` without invoking any updater. Partial drains also require reconciliation because some homes may already be stopped. Ordinary preflight failures before side effects are marked `aborted_before_install`; their claims are retained conservatively. All fake evidence references in tests are explicitly fake.

Qualification status: plan/journal implementation and injected-port local fault tests are implemented; native installation, real shared-consumer completeness, launchd multi-instance management, cross-host authentication and real dual-home upgrades remain unqualified and disabled.

import { randomBytes } from "node:crypto";
import { constants } from "node:fs";
import { link, open, realpath, stat, unlink } from "node:fs/promises";
import { connect } from "node:net";
import { isAbsolute, join } from "node:path";
import { z } from "zod";

const HomeId = z.string().regex(/^home_[a-f0-9]{32}$/);
const Digest = z.string().regex(/^[a-f0-9]{64}$/);
const boundedText = z
  .string()
  .min(1)
  .max(256)
  .regex(/^[A-Za-z0-9.+_-]+$/);
export const DeploymentIdentitySchema = z
  .object({
    schema_version: z.literal(1),
    home_id: HomeId,
    generation: z.string().regex(/^instance_[a-f0-9]{32}$/),
    version: boundedText,
    protocol_version: z.number().int().positive(),
    database_schema: z.number().int().nonnegative(),
    started_at_ms: z.number().int().nonnegative(),
    deployment_kind: z.enum(["npm", "checkout", "unknown"]),
    build_digest: Digest.nullable(),
    build_evidence: z.enum(["host_manifest", "unavailable"]),
  })
  .strict()
  .refine(
    (value) =>
      (value.build_evidence === "unavailable") ===
      (value.build_digest === null),
  );
export type DeploymentIdentity = z.infer<typeof DeploymentIdentitySchema>;
const Manifest = z
  .object({
    schema_version: z.literal(1),
    homes: z
      .array(
        z
          .object({
            home_id: HomeId,
            socket: z.string().min(1).max(4096).refine(isAbsolute),
            enabled: z.boolean(),
            desired: z
              .object({
                version: boundedText,
                protocol_version: z.number().int().positive().optional(),
                database_schema: z.number().int().nonnegative().optional(),
                deployment_kind: z.enum(["npm", "checkout"]).optional(),
                build_digest: Digest.optional(),
              })
              .strict(),
          })
          .strict(),
      )
      .min(1)
      .max(32),
  })
  .strict()
  .refine(
    (value) =>
      new Set(value.homes.map((home) => home.home_id)).size ===
        value.homes.length &&
      new Set(value.homes.map((home) => home.socket)).size ===
        value.homes.length,
  );
type HomeEntry = z.infer<typeof Manifest>["homes"][number];
type Status =
  | "matched_requested"
  | "drift"
  | "unknown"
  | "unreachable"
  | "identity_mismatch"
  | "unsafe_endpoint"
  | "disabled";
export interface HomeObservation {
  desired: HomeEntry;
  observed: DeploymentIdentity | null;
  observed_at_ms: number;
  status: Status;
  reason: string | null;
}

/** Only the authorized daemon startup calls this initializer. Inventory never does. */
export async function ensureDeploymentHomeId(home: string): Promise<string> {
  try {
    const canonicalHome = await realpath(home);
    const path = join(canonicalHome, "deployment-identity.json");
    const parent = await stat(canonicalHome);
    if (
      !parent.isDirectory() ||
      (process.getuid && parent.uid !== process.getuid())
    )
      throw new Error("identity_unavailable");
    const temporary = join(
      canonicalHome,
      `.deployment-identity-${randomBytes(16).toString("hex")}.tmp`,
    );
    // Publish only a complete, file-synced record. link is create-only, so
    // concurrent startup converges on one identity and never replaces a pin.
    const created = await open(
      temporary,
      constants.O_CREAT |
        constants.O_EXCL |
        constants.O_WRONLY |
        constants.O_NOFOLLOW,
      0o600,
    );
    try {
      try {
        await created.writeFile(
          JSON.stringify({
            schema_version: 1,
            home_id: `home_${randomBytes(16).toString("hex")}`,
          }),
        );
        await created.sync();
      } finally {
        await created.close();
      }
      try {
        await link(temporary, path);
      } catch (error) {
        if ((error as NodeJS.ErrnoException).code !== "EEXIST") throw error;
      }
      const directory = await open(canonicalHome, constants.O_RDONLY);
      try {
        await directory.sync();
      } finally {
        await directory.close();
      }
    } finally {
      await unlink(temporary).catch(() => undefined);
    }
    const file = await open(
      path,
      constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK,
    );
    try {
      const info = await file.stat();
      if (
        !info.isFile() ||
        info.size > 256 ||
        (process.getuid && info.uid !== process.getuid())
      )
        throw new Error("identity_unavailable");
      const buffer = Buffer.alloc(257);
      let bytesRead = 0;
      while (bytesRead < buffer.length) {
        const chunk = await file.read(
          buffer,
          bytesRead,
          buffer.length - bytesRead,
          bytesRead,
        );
        if (!chunk.bytesRead) break;
        bytesRead += chunk.bytesRead;
      }
      if (bytesRead > 256) throw new Error("identity_unavailable");
      return z
        .object({ schema_version: z.literal(1), home_id: HomeId })
        .strict()
        .parse(JSON.parse(buffer.subarray(0, bytesRead).toString())).home_id;
    } finally {
      await file.close();
    }
  } catch (error) {
    if (
      error instanceof SyntaxError ||
      error instanceof z.ZodError ||
      (error instanceof Error &&
        (error.message === "identity_unavailable" ||
          /^E[A-Z0-9]+$/.test((error as NodeJS.ErrnoException).code ?? "")))
    )
      throw new Error("identity_unavailable");
    throw error;
  }
}

/** Fixed read-only RPC, 64KiB total response bound before buffering. No DB or generic method port. */
export function readHomeIdentity(
  socketPath: string,
  timeoutMs = 3000,
): Promise<unknown> {
  return new Promise((resolve, reject) => {
    const socket = connect(socketPath);
    let bytes = Buffer.alloc(0);
    let settled = false;
    const finish = (error?: string, value?: unknown) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      socket.destroy();
      error ? reject(new Error(error)) : resolve(value);
    };
    const timer = setTimeout(
      () => finish("unreachable"),
      Math.min(3000, Math.max(1, timeoutMs)),
    );
    socket.on("error", () => finish("unreachable"));
    socket.on("close", () =>
      finish(bytes.length === 0 ? "unreachable" : "invalid_response"),
    );
    socket.on("connect", () => {
      const body = Buffer.from(
        JSON.stringify({ id: 1, method: "deployment.identity" }),
      );
      const header = Buffer.alloc(4);
      header.writeUInt32BE(body.length);
      socket.write(Buffer.concat([header, body]));
    });
    socket.on("data", (chunk) => {
      if (bytes.length + chunk.length > 65536) {
        finish("invalid_response");
        return;
      }
      bytes = Buffer.concat([bytes, chunk]);
      if (bytes.length < 4) return;
      const size = bytes.readUInt32BE();
      if (size > 65532) {
        finish("invalid_response");
        return;
      }
      if (bytes.length < 4 + size) return;
      if (bytes.length !== 4 + size) {
        finish("invalid_response");
        return;
      }
      try {
        const response = z
          .object({
            id: z.literal(1),
            ok: z.literal(true),
            result: DeploymentIdentitySchema,
          })
          .strict()
          .parse(JSON.parse(bytes.subarray(4).toString()));
        finish(undefined, response.result);
      } catch {
        finish("invalid_response");
      }
    });
  });
}

/** Local same-UID socket/file ownership is a trust boundary, not network authentication. */
export async function inspectHomes(raw: unknown): Promise<{
  schema_version: 1;
  mode: "readonly_inventory";
  complete: boolean;
  homes: HomeObservation[];
  limitations: string[];
}> {
  const parsed = Manifest.safeParse(raw);
  if (!parsed.success) throw new Error("invalid_manifest");
  const rows: HomeObservation[] = parsed.data.homes.map((desired) => ({
    desired,
    observed: null,
    observed_at_ms: Date.now(),
    status: "disabled",
    reason: null,
  }));
  const canonicalSockets = new Set<string>();
  for (const row of rows) {
    if (!row.desired.enabled) continue;
    const canonical = await realpath(row.desired.socket).catch(() => null);
    if (canonical !== null) {
      if (canonicalSockets.has(canonical)) throw new Error("duplicate_socket");
      canonicalSockets.add(canonical);
    }
  }
  const activeCanonicalSockets = new Set<string>();
  let next = 0;
  const worker = async () => {
    while (next < rows.length) {
      const row = rows[next++]!;
      if (!row.desired.enabled) continue;
      try {
        const canonical = await realpath(row.desired.socket);
        if (activeCanonicalSockets.has(canonical)) {
          row.status = "unsafe_endpoint";
          row.reason = "duplicate_socket";
          continue;
        }
        activeCanonicalSockets.add(canonical);
        const info = await stat(row.desired.socket);
        if (
          !info.isSocket() ||
          (process.getuid && info.uid !== process.getuid())
        ) {
          row.status = "unsafe_endpoint";
          row.reason = "canonical_same_uid_socket_required";
          continue;
        }
        let observed: DeploymentIdentity;
        try {
          observed = DeploymentIdentitySchema.parse(
            await readHomeIdentity(row.desired.socket),
          );
        } catch (error) {
          row.status =
            error instanceof Error && error.message === "unreachable"
              ? "unreachable"
              : "unknown";
          row.reason = "identity_unavailable";
          continue;
        }
        if ((await realpath(row.desired.socket)) !== canonical) {
          row.status = "unsafe_endpoint";
          row.reason = "endpoint_changed";
          continue;
        }
        row.observed_at_ms = Date.now();
        if (observed.home_id !== row.desired.home_id) {
          row.status = "identity_mismatch";
          row.reason = "pinned_home_mismatch";
          continue;
        }
        row.observed = observed;
        const desired = row.desired.desired;
        const unknown =
          (desired.build_digest !== undefined &&
            observed.build_digest === null) ||
          (desired.deployment_kind !== undefined &&
            observed.deployment_kind === "unknown");
        const drift =
          desired.version !== observed.version ||
          (desired.protocol_version !== undefined &&
            desired.protocol_version !== observed.protocol_version) ||
          (desired.database_schema !== undefined &&
            desired.database_schema !== observed.database_schema) ||
          (desired.build_digest !== undefined &&
            observed.build_digest !== null &&
            desired.build_digest !== observed.build_digest) ||
          (desired.deployment_kind !== undefined &&
            observed.deployment_kind !== "unknown" &&
            desired.deployment_kind !== observed.deployment_kind);
        row.status = drift
          ? "drift"
          : unknown
            ? "unknown"
            : "matched_requested";
        row.reason = unknown ? "deployment_evidence_unavailable" : null;
      } catch {
        row.status = "unreachable";
        row.reason = "endpoint_unavailable";
      }
    }
  };
  await Promise.all(Array.from({ length: Math.min(4, rows.length) }, worker));
  return {
    schema_version: 1,
    mode: "readonly_inventory",
    complete:
      rows.some((row) => row.desired.enabled) &&
      rows.every(
        (row) =>
          row.status === "matched_requested" || row.status === "disabled",
      ),
    homes: rows,
    limitations: [
      "Local same-UID sockets with canonical alias deduplication only; no network or peer-credential attestation.",
      "No database opened, daemon started, software installed or deployment modified.",
      "Matched_requested applies only to declared desired fields; missing build/deployment evidence is never inferred.",
      "Snapshot only; generation is reported for comparison, not an upgrade authorization.",
    ],
  };
}

export async function handleHomesCommand(
  args: string[],
  out: (value: string) => void = console.log,
): Promise<number> {
  if (args.length === 1 && args[0] === "--help") {
    out(
      "homes inspect --manifest FILE [--json]\nRead only explicitly pinned local home identities; never upgrades or opens databases.",
    );
    return 0;
  }
  if (
    args[0] !== "inspect" ||
    args[1] !== "--manifest" ||
    !args[2] ||
    (args.length !== 3 && !(args.length === 4 && args[3] === "--json"))
  )
    throw new Error("invalid_homes_arguments");
  let raw: unknown;
  try {
    const file = await open(args[2], constants.O_RDONLY | constants.O_NONBLOCK);
    try {
      const info = await file.stat();
      if (!info.isFile() || info.size > 65536)
        throw new Error("invalid_manifest");
      const buffer = Buffer.alloc(65537);
      let bytesRead = 0;
      while (bytesRead < buffer.length) {
        const chunk = await file.read(
          buffer,
          bytesRead,
          buffer.length - bytesRead,
          bytesRead,
        );
        if (!chunk.bytesRead) break;
        bytesRead += chunk.bytesRead;
      }
      if (bytesRead > 65536) throw new Error("invalid_manifest");
      raw = JSON.parse(buffer.subarray(0, bytesRead).toString());
    } finally {
      await file.close();
    }
  } catch {
    throw new Error("invalid_manifest");
  }
  const result = await inspectHomes(raw);
  out(JSON.stringify(result, null, 2));
  return result.complete ? 0 : 1;
}
