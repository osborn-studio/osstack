import { attentionDigest, type AttentionOwner, type AttentionScope, type AttentionStore } from "./attention-store.js";
import { readRunEventPage, type BoundedRunEvent } from "./run-store.js";

interface Cursor {
  binding_hash: string;
  last_seq: number;
  through_seq: number;
  last_event_hash: string | null;
  gap: string | null;
  facts_json: string;
}
interface Run {
  run_id: string;
  task_id: string;
  session_id: string;
  agent_id: string;
  capability_id: string;
  started_at: number;
  phase: string;
  outcome: string | null;
  task_created_at: number;
}
interface Observed {
  observed_status: string | null;
  unsupported_events: number;
}
const statuses = new Set(["queued", "running", "awaiting_approval", "awaiting_clarify", "paused",
  "cancelling", "cancelled", "failed", "completed", "expired"]);
const informational = new Set(["token.delta", "reasoning.delta", "reasoning.done", "progress", "tool.started",
  "tool.updated", "tool.done", "title.updated", "usage.updated", "usage.final"]);
function hash(event: BoundedRunEvent): string {
  return attentionDigest([event.event_id, event.run_id, event.session_id, event.seq, event.type,
    event.created_at, event.terminal, event.payload_bytes, event.payload_json]);
}
function object(json: string): Record<string, unknown> | null {
  try {
    const value: unknown = JSON.parse(json);
    return value !== null && typeof value === "object" && !Array.isArray(value) ? value as Record<string, unknown> : null;
  } catch { return null; }
}
function reduce(event: BoundedRunEvent, state: Observed, store: AttentionStore): string | null {
  if (event.payload_json === null) return "oversized_event_payload";
  if (!event.event_id || event.event_id.length > 4096 || !event.session_id || event.session_id.length > 4096
    || !event.type || event.type.length > 256 || !Number.isFinite(event.created_at)
    || ![0, 1].includes(event.terminal)) return "invalid_event_envelope";
  const payload = object(event.payload_json);
  if (!payload) return "invalid_event_payload";
  if (["status", "run.started", "done"].includes(event.type)) {
    if (typeof payload.status !== "string" || !statuses.has(payload.status)
      || (event.type === "run.started" && payload.status !== "running")
      || (event.type === "done" && (!event.terminal || !["completed", "failed", "cancelled", "expired"].includes(payload.status)
        || !Number.isSafeInteger(payload.last_seq) || (payload.last_seq as number) < 0 || (payload.last_seq as number) > event.seq)))
      return "invalid_event_payload";
    state.observed_status = payload.status;
  } else if (event.type.startsWith("approval.") || event.type.startsWith("clarify.")) {
    const approval = event.type.startsWith("approval.");
    const family = approval ? "approval" : "clarify";
    if (![`${family}.requested`, `${family}.resolved`].includes(event.type)) {
      state.unsupported_events++;
      return null;
    }
    const id = payload[`${family}_id`];
    if (typeof id !== "string" || !id || id.length > 4096) return "invalid_event_payload";
    // Authority stays with the original approval/clarify tables. Events never
    // grant permission or synthesize a Session answer/action.
    const table = approval ? "approvals" : "clarifies";
    const row = store.db.prepare(`SELECT status FROM ${table} WHERE ${family}_id=? AND run_id=?`)
      .get(id, event.run_id) as { status: string } | undefined;
    if (!row) return "authority_reference_missing";
    if (event.type.endsWith(".resolved") && (typeof payload.status !== "string"
      || !(approval ? ["accepted", "rejected", "timed_out"] : ["answered", "timed_out", "abandoned"]).includes(payload.status)))
      return "invalid_event_payload";
    if (event.type.endsWith(".resolved")) {
      const field = approval ? "choice" : "answer";
      if (typeof payload[field] !== "string") return "invalid_event_payload";
      const matches = store.db.prepare(`SELECT 1 FROM ${table} WHERE ${family}_id=? AND run_id=? AND status=? AND ${field} IS ?`)
        .get(id, event.run_id, payload.status, payload[field]);
      if (!matches) return "authority_resolution_mismatch";
    }
  } else if (event.type === "error") {
    if (typeof payload.code !== "string" || typeof payload.terminal !== "boolean") return "invalid_event_payload";
    if (payload.terminal) state.observed_status = "unknown";
  } else if (!informational.has(event.type)) state.unsupported_events++;
  return null;
}

/** One bounded retained-log page. Projection and cursor share the owner-fenced
 * transaction; it never invokes an adapter, resolves an approval, or pins history. */
export function projectAttentionRun(store: AttentionStore, owner: AttentionOwner,
  scope: AttentionScope, runId: string, now: number): void {
  if (!Number.isFinite(now)) throw new Error("invalid_clock");
  store.write(owner, () => {
    const persistedScope = store.getScope(scope.scope_id);
    if (!persistedScope || persistedScope.task_id !== scope.task_id || persistedScope.repo_path !== scope.repo_path)
      throw new Error("attention_scope_conflict");
    const subject = store.db.prepare("SELECT admitted,binding_hash FROM attention_subjects WHERE scope_id=? AND kind='run' AND subject_id=?")
      .get(scope.scope_id, runId) as { admitted: number; binding_hash: string } | undefined;
    if (!subject?.admitted) throw new Error("run_not_admitted");
    if (subject.binding_hash !== attentionDigest([scope.scope_id, "run", runId])) throw new Error("run_subject_binding_conflict");
    const task = store.db.prepare("SELECT created_at FROM tasks WHERE task_id=? AND tenant_id IS NULL")
      .get(scope.task_id) as { created_at: number } | undefined;
    if (!task || scope.scope_id !== attentionDigest([owner.home_id, scope.task_id, scope.repo_path, task.created_at]))
      throw new Error("attention_scope_conflict");
    const run = store.db.prepare(`SELECT r.*,t.created_at task_created_at FROM runs r JOIN tasks t ON t.task_id=r.task_id
      WHERE r.run_id=? AND r.task_id=? AND r.tenant_id IS NULL AND t.tenant_id IS NULL`).get(runId, scope.task_id) as Run | undefined;
    const saved = store.db.prepare("SELECT * FROM attention_run_cursors WHERE scope_id=? AND run_id=?")
      .get(scope.scope_id, runId) as Cursor | undefined;
    const writeSubject = (facts: Record<string, unknown>) => store.db.prepare(
      "UPDATE attention_subjects SET sampled_at=?,generation=?,payload_json=? WHERE scope_id=? AND kind='run' AND subject_id=?"
    ).run(now, owner.generation, JSON.stringify({ event_replay: { complete: false, caught_up: false,
      gap: facts.reason ?? "source_unavailable", completeness_scope: "unverified_retention" }, ...facts }), scope.scope_id, runId);
    const quarantine = (reason: string) => store.db.prepare(
      "UPDATE attention_run_cursors SET gap=COALESCE(gap,?),generation=?,updated_at=? WHERE scope_id=? AND run_id=?"
    ).run(reason, owner.generation, now, scope.scope_id, runId);
    if (!run) {
      quarantine("source_missing");
      writeSubject({ state: "source_missing", reason: "source_missing", process_liveness: "unknown", gap: "history_gap" });
      return;
    }
    const binding = attentionDigest([owner.home_id, scope.scope_id, run.run_id, run.task_id,
      run.session_id, run.agent_id, run.capability_id, run.started_at]);
    if (saved && saved.binding_hash !== binding) {
      quarantine("source_binding_conflict");
      writeSubject({ state: "unknown", reason: "source_binding_conflict", process_liveness: "unknown", gap: "history_gap" });
      return;
    }
    if (saved && (!Number.isSafeInteger(saved.last_seq) || saved.last_seq < -1
      || !Number.isSafeInteger(saved.through_seq) || saved.through_seq < -1)) throw new Error("invalid_event_cursor");
    let cursor = saved?.last_seq ?? -1;
    let anchor = saved?.last_event_hash ?? null;
    let gap = saved?.gap ?? null;
    const state: Observed = { observed_status: null, unsupported_events: 0 };
    const prior = saved ? object(saved.facts_json) : null;
    let priorValid = true;
    if (saved) {
      if (!prior || !(prior.observed_status === null || typeof prior.observed_status === "string")
        || !Number.isSafeInteger(prior.unsupported_events) || (prior.unsupported_events as number) < 0)
        { gap ??= "invalid_cursor_facts"; priorValid = false; }
      else { state.observed_status = prior.observed_status; state.unsupported_events = prior.unsupported_events as number; }
    }
    let anchorValid = true;
    if (cursor >= 0) {
      const previous = readRunEventPage(store.db, { taskId: scope.task_id, runId, afterSeq: cursor - 1, throughSeq: cursor, limit: 1 }).events[0];
      anchorValid = previous !== undefined && hash(previous) === anchor;
      if (!anchorValid) { gap ??= "event_anchor_changed"; state.observed_status = null; }
    }
    const page = readRunEventPage(store.db, { taskId: scope.task_id, runId, afterSeq: cursor, limit: 200,
      ...(saved && saved.through_seq > cursor ? { throughSeq: saved.through_seq } : {}) });
    if (page.sequenceGap) gap ??= "event_sequence_gap";
    if (anchorValid && priorValid) {
      for (const event of page.events) {
        if (event.seq !== cursor + 1) { gap ??= "event_sequence_gap"; break; }
        if (event.session_id !== run.session_id) { gap ??= "event_source_binding_conflict"; break; }
        const error = reduce(event, state, store);
        if (error) { gap ??= error; break; }
        cursor = event.seq;
        anchor = hash(event);
      }
    }
    const backlog = cursor < page.highWatermark || cursor < page.availableHighWatermark;
    const caughtUp = cursor >= 0 && !gap && !backlog && state.unsupported_events === 0;
    // A caught-up cursor verifies newly read rows and the last anchor only.
    // Mutable earlier rows are not continuously re-hashed; never certify history.
    const pending = (table: "approvals" | "clarifies", column: "approval_id" | "clarify_id") => {
      const rows = store.db.prepare(`SELECT ${column} id FROM ${table} WHERE run_id=? AND status='pending' ORDER BY ${column} LIMIT 21`)
        .all(runId) as { id: string }[];
      return { ids: rows.slice(0, 20).map((row) => row.id), truncated: rows.length > 20,
        count: (store.db.prepare(`SELECT COUNT(*) n FROM ${table} WHERE run_id=? AND status='pending'`).get(runId) as { n: number }).n };
    };
    store.db.prepare(`INSERT INTO attention_run_cursors(scope_id,run_id,binding_hash,last_seq,through_seq,last_event_hash,gap,facts_json,generation,updated_at)
      VALUES(?,?,?,?,?,?,?,?,?,?) ON CONFLICT(scope_id,run_id) DO UPDATE SET last_seq=excluded.last_seq,through_seq=excluded.through_seq,
      last_event_hash=excluded.last_event_hash,gap=excluded.gap,facts_json=excluded.facts_json,generation=excluded.generation,updated_at=excluded.updated_at`)
      .run(scope.scope_id, runId, binding, cursor, page.highWatermark, anchor, gap, priorValid ? JSON.stringify(state) : saved!.facts_json, owner.generation, now);
    writeSubject({ state: { phase: run.phase, outcome: run.outcome }, ...state,
      reason: gap ? "history_gap" : backlog ? "event_backlog" : state.unsupported_events > 0 ? "unsupported_event_type" : "snapshot_only",
      repo_binding: "unverified", process_liveness: "unknown", gap: gap ? "history_gap" : "retention_unverified",
      event_replay: { cursor, through_seq: page.highWatermark, complete: false, caught_up: caughtUp, gap, completeness_scope: "unverified_retention" },
      pending_approvals: pending("approvals", "approval_id"), pending_clarifies: pending("clarifies", "clarify_id"),
      limitations: ["No historical retention or processed-prefix integrity guarantee", "Event observations do not authorize execution"] });
  });
}

export function appendEvent(
  db: Database.Database,
  path: string,
  ev: StoredEventInput,
): void {
  withWrite(db, path, (d) => {
    d.prepare(
      "INSERT INTO events(event_id, run_id, session_id, seq, type, created_at, terminal, payload_json) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
    ).run(
      ev.event_id,
      ev.run_id,
      ev.session_id,
      ev.seq,
      ev.type,
      ev.created_at,
      ev.terminal ? 1 : 0,
      JSON.stringify(ev.payload),
    );

    // Approval / clarify events double-write to their own tables so they
    // survive a daemon restart (Hermes RFC durability requirement) and
    // can be queried via approval.list / clarify.list without scanning
    // the event log.
    if (ev.type === "approval.requested") {
      const p = ev.payload as { approval_id?: string; action_summary?: string; risk?: string };
      if (typeof p.approval_id === "string") {
        d.prepare(
          "INSERT OR IGNORE INTO approvals(approval_id, run_id, status, action_summary, risk, requested_at) VALUES (?, ?, ?, ?, ?, ?)",
        ).run(p.approval_id, ev.run_id, "pending", p.action_summary ?? "", p.risk ?? "medium", ev.created_at);
      }
    } else if (ev.type === "approval.resolved") {
      const p = ev.payload as { approval_id?: string; choice?: string; status?: string };
      if (typeof p.approval_id === "string") {
        if (!["accepted", "rejected", "timed_out"].includes(p.status ?? "")) throw new Error("invalid_approval_resolution");
        const changed = d.prepare(
          "UPDATE approvals SET status = ?, resolved_at = ?, choice = ? WHERE approval_id = ? AND run_id = ? AND status = 'pending'",
        ).run(p.status, ev.created_at, p.choice ?? null, p.approval_id, ev.run_id).changes;
        if (changed !== 1) {
          const prior = d.prepare("SELECT status,choice FROM approvals WHERE approval_id=? AND run_id=?")
            .get(p.approval_id, ev.run_id) as { status: string; choice: string | null } | undefined;
          if (!prior || prior.status !== p.status || prior.choice !== (p.choice ?? null))
            throw new Error("approval_resolution_conflict");
        }
      }
    } else if (ev.type === "clarify.requested") {
      const p = ev.payload as { clarify_id?: string; question?: string };
      if (typeof p.clarify_id === "string") {
        d.prepare(
          "INSERT OR IGNORE INTO clarifies(clarify_id, run_id, status, question, requested_at) VALUES (?, ?, ?, ?, ?)",
        ).run(p.clarify_id, ev.run_id, "pending", p.question ?? "", ev.created_at);
      }
    } else if (ev.type === "clarify.resolved") {
      const p = ev.payload as { clarify_id?: string; answer?: string; status?: string };
      if (typeof p.clarify_id === "string") {
        if (!["answered", "timed_out", "abandoned"].includes(p.status ?? "")) throw new Error("invalid_clarify_resolution");
        const changed = d.prepare(
          "UPDATE clarifies SET status = ?, resolved_at = ?, answer = ? WHERE clarify_id = ? AND run_id = ? AND status = 'pending'",
        ).run(p.status, ev.created_at, p.answer ?? null, p.clarify_id, ev.run_id).changes;
        if (changed !== 1) {
          const prior = d.prepare("SELECT status,answer FROM clarifies WHERE clarify_id=? AND run_id=?")
            .get(p.clarify_id, ev.run_id) as { status: string; answer: string | null } | undefined;
          if (!prior || prior.status !== p.status || prior.answer !== (p.answer ?? null))
            throw new Error("clarify_resolution_conflict");
        }
      }
    }
  });
}

export function finalizeRun(
  db: Database.Database,
  path: string,
  run_id: string,
  outcome: "completed" | "failed" | "cancelled" | "expired",
  completed_at: number,
): void {
  // SQLite UPDATE expressions read pre-UPDATE column values, so we must
  // compute duration in JS using the read-back started_at — not refer to
  // `completed_at` inside the same UPDATE (it would be NULL on first set).
  withWrite(db, path, (d) => {
    const row = d.prepare("SELECT started_at FROM runs WHERE run_id = ?").get(run_id) as
      | { started_at: number }
      | undefined;
    const duration_ms = row ? (completed_at - row.started_at) * 1000 : null;
    d.prepare(
      "UPDATE runs SET outcome = ?, completed_at = ?, duration_ms = ? WHERE run_id = ?",
    ).run(outcome, completed_at, duration_ms, run_id);
    d.prepare(
      "UPDATE tasks SET status = ?, updated_at = ? WHERE task_id = (SELECT task_id FROM runs WHERE run_id = ?)",
    ).run(
      outcome === "completed" ? "done" : outcome,
      completed_at,
      run_id,
    );
  });
}

/** Bounded local/default-tenant reader. Oversized payloads never enter JS. */
export interface BoundedRunEvent extends Omit<StoredEventRow, "payload_json"> {
  payload_json: string | null;
  payload_bytes: number;
}
export function readRunEventPage(db: Database.Database, input: {
  taskId: string;
  runId: string;
  afterSeq: number;
  limit: number;
  throughSeq?: number;
}): {
  events: BoundedRunEvent[];
  nextAfter: number;
  highWatermark: number;
  availableHighWatermark: number;
  hasMore: boolean;
  sequenceGap: boolean;
} {
  if (!input.taskId || !input.runId || !Number.isSafeInteger(input.afterSeq) || input.afterSeq < -1
    || !Number.isSafeInteger(input.limit) || input.limit < 1 || input.limit > 200
    || (input.throughSeq !== undefined && (!Number.isSafeInteger(input.throughSeq) || input.throughSeq < input.afterSeq)))
    throw new Error("invalid_event_page");
  const read = () => {
    const source = db.prepare(`SELECT 1 FROM runs r JOIN tasks t ON t.task_id=r.task_id
      WHERE r.run_id=? AND r.task_id=? AND r.tenant_id IS NULL AND t.tenant_id IS NULL`).get(input.runId, input.taskId);
    if (!source) throw new Error("run_source_not_found");
    const max = (db.prepare("SELECT COALESCE(MAX(seq),-1) seq FROM events WHERE run_id=?").get(input.runId) as { seq: number }).seq;
    const first = db.prepare("SELECT seq FROM events WHERE run_id=? ORDER BY seq LIMIT 1").get(input.runId) as { seq: number } | undefined;
    if (!Number.isSafeInteger(max) || max < -1 || (first && (!Number.isSafeInteger(first.seq) || first.seq < 0)))
      throw new Error("invalid_event_sequence");
    const high = input.throughSeq ?? max;
    // First select bounded metadata, then fetch payloads only for a 64KiB
    // aggregate page. Oversized records retain null payload evidence and stop
    // the reducer; small records still advance up to the 200-event limit.
    const rows = db.prepare(`WITH metadata AS (
      SELECT seq,length(CAST(payload_json AS BLOB)) payload_bytes FROM events
      WHERE run_id=? AND seq>? AND seq<=? ORDER BY seq LIMIT ?
    ), budgeted AS (
      SELECT seq,payload_bytes,SUM(CASE WHEN payload_bytes<=65536 THEN payload_bytes ELSE 0 END) OVER (ORDER BY seq) total_bytes FROM metadata
    ) SELECT substr(e.event_id,1,4097) event_id,e.run_id,substr(e.session_id,1,4097) session_id,
      e.seq,substr(e.type,1,257) type,e.created_at,e.terminal,b.payload_bytes,
      CASE WHEN b.payload_bytes<=65536 THEN e.payload_json ELSE NULL END payload_json
      FROM budgeted b JOIN events e ON e.run_id=? AND e.seq=b.seq
      WHERE b.total_bytes<=65536 ORDER BY e.seq`)
      .all(input.runId, input.afterSeq, high, input.limit, input.runId) as BoundedRunEvent[];
    const events = rows;
    const nextAfter = events.at(-1)?.seq ?? input.afterSeq;
    const hasMore = db.prepare("SELECT 1 FROM events WHERE run_id=? AND seq>? AND seq<=? LIMIT 1")
      .get(input.runId, nextAfter, high) !== undefined;
    let previous = input.afterSeq;
    let gap = max < input.afterSeq || max < high;
    for (const event of events) {
      if (!Number.isSafeInteger(event.seq) || event.seq !== previous + 1) gap = true;
      previous = event.seq;
    }
    return { events, nextAfter: events.at(-1)?.seq ?? input.afterSeq, highWatermark: high,
      availableHighWatermark: max, hasMore, sequenceGap: gap };
  };
  return db.inTransaction ? read() : db.transaction(read)();
}

/** Valid JSON with terminal control/format characters escaped in strings.
 * Use the same representation for service budgeting and CLI printing. */
export function serializeWorkerView(value: unknown): string {
  const serialized = JSON.stringify(value);
  if (typeof serialized !== "string") throw new Error("invalid_worker_response");
  return serialized.replace(/[\p{Cf}\u007f-\u009f\u2028\u2029]/gu,
    char => char.split("").map(unit => `\\u${unit.charCodeAt(0).toString(16).padStart(4, "0")}`).join(""));
}

const TokenSchema = z.object({ v: z.literal(1), kind: z.enum(["view", "cursor"]), scope: z.string(),

}).strict();

const RowSchema = z.object({
  kind: z.enum(["run", "session"]), id: sourceIdentity,
  state: StateSchema, revision: z.number().int().safe().positive().nullable(),
  updated: z.number().nullable(), ordinal: z.number().int().safe().positive().nullable(),
  turn_state: StateSchema.nullable(), decision_present: z.number(), uncertain_delivery: z.number(), unresolved_delivery: z.number(),
});
type Row = z.infer<typeof RowSchema>;

// Exact canonical paths and both tenant joins are intentional. A run alone has
// no repository column. Never infer its scope from a task title or event text.
const RELATION = `WITH bound AS (
  SELECT s.session_id, s.run_id, s.state, s.version, s.updated_at,
    r.outcome, r.completed_at, r.started_at
  FROM agent_sessions s JOIN tasks t ON t.task_id=s.task_id
  JOIN runs r ON r.run_id=s.run_id AND r.task_id=s.task_id
    AND r.session_id=s.session_id AND r.agent_id=s.agent_id
  WHERE s.repo_path=@repo AND t.tenant_id IS NULL AND r.tenant_id IS NULL
), sources AS (
  SELECT 'session' AS kind, session_id AS id, session_id, state, version AS revision, updated_at AS updated FROM bound
  UNION ALL
  SELECT 'run', run_id, session_id,
    CASE WHEN outcome IN ('completed','failed','cancelled','expired') THEN outcome
      WHEN outcome IS NULL THEN 'outcome_pending' ELSE 'unknown' END,
    NULL, coalesce(completed_at, started_at) FROM bound
)`;
const projection = (version: number) => `SELECT p.kind, substr(p.id,1,257) AS id, p.state, p.revision, p.updated,
  turn.ordinal, turn.state AS turn_state, (turn.decision_json IS NOT NULL) AS decision_present,
  EXISTS(SELECT 1 FROM session_messages m WHERE m.session_id=p.session_id
    AND m.direction='inbound' AND m.state IN ('pending','accepted','failed')) AS uncertain_delivery,
  EXISTS(SELECT 1 FROM session_messages m WHERE m.session_id=p.session_id
    AND m.direction='inbound' AND m.state IN ('pending','accepted','failed')
    ${version >= 41 ? "AND NOT EXISTS(SELECT 1 FROM attention_receipt_resolutions r WHERE r.message_id=m.message_id AND r.session_id=m.session_id AND r.disposition='abandoned_unknown')" : ""}) AS unresolved_delivery
  FROM sources p LEFT JOIN session_turns turn ON turn.session_id=p.session_id
    AND turn.ordinal=(SELECT max(ordinal) FROM session_turns WHERE session_id=p.session_id)`;

export const WORKER_VIEW_ERROR_CODES = ["invalid_request", "repository_unavailable", "invalid_cursor", "invalid_view", "source_not_found", "source_unavailable", "unsupported_schema", "unsupported_method", "response_too_large"] as const;
export class WorkerViewError extends Error {
  constructor(readonly code: typeof WORKER_VIEW_ERROR_CODES[number]) { super(code); this.name = "WorkerViewError"; }
}

/** Inject the daemon's existing connection. Construction never opens, migrates,
 * reads, or writes a database. Tokens are opaque, authenticated and ephemeral;
 * they grant no action authority and never expose provider-native identities. */
export class WorkerViewService {
  private readonly key = randomBytes(32);
  private readonly nowMs: () => number;
  constructor(private readonly db: Database.Database, options: { nowMs?: () => number } = {}) {
    this.nowMs = options.nowMs ?? Date.now;
  }

  handle(method: string, params: unknown): WorkerViewListResult | WorkerViewInspectResult {
    if (method === "worker.list") return this.list(params);
    if (method === "worker.inspect") return this.inspect(params);
    throw new WorkerViewError("unsupported_method");
  }

  list(raw: unknown): WorkerViewListResult {
    const parsed = WorkerListParamsSchema.safeParse(raw);
    if (!parsed.success) throw new WorkerViewError("invalid_request");
    const params = parsed.data;
    const common = this.context(params.repo_path);
    const previous = params.cursor ? this.decode(params.cursor, "cursor", common.scope.scope_id, common.sampled_at_ms) : null;
    const cursorContext = { ...common, sampled_at_ms: previous?.issued ?? common.sampled_at_ms };
    return this.read(version => {
      const query = { repo: common.scope.repo_path };
      const total = (this.db.prepare(`${RELATION} SELECT count(*) AS n FROM sources`).get(query) as { n: number }).n;
      const limit = params.limit ?? 25;
      const rows = this.db.prepare(`${RELATION} ${projection(version)}
        WHERE (@kind='' OR p.kind>@kind OR (p.kind=@kind AND p.id>@id))
        ORDER BY p.kind,p.id LIMIT @limit`).all({ ...query, kind: previous?.source ?? "", id: previous?.id ?? "", limit: limit + 1 });
      const result: WorkerViewListResult = { ...common, total, views: [], next_cursor: null, truncated: false };
      // Reserve room for a cursor before adding each entry; never silently drop
      // a row when the byte budget yields a shorter page than the count limit.
      let last: Row | null = null;
      for (let i = 0; i < Math.min(rows.length, limit); i++) {
        const row = RowSchema.parse(rows[i]);
        const more = i + 1 < rows.length;
        const candidate = { ...result, views: [...result.views, this.view(row, common)],
          next_cursor: more ? this.encode("cursor", row, cursorContext) : null, truncated: more };
        if (this.bytes(candidate) > WORKER_VIEW_MAX_BYTES) {
          if (!last) throw new WorkerViewError("response_too_large");
          result.next_cursor = this.encode("cursor", last, cursorContext); result.truncated = true;
          break;
        }
        Object.assign(result, candidate); last = row;
      }
      this.checkBytes(result);
      return WorkerListResultSchema.parse(result);
    });
  }

  inspect(raw: unknown): WorkerViewInspectResult {
    const parsed = WorkerInspectParamsSchema.safeParse(raw);
    if (!parsed.success) throw new WorkerViewError("invalid_request");
    const common = this.context(parsed.data.repo_path);
    const token = this.decode(parsed.data.view_id, "view", common.scope.scope_id, common.sampled_at_ms);
    return this.read(version => {
      const rawRow = this.db.prepare(`${RELATION} ${projection(version)} WHERE p.kind=@kind AND p.id=@id`)
        .get({ repo: common.scope.repo_path, kind: token.source, id: token.id });
      if (!rawRow) throw new WorkerViewError("source_not_found");
      const result = { ...common, view: this.view(RowSchema.parse(rawRow), common) };
      this.checkBytes(result);
      return WorkerInspectResultSchema.parse(result);
    });
  }

  private context(path: string) {
    let repo: string;
    try {
      repo = realpathSync(path);
      if (repo.length > 4096 || !statSync(repo).isDirectory()) throw new Error();
      accessSync(repo, constants.R_OK | constants.X_OK);
    } catch { throw new WorkerViewError("repository_unavailable"); }
    const now = this.nowMs();
    if (!Number.isSafeInteger(now) || now < 0) throw new WorkerViewError("source_unavailable");
    return { schema_version: 1 as const, mode: "readonly_state_view" as const,
      scope: { repo_path: repo, tenant: "default" as const, scope_id: createHash("sha256").update(`default\0${repo}`).digest("hex") },
      sampled_at_ms: now, coverage: {
        graph: "unsupported_authenticated_registration_required" as const,
        unbound_runs: "excluded_not_bound" as const,
        repository_binding: "canonical_stored_paths_only" as const,
        content: "structured_metadata_only" as const,
        counting: "source_views_not_unique_workers" as const,
      } };
  }

  private view(row: Row, common: ReturnType<WorkerViewService["context"]>): z.infer<typeof ViewSchema> {
    const updated = row.updated !== null && Number.isFinite(row.updated) && row.updated >= 0 && row.updated <= Number.MAX_SAFE_INTEGER / 1000 ? Math.round(row.updated * 1000) : null;
    const age = updated === null ? -1 : common.sampled_at_ms - updated;
    const terminal = ["completed", "failed", "cancelled", "expired"].includes(row.state);
    return { view_id: this.encode("view", row, common), source_kind: row.kind, state: row.state,
      revision: row.revision,
      latest_turn: row.ordinal !== null && row.turn_state !== null ? { ordinal: row.ordinal, state: row.turn_state, decision_present: row.decision_present === 1 } : null,
      last_updated_at_ms: updated, freshness: age < 0 ? "unknown" : age <= 60000 ? "recent" : "stale",
      process_liveness: "unknown", controls: "unsupported",
      uncertain_delivery_recorded: row.uncertain_delivery === 1,
      wait_reason: terminal ? "none" : row.unresolved_delivery ? "delivery_unknown" : row.state === "awaiting_decision" ? (row.decision_present ? "decision_record_present" : "unknown") : row.state === "idle" ? "awaiting_instruction" : "none",
    };
  }

  private encode(kind: Token["kind"], row: Row, common: ReturnType<WorkerViewService["context"]>): string {
    const payload: Token = { v: 1, kind, scope: common.scope.scope_id, source: row.kind, id: row.id, issued: common.sampled_at_ms };
    const iv = randomBytes(12);
    const cipher = createCipheriv("aes-256-gcm", this.key, iv);
    const data = Buffer.concat([cipher.update(JSON.stringify(payload), "utf8"), cipher.final()]);
    return Buffer.concat([iv, cipher.getAuthTag(), data]).toString("base64url");
  }

  private decode(value: string, kind: Token["kind"], scope: string, now: number): Token {
    try {
      if (!/^[A-Za-z0-9_-]+$/.test(value)) throw new Error();
      const bytes = Buffer.from(value, "base64url");
      if (bytes.toString("base64url") !== value || bytes.length < 29) throw new Error();
      const decipher = createDecipheriv("aes-256-gcm", this.key, bytes.subarray(0, 12));
      decipher.setAuthTag(bytes.subarray(12, 28));
      const token = TokenSchema.parse(JSON.parse(Buffer.concat([decipher.update(bytes.subarray(28)), decipher.final()]).toString("utf8")));
      if (token.kind !== kind || token.scope !== scope || token.issued > now || now - token.issued > TOKEN_TTL_MS) throw new Error();
      return token;
    } catch { throw new WorkerViewError(kind === "cursor" ? "invalid_cursor" : "invalid_view"); }
  }

  private read<T>(fn: (version: number) => T): T {
    try {
      return this.db.transaction(() => {
        const version = this.db.pragma("user_version", { simple: true });
        if (typeof version !== "number" || version < 24 || version > CURRENT_SCHEMA_VERSION) throw new WorkerViewError("unsupported_schema");
        return fn(version);
      })();
    } catch (error) {
      if (error instanceof WorkerViewError) throw error;
      throw new WorkerViewError("source_unavailable");
    }
  }
  private bytes(value: unknown): number { return Buffer.byteLength(serializeWorkerView(value), "utf8") + 1; }
  private checkBytes(value: unknown): void {
    if (this.bytes(value) > WORKER_VIEW_MAX_BYTES) throw new WorkerViewError("response_too_large");
  }
}
