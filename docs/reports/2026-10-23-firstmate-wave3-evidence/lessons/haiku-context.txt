        || run["binding_hash"] !== attentionDigest([run["scope_id"], "run", input.runId])) return { state: "unknown", reason: "source_not_found_or_unbound" };
      const stats = db.prepare("SELECT COUNT(*) n,COALESCE(MAX(seq),-1) last FROM events WHERE run_id=?").get(input.runId) as { n: number; last: number };
      const events = db.prepare(`SELECT substr(event_id,1,257) event_id,substr(session_id,1,257) session_id,seq,substr(type,1,257) type,created_at,terminal,
        length(CAST(payload_json AS BLOB)) bytes,CASE WHEN length(CAST(payload_json AS BLOB))<=4096 THEN payload_json ELSE NULL END payload
        FROM events WHERE run_id=? ORDER BY seq LIMIT 128`).all(input.runId) as Array<{ event_id: string; session_id: string; seq: number; type: string; created_at: number; terminal: number; bytes: number; payload: string | null }>;

        previous = event.seq; return valid;
      });
      const sourceHash = hash({ run, repo, stats, events });
      const receipt = receiptSchema.parse({ schema_version: 1, kind: "investigation", qualification: "ordinary_run_unverified", tenant: "default",
        scope_id: run["scope_id"], repo_path: repo, task_id: run["task_id"], run_id: run["run_id"],

        source_hash: sourceHash, event_count: stats.n, through_seq: stats.last, history_complete: complete, event_hashes: events.map(hash) });

    })();
  } catch { return { state: "unknown", reason: "source_unavailable" }; }
}

async function directory(repo: string, relativePath: string, create = false): Promise<string> {
  const target = resolve(repo, relativePath);
  const rel = relative(repo, target);
  if (!rel || isAbsolute(rel) || rel === ".." || rel.startsWith(".." + sep)) throw Error("unsafe_lesson_path");
  let current = repo;
  for (const part of rel.split(sep)) {
    current = join(current, part);
    try {
      const stat = await lstat(current);
      if (!stat.isDirectory() || stat.isSymbolicLink() || await realpath(current) !== current) throw Error("unsafe_lesson_path");
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code !== "ENOENT") throw error;
      if (create) { try { await mkdir(current, { mode: 0o700 }); } catch (e) { if ((e as NodeJS.ErrnoException).code !== "EEXIST") throw e; }
        if (!(await lstat(current)).isDirectory() || await realpath(current) !== current) throw Error("unsafe_lesson_path"); }
    }
  }
  return target;
}
async function safeRead(path: string): Promise<string | null> {
  let file: Awaited<ReturnType<typeof open>> | undefined;
  try {
    file = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
    const stat = await file.stat(); if (!stat.isFile() || stat.size > 256 * 1024) throw Error("unsafe_lesson_file");
    const bytes = Buffer.alloc(256 * 1024 + 1); const { bytesRead } = await file.read(bytes, 0, bytes.length, 0);
    if (bytesRead > 256 * 1024) throw Error("unsafe_lesson_file");
    return bytes.subarray(0, bytesRead).toString("utf8");
  } catch (error) { if ((error as NodeJS.ErrnoException).code === "ENOENT") return null; throw Error("unsafe_lesson_file"); }
  finally { await file?.close(); }
}

export type LessonDraftConfig = z.infer<typeof LessonDraftConfig>;

export interface LessonDraftInput {
  task: Task;
  /** Total attempts the task took, including the successful one. */
  iterations: number;
  /** Feedback derived from each failed attempt, in attempt order. */
  feedback: readonly IterationFeedback[];
  finalVerdict?: Verdict | null;
  changedFiles?: readonly string[];
  diffSummary?: string;
  commitHash?: string;
}

export interface LessonDraftResult {
  written: boolean;
  path?: string;
  reason?: string;
}

/** Drafts are only useful when the verifier actually rejected an earlier attempt. */
export function shouldDraftLesson(input: Pick<LessonDraftInput, "feedback">): boolean {
  return input.feedback.length > 0;
}

export function renderLessonDraft(input: LessonDraftInput): string {
  const { task, feedback } = input;
  const failedEvaluationIds = [...new Set(feedback.flatMap((entry) => entry.failed_evaluations))];
  const verifierCatches = feedback.map((entry) => {
    const issues = entry.blocking_issues.length === 0
      ? "verification did not pass"
      : entry.blocking_issues.map((issue) => issue.message).join("; ");
    return `- Attempt ${entry.attempt}: ${issues}`;
  });
  const focus = [...new Set(feedback.flatMap((entry) => entry.suggested_focus))];
  const suggestedEvaluations = suggestedEvaluationLines(input, failedEvaluationIds);
  const skillId = isSkillTask(task) ? skillIdOf(task) : undefined;

  const lines = [
    `# Lesson Draft: ${task.task_id}`,
    "",
    `> Generated by osstack loop: task succeeded on attempt ${input.iterations} after ${feedback.length} failed attempt(s).`,
    "> Review before adopting. Official skill files are never modified automatically.",
    "",
    "## Task Prompt",
    task.prompt,
    "",
    "## Failed Evaluations",
    ...(failedEvaluationIds.length > 0
      ? failedEvaluationIds.map((id) => `- ${id}`)
      : ["- None recorded (verification failed without evaluation specs)"]),
    "",
    "## What The Verifier Caught",
    ...verifierCatches,
    "",
    "## Final Fix Summary",
    `- Outcome: success after ${input.iterations} iteration(s)`,
    ...(input.diffSummary !== undefined ? [`- Diff: ${input.diffSummary}`] : []),
    ...(input.changedFiles !== undefined && input.changedFiles.length > 0
      ? [`- Changed files: ${input.changedFiles.join(", ")}`]
      : []),
    ...(input.commitHash !== undefined ? [`- Commit: ${input.commitHash}`] : []),
    ...(input.finalVerdict != null ? [`- Final verdict: ${input.finalVerdict.summary}`] : []),
    "",
    "## Reusable Rule",
    reusableRule(task, failedEvaluationIds, focus),
    "",
    "## Suggested Evaluation",
    ...suggestedEvaluations,
    "",
    "## Suggested Target Skill",
    skillId !== undefined ? `- ${skillId}` : "- Not inferable from task context",
    "",
  ];
  return lines.join("\n");
}

export async function writeLessonDraft(
  input: LessonDraftInput,
  options: { workspacePath: string; config?: Partial<LessonDraftConfig> },
): Promise<LessonDraftResult> {
  const config = LessonDraftConfig.parse(options.config ?? {});
  if (!config.enabled) return { written: false, reason: "lesson drafts disabled" };
  if (!shouldDraftLesson(input)) {
    return { written: false, reason: "task succeeded without failed attempts; nothing to learn" };
  }

  const dir = await resolveLessonDir(options.workspacePath, config.dir);
  const path = join(dir, `${sanitizeFileName(input.task.task_id)}.md`);
  await mkdir(dirname(path), { recursive: true });
  const file = await open(path, constants.O_WRONLY | constants.O_CREAT | constants.O_NOFOLLOW | constants.O_NONBLOCK, 0o600);
  try {
    const info = await file.stat();
    if (!info.isFile() || info.nlink !== 1) throw new Error("unsafe_lesson_file");
    await file.truncate(0);
    await file.writeFile(renderLessonDraft(input), "utf-8");
  } finally { await file.close(); }
  return { written: true, path };
}

function reusableRule(task: Task, failedEvaluationIds: string[], focus: string[]): string {
  const subject = `tasks like "${task.capability}"`;
  const evaluationClause = failedEvaluationIds.length > 0
    ? `make the ${failedEvaluationIds.join(", ")} evaluation(s) pass before claiming completion`
    : "re-run the verifier before claiming completion";
  const focusClause = focus.length > 0 ? ` Pay attention to: ${focus.join("; ")}.` : "";
  return `When working on ${subject}, ${evaluationClause}.${focusClause} (Draft rule — refine before adding to a skill.)`;
}

function suggestedEvaluationLines(input: LessonDraftInput, failedEvaluationIds: string[]): string[] {
  const lines: string[] = [];
  const specs = input.task.evaluation ?? [];
  for (const spec of specs) {
    if (!failedEvaluationIds.includes(spec.id)) continue;
    lines.push(spec.command !== undefined
      ? `- Run \`${spec.command}\` (${spec.id})`
      : `- Require ${spec.mode} review: ${spec.description} (${spec.id})`);
  }
  for (const id of failedEvaluationIds) {
    if (specs.some((spec) => spec.id === id)) continue;
    lines.push(`- Keep the ${id} check as a required evaluation`);
  }
  if (lines.length === 0) {
    lines.push("- Add an evaluation spec that would have caught the first failed attempt");
  }
  return lines;
}

function sanitizeFileName(value: string): string {
  const sanitized = value.replace(/[^a-zA-Z0-9._-]+/g, "-").replace(/^-+|-+$/g, "");
  return sanitized === "" ? "lesson" : sanitized.slice(0, 80);
}

// ── Review workflow: list / read / promote / discard ────────────────────
//
// Drafts wait in loop-lessons/drafts/ until a human reviews them.
// Promoting moves a draft to loop-lessons/promoted/ and, ONLY with an
// explicit target skill path, appends the lesson to that file. Discarding
// moves it to loop-lessons/discarded/. The loop itself never promotes.

export interface LessonDraftEntry {
  id: string;
  path: string;
  title: string;
  suggested_skill?: string;
}

export interface PromoteLessonDraftResult {
  promoted: boolean;
  from?: string;
  to?: string;
  appended_to?: string;
  reason?: string;
}

export interface DiscardLessonDraftResult {
  discarded: boolean;
  from?: string;
  to?: string;
  reason?: string;
}

export async function listLessonDrafts(
  workspacePath: string,
  dir = "loop-lessons/drafts",
): Promise<LessonDraftEntry[]> {
  let draftDir: string;
  try { draftDir = await resolveLessonDir(workspacePath, dir); }
  catch (error) { if ((error as NodeJS.ErrnoException).code === "ENOENT") return []; throw error; }
  let entries: string[];
  try {
    entries = await readdir(draftDir);
  } catch {
    return [];
  }
  const drafts: LessonDraftEntry[] = [];
  for (const entry of entries.sort()) {
    if (!entry.endsWith(".md")) continue;
    const path = join(draftDir, entry);
    try {
      drafts.push(parseLessonDraftEntry(path, await readDraftFile(path)));
    } catch {
      // Skip unreadable drafts; listing must not fail on one bad file.
    }
  }
  return drafts;
}

export async function readLessonDraft(
  workspacePath: string,
  id: string,
  dir = "loop-lessons/drafts",
): Promise<{ entry: LessonDraftEntry; content: string } | null> {
  const path = join(await resolveLessonDir(workspacePath, dir), `${sanitizeFileName(id)}.md`);
  try {
    const content = await readDraftFile(path);
    return { entry: parseLessonDraftEntry(path, content), content };
  } catch {
    return null;
  }
}

export async function promoteLessonDraft(
  workspacePath: string,
  id: string,
  options: { dir?: string; promotedDir?: string; toSkillPath?: string } = {},
): Promise<PromoteLessonDraftResult> {
  const draft = await readLessonDraft(workspacePath, id, options.dir ?? "loop-lessons/drafts");
  if (draft === null) return { promoted: false, reason: `no lesson draft found for id ${id}` };

  // Validate lifecycle destinations before touching an explicitly selected skill.
  const promotedDir = await resolveLessonDir(workspacePath, options.promotedDir ?? "loop-lessons/promoted");
  const to = join(promotedDir, basename(draft.entry.path));
  if (await existsEntry(to)) return { promoted: false, reason: "promoted lesson already exists" };
  if (options.toSkillPath !== undefined) {
    const skillPath = isAbsolute(options.toSkillPath)
      ? options.toSkillPath
      : join(workspacePath, options.toSkillPath);
    // Minimal path hardening. Skills legitimately live outside the workspace, so
    // we do NOT confine to the workspace — but we must not let a lesson be
    // appended to an arbitrary file (e.g. /etc/hosts, ~/.zshrc, a random .md
    // note). Require the target to be a SKILL.md: the canonical skill entry file
    // every skill in this ecosystem uses. This blocks obvious misfires while
    // preserving legitimate external skill targets.
    if (basename(skillPath).toLowerCase() !== "skill.md") {
      return {
        promoted: false,
        reason: `refusing to append a lesson to a non-skill target: ${options.toSkillPath}. --to-skill must point at a SKILL.md file.`,
      };
    }
    let existing: string;
    try {
      existing = await readFile(skillPath, "utf-8");
    } catch {
      return { promoted: false, reason: `target skill file not found: ${options.toSkillPath}` };
    }
    const rule = extractLessonSection(draft.content, "## Reusable Rule");
    const evaluation = extractLessonSection(draft.content, "## Suggested Evaluation");
    const appended = [
      existing.endsWith("\n") ? existing : `${existing}\n`,
      `\n## Lesson: ${draft.entry.id}\n`,
      "\n",
      rule ?? "(no reusable rule recorded)",
      "\n",
      ...(evaluation !== undefined ? ["\n### Suggested Evaluation\n\n", evaluation, "\n"] : []),
    ].join("");
    // Atomic write: a crash mid-write must never truncate an official skill.
    await writeFileAtomic(skillPath, appended);
  }

  await mkdir(promotedDir, { recursive: true });
  await rename(draft.entry.path, to);
  return {
    promoted: true,
    from: draft.entry.path,
    to,
    ...(options.toSkillPath !== undefined ? { appended_to: options.toSkillPath } : {}),
  };
}

export async function discardLessonDraft(
  workspacePath: string,
  id: string,
  options: { dir?: string; discardedDir?: string } = {},
): Promise<DiscardLessonDraftResult> {
  const draft = await readLessonDraft(workspacePath, id, options.dir ?? "loop-lessons/drafts");
  if (draft === null) return { discarded: false, reason: `no lesson draft found for id ${id}` };
  const discardedDir = await resolveLessonDir(workspacePath, options.discardedDir ?? "loop-lessons/discarded");
  await mkdir(discardedDir, { recursive: true });
  const to = join(discardedDir, basename(draft.entry.path));
  if (await existsEntry(to)) return { discarded: false, reason: "discarded lesson already exists" };
  await rename(draft.entry.path, to);
  return { discarded: true, from: draft.entry.path, to };
}

function parseLessonDraftEntry(path: string, content: string): LessonDraftEntry {
  const id = basename(path).replace(/\.md$/, "");
  const titleMatch = content.match(/^# Lesson Draft: (.+)$/m);
  const skillSection = extractLessonSection(content, "## Suggested Target Skill");
  const skillLine = skillSection?.split("\n").find((line) => line.startsWith("- "));
  const skill = skillLine?.slice(2).trim();
  return {
    id,
    path,
    title: titleMatch?.[1]?.trim() ?? id,
    ...(skill !== undefined && !skill.startsWith("Not inferable") ? { suggested_skill: skill } : {}),
  };
}

function extractLessonSection(content: string, heading: string): string | undefined {
  const lines = content.split("\n");
  const start = lines.findIndex((line) => line.trim() === heading);
  if (start === -1) return undefined;
  const rest = lines.slice(start + 1);
  const end = rest.findIndex((line) => line.startsWith("## "));
  const section = (end === -1 ? rest : rest.slice(0, end)).join("\n").trim();
  return section === "" ? undefined : section;
}

async function resolveLessonDir(workspacePath: string, dir: string): Promise<string> {
  const root = await realpath(workspacePath);
  const lexicalRoot = resolve(workspacePath);
  const lexicalTarget = resolve(lexicalRoot, dir);
  const lexicalRelative = relative(lexicalRoot, lexicalTarget);
  const insideLexicalRoot = lexicalRelative !== ".." && !lexicalRelative.startsWith(".." + sep) && !isAbsolute(lexicalRelative);
  const target = insideLexicalRoot ? resolve(root, lexicalRelative) : resolve(root, dir);
  const rel = relative(root, target);
  if (!rel || isAbsolute(rel) || rel === ".." || rel.startsWith(".." + sep)) throw new Error("unsafe_lesson_path");
  let current = root;
  for (const part of rel.split(sep)) {
    current = join(current, part);
    try {
      const info = await lstat(current);
      if (!info.isDirectory() || info.isSymbolicLink() || await realpath(current) !== current) throw new Error("unsafe_lesson_path");
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code !== "ENOENT") throw error;
    }
  }
  // Keep the caller's path spelling (e.g. /var versus /private/var) for API compatibility.
  return insideLexicalRoot ? lexicalTarget : target;
}
async function existsEntry(path: string): Promise<boolean> {
  try { await lstat(path); return true; }
  catch (error) { if ((error as NodeJS.ErrnoException).code === "ENOENT") return false; throw error; }
}
async function readDraftFile(path: string): Promise<string> {
  const file = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
  try {
    const info = await file.stat();
    if (!info.isFile() || info.size > 1024 * 1024) throw new Error("unsafe_lesson_file");
    const bytes = Buffer.alloc(1024 * 1024 + 1);
    const result = await file.read(bytes, 0, bytes.length, 0);
    if (result.bytesRead > 1024 * 1024) throw new Error("unsafe_lesson_file");
    return bytes.subarray(0, result.bytesRead).toString("utf8");
  } finally { await file.close(); }
}

# Evidence-backed lesson candidates

## Implementation contract

This bridge writes review material, never approval or published knowledge. The
real CLI `attention lesson --from-run RUN --repo PATH --default-tenant` requests
`attention.lesson-evidence` from the existing authenticated daemon. The RPC reader
uses the daemon's existing database without opening, migrating or modifying it.
The CLI never opens a database or starts a daemon. It requires a
canonical registered Attention repository scope, the default-tenant task/run and
an admitted run subject. Adapter events and ordinary run outcomes cannot prove a
verifier succeeded: these sources always yield an **investigation** candidate.
Absent, foreign, released or ambiguous source bindings fail closed without a file.

The source digest binds scope/task/run identity and bounded persisted event facts.
No raw task prompt, transcript, command output, or secret-looking log text is
copied. Oversized/truncated event history is explicitly incomplete evidence.
The stable candidate key binds scope/task/failure family/source digest. Candidate
creation is create-only and atomic; identical input is idempotent. User-edited
files are preserved and a revision candidate references the prior draft. Reviewed
promoted/discarded candidates are never silently reintroduced.

Storage uses the existing `loop-lessons/drafts` directory, so existing
`loop lessons list/show/promote/discard` controls review. Directory components
must remain under the real repository path and cannot be symlinks. The bridge
checks drafts, promoted and discarded paths; no automatic promotion or skill edit
is permitted. Missing current proof remains investigation, regardless of booleans
such as `passed` or `reviewed` in an input/event.

## Strong evidence qualification

A host-only Graph adapter can inspect authenticated registration scope, immutable
verification joins, action ledger, candidate artifacts and hashed verifier receipts.
It must call the registration's synchronous authorize callback before and after
asynchronous evidence reads. CLI callers cannot supply Graph scope or arbitrary
Graph database access. A success-after-repair candidate requires an actual prior
failed verifier command, a confirmed repair action, a different frozen candidate,
then a same-candidate successful command and independent review. Unknown actions,
missing receipts, mismatched scope/head/check, or source changes invalidate success.

Graph qualification remains pending until a real authenticated host registration
is wired. The ordinary CLI does not upgrade itself to that authority. The current
delivery contract receipt has no complete failure→repair attempt history and is
not treated as proof of recovery.

## Acceptance cases

- Real ordinary completed run + forged passed/reviewed event → investigation only.
- Foreign tenant, wrong repo, mismatched task, missing/admission-released run → no file.
- Repeated unchanged source → same candidate; edited source file → revision without overwrite.
- Source changes after inspection → retry with a new fingerprint, never reuse old proof.
- Draft/promoted/discarded symlink escape, symlink final file and unsafe output path → reject.
- Interrupted publication → no partial Markdown target, no overwriting old user content.
- Existing list/show/promote/discard sees the candidate; bridge itself never promotes.
- Missing database/schema/history and opaque execution status remain unknown/investigation.
- Strong adapter: forged passed flag, wrong scope/candidate/check, unchanged repair,
  unconfirmed stop, missing final proof or revoked authorization → no recovery claim.
