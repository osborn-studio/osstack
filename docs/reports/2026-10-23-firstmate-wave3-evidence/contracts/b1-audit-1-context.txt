const exec = promisify(execFile);
export const deliveryBytesDigest = (bytes: string | Buffer) => createHash("sha256").update(bytes).digest("hex");
export async function deliveryGit(workspace: string, args: string[]): Promise<string> {
  return (await exec("git", ["-c", "core.hooksPath=/dev/null", ...args], { cwd: workspace, encoding: "utf8", timeout: 30000, maxBuffer: 64 * 1024 * 1024 })).stdout;
}
export async function captureDeliveryCandidate(workspace: string, baseHead: string): Promise<DeliveryCandidateIdentityV2> {
  const head = (await deliveryGit(workspace, ["rev-parse", "HEAD"])).trim();
  const tree = (await deliveryGit(workspace, ["write-tree"])).trim();
  const diff = await deliveryGit(workspace, ["diff", "--cached", "--binary", baseHead, "--"]);
  return { base_head: baseHead, head, tree_id: tree, diff_digest: deliveryBytesDigest(diff) };
}
export async function persistDeliveryHostRecord(root: string, value: unknown): Promise<DeliveryReceiptReferenceV2> {
  await mkdir(root, { recursive: true, mode: 0o700 });
  const canonicalRoot = await realpath(root);
  const body = `${JSON.stringify(value)}\n`;
  const digest = deliveryBytesDigest(body);
  const path = join(canonicalRoot, `${digest}.json`);
  try { await writeFile(path, body, { flag: "wx", mode: 0o600 }); }
  catch (error) {
    if ((error as NodeJS.ErrnoException).code !== "EEXIST" || await readFile(path, "utf8") !== body) throw error;
  }
  return { path, digest };
}
export async function readDeliveryHostRecord(root: string, ref: DeliveryReceiptReferenceV2): Promise<unknown> {
  const canonicalRoot = await realpath(root);
  if (!isAbsolute(ref.path) || (await lstat(ref.path)).isSymbolicLink()) throw new Error("delivery_receipt_path_invalid");
  const canonicalPath = await realpath(ref.path);
  const local = relative(canonicalRoot, canonicalPath);
  if (local.startsWith("..") || isAbsolute(local) || !/^[a-f0-9]{64}\.json$/.test(local)
    || local !== `${ref.digest}.json`) throw new Error("delivery_receipt_path_invalid");
  const handle = await open(canonicalPath, constants.O_RDONLY | constants.O_NOFOLLOW);
  try {
    const info = await handle.stat();
    if (!info.isFile() || info.size > 1024 * 1024) throw new Error("delivery_receipt_size_invalid");
    const body = Buffer.alloc(info.size + 1);
    const { bytesRead } = await handle.read(body, 0, body.length, 0);
    if (bytesRead !== info.size) throw new Error("delivery_receipt_size_changed");
    const actual = body.subarray(0, bytesRead);
    if (deliveryBytesDigest(actual) !== ref.digest) throw new Error("delivery_receipt_digest_mismatch");
    return JSON.parse(actual.toString("utf8")) as unknown;
  } finally { await handle.close(); }
}

import { createHash } from "node:crypto";
import {
  DeliveryContractV2, FrozenDeliveryContractV2, DeliveryVerificationReceiptV2,
  type DeliveryCandidateIdentityV2,
} from "@osstack/protocol";

export function deliveryContractDigest(value: unknown): string {
  const canonical = (input: unknown): unknown => {
    if (Array.isArray(input)) return input.map(canonical);
    if (input && typeof input === "object") return Object.fromEntries(Object.entries(input)
      .sort(([a], [b]) => a.localeCompare(b, "en")).map(([key, val]) => [key, canonical(val)]));
    return input;
  };
  return createHash("sha256").update(JSON.stringify(canonical(value))).digest("hex");
}
export function deliveryCheckReference(command: string): string {
  return `sha256:${createHash("sha256").update(command).digest("hex")}`;
}
/** Canonicalization is pure: callers must separately resolve workspace/policy authority. */
export function freezeDeliveryContract(value: unknown): FrozenDeliveryContractV2 {
  const parsed = DeliveryContractV2.parse(value);
  const contract = { ...parsed,
    required_checks: [...parsed.required_checks].sort((a, b) => a.id.localeCompare(b.id, "en")),
    allowed_paths: [...new Set(parsed.allowed_paths)].sort(),
    requested_actions: [...new Set(parsed.requested_actions)].sort(),
  };
  const digest = deliveryContractDigest(contract);
  return FrozenDeliveryContractV2.parse({ contract, digest, brief: [
    `Delivery contract V2: ${contract.contract_id} (${digest})`,
    `Mode: ${contract.mode}`, `Goal: ${contract.goal}`, `Repository: ${contract.canonical_repo}`,
    `Base: ${contract.base_head}`, `Completion ceiling: ${contract.requested_completion}`,
    `Paths: ${contract.allowed_paths.join(", ")}`, `Actions: ${contract.requested_actions.join(", ")}`,
    `Required checks: ${contract.required_checks.map((check) => `${check.id}=${check.registry_ref}`).join(", ")}`,
    `Policy: ${contract.policy_digest}`, "This contract does not grant execution, publication or merge authority.",
  ].join("\n") });
}
export function assertFrozenDeliveryContract(value: unknown): FrozenDeliveryContractV2 {
  const input = FrozenDeliveryContractV2.parse(value);
  const expected = freezeDeliveryContract(input.contract);
  if (expected.digest !== input.digest || expected.brief !== input.brief) throw new Error("delivery_contract_digest_mismatch");
  return expected;
}
/** References select already accepted host/CLI commands; contract text never becomes shell. */
export function resolveDeliveryContractChecks(frozen: FrozenDeliveryContractV2, acceptedCommands: readonly string[]) {
  const { contract } = assertFrozenDeliveryContract(frozen);
  const registry = new Map(acceptedCommands.map((command) => [deliveryCheckReference(command), command]));
  return contract.required_checks.map((check) => {
    const command = registry.get(check.registry_ref);
    if (command === undefined) throw new Error(`delivery_check_not_registered:${check.id}`);
    return { ...check, command };
  });
}
export function assertDeliveryContractAdmission(frozen: FrozenDeliveryContractV2, input: {
  canonicalRepo: string; baseHead: string; taskId: string; goal: string; policyDigest: string;
  reviewPolicyRef: string; acceptedCommands: readonly string[];
}): void {
  const { contract } = assertFrozenDeliveryContract(frozen);
  if (contract.mode === "scout") throw new Error("delivery_scout_tool_guarantee_unavailable");
  if (contract.canonical_repo !== input.canonicalRepo || contract.base_head !== input.baseHead
    || contract.task_id !== input.taskId || contract.goal !== input.goal || contract.policy_digest !== input.policyDigest
    || contract.review_policy_ref !== input.reviewPolicyRef) throw new Error("delivery_contract_admission_mismatch");
  resolveDeliveryContractChecks(frozen, input.acceptedCommands);
}
export function assertDeliveryContractPaths(frozen: FrozenDeliveryContractV2, paths: readonly string[]): void {
  const { contract } = assertFrozenDeliveryContract(frozen);
  if (paths.some((path) => !contract.allowed_paths.some((allowed) => path === allowed || path.startsWith(`${allowed.replace(/\/$/, "")}/`))))
    throw new Error("delivery_contract_path_outside_scope");
}
/** Only use after a trusted host has loaded its persisted receipt by digest.
 * Parsing or calling this function on arbitrary JSON does not confer provenance. */
export function verifyDeliveryContractReceipt(frozen: FrozenDeliveryContractV2, value: unknown, expected: {
  runId: string; candidate: DeliveryCandidateIdentityV2; policyDigest: string;
}): DeliveryVerificationReceiptV2 {
  const { contract, digest } = assertFrozenDeliveryContract(frozen);
  const receipt = DeliveryVerificationReceiptV2.parse(value);
  if (receipt.contract_digest !== digest || receipt.run_id !== expected.runId || receipt.task_id !== contract.task_id
    || receipt.policy_digest !== expected.policyDigest || receipt.policy_digest !== contract.policy_digest
    || receipt.candidate.base_head !== contract.base_head
    || deliveryContractDigest(receipt.candidate) !== deliveryContractDigest(expected.candidate)) throw new Error("delivery_receipt_binding_mismatch");
  if (new Set(receipt.checks.map((check) => check.id)).size !== receipt.checks.length) throw new Error("delivery_duplicate_receipt_check");
  for (const required of contract.required_checks) {
    const check = receipt.checks.find((item) => item.id === required.id && item.registry_ref === required.registry_ref);
    if (!check || check.exit_code !== 0 || `sha256:${check.command_digest}` !== required.registry_ref) throw new Error("delivery_required_check_failed");
  }
  if (receipt.checks.some((check) => check.exit_code !== 0)) throw new Error("delivery_host_check_failed");
  const review = receipt.review;
  if (review.policy_ref !== contract.review_policy_ref || review.verdict !== "approved"
    || review.builder_agent_id === review.reviewer_agent_id
    || (review.builder_provider === review.reviewer_provider && review.builder_model === review.reviewer_model))
    throw new Error("delivery_independent_review_required");
  return receipt;
}
/** Explicit promotion creates a new contract; it carries no prior receipts or approval. */
export function promoteDeliveryScout(frozen: FrozenDeliveryContractV2, input: {
  contractId: string; baseHead: string; policyDigest: string; requestedActions: DeliveryContractV2["requested_actions"];
}): FrozenDeliveryContractV2 {
  const parent = assertFrozenDeliveryContract(frozen);
  if (parent.contract.mode !== "scout" || input.contractId === parent.contract.contract_id) throw new Error("delivery_invalid_promotion");
  return freezeDeliveryContract({ ...parent.contract, contract_id: input.contractId, parent_contract_id: parent.contract.contract_id,
    mode: "implement", base_head: input.baseHead, policy_digest: input.policyDigest, requested_actions: input.requestedActions });
}

export interface RiskClassificationContext {

  now: () => string;

export function maxRisk(
  left: AutonomyRiskLevel,
  right: AutonomyRiskLevel,
): AutonomyRiskLevel {
  return RISK_ORDER[left] >= RISK_ORDER[right] ? left : right;

export function classifyPreflightRisk(
  input: unknown,
  context: RiskClassificationContext,
): AutonomyRiskDecision {
  const intent = AutonomyTaskIntent.parse(input);
  const signals: AutonomyRiskSignal[] = [];
  const isReadOnly =
    intent.expected_paths.length === 0 &&
    intent.requested_actions.every((action) => READ_ONLY_ACTIONS.has(action));

  if (!isReadOnly) {
    addBaselineSignal(signals, intent.expected_paths, "preflight intent");
  }
  addPathSignals(signals, intent.expected_paths);
  addTextSignals(
    signals,
    [intent.goal, ...intent.requested_actions],
    "preflight request",
  );
  addProtectedPolicySignals(
    signals,
    intent.expected_paths,
    context.protected_policy_paths ?? [],
  );
  addOsstackKernelSignals(signals, intent.expected_paths, intent.repo_id);

  return decision(
    "preflight",
    signals,
    context,
    {
      intent,
      protected_policy_paths: canonicalStrings(
        context.protected_policy_paths ?? [],
      ),
    },
  );
}

export function classifyPostDiffRisk(
  input: unknown,
  context: RiskClassificationContext,
): AutonomyRiskDecision {
  const postDiff = PostDiffInput.parse(input);
  const signals: AutonomyRiskSignal[] = [];

  addBaselineSignal(signals, postDiff.changed_paths, "staged diff");
  addPathSignals(signals, postDiff.changed_paths);
  addProtectedPolicySignals(
    signals,
    postDiff.changed_paths,
    context.protected_policy_paths ?? [],
  );
  addOsstackKernelSignals(
    signals,
    postDiff.changed_paths,
    context.repo_id ?? "",
  );

  for (const change of postDiff.dependency_changes) {
    if (!change.major) continue;
    signals.push({
      rule_id: "risk.dependency.major",
      level: "R3",
      message: `major dependency change: ${change.name}`,
    });
  }

  addTextSignals(
    signals,
    [...addedDiffLines(postDiff.staged_diff_binary), ...postDiff.commands],
    "staged diff or command",
  );
  addPackageScriptSignals(signals, postDiff);

  return decision(
    "post_diff",
    signals,
    context,
    {
      post_diff: postDiff,
      repo_id: context.repo_id ?? "",
      protected_policy_paths: canonicalStrings(
        context.protected_policy_paths ?? [],
      ),
    },
  );
}

function configuredPackageScripts(commands: readonly string[]): Set<string> {
  const names = new Set(SENSITIVE_PACKAGE_SCRIPTS);
  for (const command of commands) {
    const tokens = command.trim().split(/\s+/);
    const managerIndex = tokens.findIndex((token) =>
      ["pnpm", "npm", "yarn", "bun"].includes(token)
    );
    if (managerIndex < 0) continue;
    let candidate: string | undefined;
    for (let index = managerIndex + 1; index < tokens.length; index += 1) {
      const token = tokens[index]!;
      if (token === "--") break;
      if (token === "run") {
        candidate = tokens[index + 1];
        break;
      }
      if (!token.startsWith("-")) candidate = token;
    }
    if (candidate !== undefined && /^[A-Za-z0-9_.:-]+$/.test(candidate)) {
      names.add(candidate);
    }
  }
  return names;
}

function addPackageScriptSignals(
  signals: AutonomyRiskSignal[],
  postDiff: ReturnType<typeof PostDiffInput.parse>,
): void {
  const manifests = postDiff.changed_paths.filter(
    (path) => path === "package.json" || path.endsWith("/package.json"),
  );
  if (manifests.length === 0) return;
  const scriptNames = configuredPackageScripts(postDiff.commands);
  for (const line of addedDiffLines(postDiff.staged_diff_binary)) {
    const match = line.match(/^\s*"([^"]+)"\s*:/);
    const script = match?.[1];
    if (script === undefined || !scriptNames.has(script)) continue;
    for (const path of manifests) {
      signals.push({
        rule_id: "risk.verification-script",
        level: "R3",
        message: `verification or lifecycle package script changed: ${script}`,
        path,
      });
    }
  }
}

function addBaselineSignal(
  signals: AutonomyRiskSignal[],
  paths: readonly string[],
  source: string,
): void {
  if (paths.length === 0) {
    if (source === "preflight intent") {
      signals.push({
        rule_id: "risk.baseline.source",
        level: "R2",
        message: "write intent has no bounded expected path",
      });
    }
    return;
  }

  if (paths.every(isDocumentationPath)) {
    signals.push({
      rule_id: "risk.baseline.docs",
      level: "R1",
      message: `${source} changes documentation only`,
    });
    return;
  }
  if (paths.every(isTestPath)) {
    signals.push({
      rule_id: "risk.baseline.tests",
      level: "R1",
      message: `${source} changes tests only`,
    });
    return;
  }
  signals.push({
    rule_id: "risk.baseline.source",
    level: "R2",
    message: `${source} changes source or configuration`,
  });
}

function addPathSignals(
  signals: AutonomyRiskSignal[],
  paths: readonly string[],
): void {
  for (const path of paths) {
    for (const rule of R3_PATH_RULES) {
      if (!rule.pattern.test(path)) continue;
      signals.push({
        rule_id: rule.ruleId,
        level: "R3",
        message: rule.message,
        path,
      });
    }
  }
}

function addProtectedPolicySignals(
  signals: AutonomyRiskSignal[],
  paths: readonly string[],
  protectedPaths: readonly string[],
): void {
  const normalizedProtected = new Set(
    protectedPaths.map(normalizeRepositoryPath),
  );
  for (const path of paths) {
    if (!normalizedProtected.has(normalizeRepositoryPath(path))) continue;
    signals.push({
      rule_id: "risk.policy-referenced-script",
      level: "R3",
      message: "script referenced by effective base policy changed",
      path,
    });
  }
}

function addOsstackKernelSignals(
  signals: AutonomyRiskSignal[],
  paths: readonly string[],
  repoId: string,
): void {
  if (!/(?:^|\/)osstack$/i.test(repoId)) return;
  for (const path of paths) {
    if (!OSSTACK_TRUST_KERNEL_PATH.test(path)) continue;
    signals.push({
      rule_id: "risk.osstack-trust-kernel",
      level: "R3",
      message: "osstack risk, approval, or state-machine trust kernel changed",
      path,
    });
  }
}

function addTextSignals(
  signals: AutonomyRiskSignal[],
  texts: readonly string[],
  source: string,
): void {
  for (const text of texts) {
    let matchedR4 = false;
    if (isRecursiveRootDelete(text)) {
      matchedR4 = true;
      signals.push({
        rule_id: "risk.r4.recursive-root",
        level: "R4",
        message: "explicit recursive root deletion requested",
        command: text,
      });
    }
    for (const rule of R4_TEXT_RULES) {
      if (!rule.pattern.test(text)) continue;
      matchedR4 = true;
      signals.push({
        rule_id: rule.ruleId,
        level: "R4",
        message: rule.message,
        command: text,
      });
    }
    for (const rule of R3_TEXT_RULES) {
      if (!rule.pattern.test(text)) continue;
      if (
        matchedR4 &&
        (rule.ruleId === "risk.command.destructive" ||
          rule.ruleId === "risk.diff.ddl")
      ) {
        continue;
      }
      signals.push({
        rule_id: rule.ruleId,
        level: "R3",
        message: `${rule.message} (${source})`,
        command: text,
      });
    }
  }
}

function decision(
  phase: "preflight" | "post_diff",
  rawSignals: AutonomyRiskSignal[],
  context: RiskClassificationContext,
  digestInput: unknown,
): AutonomyRiskDecision {
  const signals = dedupeAndSortSignals(rawSignals);
  const level = signals.reduce<AutonomyRiskLevel>(
    (current, signal) => maxRisk(current, signal.level),
    "R0",
  );
  return AutonomyRiskDecision.parse({
    phase,
    level,
    signals,
    policy_sha256: context.policy_sha256,
    input_sha256: sha256Canonical(digestInput),
    decided_at: context.now(),
  });
}

function dedupeAndSortSignals(
  signals: readonly AutonomyRiskSignal[],
): AutonomyRiskSignal[] {
  const byKey = new Map<string, AutonomyRiskSignal>();
  for (const signal of signals) byKey.set(signalKey(signal), signal);
  return [...byKey.values()].sort((left, right) => {
    const leftKey = signalKey(left);
    const rightKey = signalKey(right);
    return leftKey < rightKey ? -1 : leftKey > rightKey ? 1 : 0;
  });
}

function signalKey(signal: AutonomyRiskSignal): string {
  return [
    signal.rule_id,
    signal.path ?? "",
    signal.command ?? "",
    signal.level,
    signal.message,
  ].join("\u0000");
}

function addedDiffLines(diff: string): string[] {
  return diff
    .split(/\r?\n/)
    .filter((line) => line.startsWith("+") && !line.startsWith("+++"))
    .map((line) => line.slice(1));
}

function isDocumentationPath(path: string): boolean {
  return (
    /(?:^|\/)(?:docs?|documentation)(?:\/|$)/i.test(path) ||
    /\.(?:md|mdx|rst|txt)$/i.test(path)
  );
}

function isTestPath(path: string): boolean {
  return (
    /(?:^|\/)(?:tests?|__tests__)(?:\/|$)/i.test(path) ||
    /\.(?:test|spec)\.[^.]+$/i.test(path)
  );
}

function canonicalStrings(values: readonly string[]): string[] {
  return [...new Set(values)].sort();
}

function normalizeRepositoryPath(value: string): string {
  return value.replaceAll("\\", "/").replace(/^(?:\.\/)+/, "").replace(/\/+/g, "/");
}

function isRecursiveRootDelete(text: string): boolean {
  const tokens = text.trim().split(/\s+/);
  if (tokens[0] === "sudo") tokens.shift();
  if (tokens.shift() !== "rm") return false;

  let recursive = false;
  let forced = false;
  let root = false;
  for (const token of tokens) {
    if (token === "--recursive") {
      recursive = true;
      continue;
    }
    if (token === "--force") {
      forced = true;
      continue;
    }
    if (/^-[^-]/.test(token)) {
      const flags = token.slice(1);
      if (flags.includes("r") || flags.includes("R")) recursive = true;
      if (flags.includes("f")) forced = true;
      continue;
    }
    if (token === "--") continue;
    const unquoted = token.replace(/^(["'])(.*)\1$/, "$2");
    if (unquoted === "/") root = true;
    break;
  }
  return recursive && forced && root;
}

function sha256Canonical(value: unknown): string {
  return createHash("sha256").update(canonicalJson(value)).digest("hex");
}

function canonicalJson(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) {
    return `[${value.map((entry) => canonicalJson(entry)).join(",")}]`;
  }
  const record = value as Record<string, unknown>;
  const entries = Object.keys(record)
    .sort()
    .map(
      (key) => `${JSON.stringify(key)}:${canonicalJson(record[key])}`,
    );
  return `{${entries.join(",")}}`;
}

  autonomy_run_id: string;
  from_state: AutonomyRunStateType;
  to_state: AutonomyRunStateType;
  risk_level?: AutonomyRiskLevelType;
  risk_phase?: AutonomyRiskPhaseType;

function expectedClassifiedDestination(
  phase: AutonomyRiskPhaseType,
  level: AutonomyRiskLevelType,
): AutonomyRunStateType {
  if (level === "R4") return "blocked";
  if (level === "R3") return "awaiting_approval";
  if (phase === "preflight") {
    return level === "R0" ? "completed" : "preparing";
  }
  return "candidate_proof";
}

function assertClassificationGate(input: AutonomyTransitionInput): void {
  const classificationPhase =
    input.from_state === "classifying" ? "preflight" : "post_diff";
  if (input.risk_level === undefined || input.risk_phase === undefined) {
    throw new Error(
      `${input.from_state} transition requires an authoritative risk decision`,
    );
  }
  const level = AutonomyRiskLevel.parse(input.risk_level);
  const phase = AutonomyRiskPhase.parse(input.risk_phase);
  if (phase !== classificationPhase) {
    throw new Error(
      `${input.from_state} requires ${classificationPhase} risk evidence`,
    );
  }
  const expected = expectedClassifiedDestination(phase, level);
  if (input.to_state !== expected) {
    if (level === "R3") {
      throw new Error(
        `R3 must enter awaiting_approval before ${
          phase === "preflight" ? "preparing" : "candidate_proof"
        }`,
      );
    }
    if (level === "R4") {
      throw new Error("R4 can only transition to blocked and cannot be approved");
    }
    throw new Error(
      `${level} ${phase} classification must transition to ${expected}`,
    );
  }
}

function assertApprovalGate(input: AutonomyTransitionInput): void {
  const targetPhase =
    input.to_state === "preparing"
      ? "preflight"
      : input.to_state === "candidate_proof"
        ? "post_diff"
        : undefined;
  if (targetPhase === undefined) return;
  if (!isVerifiedScopedApproval(input.verified_approval)) {
    throw new Error(
      `${input.to_state} requires a cryptographically verified approval`,
    );
  }
  const approval = input.verified_approval;
  if (input.pending_approval_phase === undefined) {
    throw new Error(
      "approval resume requires the persisted pending_approval_phase",
    );
  }
  const pendingPhase = AutonomyRiskPhase.parse(input.pending_approval_phase);
  if (pendingPhase !== targetPhase) {
    throw new Error(
      `${pendingPhase} awaiting approval cannot transition to ${input.to_state}`,
    );
  }
  if (approval.request.autonomy_run_id !== input.autonomy_run_id) {
    throw new Error("verified approval belongs to a different autonomy run");
  }
  if (approval.request.phase !== targetPhase) {
    throw new Error(
      `${approval.request.phase} approval must transition to ${
        approval.request.phase === "post_diff" ? "candidate_proof" : "preparing"
      }`,
    );
  }
  if (approval.decision.payload.decision !== "approve") {
    throw new Error("rejected approval cannot authorize execution");
  }
}

function assertBlockedDeliveryResume(input: AutonomyTransitionInput): void {
  if (
    input.blocked_delivery_resume_state === undefined ||
    input.blocked_delivery_resume_state !== input.to_state
  ) {
    throw new Error(
      "blocked_delivery resume target must match the persisted delivery resume state",
    );
  }
}

export function assertAutonomyTransition(input: AutonomyTransitionInput): void {
  if (input.autonomy_run_id.length === 0) {
    throw new Error("autonomy_run_id must not be empty");
  }
  const from = AutonomyRunState.parse(input.from_state);
  const to = AutonomyRunState.parse(input.to_state);
  if (!LEGAL_TRANSITIONS[from].includes(to)) {
    throw new Error(`illegal transition from ${from} to ${to}`);
  }

  if (
    (from === "classifying" || from === "post_classifying") &&
    (!["escalated", "cancelled"].includes(to) ||
      input.risk_level !== undefined ||
      input.risk_phase !== undefined)
  ) {
    assertClassificationGate(input);
  }
  if (from === "awaiting_approval") {
    assertApprovalGate(input);
  }
  if (from === "blocked_delivery") {
    assertBlockedDeliveryResume(input);
  }
}

export function canAutonomyTransition(
  input: AutonomyTransitionInput,
): boolean {
  try {
    assertAutonomyTransition(input);
    return true;
  } catch {
    return false;
  }
