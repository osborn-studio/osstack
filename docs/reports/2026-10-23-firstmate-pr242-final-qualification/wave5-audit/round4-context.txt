import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { closeSync, constants, fstatSync, lstatSync, openSync, readSync, readlinkSync, realpathSync, type Stats } from 'node:fs';
import { join, resolve, sep } from 'node:path';

const MAX_BYTES = 256 * 1024 * 1024;
const MAX_PATHS = 100000;
const invalid = (reason: string): never => { throw Error(`GRAPH_CONTRACT_RAW_${reason}`); };
function records(bytes: Buffer): string[] {
  const text = bytes.toString('utf8');
  if (!Buffer.from(text).equals(bytes) || (text && !text.endsWith('\0'))) invalid('PATH_ENCODING');
  return text.split('\0').filter(Boolean);
}
function safeParts(name: string): string[] {
  const parts = name.split('/');
  if (name.includes('\\') || parts.some(part => !part || part === '.' || part === '..' || part.toLowerCase() === '.git')) invalid('PATH');
  return parts;
}
function stat(path: string): Stats | undefined {
  try { return lstatSync(path); } catch (error) {
    if (['ENOENT', 'ENOTDIR'].includes((error as NodeJS.ErrnoException).code ?? '')) return undefined;
    throw error;
  }
}
function sameFile(a: Stats, b: Stats): boolean {
  return a.dev === b.dev && a.ino === b.ino && a.mode === b.mode && a.size === b.size && a.mtimeMs === b.mtimeMs && a.ctimeMs === b.ctimeMs;
}

/** Read-only raw comparison with the original immutable base. Git's stat cache,
 * attributes, filters, ignore rules and core.filemode never decide membership.
 * This is a pre-freeze gate, not a filesystem snapshot; immutable candidate
 * verification still runs after capture. Bounds match the artifact store. */
export function graphWorkspaceChangedPaths(workspace: string, baseRevision: string): string[] {
  if (!/^[a-f0-9]{40}$/.test(baseRevision)) invalid('BASE');
  const root = realpathSync(workspace);
  const git = (...args: string[]) => execFileSync('git', ['-c', 'core.hooksPath=/dev/null', '-c', 'core.fsmonitor=false', ...args], {
    cwd: root, timeout: 30000, maxBuffer: 16 * 1024 * 1024,
    env: { ...Object.fromEntries(Object.entries(process.env).filter(([name]) => !name.startsWith('GIT_'))),
      GIT_CONFIG_GLOBAL: '/dev/null', GIT_CONFIG_NOSYSTEM: '1', GIT_NO_REPLACE_OBJECTS: '1', GIT_TERMINAL_PROMPT: '0', GIT_OPTIONAL_LOCKS: '0', LC_ALL: 'C' },
  });
  if (git('rev-parse', '--verify', `${baseRevision}^{commit}`).toString().trim() !== baseRevision) invalid('BASE');
  const base = new Map<string, { mode: string; oid: string }>();
  for (const entry of records(git('ls-tree', '-r', '-z', '--full-tree', baseRevision))) {
    const match = /^(100644|100755|120000) blob ([a-f0-9]{40})\t([\s\S]+)$/.exec(entry);
    if (!match) invalid('BASE_ENTRY');
    const [, mode, oid, name] = match!; safeParts(name!);
    if (base.has(name!)) invalid('DUPLICATE_PATH');
    base.set(name!, { mode: mode!, oid: oid! });
    if (base.size > MAX_PATHS) invalid('PATH_LIMIT');
  }
  for (const entry of records(git('ls-files', '--stage', '-z'))) {
    if (!/^(100644|100755|120000) [a-f0-9]{40} 0\t/.test(entry)) invalid('INDEX_ENTRY');
  }
  // No --exclude-standard: ignored files are also candidate side effects.
  const names = new Set([...base.keys(), ...records(git('ls-files', '--cached', '--others', '-z'))]);
  if (names.size > MAX_PATHS) invalid('PATH_LIMIT');
  const changed: string[] = []; let totalBytes = 0;
  for (const name of [...names].sort()) {
    const parts = safeParts(name); let ancestor = root; let missingParent = false;
    for (const part of parts.slice(0, -1)) {
      ancestor = join(ancestor, part); const info = stat(ancestor);
      if (!info) { missingParent = true; break; }
      if (info.isSymbolicLink()) invalid('SYMLINK_ANCESTOR');
      if (!info.isDirectory()) {
        if (!info.isFile()) invalid('SPECIAL_FILE');
        missingParent = true; break;
      }
    }
    const path = resolve(root, name);
    if (!path.startsWith(root + sep)) invalid('PATH');
    const before = missingParent ? undefined : stat(path); const previous = base.get(name);
    if (!before || before.isDirectory()) {
      if (before?.isDirectory() && stat(join(path, '.git'))) invalid('NESTED_REPOSITORY');
      if (previous) changed.push(name);
      continue;
    }
    if (!before.isFile() && !before.isSymbolicLink()) invalid('SPECIAL_FILE');
    let mode: string; let oid: string;
    if (before.isSymbolicLink()) {
      const bytes = readlinkSync(path, { encoding: 'buffer' }); totalBytes += bytes.length;
      if (totalBytes > MAX_BYTES) invalid('BYTE_LIMIT');
      mode = '120000'; oid = createHash('sha1').update(`blob ${bytes.length}\0`).update(bytes).digest('hex');
      const after = stat(path);
      if (!after || !sameFile(before, after) || !bytes.equals(readlinkSync(path, { encoding: 'buffer' }))) invalid('CHANGED_DURING_READ');
    } else {
      if (before.size > MAX_BYTES - totalBytes) invalid('BYTE_LIMIT');
      // O_NONBLOCK avoids hanging if a concurrent writer replaces a file with a FIFO.
      const fd = openSync(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
      try {
        const opened = fstatSync(fd);
        if (!opened.isFile() || !sameFile(before, opened)) invalid('CHANGED_DURING_READ');
        const hash = createHash('sha1').update(`blob ${opened.size}\0`); const buffer = Buffer.alloc(64 * 1024);
        let bytes = 0;
        for (;;) {
          const count = readSync(fd, buffer, 0, buffer.length, null); if (!count) break;
          bytes += count; totalBytes += count;
          if (totalBytes > MAX_BYTES || bytes > opened.size) invalid('BYTE_LIMIT');
          hash.update(buffer.subarray(0, count));
        }
        const after = stat(path);
        if (bytes !== opened.size || !sameFile(opened, fstatSync(fd)) || !after || !sameFile(opened, after)) invalid('CHANGED_DURING_READ');
        oid = hash.digest('hex');
        // Same raw mode mapping as captureArtifactTree's writer capture.
        mode = (opened.mode & 0o111) !== 0 ? '100755' : '100644';
      } finally { closeSync(fd); }
    }
    if (!previous || previous.mode !== mode || previous.oid !== oid) changed.push(name);
  }
  return changed;
}

/** Host policy is already authorized by the registration. This function grants nothing. */
export function bindGraphDeliveryContract(requestValue: StartGraphRequest, registration: {
  repositoryPath: string; task: string; acceptanceRegistry: readonly AcceptanceCommandSpec[];
  deliveryContract: GraphDeliveryContractRegistration;
  executionContext?: { configDigest: string };
}): { request: StartGraphRequest; envelope: GraphContractHostEnvelope } {
  const request = StartGraphRequest.parse(requestValue);
  const { frozen, policy } = parseGraphContractRegistration(registration.deliveryContract);
  const contract = frozen.contract;
  if (registration.executionContext && request.scope.tenantId !== 'default') throw Error('GRAPH_CONTEXT_SCOPE_UNSUPPORTED');
  if (contract.mode !== 'implement' || contract.requested_completion !== 'local_verified'
    || contract.publication_scope.mode !== 'none' || contract.merge_authority.mode !== 'none'
    || contract.required_checks.length !== 1) throw Error('GRAPH_CONTRACT_UNSUPPORTED');
  const selected = registration.acceptanceRegistry.filter(item => item.ref === request.spec.input.acceptanceRef);
  if (selected.length !== 1) throw Error('GRAPH_CONTRACT_REGISTRY_MISMATCH');
  const acceptance = acceptanceSchema.parse(selected[0]);
  if (acceptance.passExitCodes.length !== 1 || acceptance.passExitCodes[0] !== 0 || acceptance.failExitCodes.includes(0)
    || deliveryCheckReference(acceptance.command) !== contract.required_checks[0]!.registry_ref) throw Error('GRAPH_CONTRACT_REGISTRY_MISMATCH');
  assertDeliveryContractAdmission(frozen, { canonicalRepo: registration.repositoryPath, baseHead: request.spec.input.baseRevision,
    taskId: request.spec.input.taskRef, goal: registration.task, policyDigest: policy.policyDigest,
    reviewPolicyRef: policy.reviewPolicyRef, acceptedCommands: [acceptance.command] });
  for (const name of Object.keys(policy.limits) as (keyof GraphContractHostPolicy['limits'])[]) {
    if (request.spec.limits[name] > policy.limits[name]) throw Error('GRAPH_CONTRACT_BUDGET_EXCEEDED');
  }
  const envelope = freezeWorkerData(envelopeSchema.parse({ schemaVersion: 1, sourceInputArtifactRef: request.inputArtifactRef,
    frozen, policy, acceptance, ...(registration.executionContext ? { contextDigest: registration.executionContext.configDigest } : {}) }));
  return { request: freezeWorkerData(StartGraphRequest.parse({ ...request, inputArtifactRef: graphContractInputRef(envelope) })), envelope };
}

/** Parses durable host data; provenance and current grant remain the caller's responsibility. */
export function verifyGraphContractEnvelope(value: unknown, request: StartGraphRequest,
  registration: { repositoryPath: string; task: string; acceptanceRegistry: readonly AcceptanceCommandSpec[]; deliveryContract: GraphDeliveryContractRegistration; executionContext?: { configDigest: string } }): GraphContractHostEnvelope {
  const envelope = envelopeSchema.parse(value);

}

function canonical(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  return `{${Object.keys(value).sort().map(key => `${JSON.stringify(key)}:${canonical((value as Record<string, unknown>)[key])}`).join(",")}}`;

// Full-history bundles are deliberately bounded before any whole-file read.
const MAX_BUNDLE_BYTES = 256 * 1024 * 1024;
async function bundleBytes(path: string): Promise<Buffer> {
  const info = await lstat(path);
  if (!info.isFile() || info.isSymbolicLink() || info.size > MAX_BUNDLE_BYTES) throw new Error("unsupported artifact bundle size or type (maximum 256 MiB)");
  const bytes = await readFile(path);
  if (bytes.length !== info.size || bytes.length > MAX_BUNDLE_BYTES) throw new Error("artifact bundle changed during read");
  return bytes;
}

/** On-disk immutable artifacts, no scheduler and no automatic GC. Store root and
 * Git executable must be host-owned, inaccessible to untrusted producers. */
export class GraphArtifactStore {
  constructor(private readonly root: string) {
    if (!isAbsolute(root)) throw new Error("artifact root must be absolute");
  }
  private async prepare(): Promise<string> {
    // Require an existing parent so all newly created directory entries are synced.
    await realpath(dirname(this.root));
    await mkdir(this.root, { recursive: true, mode: 0o700 });
    const root = await realpath(this.root);
    for (const child of ["artifacts", "ids", "scratch", "locks"]) await mkdir(join(root, child), { recursive: true, mode: 0o700 });
    await syncDirectory(root); await syncDirectory(dirname(root));

  async freeze(raw: GraphFreezeInput, rawAuthority: GraphFreezeAuthority): Promise<CandidateReady> {
    const input = snapshot(raw); validateInput(input);
    const authority = Object.freeze({ ...rawAuthority, ...(rawAuthority.expectedTrees === undefined ? {} : { expectedTrees: snapshot(rawAuthority.expectedTrees) }) });

      ![authority.expectedTrees.base, authority.expectedTrees.candidate].every(value => typeof value === "string" && /^[a-f0-9]{40}$/.test(value)))) throw new Error("invalid freeze tree constraint");
    if (!authority.lease || typeof authority.lease !== "object") throw new Error("missing canonical lease authority");
    const check = async (): Promise<void> => {
      for (const fn of [() => authority.verifyImplementationReceipt(input), () => authority.verifyProducersStopped(input), () => authority.verifyLease(input, authority.lease)]) {
        if (await fn() !== undefined) throw new Error("nonconforming freeze authority");
      }
    };
    await check();

    const id = sha(input.artifactId); const binding = join(root, "ids", `${id}.json`);
    const lock = join(root, "locks", id);

      if (authority.expectedTrees && await gitText(input.worktreePath, ["rev-parse", `${input.baseRevision}^{tree}`]) !== authority.expectedTrees.base) throw new Error("invocation base tree mismatch");
      await check();

      const objectRepo = join(scratch, "objects.git");
      await initArtifactRepo(objectRepo); await linkSourceObjects(objectRepo, input.commonDir);
      if (await this.source(input) !== sourceIdentity) throw new Error("source identity changed before capture");

      if (authority.expectedTrees && tree !== authority.expectedTrees.candidate) throw new Error("applied tree mismatch");
