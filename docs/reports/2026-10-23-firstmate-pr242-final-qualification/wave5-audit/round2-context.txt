export const GRAPH_EXECUTION_HELP = `Usage: osstack graph <command> [--json]
  validate --spec FILE               Offline fixed-template and budget validation
  run --request FILE                Submit a StartGraphRequest to the local daemon
  status --run ID --scope FILE       Read authoritative run state
  result --run ID --scope FILE       Read artifact-only delivery and evidence references
  contract --run ID --scope FILE     Read a verified host contract completion receipt
  cancel --request FILE             Persist cancellation intent; confirmation is separate
  reconcile --request FILE          Recheck cancellation using trusted host no-child evidence
  resume --request FILE             Resume an exact interrupt with current host authorization
Request/scope files are strict v1 JSON (max 64 KiB). No implicit scope or grant.
run/cancel/reconcile/resume retain the request's idempotencyKey across transport retries.
Exit: 0 acknowledged; 1 input/RPC rejection; 2 delivery_unknown (retry the same request).
The server must have explicit trusted graph-host configuration. graph explain remains available.`;

function readJson(path: string): unknown {
  const fd = openSync(path, constants.O_RDONLY | constants.O_NONBLOCK);
  try {
    const st = fstatSync(fd);
    if (!st.isFile() || st.size > GRAPH_JSON_LIMITS.maxBytes) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    const bytes = Buffer.alloc(GRAPH_JSON_LIMITS.maxBytes + 1);
    let size = 0;
    while (size < bytes.length) {
      const count = readSync(fd, bytes, size, bytes.length - size, null);
      if (!count) break;
      size += count;
    }
    if (size > GRAPH_JSON_LIMITS.maxBytes) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    return JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(bytes.subarray(0, size))) as unknown;
  } finally { closeSync(fd); }
}
function options(args: string[], allowed: string[]): Map<string, string> {
  const found = new Map<string, string>();
  for (let index = 0; index < args.length; index++) {
    const key = args[index]!;
    if (found.has(key) || ![...allowed, '--json'].includes(key)) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    const value = key === '--json' ? 'true' : args[++index];
    if (!value || value.startsWith('--')) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    found.set(key, value);
  }
  for (const key of allowed) if (!found.has(key)) throw new GraphContractError('INPUT_CONTRACT_INVALID');
  return found;
}
function failure(code: unknown, message: string): GraphError {
  const nextAction = code === 'REVISION_CONFLICT' ? 'refresh' : code === 'UNAUTHENTICATED' || code === 'POLICY_DENIED' || code === 'GRANT_EXPIRED' ? 'reauthorize'
    : code === 'ACTION_OUTCOME_UNKNOWN' || code === 'CANCEL_UNCONFIRMED' ? 'reconcile'
    : code === 'EXECUTOR_UNSUPPORTED' || code === 'GRAPH_ADMISSION_DISABLED' ? 'stop' : 'fix_input';
  const parsed = GraphError.safeParse({ schemaVersion: 1, code, message: message.slice(0, 4096) || 'Graph request rejected', retryable: false, nextAction });
  return parsed.success ? parsed.data : { schemaVersion: 1, code: 'INTERNAL_ERROR', message: 'Graph request failed', retryable: false, nextAction: 'stop' };
}
export async function handleGraphExecutionCommand(args: string[], deps: GraphCommandDependencies = {}): Promise<number> {
  const stdout = deps.stdout ?? console.log; const stderr = deps.stderr ?? console.error;
  const json = args.includes('--json');
  const emit = (value: unknown): void => { stdout(JSON.stringify(value, null, json ? 2 : undefined)); };
  if (args.length === 2 && args[1] === '--help') { stdout(GRAPH_EXECUTION_HELP); return 0; }
  let sentMutation: GraphRequest | undefined;
  const unknownDelivery = (mutation: GraphRequest): number => {
    emit(GraphDeliveryUnknown.parse({ schemaVersion: 1, kind: 'delivery_unknown', scope: mutation.scope,
      idempotencyKey: mutation.idempotencyKey, inputDigest: computeGraphInputDigest(mutation),
      ...(mutation.kind === 'start' ? {} : { graphRunId: mutation.graphRunId }), nextAction: 'retry_same_request' }));
    return 2;
  };
  try {
    const command = args[0];
    if (command !== "validate") assertGraphPlatform();
    if (command === 'validate') {
      const opts = options(args.slice(1), ['--spec']);
      const { compileGraphSpec } = await import('@osstack/graph/spec-compiler');
      let compiled: ReturnType<typeof compileGraphSpec>;
      try { compiled = compileGraphSpec(readJson(opts.get('--spec')!)); }
      catch { throw new GraphContractError('SPEC_INVALID'); }
      emit({ schemaVersion: 1, valid: true, specHash: compiled.specHash, template: compiled.spec.template,
        compilerVersion: compiled.compilerVersion, compiledProgramDigest: compiled.compiledProgramDigest,
        nodes: compiled.plan.nodes.map(({ id, operation, inputs }) => ({ id, operation, inputs })), budget: compiled.budget });
      return 0;
    }
    let params: unknown; let method: string; let mutation: GraphRequest | undefined;
    if (command === 'status' || command === 'result' || command === 'contract') {
      const opts = options(args.slice(1), ['--run', '--scope']);
      params = GraphStatusRequest.parse({ schemaVersion: 1, scope: GraphScope.parse(readJson(opts.get('--scope')!)), graphRunId: opts.get('--run') });
      method = `graph.${command}`;
    } else if (command === 'run' || command === 'cancel' || command === 'reconcile' || command === 'resume') {
      const opts = options(args.slice(1), ['--request']);
      mutation = GraphRequest.parse(readJson(opts.get('--request')!));
      if (mutation.kind !== (command === 'run' ? 'start' : command === 'reconcile' ? 'cancel' : command)) throw new GraphContractError('INPUT_CONTRACT_INVALID');
      params = mutation; method = command === 'reconcile' ? 'graph.reconcile' : `graph.${mutation.kind}`;
    } else throw new GraphContractError('INPUT_CONTRACT_INVALID');
    if (!deps.rpc) throw new GraphContractError('UNAUTHENTICATED');
    let response: unknown;
    sentMutation = mutation;
    try { response = await deps.rpc(method, params); }
    catch {
      if (!mutation) throw new GraphContractError('INTERNAL_ERROR');
      return unknownDelivery(mutation);
    }
    if (!response || typeof response !== 'object') throw new GraphContractError('OUTPUT_CONTRACT_INVALID');
    const envelope = response as { ok?: unknown; result?: unknown; error?: { code?: unknown; message?: unknown } };
    if (envelope.ok === false) {
      const error = failure(envelope.error?.code, typeof envelope.error?.message === 'string' ? envelope.error.message : 'Graph request rejected');
      if (json) emit({ schemaVersion: 1, kind: 'graph_error', error }); else stderr(`osstack graph: ${error.code}: ${error.message}`);
      return ['ACTION_OUTCOME_UNKNOWN', 'CANCEL_UNCONFIRMED'].includes(error.code) ? 2 : 1;
    }
    if (envelope.ok !== true) throw new GraphContractError('OUTPUT_CONTRACT_INVALID');
    if (command === 'contract') {
      const { GraphContractCompletionReceipt } = await import('./graph-contract-receipt.js');
      const receipt = GraphContractCompletionReceipt.safeParse(envelope.result);
      if (!receipt.success) throw new GraphContractError('OUTPUT_CONTRACT_INVALID');
      const requested = GraphStatusRequest.parse(params);
      if (JSON.stringify(receipt.data.scope) !== JSON.stringify(requested.scope) || receipt.data.graphRunId !== requested.graphRunId) throw new GraphContractError('RESPONSE_MISMATCH');
      emit(receipt.data); return 0;
    }
    const parsedStatus = GraphDelegationStatus.safeParse(envelope.result);
    if (!parsedStatus.success) throw new GraphContractError('OUTPUT_CONTRACT_INVALID');
    const status = parsedStatus.data;
    const requested = GraphStatusRequest.safeParse(params);
    if (JSON.stringify(status.scope) !== JSON.stringify((mutation ?? requested.data)!.scope)
      || (mutation?.kind !== 'start' && status.graphRunId !== (mutation ?? requested.data)!.graphRunId)
      || (mutation?.kind === 'start' && (status.admissionBinding?.requestInputDigest ?? status.inputDigest) !== computeGraphInputDigest(mutation))) throw new GraphContractError('RESPONSE_MISMATCH');
    emit(status); return 0;
  } catch (error) {
    // A malformed or mismatched acknowledgement cannot prove non-execution.
    if (sentMutation) return unknownDelivery(sentMutation);
    const code = error instanceof GraphContractError ? error.code : 'INPUT_CONTRACT_INVALID';
    const report = failure(code, error instanceof GraphContractError ? error.message : 'Invalid graph input or response');
    if (json) emit({ schemaVersion: 1, kind: 'graph_error', error: report }); else stderr(`osstack graph: ${report.code}: ${report.message}`);
    return 1;
  }
}

const same = (a: unknown, b: unknown) => workerCanonical(a) === workerCanonical(b);
function synchronous(value: unknown): void {
  if (value !== undefined) { void Promise.resolve(value).catch(() => {}); throw new GraphContractError('POLICY_DENIED'); }
}

/** Stable public errors never disclose host paths, commands, or raw adapter text. */
export function graphRpcError(error: unknown): { code: string; message: string } {
  const raw = error instanceof Error ? error.message : '';
  const supplied = error && typeof error === 'object' && 'code' in error ? String(error.code) : raw;
  const aliases: Record<string, string> = { ENOENT: 'UNAUTHENTICATED', EACCES: 'UNAUTHENTICATED', ELOOP: 'UNAUTHENTICATED',
    RUN_NOT_FOUND: 'RUN_MISMATCH', OWNER_MISMATCH: 'STALE_GENERATION', LEASE_EXPIRED: 'GRANT_EXPIRED',
    LIVE_QUIESCENCE_REQUIRED: 'ACTION_OUTCOME_UNKNOWN', PHYSICAL_WORKER_STOP_REQUIRED: 'ACTION_OUTCOME_UNKNOWN',
    PROGRAM_IDENTITY_MISMATCH: 'CHECKPOINT_INCOMPATIBLE', TERMINAL_RUN: 'REVISION_CONFLICT', CANCEL_PENDING: 'CANCEL_UNCONFIRMED' };
  const parsed = GraphErrorCode.safeParse(aliases[supplied] ?? (supplied.startsWith('GRAPH_CONTRACT_') || supplied.startsWith('GRAPH_CONTEXT_') || supplied.startsWith('delivery_') ? 'POLICY_DENIED' : supplied));
  const code = error instanceof z.ZodError ? 'INPUT_CONTRACT_INVALID' : parsed.success ? parsed.data : 'INTERNAL_ERROR';
  return { code, message: code };
}

/** Public durable admission. Process maps hold actual supervisors, never public state. */
export class GraphRpcService {
  private readonly runs: GraphRunStore;
  private readonly registrations = new Map<string, GraphHostRegistration>();
  private readonly live = new Map<string, { driver?: GraphDelivery; done: Promise<void> }>();
  private readonly pendingRequests = new Set<Promise<unknown>>();
  private readonly owner = randomUUID();
  private closed = false;
  private closing: Promise<void> | undefined;
  private readonly options: GraphRpcOptions;
  constructor(options: GraphRpcOptions) {
    assertGraphPlatform();
    this.options = Object.freeze({ ...options, resolveCaller: options.resolveCaller, acquire: options.acquire });
    const snapshotAdapter = (a: AgentAdapter): AgentAdapter => {
      const capabilities = structuredClone(a.capabilities());
      return Object.freeze({ id: a.id, display_name: a.display_name, cost_tier: a.cost_tier,
        capabilities: () => structuredClone(capabilities), isAvailable: a.isAvailable.bind(a), invoke: a.invoke.bind(a),
        ...(a.confirmStopped ? { confirmStopped: a.confirmStopped.bind(a) } : {}) });
    };
    this.runs = new GraphRunStore(options.db);
    for (const value of options.registrations) {
      if (this.registrations.has(value.id)) throw Error('DUPLICATE_GRAPH_REGISTRATION');
      if (value.delivery.executionContext && !value.delivery.deliveryContract) throw Error('GRAPH_CONTEXT_CONTRACT_REQUIRED');
      this.registrations.set(value.id, { ...value, scope: GraphScope.parse(value.scope), delivery: { ...value.delivery,
        ...(value.delivery.deliveryContract ? { deliveryContract: parseGraphContractRegistration(value.delivery.deliveryContract) } : {}),
        ...(value.delivery.executionContext ? { executionContext: snapshotGraphExecutionContext(value.delivery.executionContext) } : {}),
        writer: { ...value.delivery.writer, adapter: snapshotAdapter(value.delivery.writer.adapter) }, reviewer: { ...value.delivery.reviewer, adapter: snapshotAdapter(value.delivery.reviewer.adapter) }, acceptanceRegistry: structuredClone(value.delivery.acceptanceRegistry) } });
    }
  }
  private caller(connection: object): GraphHostRegistration {
    if (this.closed) throw Error('GRAPH_SERVICE_CLOSED');
    const id = this.options.resolveCaller(connection);
    const registration = id === null ? undefined : this.registrations.get(id);
    if (!registration) throw new GraphContractError('UNAUTHENTICATED');
    return registration;
  }
  private row(id: string): PublicRow {
    const row = this.options.db.prepare('SELECT * FROM graph_public_delivery WHERE graph_run_id=?').get(id) as PublicRow | undefined;
    if (!row) throw new GraphContractError('RUN_MISMATCH');
    return row;
  }
  private bound(registration: GraphHostRegistration, scope: GraphScope, id: string): GraphRun {
    if (!same(scope, registration.scope)) throw new GraphContractError('SCOPE_MISMATCH');
    const run = this.runs.get(scope, id);
    if (!run || run.source.product !== registration.product) throw new GraphContractError('RUN_MISMATCH');
    const descriptor = JSON.parse(this.row(id).descriptor_json) as Descriptor;
    if (descriptor.registrationId !== registration.id) throw new GraphContractError('RUN_MISMATCH');
    if (descriptor.contractEnvelope && (graphContractInputRef(descriptor.contractEnvelope) !== descriptor.request.inputArtifactRef
      || computeGraphInputDigest(descriptor.request) !== run.identity.inputDigest)) throw new GraphContractError('CHECKPOINT_INCOMPATIBLE');
    return run;
  }
  private status(run: GraphRun, registration: GraphHostRegistration) {
    const row = this.row(run.graphRunId);
    const descriptor = JSON.parse(row.descriptor_json) as Descriptor;
    const contract = descriptor.contractEnvelope;
    if (contract && computeGraphInputDigest(descriptor.request) !== run.identity.inputDigest) throw new GraphContractError('CHECKPOINT_INCOMPATIBLE');
    const result = row.result_json ? JSON.parse(row.result_json) as StoredResult : { evidenceRefs: [] };
    const interrupt = row.interrupt_json ? JSON.parse(row.interrupt_json) as GraphDeliveryInterrupt : undefined;
    const { actions, budget } = readGraphActionState(this.options.db, run.graphRunId);
    const uncertain = ['cancel_pending', 'reconciliation_required'].includes(run.status);
    let needsAuthorization = false;
    try { synchronous(registration.authorize('execute')); } catch { needsAuthorization = true; }
    return GraphDelegationStatus.parse({ schemaVersion: 1 as const, scope: run.scope, graphRunId: run.graphRunId, source: run.source, status: run.status,
      revision: run.revision, generation: run.generation, inputDigest: run.identity.inputDigest, specHash: run.identity.specHash,
      ...(contract ? { admissionBinding: { requestInputDigest: computeGraphInputDigest({ ...descriptor.request, inputArtifactRef: contract.sourceInputArtifactRef }),
        hostInputDigest: run.identity.inputDigest, contractDigest: contract.frozen.digest, policyDigest: contract.policy.policyDigest } } : {}),
      createdAt: run.createdAt, updatedAt: run.updatedAt, publication: 'artifact_only' as const,
      phase: actions.find(a => a.state === 'running')?.effect.nodeId ?? run.status,
      budget: { limits: run.spec!.limits, ...budget },
      delivery: uncertain ? 'delivery_unknown' as const : run.status === 'ready' ? 'available' as const : 'pending' as const,
      ...result,
      ...(needsAuthorization ? { authorization: { kind: 'needs_authorization' as const, scope: run.scope, requestRef: `graph-grant:${run.graphRunId}` } } : {}),
      ...(interrupt && run.status === 'waiting_input' ? { interrupt: { interruptId: interrupt.interruptId, inputDigest: interrupt.inputDigest, kind: interrupt.kind } } : {}),
      ...(run.status === 'failed' ? { error: { schemaVersion: 1 as const, code: 'INTERNAL_ERROR' as const, message: 'Graph delivery stopped before verified publication', retryable: false as const, nextAction: 'stop' as const } } : {}),
      ...(uncertain ? { error: { schemaVersion: 1 as const, code: run.status === 'cancel_pending' ? 'CANCEL_UNCONFIRMED' as const : 'ACTION_OUTCOME_UNKNOWN' as const,
        message: 'Physical execution requires reconciliation', retryable: false as const, nextAction: 'reconcile' as const } } : {}),
    });
  }
  private replay(request: Exclude<GraphRequest, StartGraphRequest>): boolean {
    const row = this.options.db.prepare('SELECT graph_run_id,input_digest FROM graph_requests WHERE scope_json=? AND idempotency_key=?')
      .get(workerCanonical(request.scope), request.idempotencyKey) as { graph_run_id: string; input_digest: string } | undefined;
    if (!row) return false;
    if (row.graph_run_id !== request.graphRunId || row.input_digest !== computeGraphInputDigest(request)) throw new GraphContractError('IDEMPOTENCY_CONFLICT');
    return true;
  }
  async handle(connection: object, method: string, params: unknown): Promise<unknown> {
    const pending = this.handleRequest(connection, method, params);
    this.pendingRequests.add(pending);
    try { return await pending; }
    catch (error) { throw new GraphContractError(GraphErrorCode.parse(graphRpcError(error).code)); }
    finally { this.pendingRequests.delete(pending); }
  }
  private async handleRequest(connection: object, method: string, params: unknown): Promise<unknown> {
    const registration = this.caller(connection);
    if (method === 'graph.contract') {
      synchronous(registration.authorize('read'));
      const request = GraphStatusRequest.parse(params); const run = this.bound(registration, request.scope, request.graphRunId);
      const row = this.row(run.graphRunId); const descriptor = JSON.parse(row.descriptor_json) as Descriptor;
      const result = row.result_json ? JSON.parse(row.result_json) as StoredResult : undefined;
      const refs = result?.evidenceRefs.filter(ref => ref.startsWith('graph-contract:')) ?? [];
      if (!descriptor.contractEnvelope || !result?.candidate || !result.resultRef || refs.length !== 1) throw Error('GRAPH_CONTRACT_RECEIPT_UNPUBLISHED');
      const receipt = await loadGraphContractCompletion(descriptor.root, refs[0]!, { run, envelope: descriptor.contractEnvelope,
        joinRef: result.resultRef, candidateRevision: result.candidate.candidateRevision, artifactDigest: result.candidate.artifactDigest });
      synchronous(registration.authorize('read'));
      if (this.closed || workerCanonical(this.bound(registration, request.scope, request.graphRunId)) !== workerCanonical(run)) throw Error('GRAPH_CONTRACT_SNAPSHOT_CHANGED');
      return receipt;
    }
    if (method === 'graph.status' || method === 'graph.result') {
      synchronous(registration.authorize('read'));
      const request = GraphStatusRequest.parse(params);
      return this.status(this.bound(registration, request.scope, request.graphRunId), registration);
    }
    if (method === 'graph.validate') return compileGraphSpec(GraphSpec.parse(params));
    const request = GraphRequest.parse(params);
    if (!same(request.scope, registration.scope)) throw new GraphContractError('SCOPE_MISMATCH');
    if ((method === 'graph.start' || method === 'graph.submit') && request.kind === 'start') return this.submit(registration, request);
    const reconcile = method === 'graph.reconcile';
    if ((!['graph.cancel', 'graph.reconcile'].includes(method) || request.kind !== 'cancel') && (method !== 'graph.resume' || request.kind !== 'resume')) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    synchronous(registration.authorize(request.kind === 'cancel' ? 'cancel' : 'resume'));
    const run = this.bound(registration, request.scope, request.graphRunId);
    const replay = this.replay(request);
    // Only the unchanged cancellation intent can be retried as reconciliation.
    // Later revisions/generations are a lookup, never authority to stop a new owner.
    if (replay && !(reconcile && run.status === 'cancel_pending' && run.revision === request.expectedRevision + 1)) return this.status(run, registration);
    if (!replay && run.revision !== request.expectedRevision) throw new GraphContractError('REVISION_CONFLICT');
    if (request.kind === 'cancel') {
      const pending = replay ? run : this.runs.projectRequest(request, { ...fence(run), status: 'cancel_pending' });
      const entry = this.live.get(run.graphRunId);
      if (entry?.driver) void entry.driver.cancel().catch(() => {});
      else if (run.generation === 0) this.runs.project({ ...fence(pending), status: 'cancelled' });
      if (reconcile && run.generation > 0) {
        const receiptId = `public-cancel:${createHash('sha256').update(workerCanonical({ scope: request.scope,
          idempotencyKey: request.idempotencyKey, digest: computeGraphInputDigest(request) })).digest('hex')}`;
        let authorizationFailed = false;
        try {
          await cancelUnstartedGraphWorker(this.options.db, new GraphReconciler(this.options.db), { fence: fence(pending), receiptId },
            () => {
              try { synchronous(registration.authorize('cancel')); }
              catch (error) { authorizationFailed = true; throw error; }
            });
        } catch (error) {
          if (authorizationFailed) throw error;
          synchronous(registration.authorize('cancel'));
          // A finite bounded writer may have durable stopped/output evidence even
          // when the ledger ACK was lost. Recovery never dispatches it again.
          try {
            const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
            await cancelJournaledBoundedWriter(this.options.db, descriptor.root, registration.delivery.repositoryPath,
              { fence: fence(pending), receiptId }, () => {
                try { synchronous(registration.authorize('cancel')); }
                catch (failure) { authorizationFailed = true; throw failure; }
              });
          } catch (failure) {
            if (authorizationFailed) throw failure;
            try {
              const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
              await cancelUnstartedGraphPreparation(this.options.db, descriptor.root, registration.delivery.repositoryPath,
                { fence: fence(pending), receiptId }, () => {
                  try { synchronous(registration.authorize('cancel')); }
                  catch (denied) { authorizationFailed = true; throw denied; }
                });
            } catch (denied) {
              if (authorizationFailed) throw denied;
              // Operator diagnostics remain local. Never print raw filesystem,
              // adapter or SQLite error text into either RPC or daemon logs.
              const reasons = new Set(['SCOPE', 'PROGRAM', 'PREDECESSORS', 'NOT_STARTED', 'CANDIDATE',
                'NOT_CANCELLING', 'CHANGED', 'ADMISSION', 'WORKSPACE', 'LOCK_PRESENT', 'DECISION']);
              const reason = denied instanceof Error ? denied.message.replace(/^PREPARATION_RECOVERY_/, '') : '';
              console.error(JSON.stringify({ event: 'graph_recovery_blocked', graphRunId: run.graphRunId,
                stage: 'unstarted_preparation', code: reasons.has(reason) ? `PREPARATION_RECOVERY_${reason}` : graphRpcError(denied).code }));
            }
          }
          // Intent is already durable. Missing/conflicting evidence cannot undo
          // its acknowledgement; return current state without settling or releasing.
        }
        // Successful G15 commit rechecked the grant in its final transaction.
        // Later expiry cannot turn that completed operation into a rejected request.
      }
      return this.status(this.runs.get(run.scope, run.graphRunId)!, registration);
    }
    const row = this.row(run.graphRunId);
    const proof = row.interrupt_json ? JSON.parse(row.interrupt_json) as GraphDeliveryInterrupt : undefined;
    if (run.status !== 'waiting_input' || !proof || proof.interruptId !== request.interruptId) throw new GraphContractError('REVISION_CONFLICT');
    if (request.input.kind !== 'data' || !same(request.input.value, { inputDigest: run.identity.inputDigest }) || proof.inputDigest !== run.identity.inputDigest) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    const driver = this.live.get(run.graphRunId)?.driver;
    if (!driver) throw new GraphContractError('ACTION_OUTCOME_UNKNOWN');
    if (run.leaseExpiresAt === null || run.leaseExpiresAt <= Date.now() || Date.now() >= run.createdAt + run.spec!.limits.deadlineSeconds * 1000) throw new GraphContractError('GRANT_EXPIRED');
    driver.verifyResume(proof);
    this.runs.projectRequest(request, { ...fence(run), status: 'running' });
    driver.resumeQuality(proof);
    return this.status(this.runs.get(run.scope, run.graphRunId)!, registration);
  }
  private submit(registration: GraphHostRegistration, request: StartGraphRequest) {
    if (request.source.product !== registration.product) throw new GraphContractError('UNAUTHENTICATED');
    const input = request.spec.input;
    if (input.repositoryRef !== registration.repositoryRef || input.taskRef !== registration.taskRef || request.inputArtifactRef !== registration.inputArtifactRef
      || !registration.delivery.acceptanceRegistry.some(a => a.ref === input.acceptanceRef)) throw new GraphContractError('POLICY_DENIED');
    const boundContract = registration.delivery.deliveryContract ? bindGraphDeliveryContract(request,
      { ...registration.delivery, deliveryContract: registration.delivery.deliveryContract }) : undefined;
    request = boundContract?.request ?? request;
    const compiled = compileGraphSpec(request.spec);
    let created = false;
    const run = this.options.db.transaction(() => {
      synchronous(registration.authorize());
      const prior = this.options.db.prepare('SELECT graph_run_id FROM graph_requests WHERE scope_json=? AND idempotency_key=?')
        .get(workerCanonical(request.scope), request.idempotencyKey) as { graph_run_id: string } | undefined;
      if (!prior) {
        if (registration.admitNewRuns) {
          const admitted: unknown = registration.admitNewRuns();
          if (admitted !== true) {
            void Promise.resolve(admitted).catch(() => {});
            throw new GraphContractError('GRAPH_ADMISSION_DISABLED');
          }
        }
        assertGraphDeliverySupported(registration.delivery);
      }
      const run = this.runs.create(request, graphDeliveryVersion(compiled), Date.now());
      if (prior && !this.options.db.prepare('SELECT 1 FROM graph_public_delivery WHERE graph_run_id=?').get(run.graphRunId)) throw new GraphContractError('CHECKPOINT_INCOMPATIBLE');
      const descriptor: Descriptor = { registrationId: registration.id, request, root: join(this.options.root, run.graphRunId), launchOwner: this.owner,
        ...(boundContract ? { contractEnvelope: boundContract.envelope } : {}) };
      created = this.options.db.prepare('INSERT OR IGNORE INTO graph_public_delivery(graph_run_id,descriptor_json) VALUES (?,?)').run(run.graphRunId, JSON.stringify(descriptor)).changes === 1;
      return this.bound(registration, run.scope, run.graphRunId);
    }).immediate();
    if (created) {
      // Only the durable reservation winner schedules; even construction is deferred.
      const entry: { driver?: GraphDelivery; done: Promise<void> } = { done: Promise.resolve() };
      this.live.set(run.graphRunId, entry);
      entry.done = new Promise<void>(resolve => setImmediate(resolve)).then(async () => {
        if (this.closed || this.runs.get(run.scope, run.graphRunId)!.status === 'cancelled') return;
        try {
          const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
          if (descriptor.launchOwner !== this.owner) throw Error('LAUNCH_OWNER_MISMATCH');
          entry.driver = new GraphDelivery({ ...registration.delivery, db: this.options.db, root: descriptor.root, request: descriptor.request, reservedRun: run,
            ...(descriptor.contractEnvelope ? { contractEnvelope: descriptor.contractEnvelope } : {}),
            acquire: this.options.acquire, ...(this.options.authorizePolicy ? { authorizePolicy: this.options.authorizePolicy } : {}), authorize: () => { synchronous(registration.authorize()); },
            authorizeCancel: () => { synchronous(registration.authorize('cancel')); },
            publish: (candidate, feedbackRef, contractReceiptRef) => {
              const { candidateRevision, artifactRefs, artifactDigest } = candidate;
              const output: StoredResult = { candidate: { candidateRevision, artifactRefs, artifactDigest }, evidenceRefs: [feedbackRef, ...(contractReceiptRef ? [contractReceiptRef] : [])], resultRef: feedbackRef };
              this.options.db.prepare('UPDATE graph_public_delivery SET result_json=? WHERE graph_run_id=?').run(JSON.stringify(output), run.graphRunId);
            },
            qualityInterrupt: proof => {
              const current = this.runs.get(run.scope, run.graphRunId)!;
              this.options.db.transaction(() => {
                synchronous(registration.authorize());
                this.options.db.prepare('UPDATE graph_public_delivery SET interrupt_json=? WHERE graph_run_id=?').run(JSON.stringify(proof), run.graphRunId);
                this.runs.project({ ...fence(current), status: 'waiting_input' });
              }).immediate();
            } });
          await entry.driver.run();
          const stopped = this.runs.get(run.scope, run.graphRunId)!;
          if (stopped.generation === 0 && stopped.status === 'cancel_pending') this.runs.project({ ...fence(stopped), status: 'cancelled' });
        } catch {
          const current = this.runs.get(run.scope, run.graphRunId)!;
          // Before a worker exists, failure has no ambiguous external effects.
          if (current.generation === 0 && current.status === 'queued') this.runs.project({ ...fence(current), status: 'failed' });
        }
      });
    }
    return this.status(run, registration);
  }
  close(): Promise<void> {
    this.closed = true;
    this.closing ??= this.drain();
    return this.closing;
  }
  private async drain(): Promise<void> {
    const draining = [...this.live.entries()].map(async ([id, entry]) => {
      try {
        const descriptor = JSON.parse(this.row(id).descriptor_json) as Descriptor;
        const run = this.runs.get(descriptor.request.scope, id)!;
        if (!['ready', 'failed', 'cancelled', 'cancel_pending'].includes(run.status)) this.runs.project({ ...fence(run), status: 'cancel_pending' });
        if (run.generation === 0) {
          const current = this.runs.get(run.scope, id)!;
          if (current.status === 'cancel_pending') this.runs.project({ ...fence(current), status: 'cancelled' });
        }
        await entry.driver?.cancel();
      } finally {
        // run() may still be preparing the module, and a bounded worker stop
        // may leave action persistence/release pending. Neither is DB quiescence.
        try { await entry.done; } finally { await entry.driver?.drainDatabaseUsers(); }
      }
    });
    // A deadline cannot prove database users are gone. Forced process recovery
    // keeps its existing durable uncertainty; graceful shutdown waits for I/O.
    await Promise.allSettled([...draining, ...this.pendingRequests]);
  }
}

// Explicit local owner trust: owner-only config plus a private Unix socket, never
// a scope/principal taken from RPC JSON. Remote identities require resolveCaller.
const ref = z.string().min(1).max(256);
const hostConfig = z.object({ schemaVersion: z.literal(1), admissionEnabled: z.boolean().optional(), registration: z.object({ id: ref, scope: GraphScope,
  repositoryRef: ref, repositoryPath: z.string().min(1), taskRef: ref, task: z.string().min(1).max(65536), inputArtifactRef: ref,
  deliveryContract: DeliveryContractV2.optional(),
  executionContextPath: z.string().min(1).max(4096).optional(),
  writer: z.object({ agentId: ref, capabilityId: ref }).strict(), reviewer: z.object({ agentId: ref, capabilityId: ref }).strict(),
  boundedMessages: z.object({ provider: z.enum(['anthropic', 'deepseek']).optional(), baseUrl: z.string().min(1).max(2048).optional(),
    model: ref, files: z.array(ref).min(1).max(16), acceptance: z.string().min(1).max(16000) }).strict().optional(),
  acceptanceRegistry: z.array(z.object({ ref, command: z.string().min(1), passExitCodes: z.array(z.number().int()), failExitCodes: z.array(z.number().int()) }).strict()).min(1),
  grant: z.object({ scope: GraphScope, expiresAt: z.number().int().positive(), operations: z.array(z.enum(['read', 'execute', 'cancel', 'resume'])).min(1) }).strict(),
}).strict() }).strict();
function trustedConfig(path: string) {
  const fd = openSync(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
  try { const stat = fstatSync(fd);
    if (!stat.isFile() || stat.size > 131072 || (stat.mode & 0o077) !== 0 || stat.uid !== process.getuid?.()) throw Error('GRAPH_CONFIG_OWNER_REQUIRED');
    const bytes = Buffer.alloc(131073); let offset = 0;
    for (;;) { const count = readSync(fd, bytes, offset, bytes.length - offset, offset); if (!count) break; offset += count;
      if (offset === bytes.length) throw Error('GRAPH_CONFIG_LIMIT'); }
    return hostConfig.parse(JSON.parse(bytes.subarray(0, offset).toString('utf8')));
  } finally { closeSync(fd); }
}
export interface LocalGraphRegistrationOptions {
  resolveDeliveryContractPolicy?: (input: { repositoryPath: string; scope: GraphScope; contract: DeliveryContractV2 }) => Promise<{
    policy: GraphContractHostPolicy;
    /** Synchronous live policy binding; not an async best-effort recheck. */
    assertCurrent: () => void;
  }>;
  resolveExecutionContext?: (input: { configPath: string; repositoryPath: string; scope: GraphScope;
    frozen: NonNullable<GraphDeliveryOptions['deliveryContract']>['frozen'] }) => Promise<GraphExecutionContextRegistration>;
}
export async function loadLocalGraphRegistration(path: string, adapters: ReadonlyMap<string, AgentAdapter>, options: LocalGraphRegistrationOptions = {}): Promise<GraphHostRegistration> {
  const config = trustedConfig(path).registration;
  const bindingDigest = (value: typeof config) => { const { grant: _grant, ...binding } = value; return createHash('sha256').update(workerCanonical(binding)).digest('hex'); };
  const digest = bindingDigest(config);
  let policyAuthority: (() => void) | undefined;
  let executionContext: GraphExecutionContextRegistration | undefined;
  const authorize = (operation: 'read' | 'execute' | 'cancel' | 'resume' = 'execute') => {
    const fresh = trustedConfig(path).registration;
    if (bindingDigest(fresh) !== digest || !same(fresh.grant.scope, config.scope)
      || !fresh.grant.operations.includes(operation) || ((operation === 'execute' || operation === 'resume') && fresh.grant.expiresAt <= Date.now()))
      throw new GraphContractError('GRANT_EXPIRED');
    if (operation === 'execute' || operation === 'resume') synchronous(policyAuthority?.());
    if (operation === 'execute' || operation === 'resume') synchronous(executionContext?.assertCurrent());
  };
  authorize('read');
  let deliveryContract: GraphDeliveryOptions['deliveryContract'];
  if (config.deliveryContract) {
    if (!options.resolveDeliveryContractPolicy) throw Error('GRAPH_CONTRACT_POLICY_REQUIRED');
    const resolved = await options.resolveDeliveryContractPolicy({ repositoryPath: config.repositoryPath, scope: config.scope, contract: config.deliveryContract });
    if (typeof resolved.assertCurrent !== 'function') throw Error('GRAPH_CONTRACT_POLICY_REQUIRED');
    policyAuthority = resolved.assertCurrent;
    deliveryContract = parseGraphContractRegistration({ frozen: freezeDeliveryContract(config.deliveryContract), policy: resolved.policy });
    authorize('execute');
  }
  if (config.executionContextPath) {
    if (!deliveryContract || !options.resolveExecutionContext) throw Error('GRAPH_CONTEXT_REGISTRATION_REQUIRED');
    if (config.scope.tenantId !== 'default') throw Error('GRAPH_CONTEXT_SCOPE_UNSUPPORTED');
    executionContext = snapshotGraphExecutionContext(await options.resolveExecutionContext({ configPath: config.executionContextPath,
      repositoryPath: config.repositoryPath, scope: config.scope, frozen: deliveryContract.frozen }));
    authorize('execute');

      authorize: () => authorize('execute') };
    writer = await createBoundedMessagesAdapter({ ...common, role: 'writer' });
    reviewer = await createBoundedMessagesAdapter({ ...common, role: 'reviewer' });

  }
  if (!writer || !reviewer || writer.id === reviewer.id) throw Error('GRAPH_REGISTERED_AGENTS_REQUIRED');

const digest = (bytes: string | Buffer) =>
  createHash("sha256").update(bytes).digest("hex");
/** Bound synchronous authority reads: no await between final check and Graph CAS. */
function readBounded(
  path: string,
  privateFile: boolean,
  optional = false,
): Buffer | undefined {
  let fd: number | undefined;
  try {
    fd = openSync(
      path,
      constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK,
    );
    const before = fstatSync(fd);
    if (
      !before.isFile() ||
      before.size > 65536 ||
      (privateFile &&
        ((before.mode & 0o077) !== 0 || before.uid !== process.getuid?.()))
    )
      throw Error("GRAPH_HOST_FILE_INVALID");
    const buffer = Buffer.alloc(65537);
    let length = 0;
    while (length < buffer.length) {
      const count = readSync(
        fd,
        buffer,
        length,
        buffer.length - length,
        length,
      );
      if (!count) break;
      length += count;
    }
    const after = fstatSync(fd);
    const current = lstatSync(path);
    if (
      length > 65536 ||
      before.size !== after.size ||
      before.mtimeMs !== after.mtimeMs ||
      current.isSymbolicLink() ||
      current.ino !== before.ino ||
      current.dev !== before.dev
    )
      throw Error("GRAPH_HOST_FILE_CHANGED");
    return buffer.subarray(0, length);
  } catch (error) {
    if (optional && (error as NodeJS.ErrnoException).code === "ENOENT")
      return undefined;
    throw error;
  } finally {
    if (fd !== undefined) closeSync(fd);
  }
}
function overlayFingerprint(path: string): string {
  const bytes = readBounded(path, true, true);
  return bytes === undefined ? "missing" : `present:${digest(bytes)}`;
}
const hex = z.string().regex(/^[a-f0-9]{64}$/);
const graphPolicySchema = z
  .object({
    schema_version: z.literal(1),
    repo_path: z.string().refine(isAbsolute),
    scope: GraphScope,
    base_policy_digest: hex,
    review_policy_ref: z.string().min(1).max(256),

function overlayPath(
  home: string,
  repository: string,
  revision: string,
): string {
  if (!/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/.test(revision))
    throw Error("GRAPH_CONTRACT_BASE_INVALID");
  const raw = execFileSync(
    "git",
    ["-C", repository, "show", `${revision}:.osstack/repo.yaml`],
    {
      encoding: "utf8",
      maxBuffer: 65536,
      timeout: 15000,
      env: {
        ...Object.fromEntries(
          Object.entries(process.env).filter(
            ([key]) => !key.startsWith("GIT_"),
          ),
        ),
        GIT_CONFIG_GLOBAL: "/dev/null",
        GIT_CONFIG_NOSYSTEM: "1",
        GIT_NO_REPLACE_OBJECTS: "1",
        GIT_TERMINAL_PROMPT: "0",
      },
    },
  );
  const product = (parseYaml(raw) as { product?: unknown })?.product;
  const valid = (value: unknown): value is string =>
    typeof value === "string" &&
    /^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$/.test(value);
  const id = valid(product)
    ? product
    : valid(basename(repository))
      ? basename(repository)
      : `repo-${digest(repository).slice(0, 16)}`;
  return join(home, "repos", `${id}.yaml`);
}

/** Explicit local host authority; registration data alone cannot invent checks or budgets. */
export function createGraphContractHostResolvers(
  homePath: string,
): LocalGraphRegistrationOptions {
  if (!isAbsolute(homePath)) throw Error("GRAPH_HOST_HOME_REQUIRED");
  const home = realpathSync(homePath);
  return {
    resolveDeliveryContractPolicy: async ({
      repositoryPath,
      scope,
      contract,
    }) => {
      if (realpathSync(osstackHome()) !== home)
        throw Error("GRAPH_HOST_HOME_CHANGED");
      if (
        !isAbsolute(repositoryPath) ||
        realpathSync(repositoryPath) !== repositoryPath
      )
        throw Error("GRAPH_CONTRACT_REPOSITORY_INVALID");
      const path = join(home, "graph-policy.json");
      const bytes = readBounded(path, true)!;
      const config = graphPolicySchema.parse(
        JSON.parse(bytes.toString("utf8")),
      );
      const budget = GraphBudget.parse({
        ...(config.limits as object),
        schemaVersion: 1,
      });
      const { schemaVersion: _version, ...limits } = budget;
      if (
        config.repo_path !== repositoryPath ||
        workerCanonical(config.scope) !== workerCanonical(scope)
      )
        throw Error("GRAPH_CONTRACT_SCOPE_MISMATCH");
      const overlay = overlayPath(home, repositoryPath, contract.base_head);
      const overlayDigest = overlayFingerprint(overlay);
      const policyFileDigest = digest(bytes);
      const assertCurrent = () => {
        if (
          realpathSync(homePath) !== home ||
          realpathSync(osstackHome()) !== home ||
          realpathSync(repositoryPath) !== repositoryPath ||
          digest(readBounded(path, true)!) !== policyFileDigest ||
          overlayFingerprint(overlay) !== overlayDigest
        )
          throw Error("GRAPH_CONTRACT_POLICY_CHANGED");
      };
      // Reuse actual committed-base policy/overlay admission. This is a lazy Graph-only load.
      const { createDeliverCommandDeps } = await import("./bin.js");
      const policy = await createDeliverCommandDeps().deliveryContractPolicy!(
        repositoryPath,
        contract.base_head,
      );
      assertCurrent();
      if (
        policy.policyDigest !== config.base_policy_digest ||
        contract.policy_digest !== policy.policyDigest ||
        contract.review_policy_ref !== config.review_policy_ref ||
        contract.required_checks.some(
          (check) =>
            !policy.verifyCommands.some(
              (command) =>
                deliveryCheckReference(command) === check.registry_ref,
            ),
        )
      )
        throw Error("GRAPH_CONTRACT_POLICY_MISMATCH");
      return {
        policy: {
          policyDigest: policy.policyDigest,
          reviewPolicyRef: config.review_policy_ref,
          limits,
        },
        assertCurrent,
      };
    },
    resolveExecutionContext: async ({
      configPath,
      repositoryPath,
      scope,
      frozen,
    }) => {
      if (scope.tenantId !== "default")
        throw Error("GRAPH_CONTEXT_SCOPE_UNSUPPORTED");
      const repository = realpathSync(repositoryPath);
      const path = realpathSync(resolve(repository, configPath));
      const local = relative(repository, path);
      if (local.startsWith("..") || isAbsolute(local))
        throw Error("GRAPH_CONTEXT_SOURCE_ESCAPE");
      const configDigest = digest(readBounded(path, false)!);
      const {
        loadExecutionContextConfig,
        createPinnedLocalContextLoader,
        createExecutionContextAdapters,
      } = await import("./execution-context.js");
      const config = await loadExecutionContextConfig(configPath, repository);
      const assertCurrent = () => {
        if (
          realpathSync(repositoryPath) !== repository ||
          realpathSync(resolve(repository, configPath)) !== path ||
          digest(readBounded(path, false)!) !== configDigest
        )
          throw Error("GRAPH_CONTEXT_CONFIG_CHANGED");
      };
      assertCurrent();
      const receiptRoot = join(home, "graph-context-receipts", randomUUID());
      await mkdir(receiptRoot, { recursive: true, mode: 0o700 });
      assertCurrent();
      return {
        configDigest,
        receiptRoot,
        assertCurrent,
        create: async (input) => {
          assertCurrent();
          if (
            input.workspacePath !== repository ||
            input.frozen.digest !== frozen.digest
          )
            throw Error("GRAPH_CONTEXT_SCOPE_MISMATCH");
          const wrapped = createExecutionContextAdapters({
            adapters: [input.adapter],
            workspacePath: repository,
            mode: "local",
            loadContext: createPinnedLocalContextLoader(config),
            maxPromptBytes: config.max_prompt_bytes,
            maxKnowledgeBytes: config.max_knowledge_bytes,
            contractDigest: frozen.digest,
            contractBrief: frozen.brief,
            recordReceipt: async (receipt) => {
              assertCurrent();
              const ref = await persistDeliveryHostRecord(receiptRoot, receipt);
              assertCurrent();
              for (const target of [
                ref.path,
                receiptRoot,
                join(home, "graph-context-receipts"),
              ]) {
                const handle = await open(
                  target,
                  constants.O_RDONLY | constants.O_NOFOLLOW,
                );
                try {
                  await handle.sync();
                } finally {
                  await handle.close();
                }
                assertCurrent();
              }
              return ref;
            },
          });
          return {
            adapter: wrapped.adapters[0]!,
            receiptRefs: wrapped.receiptRefs,
          };
        },
      };
    },
  };
}

      return Object.freeze({ id: a.id, display_name: a.display_name, cost_tier: a.cost_tier,
        capabilities: () => structuredClone(capabilities), isAvailable: a.isAvailable.bind(a), invoke: a.invoke.bind(a),
        ...(a.confirmStopped ? { confirmStopped: a.confirmStopped.bind(a) } : {}) });
    };
    this.options = Object.freeze({ ...options, request: freezeWorkerData(StartGraphRequest.parse(options.request)),
      ...(options.deliveryContract ? { deliveryContract: parseGraphContractRegistration(options.deliveryContract) } : {}),
      ...(options.contractEnvelope ? { contractEnvelope: freezeWorkerData(structuredClone(options.contractEnvelope)) } : {}),
      ...(options.executionContext ? { executionContext: snapshotGraphExecutionContext(options.executionContext) } : {}),
      writer: Object.freeze({ ...options.writer, adapter: adapter(options.writer.adapter) }),
      reviewer: Object.freeze({ ...options.reviewer, adapter: adapter(options.reviewer.adapter) }),
      ...(options.containedTests ? { containedTests: Object.freeze({ ...options.containedTests }) } : {}),
      acceptanceRegistry: freezeWorkerData(structuredClone(options.acceptanceRegistry)) });
    options = this.options;
    if (Boolean(options.deliveryContract) !== Boolean(options.contractEnvelope)) throw Error('GRAPH_CONTRACT_ENVELOPE_REQUIRED');
    if (options.executionContext && !options.deliveryContract) throw Error('GRAPH_CONTEXT_CONTRACT_REQUIRED');
    if (options.deliveryContract) verifyGraphContractEnvelope(options.contractEnvelope, options.request,
      { ...options, deliveryContract: options.deliveryContract });
    const capability = options.writer.adapter.capabilities().find(c => c.id === options.writer.capabilityId);
    if (options.writer.identity !== options.writer.adapter.id || !capability || !['code.write','code.refactor','code.debug'].includes(capability.kind)) throw Error('WRITER_IDENTITY_REQUIRED');
    this.compiled = options.reservedRun
      ? resolveV4GraphProgram(options.request.spec, options.reservedRun.identity)
      : compileGraphSpec(options.request.spec);
    this.runs = new GraphRunStore(options.db);
    if (options.reservedRun) this.runValue = options.reservedRun;
    this.broker = new GraphActionBroker(options.db, { acquire: options.acquire,
      ...(options.authorizePolicy ? { authorizePolicy: (workspacePath: string) => options.authorizePolicy!({ policyRoot: options.repositoryPath, workspacePath }) } : {}) });
    this.artifacts = new GraphArtifactStore(join(options.root, 'artifacts'));
  }

    const createdAt = this.runValue ? this.current().createdAt : Date.now();

  private assertOwned(): void {
    if (!this.owned) return;
    const r = this.current();
    if (r.generation !== this.owned.generation || r.leaseOwner !== this.owned.leaseOwner || workerCanonical(r.identity) !== workerCanonical(this.owned.identity)) throw Error('STALE_DELIVERY_OWNER');
  }
  private fence(premise = this.current()): GraphRunFence { this.assertOwned(); const r = premise; return { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity,
    expectedGeneration: r.generation, expectedOwner: r.leaseOwner, expectedRevision: r.revision, now: Date.now() }; }
  private check(): void {
    if (this.closed) throw Error('DELIVERY_CLOSED');
    this.assertOwned();
    const value: unknown = this.options.authorize();
    if (value !== undefined) { void Promise.resolve(value).catch(() => {}); throw Error('AUTHORITY_MUST_BE_SYNCHRONOUS'); }
    if (this.options.executionContext) {
      const current: unknown = this.options.executionContext.assertCurrent();
      if (current !== undefined) { void Promise.resolve(current).catch(() => {}); throw Error('GRAPH_CONTEXT_AUTHORITY'); }
    }
  }
  private checkSettlement(): void {
    if (!this.stopping || this.current().status !== 'cancel_pending' || !this.options.authorizeCancel) { this.check(); return; }
    this.assertOwned();
    const value: unknown = this.options.authorizeCancel();
    if (value !== undefined) { void Promise.resolve(value).catch(() => {}); throw Error('AUTHORITY_MUST_BE_SYNCHRONOUS'); }
  }

  }

  private async persist(nodeId: string, workspacePath: string, evidence: GraphActionEvidence, effectSlot = 'main'): Promise<string> {
    if (this.latest(nodeId, effectSlot)?.state !== 'running') throw Error('RUNNING_EFFECT_REQUIRED');

    const action = this.latest(nodeId, effectSlot)!;
    await publishBoundedRecoveryJournal(this.options.root, action, this.current().leaseOwner!, workspacePath, ref, evidence, () => this.checkSettlement());
    return ref;
  }
  private async restoreWorkspaceMappings(): Promise<void> {
    // Functional replay may skip cached workspace callbacks. Recover mappings
    // from exact content-addressed ledger receipts, never from worker RETURNs.
    for (let epoch = 0; epoch <= this.epoch; epoch++) {
      const nodeId = `e${epoch}.workspace`; const a = this.latest(nodeId);
      if (a?.outcome?.state !== 'confirmed') continue;
      const receipt = await loadDeliveryReceipt(this.options.root, a.outcome.evidenceRef, this.receiptBinding(nodeId, this.options.repositoryPath));
      const value = receipt.workspace as { workspace?: RegisteredCandidateWorkspace } | undefined;
      const workspace = value?.workspace;
      if (!workspace || workspace.workspace_id !== this.current().scope.workspaceId || workspace.workspace_path !== join(this.options.root, `writer-${epoch}`)
        || await realpath(workspace.workspace_path) !== workspace.workspace_path) throw Error('WORKSPACE_RECEIPT_MISMATCH');
      this.check(); this.workspaces.set(epoch, Object.freeze({ ...workspace }));
    }
  }
  private workspace(epoch = this.epoch): RegisteredCandidateWorkspace {
    const w = this.workspaces.get(epoch); if (!w) throw Error('WORKSPACE_MAPPING_REQUIRED'); return w;
  }

    const epoch = this.epoch; const source = this.workspace(); const branch = p.node(kind);
    return { nodeId: n.id, effectSlot: p.preparationSlot(kind, attempt), tokenReservation: n.tokenReservation,
      action: { kind: 'workspace_create', repositoryRef: this.compiled.spec.input.repositoryRef, baseRevision: candidate.candidateRevision }, workspace: source,
      execution: { kind: 'workspace', handler: async ({ lease }) => {

          binding: this.binding(), sourceWorkspace: source, authorize: () => this.check(), authorizeCancel: () => this.checkSettlement(), acquire: this.options.acquire,
          ...(this.options.authorizePolicy ? { authorizePolicy: (workspacePath: string) => this.options.authorizePolicy!({ policyRoot: this.options.repositoryPath, workspacePath }) } : {}),
          ...(this.options.containedTests ? { containedTests: this.options.containedTests } : {}),

          ...(this.options.deliveryContract ? { deliveryContract: this.options.deliveryContract.frozen } : {}),

          reviewer: this.options.reviewer, verifyReviewer: this.options.verifyReviewer,

        const action = prepared.actions[0]!; this.branches.set(branch.id, action);
        return { status: 'completed', stopped: true, evidence: { workspace: action.workspace, workspaceLeaseRef: `e${epoch}.${kind}` } };
      } }, persist: async evidence => ({ evidenceRef: await this.persist(n.id, source.workspace_path, evidence, p.preparationSlot(kind, attempt)), actualTokens: 0,
        result: { schemaVersion: 1, kind: 'workspace', workspaceLeaseRef: `e${epoch}.${kind}`, receiptRef: `e${epoch}.prepared.${kind}.${attempt}` } }) };
  }
  private async registration(name: string): Promise<GraphRegisteredAction> {
    const p = deliveryEpoch(this.compiled, this.epoch); const n = p.node(name); const r = this.current(); const epoch = this.epoch;
    const common = { nodeId: n.id, effectSlot: n.effectSlot, tokenReservation: n.tokenReservation };
    const workspaceLeaseRef = `e${epoch}.writer`;
    if (name === 'workspace') {
      const path = join(this.options.root, `writer-${epoch}`); const workspace = { workspace_id: r.scope.workspaceId, workspace_path: path };
      const baseRevision = epoch === 0 ? this.compiled.spec.input.baseRevision : this.candidate(epoch - 1).candidateRevision;
      return { ...common, workspace: { workspace_id: r.scope.workspaceId, workspace_path: this.options.repositoryPath },
        action: { kind: 'workspace_create', repositoryRef: this.compiled.spec.input.repositoryRef, baseRevision },
        execution: { kind: 'workspace', handler: async () => {
          this.check();
          if (epoch === 0) { if (git(this.options.repositoryPath, 'rev-parse', `${baseRevision}^{commit}`) !== baseRevision) throw Error('BASE_REVISION_MISMATCH'); git(this.options.root, 'clone', '--no-local', '--no-hardlinks', '--', this.options.repositoryPath, path); git(path, 'checkout', '--detach', baseRevision); }
          else { const restored = await this.artifacts.restore(this.candidate(epoch - 1).artifactDigest, { rootPath: this.options.root, directoryName: `writer-${epoch}` }); if (restored.worktreePath !== path) throw Error('RESTORE_PATH_MISMATCH'); }
          this.workspaces.set(epoch, workspace);
          return { status: 'completed', stopped: true, evidence: { workspace, baseRevision } };
        } }, persist: async evidence => ({ evidenceRef: await this.persist(n.id, this.options.repositoryPath, evidence), actualTokens: 0,
          result: { schemaVersion: 1, kind: 'workspace', workspaceLeaseRef, receiptRef: `e${epoch}.workspace` } }) };
    }
    if (name === 'implement' || name === 'repair') {
      this.result(`e${epoch}.workspace`); const workspace = this.workspace();
      const feedback = epoch ? this.options.db.prepare('SELECT bundle_json FROM graph_verification_joins WHERE receipt_ref=? AND graph_run_id=?').get(this.feedbackRef, r.graphRunId) as { bundle_json: string } | undefined : undefined;
      if (epoch && !feedback) throw Error('IMMUTABLE_FEEDBACK_REQUIRED');
      const action = epoch ? { kind: 'repair' as const, inputArtifactRef: this.options.request.inputArtifactRef, workspaceLeaseRef,
        candidateRevision: this.candidate(epoch - 1).candidateRevision, feedbackRef: this.feedbackRef! } : { kind: 'implement' as const, inputArtifactRef: this.options.request.inputArtifactRef, workspaceLeaseRef };
      const rawAdapter = this.options.writer.adapter;
      const context = this.options.executionContext ? await prepareGraphExecutionContext({ registration: this.options.executionContext,
        adapter: rawAdapter, workspacePath: this.options.repositoryPath, frozen: this.options.deliveryContract!.frozen, authorize: () => this.check() }) : undefined;
      const adapter = context?.adapter ?? rawAdapter;
      const capability = adapter.capabilities().find(c => c.id === this.options.writer.capabilityId); if (!capability) throw Error('WRITER_CAPABILITY_REQUIRED');
      return { ...common, workspace, action, execution: { kind: 'adapter', adapter, capabilityId: capability.id, plan: {
        plan_id: `delivery-${r.graphRunId}-${epoch}`, agent_id: adapter.id, max_iterations: 1,
        task: { task_id: `delivery-${epoch}`, source: 'manual', capability: capability.kind, priority: 100,
          prompt: `${this.options.task}${this.options.deliveryContract && !context ? `\n\n${this.options.deliveryContract.frozen.brief}` : ''}${feedback ? `\nRepair frozen candidate ${action.kind === 'repair' ? action.candidateRevision : ''}. Complete host quality feedback:\n${feedback.bundle_json}` : ''}` },
        stop_condition: { type: 'tests_pass' },
      } }, persist: async evidence => {
        if (context && evidence.status === 'completed' && evidence.stopped) {
          const action = this.latest(n.id)!; const recorded = await context.evidence(evidence.candidate?.execution?.run_id ?? '', { effect: action.effect, effectKey: action.effectKey });
          this.check(); this.contextEvidence.set(action.effectKey, recorded);
        }
        return { evidenceRef: await this.persist(n.id, workspace.workspace_path, evidence),
          ...(evidence.status === 'completed' && evidence.stopped ? { result: { schemaVersion: 1 as const, kind: 'implementation_complete' as const, workspaceLeaseRef, receiptRef: `e${epoch}.implementation` } } : {}) };
      } };
    }
    if (name === 'freeze') {
      const implementation = this.latest(`e${epoch}.${epoch ? 'repair' : 'implement'}`);
      if (implementation?.outcome?.state !== 'confirmed' || implementation.outcome.result.kind !== 'implementation_complete') throw Error('IMPLEMENTATION_REQUIRED');
      const workspace = this.workspace(); const scope = await resolveRepositoryScope(workspace.workspace_path);
      const checkProducer = async () => { const evidence = await loadDeliveryReceipt(this.options.root, implementation.outcome!.evidenceRef,
        this.receiptBinding(implementation.effect.nodeId, workspace.workspace_path));
        if (evidence.stopped !== true || workerCanonical(evidence.candidate?.workspace) !== workerCanonical(workspace)) throw Error('PRODUCER_STOP_REQUIRED'); this.check();
        if (this.options.deliveryContract) assertGraphContractPaths(this.options.deliveryContract.frozen, workspace.workspace_path); };
      return { ...common, workspace, action: { kind: 'freeze', workspaceLeaseRef, receiptRef: implementation.outcome.result.receiptRef },
        execution: { kind: 'freeze', store: this.artifacts, input: { artifactId: `delivery-${r.graphRunId}-${epoch}`, scope: r.scope, runId: r.graphRunId,
          generation: r.generation, owner: r.leaseOwner!, workspaceLeaseRef, repoId: scope.repoId, commonDir: scope.commonDir, worktreePath: workspace.workspace_path,
          baseRevision: epoch ? this.candidate(epoch - 1).candidateRevision : this.compiled.spec.input.baseRevision,
          implementationReceiptRef: implementation.outcome.result.receiptRef, producerStopEvidenceRef: implementation.outcome.evidenceRef },
        authority: { verifyLease: async (_input, lease) => { this.check(); await (lease as ExecutionLease).assertActive!(); }, verifyImplementationReceipt: checkProducer, verifyProducersStopped: checkProducer } },
        persist: async evidence => { return { evidenceRef: await this.persist(n.id, workspace.workspace_path, evidence), actualTokens: 0, result: evidence.frozenCandidate! }; } };
    }
    if (name === 'prepare_review' || name === 'prepare_tests') return this.preparation(name === 'prepare_review' ? 'review' : 'tests', 1);
    const action = this.branches.get(n.id); if (!action) throw Error('PREPARATION_REQUIRED'); return action;
  }
  private request(action: GraphRegisteredAction) { const r = this.current(); return { schemaVersion: 1 as const, kind: 'worker_action' as const, requestId: `host:${randomUUID()}`,
    scope: r.scope, graphRunId: r.graphRunId, specHash: r.identity.specHash, generation: r.generation, epoch: this.epoch,
    nodeId: action.nodeId, effectSlot: action.effectSlot, action: action.action }; }
  private async dispatch(connection: object, value: WorkerActionRequest, options: { signal?: AbortSignal }): Promise<WorkerActionResponse> {
    if (this.stopping) throw Error('DELIVERY_STOPPING');
    this.check(); const request = WorkerActionRequest.parse(value); const r = this.current();
    if (request.epoch !== this.epoch || request.generation !== r.generation || request.graphRunId !== r.graphRunId || request.specHash !== r.identity.specHash || workerCanonical(request.scope) !== workerCanonical(r.scope)) throw Error('DELIVERY_IDENTITY_MISMATCH');
    const name = request.nodeId.replace(`e${this.epoch}.`, ''); const n = deliveryEpoch(this.compiled, this.epoch).node(name);
    if (request.effectSlot !== n.effectSlot) throw Error('DELIVERY_SLOT_MISMATCH');
    if (!this.installed.has(n.id)) this.installed.set(n.id, this.registration(name).then(action => { this.check(); this.broker.installActions(connection, [action]); }));
    await this.installed.get(n.id);
    let response = await this.broker.handle(connection, request, options);
    if (name !== 'review' && name !== 'tests') return response;
    // The host inspects the actual attested failure. The task never caches failed RETURN.
    while (response.kind === 'worker_action_response' && response.outcome.state === 'failed' && response.outcome.error.retryable) {
      const prior = this.latest(n.id)!;
      if (prior.effect.logicalAttempt >= n.maxLedgerAttempts) break;
      if (!prior.outcome?.evidenceRef.startsWith('graph-verification:')) break;
      const envelope = await loadGraphVerificationEvidence(join(this.options.root, 'verification'), prior.outcome!.evidenceRef, this.binding(), { effect: prior.effect, effectKey: prior.effectKey });
      if (!envelope.verificationFailure?.retrySafe) break;
      const snapshot = new GraphVerificationStore(this.options.db, () => {}).snapshot(this.fence());
      const cost = deliveryEpoch(this.compiled, this.epoch).nextCost[name];
      if (snapshot.budget.accountingBlocked || snapshot.budget.actionStarts + cost.actions > this.compiled.spec.limits.maxActionStarts
        || BigInt(snapshot.budget.chargedTokens) + BigInt(snapshot.budget.reservedTokens) + BigInt(cost.tokens) > BigInt(this.compiled.spec.limits.tokenReservationLimit)) break;
      const prep = await this.preparation(name, prior.effect.logicalAttempt + 1);
      this.broker.installActions(connection, [prep]);
      const prepared = await this.broker.handle(connection, this.request(prep), options);
      if (prepared.kind !== 'worker_action_response' || prepared.outcome.state !== 'confirmed') return response;
      response = await this.broker.retryAuthorized(connection, prior.effectKey, () => { this.check(); }, { ...options, replacement: this.branches.get(n.id)! });
      if (response.kind === 'worker_action_response') response = { ...response, requestId: request.requestId };
    }
    return response;
  }
  private async evaluate(): Promise<GraphVerificationDecision> {
    const p = deliveryEpoch(this.compiled, this.epoch);
    return new GraphVerificationJoinService(this.options.db, { binding: this.binding(), artifacts: this.artifacts, evidenceRoot: join(this.options.root, 'verification'),
      slots: p.slots, nextCost: p.nextCost, authorize: () => this.check() }).evaluate(this.fence());
  }
  private async takeover(handle: GraphRuntimeHandle, stopped: GraphRuntimeStop): Promise<void> {
    const r = this.current(); const key = {};
    const inspection = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); });
    let inventory: GraphExecutorInventory | undefined;
    const verify = () => { const observed = this.runtime.observedExit(handle.connection);
      if (!stopped.actionsSettled || stopped.worker.kind !== 'exited' || stopped.worker.journalError || workerCanonical(observed) !== workerCanonical(stopped.worker)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
      return verifyJournaledGraphWorkerExit(this.options.db, handle.binding);
    };
    this.runtime.reconciler.register(key, { binding: { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity, generation: r.generation, owner: r.leaseOwner },
      authorize: () => this.check(), stopWorker: async () => stopped,
      verifyWorker: async () => verify(),
      verifyQuiescence: async snapshot => { verify(); const decision = await this.evaluate(); if (decision.snapshot.pendingAdmissions || decision.snapshot.unresolvedActions) throw Error('QUIESCENCE_REQUIRED');
        inventory = inspection.inspectExecutors(snapshot.fence);
        if (inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
        return { executorEvidenceRef: decision.publication!.record.receiptRef, workspaceEvidenceRef: this.candidate().artifactRefs[0]! }; },
      verifyOutcome: async () => { throw Error('NO_EFFECT_RECONCILIATION'); }, recheck: (_snapshot, proof) => {
        this.check(); if (workerCanonical(verify()) !== workerCanonical(proof.worker) || !inventory) throw Error('QUIESCENCE_REQUIRED');
        inspection.assertExecutorInventory(inventory);
      }, release: async () => {},
    });
    try { const takeover = await this.runtime.reconciler.reconcile(key, { kind: 'takeover', fence: this.fence(), receiptId: `delivery-takeover-${this.epoch}`, owner: `delivery-${r.generation + 1}`, leaseExpiresAt: this.leaseDeadline() }); this.owned = takeover.result.run; }
    finally { this.runtime.reconciler.unregister(key); }
  }
  private checkpointDigest(): string {
    const rows = this.options.db.prepare('SELECT namespace,checkpoint_id,checkpoint FROM graph_checkpoints WHERE graph_run_id=? ORDER BY namespace,checkpoint_id').all(this.current().graphRunId);
    const writes = this.options.db.prepare("SELECT namespace,checkpoint_id,task_id,channel,write_index,value FROM graph_checkpoint_writes WHERE graph_run_id=? ORDER BY namespace,checkpoint_id,task_id,write_index").all(this.current().graphRunId);
    if (!rows.length || !writes.some(row => (row as { channel: string }).channel === '__interrupt__')) throw Error('DURABLE_INTERRUPT_REQUIRED');
    return createHash('sha256').update(workerCanonical({ rows, writes })).digest('hex');
  }
  verifyResume(proof: GraphDeliveryInterrupt): void {
    if (!this.waiting || workerCanonical(proof) !== workerCanonical(this.waiting.proof) || this.stopping) throw Error('LIVE_QUIESCENCE_REQUIRED');
    this.waiting.verify();
  }
  resumeQuality(proof: GraphDeliveryInterrupt): void { this.verifyResume(proof); this.waiting!.resolve(); }
  /** Actual worker and broker signal chain. A drained Promise never clears holds. */
  async cancel(): Promise<void> {
    this.stopping = true; this.waiting?.reject(Error('DELIVERY_CANCELLED'));
    const handle = this.active;
    if (!handle) return;
    const stopped = await handle.stop('GRAPH_CANCEL_REQUESTED');
    if (!this.owned || stopped.worker.kind !== 'exited' || !stopped.actionsSettled) return;
    const r = this.current(); if (r.status !== 'cancel_pending') return;
    const inspection = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); });
    let inventory: GraphExecutorInventory | undefined;
    const verify = () => {
      this.assertOwned();
      if (stopped.worker.journalError || workerCanonical(this.runtime.observedExit(handle.connection)) !== workerCanonical(stopped.worker)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
      return verifyJournaledGraphWorkerExit(this.options.db, handle.binding);
    };
    const key = {};
    this.runtime.reconciler.register(key, { binding: { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity, generation: r.generation, owner: r.leaseOwner },
      authorize: () => { this.assertOwned(); const result: unknown = (this.options.authorizeCancel ?? this.options.authorize)();
        if (result !== undefined) { void Promise.resolve(result).catch(() => {}); throw Error('AUTHORITY_MUST_BE_SYNCHRONOUS'); } }, stopWorker: async () => stopped,
      verifyWorker: async () => verify(),
      verifyQuiescence: async snapshot => {
        const exit = verify(); inventory = inspection.inspectExecutors(snapshot.fence);
        if (inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
        return { executorEvidenceRef: `graph-executor-inventory:${inventory.digest}`, workspaceEvidenceRef: exit.evidenceRef };
      },
      verifyOutcome: async () => { throw Error('NO_EFFECT_RECONCILIATION'); }, recheck: (_snapshot, proof) => {
        if (workerCanonical(verify()) !== workerCanonical(proof.worker) || !inventory) throw Error('QUIESCENCE_REQUIRED');
        inspection.assertExecutorInventory(inventory);
      }, release: async () => {},
    });

          // A sealed failure must never advertise a retry that cannot be resumed.

            if (current.digest !== decision.snapshot.digest) throw Error('GRAPH_CONTRACT_SNAPSHOT_CHANGED');

            root: this.options.root, envelope: this.options.contractEnvelope, decision, artifacts: this.artifacts, assertCurrent: assertContractCurrent,
            contextEvidence: [...this.contextEvidence.values()],
            ...(this.options.executionContext ? { contextReceiptRoot: this.options.executionContext.receiptRoot } : {}),
          }) : undefined;

          const inventory = inspection.inspectExecutors(this.fence(decision.snapshot.run));
          if (route.kind !== 'reconcile' && inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
          if (this.current().status !== 'cancel_pending') this.options.db.transaction(() => {
            if (contractReceiptRef) {
              this.check();
              new GraphVerificationStore(this.options.db, () => {}).assertCurrentInTransaction(decision.snapshot, Date.now());
            }
            if (workerCanonical(verifyDurableExit()) !== workerCanonical(workerProof)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
            inspection.assertExecutorInventory(inventory);
            this.runs.project({ ...this.fence(decision.snapshot.run), status: route.kind === 'ready' ? 'ready' : route.kind === 'reconcile' ? 'reconciliation_required' : 'failed' });
            if (decision.publication) this.options.publish?.(candidate, decision.publication.record.receiptRef, contractReceiptRef);
          }).immediate();
          return { route, candidate, ...(decision.publication ? { feedbackRef: decision.publication.record.receiptRef } : {}) };
        }
        if (workerError || !decision.publication) throw workerError ?? Error('IMMUTABLE_JOIN_REQUIRED');
        this.check();
        this.feedbackRef = decision.publication.record.receiptRef;
        this.options.publish?.(candidate, this.feedbackRef);
        if (this.options.qualityInterrupt) {
          const proof: GraphDeliveryInterrupt = { interruptId: `quality:${r.graphRunId}:${this.epoch}:${r.generation}`, inputDigest: r.identity.inputDigest,
            kind: 'quality', checkpointDigest: this.checkpointDigest(), feedbackRef: this.feedbackRef };
          await new Promise<void>((resolve, reject) => {
            this.waiting = { proof, resolve, reject, verify: () => {
              this.check();
              if (workerCanonical(verifyDurableExit()) !== workerCanonical(workerProof)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
              if (this.checkpointDigest() !== proof.checkpointDigest || workerCanonical(this.runtime.observedExit(handle.connection)) !== workerCanonical(stopped.worker)
                || stopped.worker.kind !== 'exited' || !stopped.actionsSettled) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
              if (!this.options.db.prepare('SELECT 1 FROM graph_verification_joins WHERE graph_run_id=? AND receipt_ref=?').get(r.graphRunId, proof.feedbackRef)) throw Error('IMMUTABLE_JOIN_REQUIRED');
            } };
            this.options.qualityInterrupt!(proof);
          });
          this.waiting = undefined;
        }
        await this.takeover(handle, stopped);
        const next = this.epoch + 1;
        new GraphDeliveryEpochStore(this.options.db).activate(this.fence(), next, deliveryEpoch(this.compiled, next).cost, this.feedbackRef);
        resume = { kind: 'interrupt', value: { ...decision.route, feedbackRef: this.feedbackRef } };
        this.epoch = next;
      }
    } catch (error) {
      // An idempotent prepare may find another owner's run. Failed acquisition
      // grants no authority to project that run's status.
      if (!this.owned) throw error;
      const current = this.current();
      let inventoryUnknown = true;
      try { inventoryUnknown = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); })
        .inspectExecutors(this.fence(current)).requiredEffectKeys.length > 0; } catch { /* An unreadable inventory is not quiescence. */ }
      const reconcile = inventoryUnknown || !durableWorkerStopped || !lastStopped?.actionsSettled || lastStopped.worker.kind !== 'exited';
      try {
        this.check();
        if (!['ready','failed','cancelled','cancel_pending'].includes(current.status) && current.leaseExpiresAt! > Date.now()) {
          this.runs.project({ ...this.fence(current), status: reconcile ? 'reconciliation_required' : 'failed' });
        }
      } catch { /* Revoked/stale hosts cannot project replacement or cancellation state. */ }
      return { route: { kind: reconcile ? 'reconcile' : 'stop', epoch: this.epoch, reason: String(error).slice(0, 4096) } };
    } finally { this.closed = true; await this.runtime.close(); if (this.connection) this.broker.unregister(this.connection); }
  }
}

        // promoted to retryable structured review/code results.

          const error = parsed.success && !parsed.data.retryable ? parsed.data : GraphError.parse({ schemaVersion: 1, code: 'POLICY_DENIED',
            message: 'Stopped action has no attested verification report; host must inspect admission/execution failure', retryable: false, nextAction: 'stop' });
          branches.push({ ...base, state: 'failed', error }); continue;

          || envelope.workspaceLeaseRef !== action.action.workspaceLeaseRef) throw Error('JOIN_EVIDENCE_MISMATCH');

            evidenceRefs: [outcome.evidenceRef], evidenceDigest: outcome.evidenceRef.slice('graph-verification:'.length),

          if (verificationCanonical(outcome.result) !== verificationCanonical(result)) throw Error('JOIN_RESULT_MISMATCH');
          branches.push({ ...base, state: 'confirmed', result });
        } else {
          const failure = envelope.verificationFailure;
          // A cancelled settlement can retain a valid receipt yet have no G19
          // classification. Preserve the ledger failure instead of promoting the
          // receipt's code verdict, and leave host retry policy to G21.
          const error = failure ? GraphError.parse(outcome.error) : GraphError.parse({ schemaVersion: 1, code: 'INTERNAL_ERROR',
            message: 'Stopped invocation has no retryable verification classification', retryable: false, nextAction: 'stop' });
          if (failure && (error.code !== failure.code || error.message !== failure.message || error.retryable !== failure.retrySafe || outcome.retrySafe !== failure.retrySafe)) throw Error('JOIN_FAILURE_MISMATCH');
          branches.push({ ...base, state: 'failed', error });
        }

        branches.push({ ...base, state: cancelling ? 'cancel_pending' : 'unknown', error: { schemaVersion: 1, code: cancelling ? 'CANCEL_UNCONFIRMED' : 'ACTION_OUTCOME_UNKNOWN',
          message: 'Ledger execution requires reconciliation', retryable: false, nextAction: 'reconcile' } });
      }
    }
    // Reject races after asynchronous evidence reads without refreshing away the
    // premise that was verified. Admission releases may change without revision.
    this.check(); const time = now(); const current = store.snapshot({ ...fence, now: time });
    if (current.digest !== snapshot.digest) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
    const joined = joinVerificationBranches({ binding, expected, branches }, branch => {
      if (!branches.some(original => verificationCanonical(branch) === verificationCanonical(original))) throw Error('JOIN_BRANCH_MISMATCH');
    });
    const spent = (kind: Kind) => selected[kind] ? 1 + snapshot.actions.filter(a => a.effect.epoch === binding.epoch
      && a.effect.nodeId === slots[kind].nodeId && a.effect.logicalAttempt > 1).length : 0;
    const route = routeVerification(joined, { binding, spec: snapshot.run.spec, runStatus: snapshot.run.status, runCreatedAt: snapshot.run.createdAt, now: time,
      ...snapshot.budget, pendingAdmissions: snapshot.pendingAdmissions, unresolvedActions: snapshot.unresolvedActions,
      branchAttemptsSpent: { review: spent('review'), tests: spent('tests') }, nextCost: this.options.nextCost });
    if (joined.status !== 'complete' || snapshot.unresolvedActions || snapshot.pendingAdmissions || snapshot.run.status !== 'running') return freezeWorkerData({ joined, route, snapshot });
    attested = { digest: snapshot.digest, joined };
    const committed = store.commit(snapshot, joined, time);
    return freezeWorkerData({ joined, route, snapshot: store.snapshot({ ...fence, expectedRevision: committed.run.revision, now: time }),
      publication: { record: committed.record, created: committed.created } });
  }
}

    if (before.size > maximum) throw new ExecutionContextError("execution_context_source_too_large");

    while (length < buffer.length) { const read = await file.read(buffer, length, buffer.length - length, null); if (!read.bytesRead) break; length += read.bytesRead; }

    if (length > maximum || before.size !== after.size || before.mtimeMs !== after.mtimeMs) throw new ExecutionContextError("execution_context_source_changed");

    if (!inside(root, current) || before.ino !== identity.ino || before.dev !== identity.dev) throw new ExecutionContextError("execution_context_source_escape");
    return buffer.subarray(0, length);
  } catch (error) { if (error instanceof ExecutionContextError) throw error; throw new ExecutionContextError("execution_context_source_unavailable"); }

  if (!review || input.reviews.some((item) => item.verdict !== "approved") || input.builderIdentity.evidence_source !== "reported"
    || review.runtime_identity.evidence_source !== "reported") throw new Error("delivery_reported_review_missing");
  const checks = input.commands.map((command, index) => {
    const registryRef = deliveryCheckReference(command.command);
    return { id: contract.required_checks.find((check) => check.registry_ref === registryRef)?.id ?? `host:${index}`,
      registry_ref: registryRef, command_digest: registryRef.slice(7), exit_code: command.exit_code, duration_ms: command.duration_ms };
  });
  return DeliveryVerificationReceiptV2.parse({ schema_version: 2, source: "osstack_loop_host", run_id: input.runId,
    task_id: contract.task_id, run_kind: input.runKind, contract_digest: digest, policy_digest: contract.policy_digest,
    candidate: input.candidate, workspace_state_digest: input.workspaceStateDigest, checks, review: { policy_ref: contract.review_policy_ref, verdict: review.verdict,
      builder_agent_id: input.builderAgentId, reviewer_agent_id: review.reviewer_agent_id,
      builder_provider: input.builderIdentity.provider_id, builder_model: input.builderIdentity.model_id,
      reviewer_provider: review.runtime_identity.provider_id, reviewer_model: review.runtime_identity.model_id,
      identity_source: "reported", artifact_digest: deliveryContractDigest(input.reviews) },
    observed_at: new Date().toISOString(), context_receipt_refs: input.contextReceiptRefs ?? [] });
}
/** Trusted host port: load its own immutable record and re-sample actual Git state.
 * A committed candidate may change HEAD, but its verified tree and base-relative diff must not change. */
export async function verifyPersistedDeliveryReceipt(input: {
  root: string; ref: DeliveryReceiptReferenceV2; frozen: FrozenDeliveryContractV2; runId: string; workspace: string; expectedHeads?: readonly string[];
}) {
  const raw = await readDeliveryHostRecord(input.root, input.ref);
  const receipt = DeliveryVerificationReceiptV2.parse(raw);
  const current = await captureDeliveryCandidate(input.workspace, input.frozen.contract.base_head);
  if (receipt.candidate.head !== input.frozen.contract.base_head
    || !(input.expectedHeads ?? [input.frozen.contract.base_head]).includes(current.head)) throw new Error("delivery_candidate_head_mismatch");
  if (current.tree_id !== receipt.candidate.tree_id || current.diff_digest !== receipt.candidate.diff_digest)
    throw new Error("delivery_candidate_changed_after_verification");
  if ((await snapshotDeliveryWorkspace(input.workspace, input.frozen.contract.base_head)).digest !== receipt.workspace_state_digest)
    throw new Error("delivery_workspace_changed_after_verification");
  // Unstaged/untracked edits were never part of the reviewed tree.
  if ((await deliveryGit(input.workspace, ["diff", "--name-only"])).trim()
    || (await deliveryGit(input.workspace, ["ls-files", "--others", "--exclude-standard"])).trim())
    throw new Error("delivery_candidate_has_unverified_changes");
  verifyDeliveryContractReceipt(input.frozen, raw, { runId: input.runId, candidate: { ...current, head: receipt.candidate.head },
    policyDigest: input.frozen.contract.policy_digest });
  return { receipt, current };
}

async function deliveryConfig(workspace: string): Promise<string> {
  const config = await deliveryGit(workspace, ["config", "--local", "--null", "--list"]);
  if (config.split("\0").some(entry => /^filter\..*\.(?:clean|smudge|process)\n/i.test(entry)))
    throw new Error("delivery_git_filter_unsupported");
  return config;
}
async function snapshotDeliveryWorkspace(workspace: string, baseHead: string, ignoredDirectories = false) {
  const canonicalWorkspace = await realpath(workspace);
  await deliveryConfig(workspace);
  const flags = (await deliveryGit(workspace, ["ls-files", "-v", "-z"])).split("\0").filter(Boolean);
  if (flags.some(entry => /^[a-zS]/.test(entry))) throw new Error("delivery_hidden_index_flags");
  const tracked = (await deliveryGit(workspace, ["diff", "--no-renames", "--name-only", "-z", baseHead, "--"])).split("\0").filter(Boolean);
  const untracked = (await deliveryGit(workspace, ["ls-files", "--others", "--exclude-standard", "-z"])).split("\0").filter(Boolean);
  const ignored = (await deliveryGit(workspace, ["ls-files", "--others", "--ignored", "--exclude-standard", ...(ignoredDirectories ? ["--directory", "--no-empty-directory"] : []), "-z"])).split("\0").filter(Boolean).sort();
  const paths = [...new Set([...tracked, ...untracked])].sort();
  const allFiles = [...new Set([...flags.map(entry => entry.slice(2)), ...paths])].sort();
  if (allFiles.length + ignored.length > 100000) throw new Error("delivery_candidate_too_many_files");
  const changes: Array<{ path: string; digest: string | null; mode: number | null }> = [];
  let bytes = 0;
  for (const path of allFiles) {
    const absolute = join(workspace, path);
    const parent = await realpath(dirname(absolute)).catch(error => { if ((error as NodeJS.ErrnoException).code === "ENOENT") return undefined; throw error; });
    if (parent !== undefined) { const rel = relative(canonicalWorkspace, parent); if (rel === ".." || rel.startsWith("../") || isAbsolute(rel)) throw new Error("delivery_candidate_parent_escape"); }
    const info = await lstat(absolute).catch(error => { if ((error as NodeJS.ErrnoException).code === "ENOENT") return undefined; throw error; });
    if (!info) { changes.push({ path, digest: null, mode: null }); continue; }
    if (!info.isFile() && !info.isSymbolicLink()) throw new Error("delivery_candidate_file_invalid");
    bytes += info.size;
    if (bytes > 64 * 1024 * 1024) throw new Error("delivery_candidate_too_large");
    let body: string | Buffer;
    if (info.isSymbolicLink()) body = await readlink(absolute);
    else {
      const file = await open(absolute, constants.O_RDONLY | constants.O_NOFOLLOW);
      try {
        const opened = await file.stat();
        if (!opened.isFile() || opened.size !== info.size) throw new Error("delivery_candidate_file_changed");
        body = Buffer.alloc(info.size + 1);
        const { bytesRead } = await file.read(body, 0, body.length, 0);
        if (bytesRead !== info.size) throw new Error("delivery_candidate_file_changed");
        body = body.subarray(0, bytesRead);
      } finally { await file.close(); }
    }
    changes.push({ path, digest: deliveryBytesDigest(body), mode: info.isSymbolicLink() ? 0o120000 : info.mode & 0o111 ? 0o100755 : 0o100644 });
  }
  return { digest: deliveryContractDigest({ baseHead, changes, ignored }), paths, ignored };
}
/** One invocation owns its pre-build baseline and exact check/review snapshot.
 * The same snapshot must survive staging and receipt capture; it is not re-baselined. */
export function createDeliveryVerificationFence(frozen: FrozenDeliveryContractV2) {
  const baselines = new Map<string, { ignored: Set<string>; config: string }>();
  const verified = new Map<string, string>();
  const sample = async (workspace: string) => {
    const baseline = baselines.get(workspace);
    if (!baseline) throw new Error("delivery_verification_baseline_missing");
    if ((await deliveryGit(workspace, ["rev-parse", "HEAD"])).trim() !== frozen.contract.base_head)
      throw new Error("delivery_candidate_base_changed");
    if (await deliveryConfig(workspace) !== baseline.config) throw new Error("delivery_git_config_changed");
    const snapshot = await snapshotDeliveryWorkspace(workspace, frozen.contract.base_head);
    if (snapshot.ignored.some(path => !baseline.ignored.has(path))) throw new Error("delivery_new_ignored_file");
    assertDeliveryContractPaths(frozen, snapshot.paths);
    return snapshot.digest;
  };
  const fence = async (workspace: string): Promise<() => Promise<void>> => {
    verified.delete(workspace);
    const before = await sample(workspace);
    return async () => {
      if (await sample(workspace) !== before) throw new Error("delivery_candidate_changed_during_verification");
      verified.set(workspace, before);
    };
  };
  return Object.assign(fence, {
    prepare: async (workspace: string) => {
      const snapshot = await snapshotDeliveryWorkspace(workspace, frozen.contract.base_head);
      baselines.set(workspace, { ignored: new Set(snapshot.ignored), config: await deliveryConfig(workspace) });
      verified.delete(workspace);
    },
    assertVerified: async (workspace: string) => {
      const digest = verified.get(workspace);
      if (!digest || await sample(workspace) !== digest) throw new Error("delivery_reviewed_candidate_changed");
      return digest;
    },
  });
}

/** Detect changes to the original checkout; this is not a filesystem sandbox.
 * Existing ignored file contents and arbitrary paths outside the checkout are not sampled.
 */
export async function createDeliverySourceFence(workspace: string, baseHead: string): Promise<() => Promise<void>> {
  const sample = async () => {
    const [head, index, snapshot] = await Promise.all([
      deliveryGit(workspace, ["rev-parse", "HEAD"]),
      deliveryGit(workspace, ["ls-files", "--stage", "-z"]),
      snapshotDeliveryWorkspace(workspace, baseHead, true),
    ]);
    return deliveryContractDigest({ head, index, snapshot: snapshot.digest });
  };
  const baseline = await sample();
  return async () => { if (await sample() !== baseline) throw new Error("delivery_source_changed"); };
}

import { constants } from "node:fs";
import { open, realpath, lstat } from "node:fs/promises";
import { isAbsolute, relative } from "node:path";
import { LoopState } from "@osstack/loop";
/** No replay or cleanup port. A Loop JSON file is a lossy snapshot source. */
export async function readAttentionLoop(input: {
  repo: string;
  path: string;
  loop_id: string;
  signal?: AbortSignal;
}) {
  input.signal?.throwIfAborted();
  if (!isAbsolute(input.path) || !isAbsolute(input.repo) || !input.loop_id)
    throw new Error("explicit_loop_binding_required");
  const repo = await realpath(input.repo);
  input.signal?.throwIfAborted();
  let handle: Awaited<ReturnType<typeof open>> | undefined;
  try {
    const path = await realpath(input.path);
    input.signal?.throwIfAborted();
    const rel = relative(repo, path);
    if (rel === ".." || rel.startsWith("../") || isAbsolute(rel))
      throw new Error("loop_outside_scope");
    const before = await lstat(path);
    input.signal?.throwIfAborted();
    if (!before.isFile())
      throw new Error("regular_loop_file_required");
    handle = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
    input.signal?.throwIfAborted();
    const stat = await handle.stat();
    input.signal?.throwIfAborted();
    const resolvedAfterOpen = await realpath(path);
    input.signal?.throwIfAborted();
    if (resolvedAfterOpen !== path || before.dev !== stat.dev || before.ino !== stat.ino)
      throw new Error("loop_source_changed");
    if (!stat.isFile() || stat.size > 1024 * 1024)
      throw new Error("unsupported_loop_source");
    const buffer = Buffer.alloc(1024 * 1024 + 1);
    const { bytesRead } = await handle.read(buffer, 0, buffer.length, 0);
    input.signal?.throwIfAborted();
    if (bytesRead > 1024 * 1024) throw new Error("oversized_loop_source");
    const state = LoopState.parse(
      JSON.parse(buffer.subarray(0, bytesRead).toString("utf8"))
    );
    if (state.loop_id !== input.loop_id)
      throw new Error("loop_identity_conflict");
    return {
      loop_id: input.loop_id,
      path,
      state: "snapshot_available",
      updated_at: state.updated_at,
      in_progress: state.in_progress ?? [],
      pending: state.pending,
      completed_count: state.completed_tasks.length,
      wait: state.wait_reason ?? null,
      process_liveness: "unknown",
      coverage: "snapshot_only",
      gap: "best_effort_snapshot_no_event_replay",
      recovery_required: (state.in_progress ?? []).length > 0,
    };
  } catch (error) {
    input.signal?.throwIfAborted();
    if ((error as NodeJS.ErrnoException).code === "ENOENT")
      return {
        loop_id: input.loop_id,
        path: input.path,
        state: "source_missing",
        process_liveness: "unknown",
        coverage: "uncovered",
        gap: "source_missing",
      };
    throw error;
  } finally {
    await handle?.close();
  }
}

    maxDurationMs?: number;
    signal?: AbortSignal;
  } = {}) {
    const maxItems = options.maxItems ?? 20;
    const maxDurationMs = options.maxDurationMs ?? 1000;
    if (!Number.isInteger(maxItems) || maxItems < 1 || maxItems > 100 ||
        !Number.isInteger(maxDurationMs) || maxDurationMs < 1 || maxDurationMs > 10_000) {
      throw new Error("invalid_delivery_batch_bounds");

    const result = { processed: 0, attempted: 0, delivered: 0, retried: 0,
      deadLetters: 0, superseded: 0, stale: 0, cancelled: 0,

    });

          : "delivery_failed";

      if (onAbort) signal?.removeEventListener("abort", onAbort);

          cancelled ? 1 : 0,
          row.delivery_id,
          row.attempts + 1
        )
    );
    changed = updated.changes;

      if (e instanceof Error && e.message === "stale_owner") return "stale";
      throw e;
    }

          process_liveness: "unknown",
          valid_until: (prior as { valid_until: number }).valid_until,
          expired: now > (prior as { valid_until: number }).valid_until,
          currently_valid:
            currentlyValid &&
            now <= (prior as { valid_until: number }).valid_until,
        };
      }
      if (row.expires_at < now) throw new Error("invalid_foreground");
      const s = this.scope(scope);
      if (
        new AttentionObserver(this.store, this.owner).inventory(scope, now)
          .coverage !== "covered"
      )
        throw new Error("coverage_not_ready");
      if (
        !s.scan_complete ||
        s.generation !== this.owner.generation ||
        s.deadline < now ||
        s.sampled_at > now ||
        s.error ||
        this.store.db
          .prepare(
            "SELECT 1 FROM attention_subjects WHERE scope_id=? AND (admitted=0 OR sampled_at<?) LIMIT 1"
          )
          .get(scope, now - 65000)
      )
        throw new Error("coverage_not_ready");
      const subjects = JSON.parse(row.subjects_json) as {
        subject_id: string;
        source_version: number;
      }[];
      this.validateSubjects(scope, subjects);
      const receipt = {
        receipt_id: randomUUID(),
        client_id: client,
        scope_id: scope,
        generation: this.owner.generation,
        subjects_json: row.subjects_json,
        created_at: now,
        valid_until: s.deadline,
      };
      this.store.db
        .prepare("INSERT INTO attention_handoffs VALUES(?,?,?,?,?,?,?)")
        .run(
          receipt.receipt_id,
          client,
          scope,
          this.owner.generation,
          row.subjects_json,
          now,
          s.deadline
        );
      this.store.db
        .prepare(
          "UPDATE attention_foregrounds SET handed_off=1 WHERE client_id=?"
        )
        .run(client);
      return {
        ...receipt,
        execution_takeover: false,
        process_liveness: "unknown",
        valid_until: s.deadline,
      };
    });
  }
  turnEnd(
    scope: string,
    client: string,
    token: string,
    role: string,
    now = Date.now()
  ) {
    return this.store.write(this.owner, () => {
      if (role !== "foreground_owner") throw new Error("wrong_role");
      const row = this.authenticate(scope, client, token, now, true);
      if (row.reminder_sent || row.handed_off) return { reminder: null };
      this.store.db
        .prepare(
          "UPDATE attention_foregrounds SET reminder_sent=1 WHERE client_id=?"
        )
        .run(client);
      return {
        reminder:
          "Foreground ended. Check attention coverage and explicitly hand off unresolved work.",
        continues: false,
        offline_detection: false,
        foreground_expired: row.expires_at < now,
      };
    });
  }
  private authenticate(
    scope: string,
    client: string,
    token: string,
    now: number,
    allowExpired = false
  ) {
    const row = this.store.db
      .prepare(
        "SELECT * FROM attention_foregrounds WHERE client_id=? AND scope_id=?"
      )
      .get(client, scope) as
      | {
          token_hash: string;
          generation: number;
          expires_at: number;
          subjects_json: string;
          handed_off: number;
          reminder_sent: number;
        }
      | undefined;
    if (
      typeof token !== "string" ||
      token.length !== 64 ||
      !row ||
      !timingSafeEqual(
        Buffer.from(row.token_hash),
        Buffer.from(attentionDigest(token))
      ) ||
      row.generation !== this.owner.generation ||
      (!allowExpired && row.expires_at < now)
    )
      throw new Error("invalid_foreground");
    return row;
  }
  private validateSubjects(
    scope: string,
    subjects: { subject_id: string; source_version: number }[]
  ): void {
    for (const s of subjects) {
      const row = this.store.db
        .prepare(
          "SELECT source_version,admitted,payload_json FROM attention_subjects WHERE scope_id=? AND kind='session' AND subject_id=?"
        )
        .get(scope, s.subject_id) as
        | { source_version: number; admitted: number; payload_json: string }
        | undefined;
      const live = this.store.db
        .prepare("SELECT version FROM agent_sessions WHERE session_id=?")
        .get(s.subject_id) as { version: number } | undefined;
      if (
        !row ||
        row.admitted !== 1 ||
        row.source_version !== s.source_version ||
        live?.version !== s.source_version ||
        JSON.parse(row.payload_json).reason ||
        JSON.parse(row.payload_json).gap === "history_gap" ||
        (JSON.parse(row.payload_json).unresolved_receipts ?? []).length > 0
      )
        throw new Error("subject_not_covered");
    }
  }
  private scope(scope: string) {
    const s = this.store.getScope(scope);
    if (!s) throw new Error("not_found");
    return s;
  }
  private actor(actor: string): void {
    if (typeof actor !== "string" || actor.length < 1 || actor.length > 256)
      throw new Error("invalid_actor");
  }

function fence(run: GraphRun, now: number): GraphRunFence {

/** A durable immutable evidence publication, not an epoch scheduler or node FSM.
 * The trusted host verifier must bind the bundle to current ledger attempts and
 * independent artifact/executor evidence. No worker JSON port calls this store. */
export class GraphVerificationStore {
  constructor(private readonly db: Database.Database,
    private readonly verify: (snapshot: GraphVerificationSnapshot, bundle: GraphJsonValue) => void) {}
  private capture(run: GraphRun): GraphVerificationSnapshot {
    const state = readGraphActionState(this.db, run.graphRunId);
    const admissions = this.db.prepare('SELECT effect_key,released_at FROM graph_action_admissions WHERE graph_run_id=? ORDER BY effect_key').all(run.graphRunId) as Array<{ effect_key: string; released_at: number | null }>;
    const body = { run, ...state, pendingAdmissions: admissions.filter(a => a.released_at === null).length,
      unresolvedActions: state.actions.filter(a => ['intent','running','unknown','cancel_pending'].includes(a.state)).length };
    return freezeWorkerData({ ...body, digest: hash({ body, admissions }) });
  }
  snapshot(input: GraphRunFence): GraphVerificationSnapshot {
    return fencedGraphTransaction(this.db, input, run => ({ changed: false, value: this.capture(run) })).value;
  }
  /** Final synchronous publication check inside the caller's existing IMMEDIATE transaction.
   * No refreshed premise: includes admissions which can change without run revision. */
  assertCurrentInTransaction(snapshot: GraphVerificationSnapshot, now: number): void {
    if (!this.db.inTransaction) throw Error('GRAPH_PUBLICATION_TRANSACTION_REQUIRED');
    const input = fence(snapshot.run, now);
    const run = checkGraphRunFence(new GraphRunStore(this.db).get(input.scope, input.graphRunId), input, true);
    const current = this.capture(run);
    if (current.digest !== snapshot.digest || workerCanonical(current) !== workerCanonical(snapshot)) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
  }
  commit(snapshotValue: GraphVerificationSnapshot, bundleValue: unknown, now: number): { record: GraphVerificationRecord; created: boolean; run: GraphRun } {
    const snapshot = freezeWorkerData(structuredClone(snapshotValue)); const bundle = freezeWorkerData(GraphJsonValue.parse(bundleValue));
    const result = fencedGraphTransaction(this.db, fence(snapshot.run, now), run => {
      const current = this.capture(run);
      if (current.digest !== snapshot.digest || workerCanonical(current) !== workerCanonical(snapshot)) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
      if (current.unresolvedActions) throw Error('UNRESOLVED_ACTION');
      if (current.pendingAdmissions) throw Error('ADMISSION_RELEASE_PENDING');
      if (!bundle || typeof bundle !== 'object' || Array.isArray(bundle)) throw Error('INVALID_JOIN');
      const object = bundle as Record<string, GraphJsonValue>; const binding = object.binding;
      if (!binding || typeof binding !== 'object' || Array.isArray(binding)) throw Error('INVALID_JOIN_BINDING');
      const b = binding as Record<string, GraphJsonValue>;
      const candidate = GraphResult.parse({ schemaVersion: 1, kind: 'candidate_ready', candidateRevision: b.candidateRevision, artifactDigest: b.artifactDigest, artifactRefs: ['join-validation'] });
      if (candidate.kind !== 'candidate_ready' || object.schemaVersion !== 1 || object.status !== 'complete'
        || b.graphRunId !== run.graphRunId || workerCanonical(b.scope) !== workerCanonical(run.scope) || b.specHash !== run.identity.specHash
        || !Number.isSafeInteger(b.epoch) || (b.epoch as number) < 0 || (b.epoch as number) >= (run.spec?.limits.maxEpochs ?? 0)
        || b.acceptanceRef !== run.spec?.input.acceptanceRef || typeof object.evidenceDigest !== 'string' || !/^[a-f0-9]{64}$/.test(object.evidenceDigest)) throw Error('INVALID_JOIN_BINDING');
      if (typeof this.verify !== 'function') throw Error('trusted verifier required');
      const verified: unknown = this.verify(current, bundle);
      if (verified !== undefined) { void Promise.resolve(verified).catch(() => {}); throw Error('join verifier must be synchronous'); }
      const receiptRef = `graph-join:${hash(binding)}`;
      const saved = this.db.prepare('SELECT bundle_json,generation,owner,created_at,receipt_ref FROM graph_verification_joins WHERE graph_run_id=? AND epoch=? AND candidate_revision=?')
        .get(run.graphRunId, b.epoch, candidate.candidateRevision) as { bundle_json: string; generation: number; owner: string; created_at: number; receipt_ref: string } | undefined;
      if (saved && (saved.bundle_json !== workerCanonical(bundle) || saved.receipt_ref !== receiptRef)) throw Error('JOIN_PUBLICATION_CONFLICT');
      if (!saved) this.db.prepare('INSERT INTO graph_verification_joins(receipt_ref,graph_run_id,epoch,candidate_revision,generation,owner,bundle_json,created_at) VALUES(?,?,?,?,?,?,?,?)')
        .run(receiptRef, run.graphRunId, b.epoch, candidate.candidateRevision, run.generation, run.leaseOwner, workerCanonical(bundle), now);
      const record: GraphVerificationRecord = { receiptRef, graphRunId: run.graphRunId, epoch: b.epoch as number, candidateRevision: candidate.candidateRevision,
        generation: saved?.generation ?? run.generation, owner: saved?.owner ?? run.leaseOwner!, bundle, createdAt: saved?.created_at ?? now };
      return { changed: !saved, value: { record, created: !saved } };
    });
    return freezeWorkerData({ ...result.value, run: result.run });
  }
}

export class GraphDeliveryEpochStore {
  constructor(private readonly db: Database.Database) {}
  activate(input: GraphRunFence, epoch: number, cost: { actions: number; tokens: number }, joinRef?: string): GraphRun {
    return fencedGraphTransaction(this.db, input, run => {
      if (!run.spec || run.status !== 'running') throw Error('RUN_NOT_RUNNING');
      const state = readGraphActionState(this.db, run.graphRunId);
      if (state.actions.some(a => ['intent','running','unknown','cancel_pending'].includes(a.state))) throw Error('UNRESOLVED_ACTION');
      if (this.db.prepare('SELECT 1 FROM graph_action_admissions WHERE graph_run_id=? AND released_at IS NULL').get(run.graphRunId)) throw Error('ADMISSION_RELEASE_PENDING');
      const row = this.db.prepare('SELECT max(epoch) AS epoch FROM graph_delivery_epochs WHERE graph_run_id=?').get(run.graphRunId) as { epoch: number | null };
      if (row.epoch === epoch) {
        const prior = this.db.prepare('SELECT join_ref FROM graph_delivery_epochs WHERE graph_run_id=? AND epoch=?').get(run.graphRunId, epoch) as { join_ref: string | null };
        if (prior.join_ref !== (joinRef ?? null)) throw Error('EPOCH_PERMIT_CONFLICT');
        return { changed: false, value: undefined };
      }
      if (epoch !== (row.epoch ?? -1) + 1 || epoch >= run.spec.limits.maxEpochs) throw Error('EPOCH_MISMATCH');
      if (epoch === 0 && (joinRef || state.actions.length)) throw Error('INITIAL_EPOCH_ALREADY_CONSUMED');
      if (epoch > 0) {
        const saved = this.db.prepare('SELECT bundle_json,epoch FROM graph_verification_joins WHERE graph_run_id=? AND receipt_ref=?').get(run.graphRunId, joinRef) as { bundle_json: string; epoch: number } | undefined;
        if (!saved || saved.epoch !== epoch - 1) throw Error('IMMUTABLE_JOIN_REQUIRED');
        const bundle = JSON.parse(saved.bundle_json);
        if (bundle.status !== 'complete' || bundle.branches.length !== 2 || !bundle.branches.every((b: { state: string }) => b.state === 'confirmed')
          || !bundle.branches.some((b: { result: { verdict?: string; passed?: boolean } }) => b.result.verdict === 'changes_requested' || b.result.passed === false)) throw Error('NEGATIVE_JOIN_REQUIRED');
        for (const branch of bundle.branches) {
          const latest = state.actions.filter(a => a.effect.epoch === epoch - 1 && a.effect.nodeId === branch.nodeId && a.effect.effectSlot === branch.effectSlot).sort((a,b) => b.effect.logicalAttempt - a.effect.logicalAttempt)[0];
          if (!latest || latest.effectKey !== branch.effectKey || latest.state !== 'confirmed' || latest.outcome?.state !== 'confirmed' || workerCanonical(latest.outcome.result) !== workerCanonical(branch.result)) throw Error('STALE_JOIN');
        }
      }
      if (!Number.isSafeInteger(cost.actions) || cost.actions < 1 || !Number.isSafeInteger(cost.tokens) || cost.tokens < 0) throw Error('INVALID_COMPILED_COST');
      const budget = state.budget;
      if (input.now - run.createdAt >= run.spec.limits.deadlineSeconds * 1000 || budget.accountingBlocked
        || budget.actionStarts + cost.actions > run.spec.limits.maxActionStarts
        || BigInt(budget.chargedTokens) + BigInt(budget.reservedTokens) + BigInt(cost.tokens) > BigInt(run.spec.limits.tokenReservationLimit)) throw Error('BUDGET_EXHAUSTED');
      this.db.prepare('INSERT INTO graph_delivery_epochs(graph_run_id,epoch,join_ref,generation,created_at) VALUES(?,?,?,?,?)').run(run.graphRunId, epoch, joinRef ?? null, run.generation, input.now);
      return { changed: true, value: undefined };
    }).run;
  }
}

function guardDatabasePath(dbPath: string) {
  const checkoutRoot = dirname(realpathSync(fileURLToPath(import.meta.url)));
  const defaultPath = defaultOsstackHome();
  const defaultHome = existsSync(defaultPath) ? realpathSync(defaultPath) : defaultPath;

  // below a protected home. Check protected ancestors as well as the leaf home.
  for (let ancestor = home; ; ancestor = dirname(ancestor)) {

function readState(path: string) {
  const fd = openSync(path, constants.O_RDONLY | constants.O_NONBLOCK);
  try {
    const stat = fstatSync(fd);
    if (!stat.isFile()) throw new Error("Loop state must be a regular file");
    if (stat.size > MAX_STATE_BYTES) throw new Error("Loop state exceeds 4 MiB");
    // Read at most the bound + sentinel even if a producer grows the file.
    const buffer = Buffer.alloc(MAX_STATE_BYTES + 1);
    let length = 0;
    while (length < buffer.length) {
      const count = readSync(fd, buffer, length, buffer.length - length, null);
      if (count === 0) break;
      length += count;
    }
    if (length > MAX_STATE_BYTES) throw new Error("Loop state exceeds 4 MiB");
    const bytes = buffer.subarray(0, length);
    let state: LoopState;
    try { state = LoopState.parse(JSON.parse(bytes.toString("utf8"))); }
    catch { throw new Error("Invalid Loop state JSON or schema"); }
    return { state, sha256: createHash("sha256").update(bytes).digest("hex") };
  } finally { closeSync(fd); }
}

export function explainGraph(input: GraphExplainInput) {
  if (!isAbsolute(input.dbPath) || !isAbsolute(input.statePath)) throw new Error("--db and --state must be absolute paths");
  if (!input.taskId || !input.board || input.tenantId === "" || input.tenantId === undefined) throw new Error("Explicit task, board and tenant scope required");
  const dbPath = realpathSync(input.dbPath);
  if (!statSync(dbPath).isFile()) throw new Error("Database must be an existing regular file");
  guardDatabasePath(dbPath);
  const statePath = realpathSync(input.statePath);
  const { state, sha256 } = readState(statePath);
  // Do not use core openDb: even its readonly mode sets database pragmas.
  const db = new Database(dbPath, { readonly: true, fileMustExist: true, timeout: 1000 });
  let facts: ReturnType<typeof readFacts>;
  try { facts = db.transaction(() => readFacts(db, input))(); }
  catch (error) {
    if (error instanceof Database.SqliteError) throw new Error(`Cannot read graph source tables/columns: ${error.message}`);
    throw error;
  }
  finally { db.close(); }
  const matches: LoopMatch[] = [];
  const locationCounts = { completed_tasks: 0, in_progress: 0, pending: 0, deferred_tasks: 0 };
  function addMatch(match: LoopMatch) {
    locationCounts[match.location] += 1;
    if (matches.length <= LIMIT) matches.push(match);
  }
  // Keep only the bounded projection; do not expose prompts or task context.
  for (const [index, task] of state.completed_tasks.entries()) {
    if (task.task_id !== input.taskId) continue;
    addMatch({ location: "completed_tasks", index, task_id: task.task_id, outcome: task.outcome,
      ...(task.run_id === undefined ? {} : { run_id: task.run_id, run_match: facts.runs.items.some((run) => run.run_id === task.run_id) ? "scoped_run" as const : "not_in_returned_runs" as const }) });
  }
  for (const location of ["in_progress", "pending"] as const) {
    for (const [index, task] of (state[location] ?? []).entries()) {
      if (task.task_id === input.taskId) addMatch({ location, index, task_id: task.task_id });
    }
  }
  for (const [index, entry] of (state.deferred_tasks ?? []).entries()) {
    if (entry.task.task_id === input.taskId) addMatch({ location: "deferred_tasks", index, task_id: entry.task.task_id, authorization: entry.authorization, reason: entry.reason });
  }
  const loopMatches = bounded(matches);
  const locations = Object.entries(locationCounts).map(([location, total]) => {
    const returned = loopMatches.items.filter((match) => match.location === location).length;
    return { location, total, returned, truncated: returned < total };
  });
  const truncated = [facts.cards, facts.runs, facts.relations, loopMatches].some((collection) => collection.truncated);
  return {
    schema_version: 1,
    scope: { task_id: input.taskId, board: input.board, tenant_id: input.tenantId },
    sources: {
      database: { path: dbPath, observed_at: facts.observedAt, consistency: "single_read_transaction", authority: "stored_task_run_card_facts", tables: ["tasks", "runs", "kanban_cards", "kanban_card_links"] },
      loop: { path: statePath, authority: "caller_supplied_file_content" },
    },
    task: facts.task, cards: facts.cards, runs: facts.runs, relations: facts.relations,
    loop: { loop_id: state.loop_id, updated_at: state.updated_at, sha256, binding: "caller_supplied_unverified", matches: loopMatches, locations },
    coverage: { scope: "direct_card_links_in_selected_board_and_tenant", limit_per_collection: LIMIT, truncated, loop_task_found: matches.length > 0 },
    notices: [...(matches.length === 0 ? ["loop_task_not_found"] : []), ...(matches.length > 1 ? ["multiple_loop_records"] : []), ...(truncated ? ["results_truncated"] : [])],
    limitations: [
      "Stored card links are associations, not scheduler dependencies or proof of blocking. Only stored blocked status/reasons are evidence of an existing blocker.",
      "Database and Loop file have no shared snapshot. Their statuses are separate observations; disagreement does not establish a cause or a current unified state.",
      "Loop files carry no tenant/repository identity. Task and run ID matches do not authenticate the caller-supplied file.",
      "Only direct links within the requested scope are shown. Empty or truncated results do not prove no impact. not_in_returned_runs is not proof that a run never existed.",
    ],
  };
}

function readFacts(db: Database.Database, input: GraphExplainInput) {
  const params = { task: input.taskId, board: input.board, tenant: input.tenantId, limit: LIMIT + 1 };
  const task = db.prepare("SELECT task_id,status,updated_at FROM tasks WHERE task_id=@task AND tenant_id IS @tenant").get(params) as TaskFact | undefined;
  if (!task) throw new Error("Task not found in the explicit tenant scope");
  const cards = db.prepare("SELECT card_id,task_id,status,blocked_reason,updated_at FROM kanban_cards WHERE task_id=@task AND board=@board AND tenant_id IS @tenant ORDER BY card_id LIMIT @limit").all(params) as CardFact[];
  const runs = db.prepare("SELECT run_id,task_id,phase,outcome,started_at,completed_at FROM runs WHERE task_id=@task AND tenant_id IS @tenant ORDER BY started_at DESC,run_id LIMIT @limit").all(params) as RunFact[];
  // Query the full scoped task relation, not the truncated cards projection.
  const relations = db.prepare(`WITH task_cards AS (
      SELECT card_id FROM kanban_cards WHERE task_id=@task AND board=@board AND tenant_id IS @tenant
    ), candidate_links AS (
      SELECT l.parent_id,l.child_id FROM task_cards t JOIN kanban_card_links l ON l.parent_id=t.card_id
      UNION
      SELECT l.parent_id,l.child_id FROM task_cards t JOIN kanban_card_links l ON l.child_id=t.card_id
    )
    SELECT l.parent_id,l.child_id,p.task_id AS parent_task_id,c.task_id AS child_task_id,
    p.status AS parent_status,c.status AS child_status,p.blocked_reason AS parent_blocked_reason,c.blocked_reason AS child_blocked_reason
    FROM candidate_links l JOIN kanban_cards p ON p.card_id=l.parent_id JOIN kanban_cards c ON c.card_id=l.child_id
    WHERE p.board=@board AND c.board=@board AND p.tenant_id IS @tenant AND c.tenant_id IS @tenant
    ORDER BY l.parent_id,l.child_id LIMIT @limit`).all(params) as RelationFact[];
  return { task, cards: bounded(cards), runs: bounded(runs), relations: bounded(relations), observedAt: new Date().toISOString() };
}

function parseArgs(args: string[]): GraphExplainInput & { json: boolean } {
  if (args[0] !== "explain") throw new Error(HELP);
  const values = new Map<string, string>();
  const flags = new Set<string>();
  for (let index = 1; index < args.length; index++) {
    const key = args[index] as string;
    if (values.has(key) || flags.has(key)) throw new Error(`Repeated flag: ${key}`);
    if (key === "--json" || key === "--default-tenant") { flags.add(key); continue; }
    if (!["--task", "--db", "--board", "--state", "--tenant"].includes(key)) throw new Error(`Unknown argument: ${key}`);
    const value = args[++index];
    if (!value || value.startsWith("--")) throw new Error(`Missing value: ${key}`);
    values.set(key, value);
  }
  function required(key: string) { const value = values.get(key); if (!value) throw new Error(`Required: ${key}`); return value; }
  if (values.has("--tenant") === flags.has("--default-tenant")) throw new Error("Specify exactly one of --tenant ID or --default-tenant");
  return { taskId: required("--task"), dbPath: required("--db"), board: required("--board"), statePath: required("--state"), tenantId: flags.has("--default-tenant") ? null : required("--tenant"), json: flags.has("--json") };
}

function renderText(result: ReturnType<typeof explainGraph>) {
  // JSON quoting keeps newlines/terminal controls in stored text inert.
  const show = (value: unknown) => JSON.stringify(value);
  return [
    `Task ${show(result.task.task_id)}: DB status=${show(result.task.status)}`,
    `Scope: board=${show(result.scope.board)} tenant=${show(result.scope.tenant_id)}`,
    `Database: ${show(result.sources.database.path)} (${result.sources.database.observed_at})`,
    `Loop: ${show(result.sources.loop.path)} updated=${show(result.loop.updated_at)} sha256=${result.loop.sha256}`,
    `Loop binding: ${result.loop.binding}`,
    `Loop coverage by location: ${show(result.loop.locations)}`,
    ...result.loop.matches.items.map((match) => `  Loop ${show(match)}`),
    ...result.cards.items.map((card) => `  Card ${show(card)}`),
    ...result.runs.items.map((run) => `  Run ${show(run)}`),
    ...result.relations.items.map((relation) => `  Stored link ${show(relation)}`),
    `Coverage: ${show(result.coverage)}`,
    `Notices: ${show(result.notices)}`,
    ...result.limitations.map((limitation) => `Note: ${limitation}`),
  ].join("\n");
}
