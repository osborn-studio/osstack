export const RemoteCancelGrantBody = z
  .object({
    domain: z.literal("osstack.remote-graph-cancel.v1"),
    grant_id: id,
    request_id: id,
    actor_id: ref,
    home_id: z.string().regex(/^home_[a-f0-9]{32}$/),
    daemon_generation: z.string().regex(/^instance_[a-f0-9]{32}$/),
    server_registry_sha256: digest,
    scope_alias: ref,
    repo_path: z.string().min(1).max(4096),
    graph_scope: GraphScope,
    action: z.literal("graph.cancel"),
    graph_run_id: ref,
    graph_generation: z.number().int().safe().nonnegative(),
    expected_revision: z.number().int().safe().nonnegative(),
    nonce: digest,
    issued_at_ms: z.number().int().safe().nonnegative(),
    expires_at_ms: z.number().int().safe().nonnegative(),
    max_calls: z.literal(1),
  })
  .strict()
  .refine((v) => v.expires_at_ms > v.issued_at_ms && v.expires_at_ms - v.issued_at_ms <= 300000);
export const SignedRemoteCancelGrant = z
  .object({
    body: RemoteCancelGrantBody,
    key_id: ref,
    signature: z.string().regex(/^[A-Za-z0-9+/]{85}[AQgw]==$/),

export const remoteCancelDigest = (value: unknown) =>
  createHash("sha256").update(deliveryCanonical(value)).digest("hex");
export function verifyRemoteCancelGrant(
  grant: SignedRemoteCancelGrant,
  registry: RemoteCancelServerRegistry,
  registryDigest: string,
  forWrite: boolean,
  now = Date.now(),
): void {
  const b = grant.body,
    key = registry.keys.find((k) => k.key_id === grant.key_id),
    scope = registry.scopes.find((s) => s.alias === b.scope_alias);
  if (
    !key ||
    !scope ||
    b.home_id !== registry.home_id ||
    b.server_registry_sha256 !== registryDigest ||
    b.actor_id !== key.actor_id ||
    !key.scope_aliases.includes(b.scope_alias) ||
    scope.repo_path !== b.repo_path ||
    deliveryCanonical(scope.graph_scope) !== deliveryCanonical(b.graph_scope) ||
    b.expires_at_ms - b.issued_at_ms > key.max_validity_ms ||
    !key.public_key_pem.startsWith("-----BEGIN PUBLIC KEY-----")
  )
    throw Error("remote_cancel_authorization_refused");
  const publicKey = createPublicKey(key.public_key_pem);
  if (
    publicKey.asymmetricKeyType !== "ed25519" ||
    !verify(null, Buffer.from(deliveryCanonical(b)), publicKey, Buffer.from(grant.signature, "base64"))
  )
    throw Error("remote_cancel_authorization_refused");
  // Exact historical receipt query remains possible after expiry; it never replays an effect.
  if (forWrite && (!Number.isSafeInteger(now) || now < b.issued_at_ms || now >= b.expires_at_ms))
    throw Error("remote_cancel_grant_expired");
}
export async function assertRemoteJournal(path: string): Promise<void> {
  if (!isAbsolute(path) || resolve(path) !== path || (await realpath(path)) !== path)
    throw Error("remote_cancel_journal_unavailable");
  const info = await lstat(path);
  if (!info.isDirectory() || info.uid !== process.getuid?.() || (info.mode & 0o077) !== 0)
    throw Error("remote_cancel_journal_unavailable");
  for (let p = dirname(path); ; p = dirname(p)) {
    const st = await lstat(p);
    if (
      !st.isDirectory() ||
      st.isSymbolicLink() ||
      (st.uid !== 0 && st.uid !== process.getuid?.()) ||
      ((st.mode & 0o022) !== 0 && !(st.uid === 0 && (st.mode & 0o1000) !== 0))
    )
      throw Error("remote_cancel_journal_unavailable");
    if (dirname(p) === p) break;
  }
}
export async function initializeRemoteJournal(home: string): Promise<string> {
  const dir = join(home, "remote-actions");
  await mkdir(dir, { mode: 0o700 }).catch((e) => {
    if ((e as NodeJS.ErrnoException).code !== "EEXIST") throw e;
  });
  await assertRemoteJournal(dir);
  return dir;
}
/** Create-only, file + directory fsync; a partial/crashed record is never overwritten or rearmed. */
export async function writeRemoteJournal(path: string, value: unknown): Promise<boolean> {
  await assertRemoteJournal(dirname(path));
  const entries = await opendir(dirname(path));
  let count = 0;
  try {
    for await (const _ of entries) {
      if (++count > 2048) throw Error("remote_cancel_journal_full");
    }
  } finally {
    await entries.close().catch(() => {});
  }
  const bytes = Buffer.from(JSON.stringify(value));
  if (bytes.length > 32768) throw Error("remote_cancel_journal_too_large");
  let file: Awaited<ReturnType<typeof open>>;
  try {
    file = await open(
      path,
      constants.O_WRONLY | constants.O_CREAT | constants.O_EXCL | constants.O_NOFOLLOW,
      0o600,
    );
  } catch (e) {
    if ((e as NodeJS.ErrnoException).code === "EEXIST") return false;
    throw e;
  }
  try {
    await file.writeFile(bytes);
    await file.sync();
  } finally {
    await file.close();
  }
  const directory = await open(dirname(path), constants.O_RDONLY);
  try {
    await directory.sync();
  } finally {
    await directory.close();
  }
  return true;
}
export async function readRemoteJournal(path: string): Promise<unknown | null> {
  try {
    return JSON.parse(
      new TextDecoder("utf-8", { fatal: true }).decode(await readPrivateDeliveryFile(path, 32768)),
    );
  } catch (e) {
    if ((e as NodeJS.ErrnoException).code === "ENOENT") return null;
    throw Error("remote_cancel_journal_unavailable");
  }
}

async function registry(path: string, digest: string): Promise<RemoteCancelServerRegistry> {
  if (!/^[a-f0-9]{64}$/.test(digest)) throw Error("remote_cancel_registry_invalid");
  const raw = await readPrivateDeliveryFile(path, 32768);
  if (remoteHash(raw) !== digest) throw Error("remote_cancel_registry_changed");
  const value = RemoteCancelServerRegistry.parse(JSON.parse(raw.toString("utf8")));
  await assertRemoteJournal(value.journal_dir);
  for (const scope of value.scopes) {
    for (const protectedPath of [path, value.journal_dir, value.socket]) {
      const rel = relative(scope.repo_path, protectedPath);
      if (rel === "" || (!isAbsolute(rel) && rel !== ".." && !rel.startsWith(`..${sep}`)))
        throw Error("remote_cancel_registry_in_repository");
    }
    if (
      (await realpath(scope.repo_path)) !== scope.repo_path ||
      (await lstat(scope.repo_path)).isDirectory() !== true
    )
      throw Error("remote_cancel_scope_unavailable");
  }
  return value;
}
/** One connected existing daemon: all identity/status/cancel frames stay bound to that same process. */
async function graphConnection(socketPath: string) {
  if ((await realpath(socketPath)) !== socketPath) throw Error("remote_cancel_endpoint_invalid");
  const before = await lstat(socketPath);
  if (!before.isSocket() || before.uid !== process.getuid?.() || (before.mode & 0o022) !== 0)
    throw Error("remote_cancel_endpoint_invalid");
  const socket = connect(socketPath),
    channel = new RemoteChannel(socket, socket, 65536 + 1024, 8);
  let id = 0;
  return {
    async call(
      method: "deployment.identity" | "graph.status" | "graph.cancel",
      params?: unknown,
    ): Promise<unknown> {
      const requestId = ++id;
      await channel.send({ id: requestId, method, ...(params === undefined ? {} : { params }) });
      const raw = await channel.receive();
      const current = await lstat(socketPath);
      if (current.ino !== before.ino || current.dev !== before.dev)
        throw Error("remote_cancel_endpoint_changed");
      return z
        .object({ id: z.literal(requestId), ok: z.literal(true), result: z.unknown() })
        .strict()
        .parse(raw).result;
    },
    close() {
      channel.close();
      socket.destroy();
    },
  };
}
export async function handleRemoteControlExportCommand(
  args: string[],
  deps: { input?: Readable; output?: Writable; env?: NodeJS.ProcessEnv } = {},
): Promise<number> {
  if (args.length !== 1 || args[0] !== "--stdio") return 1;
  const env = deps.env ?? process.env,
    input = deps.input ?? process.stdin,
    output = deps.output ?? process.stdout;
  const path = env["OSSTACK_REMOTE_CONTROL_REGISTRY"] ?? "",
    pin = env["OSSTACK_REMOTE_CONTROL_SHA256"] ?? "";
  let rpc: Awaited<ReturnType<typeof graphConnection>> | undefined;
  let channel: RemoteChannel | undefined;
  let expired = false;
  const timer = setTimeout(() => {
    expired = true;
    rpc?.close();
    channel?.close();
    input.pause();
  }, 10000);
  const active = () => {
    if (expired) throw Error("remote_cancel_timeout");
  };
  try {
    const config = await registry(path, pin);
    channel = new RemoteChannel(input, output);
    const hello = RemoteCancelHello.parse(await channel.receive());
    if (hello.home_id !== config.home_id || hello.registry_sha256 !== pin)
      throw Error("remote_cancel_identity_mismatch");
    const token = randomBytes(32).toString("hex");
    await channel.send({
      type: "control_grant",
      protocol: 1,
      nonce: hello.nonce,
      home_id: config.home_id,
      daemon_generation: "unobserved",
      registry_sha256: pin,
      token,
      methods: ["cancel", "query"],
    });
    const request = RemoteCancelRequest.parse(await channel.receive());
    if (request.token !== token) throw Error("remote_cancel_token_invalid");
    const grant = request.grant,
      b = grant.body,
      hash = remoteCancelDigest(grant);
    verifyRemoteCancelGrant(grant, config, pin, false);
    const startedPath = join(config.journal_dir, `${b.request_id}.started.json`),
      resultPath = join(config.journal_dir, `${b.request_id}.result.json`);
    const base = {
      type: "control_receipt" as const,
      protocol: 1 as const,
      nonce: hello.nonce,
      home_id: config.home_id,
      request_id: b.request_id,
      request_sha256: hash,
      daemon_generation: b.daemon_generation,
      graph_generation: b.graph_generation,
      graph_run_id: b.graph_run_id,
      physical_stop: "not_attested" as const,
      observed_revision: null,
    };
    const existing = await readRemoteJournal(startedPath);
    if (existing && deliveryCanonical(existing) !== deliveryCanonical({ request_sha256: hash, grant }))
      throw Error("remote_cancel_idempotency_conflict");
    const prior = await readRemoteJournal(resultPath);
    if (prior) {
      const receipt = RemoteCancelReceipt.parse(prior);
      if (
        !existing ||
        receipt.request_sha256 !== hash ||
        receipt.home_id !== b.home_id ||
        receipt.request_id !== b.request_id ||
        receipt.daemon_generation !== b.daemon_generation ||
        receipt.graph_generation !== b.graph_generation ||
        receipt.graph_run_id !== b.graph_run_id
      )
        throw Error("remote_cancel_receipt_invalid");
      await channel.send({ ...receipt, nonce: hello.nonce });
      return 0;
    }
    if (request.type === "query" || existing) {
      await channel.send({ ...base, outcome: existing ? "delivery_unknown" : "not_found" });
      return 0;
    }
    verifyRemoteCancelGrant(grant, config, pin, true);
    active();
    rpc = await graphConnection(config.socket);
    active();
    const identity = DeploymentIdentitySchema.parse(await rpc.call("deployment.identity"));
    if (
      identity.home_id !== b.home_id ||
      identity.generation !== b.daemon_generation ||
      identity.protocol_version !== config.daemon_protocol
    )
      throw Error("remote_cancel_identity_mismatch");
    const status = GraphDelegationStatus.parse(
      await rpc.call("graph.status", { schemaVersion: 1, scope: b.graph_scope, graphRunId: b.graph_run_id }),
    );
    if (
      status.graphRunId !== b.graph_run_id ||
      deliveryCanonical(status.scope) !== deliveryCanonical(b.graph_scope) ||
      status.generation !== b.graph_generation ||
      status.revision !== b.expected_revision
    )
      throw Error("remote_cancel_generation_mismatch");
    // Registry revocation and grant time are rechecked immediately before durable reserve.
    await registry(path, pin);
    verifyRemoteCancelGrant(grant, config, pin, true);
    active();
    if (!(await writeRemoteJournal(startedPath, { request_sha256: hash, grant }))) {
      const other = await readRemoteJournal(startedPath);
      if (deliveryCanonical(other) !== deliveryCanonical({ request_sha256: hash, grant }))
        throw Error("remote_cancel_idempotency_conflict");
      await channel.send({ ...base, outcome: "delivery_unknown" });
      return 0;
    }
    let receipt: RemoteCancelReceipt = { ...base, outcome: "delivery_unknown" };
    try {
      await registry(path, pin);
      verifyRemoteCancelGrant(grant, config, pin, true);
      active();
      const result = GraphDelegationStatus.parse(
        await rpc.call("graph.cancel", {
          schemaVersion: 1,
          kind: "cancel",
          scope: b.graph_scope,
          graphRunId: b.graph_run_id,
          expectedRevision: b.expected_revision,
          idempotencyKey: `remote-cancel:${b.request_id}`,
        }),
      );
      if (
        result.graphRunId !== b.graph_run_id ||
        deliveryCanonical(result.scope) !== deliveryCanonical(b.graph_scope) ||
        result.generation !== b.graph_generation ||
        result.revision < b.expected_revision
      )
        throw Error("remote_cancel_receipt_invalid");
      receipt = { ...base, outcome: "stop_requested", observed_revision: result.revision };
    } catch {
      /* The durable reserve never re-arms, including a crash or lost daemon response. */
    }
    await writeRemoteJournal(resultPath, receipt);
    await channel.send(receipt);
    return 0;
  } catch {
    return 1;
  } finally {
    clearTimeout(timer);
    rpc?.close();
    channel?.close();
    input.pause();
  }
}

export const REMOTE_CANCEL_HELP = `remote-home cancel|query --registry FILE --sha256 HASH --alias NAME --grant FILE\nSigned exact Graph cancellation only. Local durable attempt precedes transport; repeat cancel becomes query, never resend.\nRemote stop_requested is not physical stopped proof. Missing/offline receipts remain unknown.\n`;
export async function handleRemoteCancelCommand(
  args: string[],
  deps: RemoteCancelClientDependencies,
): Promise<number> {
  const out =
      deps.stdout ??
      ((text) => {
        process.stdout.write(text);
      }),
    err =
      deps.stderr ??
      ((text) => {
        process.stderr.write(text);
      });
  if (args.length === 1 && args[0] === "--help") {
    out(REMOTE_CANCEL_HELP);
    return 0;
  }
  let child: ChildProcessWithoutNullStreams | undefined,
    channel: RemoteChannel | undefined,
    dir: string | undefined,
    timer: NodeJS.Timeout | undefined;
  let unknown: Record<string, unknown> | undefined;
  try {
    if (
      !["cancel", "query"].includes(args[0]!) ||
      args.length !== 9 ||
      args.some((a) => Buffer.byteLength(a) > 4096)
    )
      throw Error();
    const flags = new Map<string, string>();
    for (let i = 1; i < args.length; i += 2) {
      const key = args[i]!,
        value = args[i + 1];
      if (!["--registry", "--sha256", "--alias", "--grant"].includes(key) || flags.has(key) || !value)
        throw Error();
      flags.set(key, value);
    }
    const registry = RemoteClientRegistry.parse(
      await readPinnedRemoteRegistry(flags.get("--registry") ?? "", flags.get("--sha256") ?? ""),
    );
    const entry = registry.homes.find((e) => e.alias === flags.get("--alias"));
    if (!entry) throw Error();
    const grant = SignedRemoteCancelGrant.parse(
        JSON.parse((await readPrivateDeliveryFile(flags.get("--grant") ?? "", 32768)).toString("utf8")),
      ),
      b = grant.body,
      digest = remoteCancelDigest(grant);
    if (
      b.home_id !== entry.home_id ||
      b.server_registry_sha256 !== entry.registry_sha256 ||
      b.scope_alias !== entry.scope
    )
      throw Error();
    const journal = await initializeRemoteJournal(deps.home),
      recordPath = join(journal, `${b.request_id}.local.json`);
    const record = {
      alias: entry.alias,
      endpoint_sha256: remoteCancelDigest(entry),
      request_sha256: digest,
      grant,
    };
    const previous = await readRemoteJournal(recordPath);
    if (previous && deliveryCanonical(previous) !== deliveryCanonical(record)) throw Error();
    let action = args[0] as "cancel" | "query";
    if (action === "cancel") {
      if (previous) action = "query";
      else if (!(await writeRemoteJournal(recordPath, record))) {
        if (deliveryCanonical(await readRemoteJournal(recordPath)) !== deliveryCanonical(record))
          throw Error();
        action = "query";
      }
    }
    unknown = {
      schema_version: 1,
      mode: "remote_graph_cancel",
      alias: entry.alias,
      request_id: b.request_id,
      request_sha256: digest,
      outcome: "delivery_unknown",
      physical_stop: "not_attested",
      qualification: "remote_host_pending",
    };
    await assertPrivateRemoteFile(entry.identity_file);
    const { createScratchDir } = await import("@osstack/core");
    dir = await createScratchDir("runtime", "remote-cancel");
    const known = join(dir, "known_hosts");
    await writeFile(
      known,
      `${entry.port === 22 ? entry.host : `[${entry.host}]:${entry.port}`} ${entry.host_key}\n`,
      { mode: 0o600, flag: "wx" },
    );
    child = (deps.spawn ?? spawn)("/usr/bin/ssh", buildRemoteControlSshArgs(entry, known), {
      stdio: ["pipe", "pipe", "pipe"],
      shell: false,
      env: { PATH: "/usr/bin:/bin", LANG: "C", LC_ALL: "C" },
    }) as ChildProcessWithoutNullStreams;
    channel = new RemoteChannel(child.stdout, child.stdin);
    let stderrBytes = 0;
    child.stderr.on("error", () => {
      child?.stdin.destroy();
      child?.stdout.destroy();
    });
    child.stderr.on("data", (chunk: Buffer) => {
      stderrBytes += chunk.length;
      if (stderrBytes > 8192) child?.kill("SIGKILL");
    });
    child.on("error", () => {
      child?.stdin.destroy();
      child?.stdout.destroy();
    });
    const active = channel,
      nonce = randomBytes(32).toString("hex"),
      timeout = Math.max(1, Math.min(10000, deps.timeoutMs ?? 10000));
    const exchange = async () => {
      await active.send({
        type: "control_hello",
        protocol: 1,
        nonce,
        home_id: entry.home_id,
        registry_sha256: entry.registry_sha256,
      });
      const handshake = RemoteCancelHandshake.parse(await active.receive(timeout));
      if (
        handshake.nonce !== nonce ||
        handshake.home_id !== entry.home_id ||
        handshake.registry_sha256 !== entry.registry_sha256
      )
        throw Error();
      await active.send({ type: action, token: handshake.token, grant });
      const receipt = RemoteCancelReceipt.parse(await active.receive(timeout));
      if (
        receipt.nonce !== nonce ||
        receipt.home_id !== b.home_id ||
        receipt.request_id !== b.request_id ||
        receipt.request_sha256 !== digest ||
        receipt.daemon_generation !== b.daemon_generation ||
        receipt.graph_generation !== b.graph_generation ||
        receipt.graph_run_id !== b.graph_run_id
      )
        throw Error();
      await writeRemoteJournal(
        join(journal, `${b.request_id}.observed-${remoteCancelDigest({ ...receipt, nonce: "" })}.json`),
        receipt,
      );
      return receipt;
    };
    const receipt = await Promise.race([
      exchange(),
      new Promise<never>((_, reject) => {
        timer = setTimeout(() => reject(Error("remote_cancel_timeout")), timeout);
      }),
    ]);
    out(`${JSON.stringify({ ...unknown, outcome: receipt.outcome, receipt })}\n`);
    return receipt.outcome === "stop_requested" ? 0 : 1;
  } catch {
    if (unknown) out(`${JSON.stringify(unknown)}\n`);
    else err("invalid_remote_cancel_request\n");
    return 1;
  } finally {
    if (timer) clearTimeout(timer);
    channel?.close();
    if (child) {
      child.stdin.destroy();
      child.stdout.destroy();
      child.stderr.destroy();
      if (child.exitCode === null) child.kill("SIGKILL");
    }
    if (dir) await rm(dir, { recursive: true, force: true }).catch(() => {});
  }
}

export async function handleRemoteCancelIssuerCommand(
  args: string[],
  deps: RemoteCancelIssuerDependencies,
): Promise<number> {
  const out =
      deps.stdout ??
      ((s) => {
        process.stdout.write(s);
      }),
    err =
      deps.stderr ??
      ((s) => {
        process.stderr.write(s);
      });

    out(REMOTE_CANCEL_ISSUER_HELP);
    return 0;
  }
  try {
    if (
      args[0] !== "issue-cancel-grant" ||
      args.length !== 13 ||
      args.some((a) => Buffer.byteLength(a) > 4096)
    )
      throw Error();
    const flags = new Map<string, string>();
    for (let i = 1; i < args.length; i += 2) {
      const k = args[i]!,
        v = args[i + 1];
      if (
        ![
          "--target",
          "--daemon-generation",
          "--graph-run",
          "--graph-generation",
          "--revision",
          "--output",
        ].includes(k) ||
        !v ||
        flags.has(k)
      )
        throw Error();
      flags.set(k, v);
    }
    const configPath = join(deps.home, "remote-cancel-issuer.json"),
      raw = await readPrivateDeliveryFile(configPath, 32768),
      config = RemoteCancelIssuerConfig.parse(JSON.parse(raw.toString("utf8")));
    const target = config.targets.find((t) => t.name === flags.get("--target")),
      output = flags.get("--output");
    if (!target || !output || !isAbsolute(output)) throw Error();
    await assertRemoteJournal(dirname(output));
    const numeric = (key: string) => {
      const v = flags.get(key) ?? "";
      if (!/^(0|[1-9][0-9]*)$/.test(v)) throw Error();
      return Number(v);
    };
    const issued = Date.now();
    const body = RemoteCancelGrantBody.parse({
      domain: "osstack.remote-graph-cancel.v1",
      grant_id: randomUUID(),
      request_id: randomUUID(),
      actor_id: config.actor_id,

      server_registry_sha256: target.server_registry_sha256,
      scope_alias: target.scope_alias,
      repo_path: target.repo_path,
      graph_scope: target.graph_scope,
      action: "graph.cancel",
      graph_run_id: flags.get("--graph-run"),
      graph_generation: numeric("--graph-generation"),
      expected_revision: numeric("--revision"),
      nonce: randomBytes(32).toString("hex"),
      issued_at_ms: issued,
      expires_at_ms: issued + config.max_validity_ms,
      max_calls: 1,
    });
    const preview = {
        action: "remote Graph cancellation",
        target: target.name,
        key_id: config.key_id,
        grant: body,
        physical_stop: "not_attested",
        retry: "never; query same request after uncertainty",
      },

    if (
      !(await (deps.confirm ?? terminalConfirm)(`${JSON.stringify(preview, null, 2)}\n`, `APPROVE ${digest}`))
    )
      throw Error();
    if (
      !raw.equals(await readPrivateDeliveryFile(configPath, 32768)) ||
      Date.now() >= body.expires_at_ms ||
      Date.now() < body.issued_at_ms
    )
      throw Error();
    const key = createPrivateKey(
      await readPrivateDeliveryFile(join(deps.home, `issuer-${config.key_id}.pem`), 4096),
    );
    const expected = createPublicKey(config.public_key_pem);
    if (
      key.asymmetricKeyType !== "ed25519" ||
      expected.asymmetricKeyType !== "ed25519" ||
      !createPublicKey(key)
        .export({ format: "der", type: "spki" })
        .equals(expected.export({ format: "der", type: "spki" }))
    )
      throw Error();
    if (!raw.equals(await readPrivateDeliveryFile(configPath, 32768))) throw Error();

      body,
      key_id: config.key_id,
      signature: sign(null, Buffer.from(deliveryCanonical(body)), key).toString("base64"),
    });
    if (!(await writeRemoteJournal(output, grant))) throw Error();
    out(
      `${JSON.stringify({ schema_version: 1, status: "grant_created", request_id: body.request_id, preview_sha256: digest, output })}\n`,
    );
    return 0;
  } catch {
    err("remote_cancel_issuer_refused\n");
    return 1;
  }
}

/** Bounded channel: two sequential messages per direction, no generic RPC or buffering backlog. */
export class RemoteChannel {
  private bytes=Buffer.alloc(0);
  private queue: unknown[]=[];
  private failure: Error | null=null;
  private count=0;
  private sent=0;
  private closed=false;
  private pending: {resolve:(value:unknown)=>void;reject:(error:Error)=>void;timer:NodeJS.Timeout} | null=null;
  constructor(private readonly input:Readable,private readonly output:Writable,private readonly inputLimit=REMOTE_MAX_BYTES,private readonly maxFrames=2) {
    if(!Number.isInteger(maxFrames)||maxFrames<2||maxFrames>8)throw new Error("invalid_frame");
    if(!Number.isInteger(inputLimit)||inputLimit<REMOTE_MAX_BYTES||inputLimit>REMOTE_MAX_BYTES+1024)throw new Error("invalid_frame");
    input.on("data",this.data);input.on("end",this.end);input.on("error",this.error);output.on("error",this.error);
  }
  private error=()=>this.fail("transport_unavailable");
  private end=()=>this.fail("transport_closed");
  private data=(chunk:Buffer)=>{
    if(this.failure)return;
    if(this.bytes.length+chunk.length>this.inputLimit){this.fail("invalid_frame");return;}
    this.bytes=Buffer.concat([this.bytes,chunk]);
    if(this.bytes.length<4)return;
    const length=this.bytes.readUInt32BE();
    if(length===0||length>this.inputLimit-4){this.fail("invalid_frame");return;}
    if(this.bytes.length<length+4)return;
    if(this.bytes.length!==length+4||++this.count>this.maxFrames){this.fail("invalid_frame");return;}
    let parsed:unknown;
    try {parsed=JSON.parse(new TextDecoder("utf-8",{fatal:true}).decode(this.bytes.subarray(4)));}
    catch {this.fail("invalid_frame");return;}
    this.bytes=Buffer.alloc(0);
    if(this.pending){const p=this.pending;this.pending=null;clearTimeout(p.timer);p.resolve(parsed);}
    else if(this.queue.length===0)this.queue.push(parsed);else this.fail("invalid_frame");
  };
  private fail(reason:string){
    if(this.failure)return;this.failure=new Error(reason);
    if(this.pending){clearTimeout(this.pending.timer);this.pending.reject(this.failure);this.pending=null;}
  }
  async receive(timeoutMs=3000):Promise<unknown>{
    if(this.failure)throw this.failure;
    if(this.queue.length)return this.queue.shift();
    if(this.pending)throw new Error("invalid_frame");
    return new Promise((resolve,reject)=>{const timer=setTimeout(()=>this.fail("transport_timeout"),Math.max(1,Math.min(10000,timeoutMs)));this.pending={resolve,reject,timer};});
  }
  async send(value:unknown):Promise<void>{
    if(this.failure)throw this.failure;
    if(++this.sent>this.maxFrames)throw new Error("invalid_frame");
    const frame=encodeRemoteFrame(value);
    await new Promise<void>((resolve,reject)=>{
      const timer=setTimeout(()=>{this.fail("transport_timeout");reject(new Error("transport_timeout"));},3000);
      this.output.write(frame,error=>{clearTimeout(timer);error?reject(new Error("transport_unavailable")):resolve();});
    });
  }
  close(){if(this.closed)return;this.closed=true;this.fail("transport_closed");this.input.off("data",this.data);this.input.off("end",this.end);this.input.off("error",this.error);this.output.off("error",this.error);this.input.on("error",ignoreLateStreamError);if((this.output as object)!==(this.input as object))this.output.on("error",ignoreLateStreamError);}
}
