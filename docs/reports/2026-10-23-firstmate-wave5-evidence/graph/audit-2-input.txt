## Findings

**P0: 0 demonstrated.** I found no path that reaches `local_verified` without a DB-bound, host-written completion receipt, or that replaces a sealed contract run with legacy behavior.

**P1: 1 demonstrated, conditional on adapter behavior**

1. **The hashed prompt is not the object the adapter receives.**
   - Trigger: an adapter mutates `task.prompt` (or any field) during `invoke`.
   - Location: `graph-execution-context.ts`, `prepareGraphExecutionContext`, `observed.invoke`. It sets `exact = structuredClone(task)`, records it, then calls `yield* raw.invoke(task, signal)`, passing the mutable original. `execution-context.ts`, `createExecutionContextAdapters`, also passes an unfrozen `{ ...invocation, prompt }`.
   - Consequence: `evidence()` checks `promptDigest` against the clone, so `prompt_sha256`, `GraphContextEvidence.promptDigest`, and the completion receipt attest a prompt that may differ from what was sent. This is the exact "actual prompt vs stored hash" invariant the review asks about, and the current check cannot detect it.
   - Minimal correction: deep-freeze the clone and pass that frozen object to `raw.invoke`. Deep-freeze the assembled task in the wrapper before `adapter.invoke`.

**P2: 4 demonstrated**

2. **Zero-event adapter streams break a successful action.**
   - Trigger: an adapter whose generator completes without yielding.
   - Location: `execution-context.ts` wrapper. `adapter_observed` is persisted only on the first yielded step. In that case `receiptRefs()` has one entry, so `evidence()` throws `GRAPH_CONTEXT_RECEIPT_COUNT` inside writer or review `persist`.
   - Correction: after the loop, persist the observed receipt if it has not been written yet, before returning.

3. **Receipt counting ignores the run.**
   - Trigger: a second invocation through the same wrapper instance.
   - Location: `graph-execution-context.ts`, `evidence()`. It calls `context.receiptRefs()` without a filter and requires `refs.length === 2`.
   - Consequence: a valid pair for the current run fails, or the count is coupled to the instance rather than the attempt.
   - Correction: call `receiptRefs({ run_id: runId })`, and widen the registration's `receiptRefs` type to accept the filter.

4. **Context receipts do not bind to the effect.**
   - Trigger: `run_id` reuse across attempts, or any restart that drops the in-memory `GraphDelivery.contextEvidence` map.
   - Location: `execution-context.ts` `ExecutionContextReceipt` has no `effectKey`, `epoch`, or `logicalAttempt`. `graph-contract-receipt.ts`, `publishGraphContractCompletion`, checks only run, task, session, prompt, and contract digest.
   - Consequence: G2.6 is only partly met. The effect link lives in host memory and the completion row, not in the verifiable receipt bytes.
   - Correction: pass the effect identity into the wrapper and include `effectKey`, `epoch`, and `logicalAttempt` in both phase receipts. Check them in `publishGraphContractCompletion`.

5. **Public error mapping is inconsistent.**
   - Location: `graph-rpc.ts`, `graphRpcError`. `GRAPH_CONTRACT_*` maps to `POLICY_DENIED`, while `GRAPH_CONTEXT_*` falls through to `INTERNAL_ERROR`. A context authority revoke therefore looks like an internal fault.
   - Correction: map `GRAPH_CONTEXT_AUTHORITY` to the same authority code as the contract path.

    review: { verdict: 'accept', writerIdentity: review.envelope.writerIdentity, reviewerIdentity: review.envelope.reviewerIdentity,
      authority: 'registered_host_reviewer', promptDigest: reviewReceipt.promptDigest, effectKey: review.effectKey, evidenceRef: review.ref, logicalAttempt: review.logicalAttempt },
    achievedCompletion: 'local_verified', publication: 'artifact_only', contextEvidence: selectedContext,
    ...(envelope.contextDigest ? { contextDigest: envelope.contextDigest } : {}) });
  check();
  const root = join(input.root, 'contract-receipts'); const saved = await persistDeliveryHostRecord(root, receipt); check();
  // This record must be durable before the enclosing host transaction publishes its reference.
  for (const path of [saved.path, root, input.root]) {
    const handle = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW);
    try { check(); await handle.sync(); check(); } finally { await handle.close(); }
  }
  const actual = await readDeliveryHostRecord(root, saved); check();
  if (workerCanonical(actual) !== workerCanonical(receipt)) throw Error('GRAPH_CONTRACT_RECEIPT_CHANGED');
  return `graph-contract:${saved.digest}`;
}

/** A receipt file alone is not completion: caller must obtain ref from the ready DB projection. */
export async function loadGraphContractCompletion(root: string, ref: string, expected: {
  run: GraphRun; envelope: GraphContractHostEnvelope; joinRef: string; candidateRevision: string; artifactDigest: string;
}): Promise<GraphContractCompletionReceipt> {
  if (!/^graph-contract:[a-f0-9]{64}$/.test(ref) || expected.run.status !== 'ready') throw Error('GRAPH_CONTRACT_RECEIPT_UNPUBLISHED');
  const digest = ref.slice('graph-contract:'.length); const directory = join(root, 'contract-receipts');
  const receipt = GraphContractCompletionReceipt.parse(await readDeliveryHostRecord(directory, { path: join(directory, `${digest}.json`), digest }));
  if (receipt.graphRunId !== expected.run.graphRunId || receipt.inputDigest !== expected.run.identity.inputDigest
    || receipt.specHash !== expected.run.identity.specHash || receipt.generation !== expected.run.generation
    || workerCanonical(receipt.scope) !== workerCanonical(expected.run.scope)
    || receipt.contractDigest !== expected.envelope.frozen.digest || receipt.policyDigest !== expected.envelope.policy.policyDigest
    || receipt.contextDigest !== expected.envelope.contextDigest || receipt.contextEvidence.length !== (expected.envelope.contextDigest ? 2 : 0)
    || receipt.joinRef !== expected.joinRef || receipt.candidate.revision !== expected.candidateRevision || receipt.candidate.artifactDigest !== expected.artifactDigest)
    throw Error('GRAPH_CONTRACT_RECEIPT_BINDING');
  return freezeWorkerData(receipt);
}

/** Host policy is already authorized by the registration. This function grants nothing. */
export function bindGraphDeliveryContract(requestValue: StartGraphRequest, registration: {
  repositoryPath: string; task: string; acceptanceRegistry: readonly AcceptanceCommandSpec[];
  deliveryContract: GraphDeliveryContractRegistration;
  executionContext?: { configDigest: string };

    || contract.publication_scope.mode !== 'none' || contract.merge_authority.mode !== 'none'
    || contract.required_checks.length !== 1) throw Error('GRAPH_CONTRACT_UNSUPPORTED');

        } }, persist: async evidence => ({ evidenceRef: await this.persist(n.id, this.options.repositoryPath, evidence), actualTokens: 0,
          result: { schemaVersion: 1, kind: 'workspace', workspaceLeaseRef, receiptRef: `e${epoch}.workspace` } }) };

    if (name === 'implement' || name === 'repair') {

      if (epoch && !feedback) throw Error('IMMUTABLE_FEEDBACK_REQUIRED');

        candidateRevision: this.candidate(epoch - 1).candidateRevision, feedbackRef: this.feedbackRef! } : { kind: 'implement' as const, inputArtifactRef: this.options.request.inputArtifactRef, workspaceLeaseRef };
      const rawAdapter = this.options.writer.adapter;
      const context = this.options.executionContext ? await prepareGraphExecutionContext({ registration: this.options.executionContext,
        adapter: rawAdapter, workspacePath: this.options.repositoryPath, frozen: this.options.deliveryContract!.frozen, authorize: () => this.check() }) : undefined;
      const adapter = context?.adapter ?? rawAdapter;
      const capability = adapter.capabilities().find(c => c.id === this.options.writer.capabilityId); if (!capability) throw Error('WRITER_CAPABILITY_REQUIRED');
      return { ...common, workspace, action, execution: { kind: 'adapter', adapter, capabilityId: capability.id, plan: {
        plan_id: `delivery-${r.graphRunId}-${epoch}`, agent_id: adapter.id, max_iterations: 1,
        task: { task_id: `delivery-${epoch}`, source: 'manual', capability: capability.kind, priority: 100,
          prompt: `${this.options.task}${this.options.deliveryContract && !context ? `\n\n${this.options.deliveryContract.frozen.brief}` : ''}${feedback ? `\nRepair frozen candidate ${action.kind === 'repair' ? action.candidateRevision : ''}. Complete host quality feedback:\n${feedback.bundle_json}` : ''}` },
        stop_condition: { type: 'tests_pass' },
      } }, persist: async evidence => {
        if (context && evidence.status === 'completed' && evidence.stopped) {
          const action = this.latest(n.id)!; const recorded = await context.evidence(evidence.candidate?.execution?.run_id ?? '', { effect: action.effect, effectKey: action.effectKey });
          this.check(); this.contextEvidence.set(action.effectKey, recorded);
        }
        return { evidenceRef: await this.persist(n.id, workspace.workspace_path, evidence),
          ...(evidence.status === 'completed' && evidence.stopped ? { result: { schemaVersion: 1 as const, kind: 'implementation_complete' as const, workspaceLeaseRef, receiptRef: `e${epoch}.implementation` } } : {}) };
      } };
    }
    if (name === 'freeze') {
      const implementation = this.latest(`e${epoch}.${epoch ? 'repair' : 'implement'}`);
      if (implementation?.outcome?.state !== 'confirmed' || implementation.outcome.result.kind !== 'implementation_complete') throw Error('IMPLEMENTATION_REQUIRED');
      const workspace = this.workspace(); const scope = await resolveRepositoryScope(workspace.workspace_path);
      const checkProducer = async () => { const evidence = await loadDeliveryReceipt(this.options.root, implementation.outcome!.evidenceRef,
        this.receiptBinding(implementation.effect.nodeId, workspace.workspace_path));
        if (evidence.stopped !== true || workerCanonical(evidence.candidate?.workspace) !== workerCanonical(workspace)) throw Error('PRODUCER_STOP_REQUIRED'); this.check();
        if (this.options.deliveryContract) assertGraphContractPaths(this.options.deliveryContract.frozen, workspace.workspace_path); };
      return { ...common, workspace, action: { kind: 'freeze', workspaceLeaseRef, receiptRef: implementation.outcome.result.receiptRef },
        execution: { kind: 'freeze', store: this.artifacts, input: { artifactId: `delivery-${r.graphRunId}-${epoch}`, scope: r.scope, runId: r.graphRunId,
          generation: r.generation, owner: r.leaseOwner!, workspaceLeaseRef, repoId: scope.repoId, commonDir: scope.commonDir, worktreePath: workspace.workspace_path,
          baseRevision: epoch ? this.candidate(epoch - 1).candidateRevision : this.compiled.spec.input.baseRevision,
          implementationReceiptRef: implementation.outcome.result.receiptRef, producerStopEvidenceRef: implementation.outcome.evidenceRef },
        authority: { verifyLease: async (_input, lease) => { this.check(); await (lease as ExecutionLease).assertActive!(); }, verifyImplementationReceipt: checkProducer, verifyProducersStopped: checkProducer } },
        persist: async evidence => { return { evidenceRef: await this.persist(n.id, workspace.workspace_path, evidence), actualTokens: 0, result: evidence.frozenCandidate! }; } };
    }
    if (name === 'prepare_review' || name === 'prepare_tests') return this.preparation(name === 'prepare_review' ? 'review' : 'tests', 1);
    const action = this.branches.get(n.id); if (!action) throw Error('PREPARATION_REQUIRED'); return action;
  }
  private request(action: GraphRegisteredAction) { const r = this.current(); return { schemaVersion: 1 as const, kind: 'worker_action' as const, requestId: `host:${randomUUID()}`,
    scope: r.scope, graphRunId: r.graphRunId, specHash: r.identity.specHash, generation: r.generation, epoch: this.epoch,
    nodeId: action.nodeId, effectSlot: action.effectSlot, action: action.action }; }
  private async dispatch(connection: object, value: WorkerActionRequest, options: { signal?: AbortSignal }): Promise<WorkerActionResponse> {
    if (this.stopping) throw Error('DELIVERY_STOPPING');
    this.check(); const request = WorkerActionRequest.parse(value); const r = this.current();
    if (request.epoch !== this.epoch || request.generation !== r.generation || request.graphRunId !== r.graphRunId || request.specHash !== r.identity.specHash || workerCanonical(request.scope) !== workerCanonical(r.scope)) throw Error('DELIVERY_IDENTITY_MISMATCH');
    const name = request.nodeId.replace(`e${this.epoch}.`, ''); const n = deliveryEpoch(this.compiled, this.epoch).node(name);
    if (request.effectSlot !== n.effectSlot) throw Error('DELIVERY_SLOT_MISMATCH');
    if (!this.installed.has(n.id)) this.installed.set(n.id, this.registration(name).then(action => { this.check(); this.broker.installActions(connection, [action]); }));
    await this.installed.get(n.id);
    let response = await this.broker.handle(connection, request, options);
    if (name !== 'review' && name !== 'tests') return response;
    // The host inspects the actual attested failure. The task never caches failed RETURN.
    while (response.kind === 'worker_action_response' && response.outcome.state === 'failed' && response.outcome.error.retryable) {
      const prior = this.latest(n.id)!;
      if (prior.effect.logicalAttempt >= n.maxLedgerAttempts) break;
      if (!prior.outcome?.evidenceRef.startsWith('graph-verification:')) break;
      const envelope = await loadGraphVerificationEvidence(join(this.options.root, 'verification'), prior.outcome!.evidenceRef, this.binding(), { effect: prior.effect, effectKey: prior.effectKey });
      if (!envelope.verificationFailure?.retrySafe) break;
      const snapshot = new GraphVerificationStore(this.options.db, () => {}).snapshot(this.fence());
      const cost = deliveryEpoch(this.compiled, this.epoch).nextCost[name];
      if (snapshot.budget.accountingBlocked || snapshot.budget.actionStarts + cost.actions > this.compiled.spec.limits.maxActionStarts
        || BigInt(snapshot.budget.chargedTokens) + BigInt(snapshot.budget.reservedTokens) + BigInt(cost.tokens) > BigInt(this.compiled.spec.limits.tokenReservationLimit)) break;
      const prep = await this.preparation(name, prior.effect.logicalAttempt + 1);
      this.broker.installActions(connection, [prep]);
      const prepared = await this.broker.handle(connection, this.request(prep), options);
      if (prepared.kind !== 'worker_action_response' || prepared.outcome.state !== 'confirmed') return response;
      response = await this.broker.retryAuthorized(connection, prior.effectKey, () => { this.check(); }, { ...options, replacement: this.branches.get(n.id)! });
      if (response.kind === 'worker_action_response') response = { ...response, requestId: request.requestId };
    }
    return response;
  }
  private async evaluate(): Promise<GraphVerificationDecision> {
    const p = deliveryEpoch(this.compiled, this.epoch);
    return new GraphVerificationJoinService(this.options.db, { binding: this.binding(), artifacts: this.artifacts, evidenceRoot: join(this.options.root, 'verification'),
      slots: p.slots, nextCost: p.nextCost, authorize: () => this.check() }).evaluate(this.fence());
  }
  private async takeover(handle: GraphRuntimeHandle, stopped: GraphRuntimeStop): Promise<void> {
    const r = this.current(); const key = {};
    const inspection = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); });
    let inventory: GraphExecutorInventory | undefined;
    const verify = () => { const observed = this.runtime.observedExit(handle.connection);
      if (!stopped.actionsSettled || stopped.worker.kind !== 'exited' || stopped.worker.journalError || workerCanonical(observed) !== workerCanonical(stopped.worker)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
      return verifyJournaledGraphWorkerExit(this.options.db, handle.binding);
    };
    this.runtime.reconciler.register(key, { binding: { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity, generation: r.generation, owner: r.leaseOwner },
      authorize: () => this.check(), stopWorker: async () => stopped,
      verifyWorker: async () => verify(),
      verifyQuiescence: async snapshot => { verify(); const decision = await this.evaluate(); if (decision.snapshot.pendingAdmissions || decision.snapshot.unresolvedActions) throw Error('QUIESCENCE_REQUIRED');
        inventory = inspection.inspectExecutors(snapshot.fence);
        if (inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
        return { executorEvidenceRef: decision.publication!.record.receiptRef, workspaceEvidenceRef: this.candidate().artifactRefs[0]! }; },
      verifyOutcome: async () => { throw Error('NO_EFFECT_RECONCILIATION'); }, recheck: (_snapshot, proof) => {
        this.check(); if (workerCanonical(verify()) !== workerCanonical(proof.worker) || !inventory) throw Error('QUIESCENCE_REQUIRED');
        inspection.assertExecutorInventory(inventory);
      }, release: async () => {},
    });
    try { const takeover = await this.runtime.reconciler.reconcile(key, { kind: 'takeover', fence: this.fence(), receiptId: `delivery-takeover-${this.epoch}`, owner: `delivery-${r.generation + 1}`, leaseExpiresAt: this.leaseDeadline() }); this.owned = takeover.result.run; }
    finally { this.runtime.reconciler.unregister(key); }
  }
  private checkpointDigest(): string {
    const rows = this.options.db.prepare('SELECT namespace,checkpoint_id,checkpoint FROM graph_checkpoints WHERE graph_run_id=? ORDER BY namespace,checkpoint_id').all(this.current().graphRunId);
    const writes = this.options.db.prepare("SELECT namespace,checkpoint_id,task_id,channel,write_index,value FROM graph_checkpoint_writes WHERE graph_run_id=? ORDER BY namespace,checkpoint_id,task_id,write_index").all(this.current().graphRunId);
    if (!rows.length || !writes.some(row => (row as { channel: string }).channel === '__interrupt__')) throw Error('DURABLE_INTERRUPT_REQUIRED');
    return createHash('sha256').update(workerCanonical({ rows, writes })).digest('hex');
  }
  verifyResume(proof: GraphDeliveryInterrupt): void {
    if (!this.waiting || workerCanonical(proof) !== workerCanonical(this.waiting.proof) || this.stopping) throw Error('LIVE_QUIESCENCE_REQUIRED');
    this.waiting.verify();
  }
  resumeQuality(proof: GraphDeliveryInterrupt): void { this.verifyResume(proof); this.waiting!.resolve(); }
  /** Actual worker and broker signal chain. A drained Promise never clears holds. */
  async cancel(): Promise<void> {
    this.stopping = true; this.waiting?.reject(Error('DELIVERY_CANCELLED'));
    const handle = this.active;
    if (!handle) return;
    const stopped = await handle.stop('GRAPH_CANCEL_REQUESTED');
    if (!this.owned || stopped.worker.kind !== 'exited' || !stopped.actionsSettled) return;
    const r = this.current(); if (r.status !== 'cancel_pending') return;
    const inspection = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); });
    let inventory: GraphExecutorInventory | undefined;
    const verify = () => {
      this.assertOwned();
      if (stopped.worker.journalError || workerCanonical(this.runtime.observedExit(handle.connection)) !== workerCanonical(stopped.worker)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
      return verifyJournaledGraphWorkerExit(this.options.db, handle.binding);
    };
    const key = {};
    this.runtime.reconciler.register(key, { binding: { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity, generation: r.generation, owner: r.leaseOwner },
      authorize: () => { this.assertOwned(); const result: unknown = (this.options.authorizeCancel ?? this.options.authorize)();
        if (result !== undefined) { void Promise.resolve(result).catch(() => {}); throw Error('AUTHORITY_MUST_BE_SYNCHRONOUS'); } }, stopWorker: async () => stopped,
      verifyWorker: async () => verify(),
      verifyQuiescence: async snapshot => {
        const exit = verify(); inventory = inspection.inspectExecutors(snapshot.fence);
        if (inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
        return { executorEvidenceRef: `graph-executor-inventory:${inventory.digest}`, workspaceEvidenceRef: exit.evidenceRef };
      },
      verifyOutcome: async () => { throw Error('NO_EFFECT_RECONCILIATION'); }, recheck: (_snapshot, proof) => {
        if (workerCanonical(verify()) !== workerCanonical(proof.worker) || !inventory) throw Error('QUIESCENCE_REQUIRED');
        inspection.assertExecutorInventory(inventory);
      }, release: async () => {},
    });
    try { await this.runtime.reconciler.reconcile(key, { kind: 'project', status: 'cancelled', receiptId: `delivery-cancel-${r.generation}`, fence: this.fence() }); }
    finally { this.runtime.reconciler.unregister(key); }
  }
  async run(): Promise<{ route: GraphVerificationDecision['route']; candidate?: Candidate; feedbackRef?: string }> {
    if (this.started) throw Error('DELIVERY_ALREADY_STARTED'); this.started = true;
    this.check(); await mkdir(this.options.root, { recursive: true, mode: 0o700 }); this.check();
    if (this.options.root === this.options.repositoryPath || this.options.root.startsWith(`${this.options.repositoryPath}/`) || await realpath(this.options.root) !== this.options.root || await realpath(this.options.repositoryPath) !== this.options.repositoryPath) throw Error('CANONICAL_HOST_PATHS_REQUIRED');
    this.check();
    const modulePath = join(this.options.root, 'delivery.mjs'); await writeFile(modulePath, this.compiled.source, { flag: 'wx', mode: 0o600 }); this.check();
    const version = graphDeliveryVersion(this.compiled);
    this.runtime = new GraphRuntime({ db: this.options.db, broker: this.broker,
      checkpoints: new GraphCheckpointService({ runs: this.runs, checkpoints: new GraphCheckpointStore(this.options.db), bindings: new Map() }),
      programs: [{ id: 'code-delivery-v1', modulePath, digest: this.compiled.compiledProgramDigest }] });
    let resume: GraphWorkerResume | undefined;
    let lastStopped: GraphRuntimeStop | undefined;
    let durableWorkerStopped = false;
    try {
      this.runValue = this.options.reservedRun ?? this.runtime.prepare(this.options.request, version);
      if (this.stopping || this.current().status === 'cancel_pending') return { route: { kind: 'stop', epoch: 0, reason: 'cancel_requested' } };
      this.runValue = this.runtime.claim(this.runValue, 'delivery-1', this.leaseDeadline()); this.owned = this.runValue;
      this.runs.project({ ...this.fence(), status: 'running' });
      new GraphDeliveryEpochStore(this.options.db).activate(this.fence(), 0, deliveryEpoch(this.compiled, 0).cost);
      for (;;) {
        durableWorkerStopped = false;
        await this.restoreWorkspaceMappings();
        const r = this.current(); this.installed.clear();
        const handle = this.runtime.start({ request: this.options.request, run: r, programId: 'code-delivery-v1', leaseDurationMs: 60000, ...(resume ? { resume } : {}),
          authorize: () => this.check(), registration: { scope: r.scope, graphRunId: r.graphRunId, identity: r.identity, generation: r.generation, owner: r.leaseOwner!, epoch: this.epoch,
            workspace: { workspace_id: r.scope.workspaceId, workspace_path: this.options.repositoryPath }, authorize: () => this.check(), actions: [] },
          dispatch: (key, request, options) => this.dispatch(key, request, options) });
        this.active = handle; this.connection = handle.connection;
        let workerError: unknown;
        try { await handle.result; } catch (error) { workerError = error; }
        const stopped = await handle.stop(); lastStopped = stopped;
        if (this.stopping || this.current().status === 'cancel_pending') return { route: { kind: 'reconcile', epoch: this.epoch, reason: 'cancel_requested' } };
        if (!stopped.actionsSettled || stopped.worker.kind !== 'exited' || stopped.worker.journalError) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
        const verifyDurableExit = () => {
          durableWorkerStopped = false; const proof = verifyJournaledGraphWorkerExit(this.options.db, handle.binding); durableWorkerStopped = true; return proof;
        };
        const workerProof = verifyDurableExit();
        // No heartbeat now. Extend only the live lease, then capture a fresh immutable premise.
        const current = this.current(); this.runs.renewLease({ ...this.fence(), leaseExpiresAt: Math.max(current.leaseExpiresAt!, this.leaseDeadline()) });
        if (!this.latest(`e${this.epoch}.freeze`)) throw workerError ?? Error('FREEZE_REQUIRED');
        const decision = await this.evaluate(); const candidate = this.candidate();
        if (decision.route.kind !== 'repair_required') {
          // Dispatch already exhausted or could not admit the explicit retry.
          // A sealed failure must never advertise a retry that cannot be resumed.
          const route = decision.route.kind === 'retry_required'
            ? { kind: 'stop' as const, epoch: this.epoch, reason: 'retry_dispatch_incomplete' } : decision.route;
          this.check();
          const assertContractCurrent = () => {
            this.check();
            const current = new GraphVerificationStore(this.options.db, () => {}).snapshot(this.fence(decision.snapshot.run));
            if (current.digest !== decision.snapshot.digest) throw Error('GRAPH_CONTRACT_SNAPSHOT_CHANGED');
          };
          const contractReceiptRef = route.kind === 'ready' && this.options.contractEnvelope ? await publishGraphContractCompletion({
            root: this.options.root, envelope: this.options.contractEnvelope, decision, artifacts: this.artifacts, assertCurrent: assertContractCurrent,
            contextEvidence: [...this.contextEvidence.values()],
            ...(this.options.executionContext ? { contextReceiptRoot: this.options.executionContext.receiptRoot } : {}),
          }) : undefined;
          const inspection = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); });
          const inventory = inspection.inspectExecutors(this.fence(decision.snapshot.run));
          if (route.kind !== 'reconcile' && inventory.requiredEffectKeys.length) throw Error('QUIESCENCE_REQUIRED');
          if (this.current().status !== 'cancel_pending') this.options.db.transaction(() => {
            if (contractReceiptRef) {
              this.check();
              new GraphVerificationStore(this.options.db, () => {}).assertCurrentInTransaction(decision.snapshot, Date.now());
            }
            if (workerCanonical(verifyDurableExit()) !== workerCanonical(workerProof)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
            inspection.assertExecutorInventory(inventory);
            this.runs.project({ ...this.fence(decision.snapshot.run), status: route.kind === 'ready' ? 'ready' : route.kind === 'reconcile' ? 'reconciliation_required' : 'failed' });
            if (decision.publication) this.options.publish?.(candidate, decision.publication.record.receiptRef, contractReceiptRef);
          }).immediate();
          return { route, candidate, ...(decision.publication ? { feedbackRef: decision.publication.record.receiptRef } : {}) };
        }
        if (workerError || !decision.publication) throw workerError ?? Error('IMMUTABLE_JOIN_REQUIRED');
        this.check();
        this.feedbackRef = decision.publication.record.receiptRef;
        this.options.publish?.(candidate, this.feedbackRef);
        if (this.options.qualityInterrupt) {
          const proof: GraphDeliveryInterrupt = { interruptId: `quality:${r.graphRunId}:${this.epoch}:${r.generation}`, inputDigest: r.identity.inputDigest,
            kind: 'quality', checkpointDigest: this.checkpointDigest(), feedbackRef: this.feedbackRef };
          await new Promise<void>((resolve, reject) => {
            this.waiting = { proof, resolve, reject, verify: () => {
              this.check();
              if (workerCanonical(verifyDurableExit()) !== workerCanonical(workerProof)) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
              if (this.checkpointDigest() !== proof.checkpointDigest || workerCanonical(this.runtime.observedExit(handle.connection)) !== workerCanonical(stopped.worker)
                || stopped.worker.kind !== 'exited' || !stopped.actionsSettled) throw Error('PHYSICAL_WORKER_STOP_REQUIRED');
              if (!this.options.db.prepare('SELECT 1 FROM graph_verification_joins WHERE graph_run_id=? AND receipt_ref=?').get(r.graphRunId, proof.feedbackRef)) throw Error('IMMUTABLE_JOIN_REQUIRED');
            } };
            this.options.qualityInterrupt!(proof);
          });
          this.waiting = undefined;
        }
        await this.takeover(handle, stopped);
        const next = this.epoch + 1;
        new GraphDeliveryEpochStore(this.options.db).activate(this.fence(), next, deliveryEpoch(this.compiled, next).cost, this.feedbackRef);
        resume = { kind: 'interrupt', value: { ...decision.route, feedbackRef: this.feedbackRef } };
        this.epoch = next;
      }
    } catch (error) {
      // An idempotent prepare may find another owner's run. Failed acquisition
      // grants no authority to project that run's status.
      if (!this.owned) throw error;
      const current = this.current();
      let inventoryUnknown = true;
      try { inventoryUnknown = new GraphRecoveryStore(this.options.db, () => { throw Error('INSPECTION_ONLY'); })
        .inspectExecutors(this.fence(current)).requiredEffectKeys.length > 0; } catch { /* An unreadable inventory is not quiescence. */ }
      const reconcile = inventoryUnknown || !durableWorkerStopped || !lastStopped?.actionsSettled || lastStopped.worker.kind !== 'exited';
      try {
        this.check();
        if (!['ready','failed','cancelled','cancel_pending'].includes(current.status) && current.leaseExpiresAt! > Date.now()) {
          this.runs.project({ ...this.fence(current), status: reconcile ? 'reconciliation_required' : 'failed' });
        }
      } catch { /* Revoked/stale hosts cannot project replacement or cancellation state. */ }
      return { route: { kind: reconcile ? 'reconcile' : 'stop', epoch: this.epoch, reason: String(error).slice(0, 4096) } };
    } finally { this.closed = true; await this.runtime.close(); if (this.connection) this.broker.unregister(this.connection); }
  }
}

interface Descriptor { registrationId: string; request: StartGraphRequest; root: string; launchOwner: string; contractEnvelope?: GraphContractHostEnvelope }
interface StoredResult {
  candidate?: { candidateRevision: string; artifactRefs: string[]; artifactDigest: string };
  evidenceRefs: string[]; resultRef?: string;
}
interface PublicRow { descriptor_json: string; result_json: string | null; interrupt_json: string | null }
const fence = (r: GraphRun) => ({ scope: r.scope, graphRunId: r.graphRunId, identity: r.identity,
  expectedGeneration: r.generation, expectedOwner: r.leaseOwner, expectedRevision: r.revision, now: Date.now() });
const same = (a: unknown, b: unknown) => workerCanonical(a) === workerCanonical(b);
function synchronous(value: unknown): void {
  if (value !== undefined) { void Promise.resolve(value).catch(() => {}); throw new GraphContractError('POLICY_DENIED'); }
}

/** Stable public errors never disclose host paths, commands, or raw adapter text. */
export function graphRpcError(error: unknown): { code: string; message: string } {
  const raw = error instanceof Error ? error.message : '';
  const supplied = error && typeof error === 'object' && 'code' in error ? String(error.code) : raw;
  const aliases: Record<string, string> = { ENOENT: 'UNAUTHENTICATED', EACCES: 'UNAUTHENTICATED', ELOOP: 'UNAUTHENTICATED',
    RUN_NOT_FOUND: 'RUN_MISMATCH', OWNER_MISMATCH: 'STALE_GENERATION', LEASE_EXPIRED: 'GRANT_EXPIRED',
    LIVE_QUIESCENCE_REQUIRED: 'ACTION_OUTCOME_UNKNOWN', PHYSICAL_WORKER_STOP_REQUIRED: 'ACTION_OUTCOME_UNKNOWN',
    PROGRAM_IDENTITY_MISMATCH: 'CHECKPOINT_INCOMPATIBLE', TERMINAL_RUN: 'REVISION_CONFLICT', CANCEL_PENDING: 'CANCEL_UNCONFIRMED' };
  const parsed = GraphErrorCode.safeParse(aliases[supplied] ?? (supplied.startsWith('GRAPH_CONTRACT_') || supplied.startsWith('GRAPH_CONTEXT_') || supplied.startsWith('delivery_') ? 'POLICY_DENIED' : supplied));
  const code = error instanceof z.ZodError ? 'INPUT_CONTRACT_INVALID' : parsed.success ? parsed.data : 'INTERNAL_ERROR';
  return { code, message: code };
}

/** Public durable admission. Process maps hold actual supervisors, never public state. */
export class GraphRpcService {
  private readonly runs: GraphRunStore;
  private readonly registrations = new Map<string, GraphHostRegistration>();
  private readonly live = new Map<string, { driver?: GraphDelivery; done: Promise<void> }>();
  private readonly pendingRequests = new Set<Promise<unknown>>();
  private readonly owner = randomUUID();
  private closed = false;
  private closing: Promise<void> | undefined;
  private readonly options: GraphRpcOptions;
  constructor(options: GraphRpcOptions) {
    assertGraphPlatform();
    this.options = Object.freeze({ ...options, resolveCaller: options.resolveCaller, acquire: options.acquire });
    const snapshotAdapter = (a: AgentAdapter): AgentAdapter => {
      const capabilities = structuredClone(a.capabilities());
      return Object.freeze({ id: a.id, display_name: a.display_name, cost_tier: a.cost_tier,
        capabilities: () => structuredClone(capabilities), isAvailable: a.isAvailable.bind(a), invoke: a.invoke.bind(a),
        ...(a.confirmStopped ? { confirmStopped: a.confirmStopped.bind(a) } : {}) });
    };
    this.runs = new GraphRunStore(options.db);
    for (const value of options.registrations) {
      if (this.registrations.has(value.id)) throw Error('DUPLICATE_GRAPH_REGISTRATION');
      if (value.delivery.executionContext && !value.delivery.deliveryContract) throw Error('GRAPH_CONTEXT_CONTRACT_REQUIRED');
      this.registrations.set(value.id, { ...value, scope: GraphScope.parse(value.scope), delivery: { ...value.delivery,
        ...(value.delivery.deliveryContract ? { deliveryContract: parseGraphContractRegistration(value.delivery.deliveryContract) } : {}),
        ...(value.delivery.executionContext ? { executionContext: snapshotGraphExecutionContext(value.delivery.executionContext) } : {}),
        writer: { ...value.delivery.writer, adapter: snapshotAdapter(value.delivery.writer.adapter) }, reviewer: { ...value.delivery.reviewer, adapter: snapshotAdapter(value.delivery.reviewer.adapter) }, acceptanceRegistry: structuredClone(value.delivery.acceptanceRegistry) } });
    }
  }
  private caller(connection: object): GraphHostRegistration {
    if (this.closed) throw Error('GRAPH_SERVICE_CLOSED');
    const id = this.options.resolveCaller(connection);
    const registration = id === null ? undefined : this.registrations.get(id);
    if (!registration) throw new GraphContractError('UNAUTHENTICATED');
    return registration;
  }
  private row(id: string): PublicRow {
    const row = this.options.db.prepare('SELECT * FROM graph_public_delivery WHERE graph_run_id=?').get(id) as PublicRow | undefined;
    if (!row) throw new GraphContractError('RUN_MISMATCH');
    return row;
  }
  private bound(registration: GraphHostRegistration, scope: GraphScope, id: string): GraphRun {
    if (!same(scope, registration.scope)) throw new GraphContractError('SCOPE_MISMATCH');
    const run = this.runs.get(scope, id);
    if (!run || run.source.product !== registration.product) throw new GraphContractError('RUN_MISMATCH');
    if ((JSON.parse(this.row(id).descriptor_json) as Descriptor).registrationId !== registration.id) throw new GraphContractError('RUN_MISMATCH');
    return run;
  }
  private status(run: GraphRun, registration: GraphHostRegistration) {
    const row = this.row(run.graphRunId);
    const result = row.result_json ? JSON.parse(row.result_json) as StoredResult : { evidenceRefs: [] };
    const interrupt = row.interrupt_json ? JSON.parse(row.interrupt_json) as GraphDeliveryInterrupt : undefined;
    const { actions, budget } = readGraphActionState(this.options.db, run.graphRunId);
    const uncertain = ['cancel_pending', 'reconciliation_required'].includes(run.status);
    let needsAuthorization = false;
    try { synchronous(registration.authorize('execute')); } catch { needsAuthorization = true; }
    return GraphDelegationStatus.parse({ schemaVersion: 1 as const, scope: run.scope, graphRunId: run.graphRunId, source: run.source, status: run.status,
      revision: run.revision, generation: run.generation, inputDigest: run.identity.inputDigest, specHash: run.identity.specHash,
      createdAt: run.createdAt, updatedAt: run.updatedAt, publication: 'artifact_only' as const,
      phase: actions.find(a => a.state === 'running')?.effect.nodeId ?? run.status,
      budget: { limits: run.spec!.limits, ...budget },
      delivery: uncertain ? 'delivery_unknown' as const : run.status === 'ready' ? 'available' as const : 'pending' as const,
      ...result,
      ...(needsAuthorization ? { authorization: { kind: 'needs_authorization' as const, scope: run.scope, requestRef: `graph-grant:${run.graphRunId}` } } : {}),
      ...(interrupt && run.status === 'waiting_input' ? { interrupt: { interruptId: interrupt.interruptId, inputDigest: interrupt.inputDigest, kind: interrupt.kind } } : {}),
      ...(run.status === 'failed' ? { error: { schemaVersion: 1 as const, code: 'INTERNAL_ERROR' as const, message: 'Graph delivery stopped before verified publication', retryable: false as const, nextAction: 'stop' as const } } : {}),
      ...(uncertain ? { error: { schemaVersion: 1 as const, code: run.status === 'cancel_pending' ? 'CANCEL_UNCONFIRMED' as const : 'ACTION_OUTCOME_UNKNOWN' as const,
        message: 'Physical execution requires reconciliation', retryable: false as const, nextAction: 'reconcile' as const } } : {}),
    });
  }
  private replay(request: Exclude<GraphRequest, StartGraphRequest>): boolean {
    const row = this.options.db.prepare('SELECT graph_run_id,input_digest FROM graph_requests WHERE scope_json=? AND idempotency_key=?')
      .get(workerCanonical(request.scope), request.idempotencyKey) as { graph_run_id: string; input_digest: string } | undefined;
    if (!row) return false;
    if (row.graph_run_id !== request.graphRunId || row.input_digest !== computeGraphInputDigest(request)) throw new GraphContractError('IDEMPOTENCY_CONFLICT');
    return true;
  }
  async handle(connection: object, method: string, params: unknown): Promise<unknown> {
    const pending = this.handleRequest(connection, method, params);
    this.pendingRequests.add(pending);
    try { return await pending; }
    catch (error) { throw new GraphContractError(GraphErrorCode.parse(graphRpcError(error).code)); }
    finally { this.pendingRequests.delete(pending); }
  }
  private async handleRequest(connection: object, method: string, params: unknown): Promise<unknown> {
    const registration = this.caller(connection);
    if (method === 'graph.contract') {
      synchronous(registration.authorize('read'));
      const request = GraphStatusRequest.parse(params); const run = this.bound(registration, request.scope, request.graphRunId);
      const row = this.row(run.graphRunId); const descriptor = JSON.parse(row.descriptor_json) as Descriptor;
      const result = row.result_json ? JSON.parse(row.result_json) as StoredResult : undefined;
      const refs = result?.evidenceRefs.filter(ref => ref.startsWith('graph-contract:')) ?? [];
      if (!descriptor.contractEnvelope || !result?.candidate || !result.resultRef || refs.length !== 1) throw Error('GRAPH_CONTRACT_RECEIPT_UNPUBLISHED');
      const receipt = await loadGraphContractCompletion(descriptor.root, refs[0]!, { run, envelope: descriptor.contractEnvelope,
        joinRef: result.resultRef, candidateRevision: result.candidate.candidateRevision, artifactDigest: result.candidate.artifactDigest });
      synchronous(registration.authorize('read'));
      if (this.closed || workerCanonical(this.bound(registration, request.scope, request.graphRunId)) !== workerCanonical(run)) throw Error('GRAPH_CONTRACT_SNAPSHOT_CHANGED');
      return receipt;
    }
    if (method === 'graph.status' || method === 'graph.result') {
      synchronous(registration.authorize('read'));
      const request = GraphStatusRequest.parse(params);
      return this.status(this.bound(registration, request.scope, request.graphRunId), registration);
    }
    if (method === 'graph.validate') return compileGraphSpec(GraphSpec.parse(params));
    const request = GraphRequest.parse(params);
    if (!same(request.scope, registration.scope)) throw new GraphContractError('SCOPE_MISMATCH');
    if ((method === 'graph.start' || method === 'graph.submit') && request.kind === 'start') return this.submit(registration, request);
    const reconcile = method === 'graph.reconcile';
    if ((!['graph.cancel', 'graph.reconcile'].includes(method) || request.kind !== 'cancel') && (method !== 'graph.resume' || request.kind !== 'resume')) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    synchronous(registration.authorize(request.kind === 'cancel' ? 'cancel' : 'resume'));
    const run = this.bound(registration, request.scope, request.graphRunId);
    const replay = this.replay(request);
    // Only the unchanged cancellation intent can be retried as reconciliation.
    // Later revisions/generations are a lookup, never authority to stop a new owner.
    if (replay && !(reconcile && run.status === 'cancel_pending' && run.revision === request.expectedRevision + 1)) return this.status(run, registration);
    if (!replay && run.revision !== request.expectedRevision) throw new GraphContractError('REVISION_CONFLICT');
    if (request.kind === 'cancel') {
      const pending = replay ? run : this.runs.projectRequest(request, { ...fence(run), status: 'cancel_pending' });
      const entry = this.live.get(run.graphRunId);
      if (entry?.driver) void entry.driver.cancel().catch(() => {});
      else if (run.generation === 0) this.runs.project({ ...fence(pending), status: 'cancelled' });
      if (reconcile && run.generation > 0) {
        const receiptId = `public-cancel:${createHash('sha256').update(workerCanonical({ scope: request.scope,
          idempotencyKey: request.idempotencyKey, digest: computeGraphInputDigest(request) })).digest('hex')}`;
        let authorizationFailed = false;
        try {
          await cancelUnstartedGraphWorker(this.options.db, new GraphReconciler(this.options.db), { fence: fence(pending), receiptId },
            () => {
              try { synchronous(registration.authorize('cancel')); }
              catch (error) { authorizationFailed = true; throw error; }
            });
        } catch (error) {
          if (authorizationFailed) throw error;
          synchronous(registration.authorize('cancel'));
          // A finite bounded writer may have durable stopped/output evidence even
          // when the ledger ACK was lost. Recovery never dispatches it again.
          try {
            const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
            await cancelJournaledBoundedWriter(this.options.db, descriptor.root, registration.delivery.repositoryPath,
              { fence: fence(pending), receiptId }, () => {
                try { synchronous(registration.authorize('cancel')); }
                catch (failure) { authorizationFailed = true; throw failure; }
              });
          } catch (failure) {
            if (authorizationFailed) throw failure;
            try {
              const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
              await cancelUnstartedGraphPreparation(this.options.db, descriptor.root, registration.delivery.repositoryPath,
                { fence: fence(pending), receiptId }, () => {
                  try { synchronous(registration.authorize('cancel')); }
                  catch (denied) { authorizationFailed = true; throw denied; }
                });
            } catch (denied) {
              if (authorizationFailed) throw denied;
              // Operator diagnostics remain local. Never print raw filesystem,
              // adapter or SQLite error text into either RPC or daemon logs.
              const reasons = new Set(['SCOPE', 'PROGRAM', 'PREDECESSORS', 'NOT_STARTED', 'CANDIDATE',
                'NOT_CANCELLING', 'CHANGED', 'ADMISSION', 'WORKSPACE', 'LOCK_PRESENT', 'DECISION']);
              const reason = denied instanceof Error ? denied.message.replace(/^PREPARATION_RECOVERY_/, '') : '';
              console.error(JSON.stringify({ event: 'graph_recovery_blocked', graphRunId: run.graphRunId,
                stage: 'unstarted_preparation', code: reasons.has(reason) ? `PREPARATION_RECOVERY_${reason}` : graphRpcError(denied).code }));
            }
          }
          // Intent is already durable. Missing/conflicting evidence cannot undo
          // its acknowledgement; return current state without settling or releasing.
        }
        // Successful G15 commit rechecked the grant in its final transaction.
        // Later expiry cannot turn that completed operation into a rejected request.
      }
      return this.status(this.runs.get(run.scope, run.graphRunId)!, registration);
    }
    const row = this.row(run.graphRunId);
    const proof = row.interrupt_json ? JSON.parse(row.interrupt_json) as GraphDeliveryInterrupt : undefined;
    if (run.status !== 'waiting_input' || !proof || proof.interruptId !== request.interruptId) throw new GraphContractError('REVISION_CONFLICT');
    if (request.input.kind !== 'data' || !same(request.input.value, { inputDigest: run.identity.inputDigest }) || proof.inputDigest !== run.identity.inputDigest) throw new GraphContractError('INPUT_CONTRACT_INVALID');
    const driver = this.live.get(run.graphRunId)?.driver;
    if (!driver) throw new GraphContractError('ACTION_OUTCOME_UNKNOWN');
    if (run.leaseExpiresAt === null || run.leaseExpiresAt <= Date.now() || Date.now() >= run.createdAt + run.spec!.limits.deadlineSeconds * 1000) throw new GraphContractError('GRANT_EXPIRED');
    driver.verifyResume(proof);
    this.runs.projectRequest(request, { ...fence(run), status: 'running' });
    driver.resumeQuality(proof);
    return this.status(this.runs.get(run.scope, run.graphRunId)!, registration);
  }
  private submit(registration: GraphHostRegistration, request: StartGraphRequest) {
    if (request.source.product !== registration.product) throw new GraphContractError('UNAUTHENTICATED');
    const input = request.spec.input;
    if (input.repositoryRef !== registration.repositoryRef || input.taskRef !== registration.taskRef || request.inputArtifactRef !== registration.inputArtifactRef
      || !registration.delivery.acceptanceRegistry.some(a => a.ref === input.acceptanceRef)) throw new GraphContractError('POLICY_DENIED');
    const boundContract = registration.delivery.deliveryContract ? bindGraphDeliveryContract(request,
      { ...registration.delivery, deliveryContract: registration.delivery.deliveryContract }) : undefined;
    request = boundContract?.request ?? request;
    const compiled = compileGraphSpec(request.spec);
    let created = false;
    const run = this.options.db.transaction(() => {
      synchronous(registration.authorize());
      const prior = this.options.db.prepare('SELECT graph_run_id FROM graph_requests WHERE scope_json=? AND idempotency_key=?')
        .get(workerCanonical(request.scope), request.idempotencyKey) as { graph_run_id: string } | undefined;
      if (!prior) {
        if (registration.admitNewRuns) {
          const admitted: unknown = registration.admitNewRuns();
          if (admitted !== true) {
            void Promise.resolve(admitted).catch(() => {});
            throw new GraphContractError('GRAPH_ADMISSION_DISABLED');
          }
        }
        assertGraphDeliverySupported(registration.delivery);
      }
      const run = this.runs.create(request, graphDeliveryVersion(compiled), Date.now());
      if (prior && !this.options.db.prepare('SELECT 1 FROM graph_public_delivery WHERE graph_run_id=?').get(run.graphRunId)) throw new GraphContractError('CHECKPOINT_INCOMPATIBLE');
      const descriptor: Descriptor = { registrationId: registration.id, request, root: join(this.options.root, run.graphRunId), launchOwner: this.owner,
        ...(boundContract ? { contractEnvelope: boundContract.envelope } : {}) };
      created = this.options.db.prepare('INSERT OR IGNORE INTO graph_public_delivery(graph_run_id,descriptor_json) VALUES (?,?)').run(run.graphRunId, JSON.stringify(descriptor)).changes === 1;
      return this.bound(registration, run.scope, run.graphRunId);
    }).immediate();
    if (created) {
      // Only the durable reservation winner schedules; even construction is deferred.
      const entry: { driver?: GraphDelivery; done: Promise<void> } = { done: Promise.resolve() };
      this.live.set(run.graphRunId, entry);
      entry.done = new Promise<void>(resolve => setImmediate(resolve)).then(async () => {
        if (this.closed || this.runs.get(run.scope, run.graphRunId)!.status === 'cancelled') return;
        try {
          const descriptor = JSON.parse(this.row(run.graphRunId).descriptor_json) as Descriptor;
          if (descriptor.launchOwner !== this.owner) throw Error('LAUNCH_OWNER_MISMATCH');
          entry.driver = new GraphDelivery({ ...registration.delivery, db: this.options.db, root: descriptor.root, request: descriptor.request, reservedRun: run,
            ...(descriptor.contractEnvelope ? { contractEnvelope: descriptor.contractEnvelope } : {}),
            acquire: this.options.acquire, ...(this.options.authorizePolicy ? { authorizePolicy: this.options.authorizePolicy } : {}), authorize: () => { synchronous(registration.authorize()); },
            authorizeCancel: () => { synchronous(registration.authorize('cancel')); },
            publish: (candidate, feedbackRef, contractReceiptRef) => {
              const { candidateRevision, artifactRefs, artifactDigest } = candidate;
              const output: StoredResult = { candidate: { candidateRevision, artifactRefs, artifactDigest }, evidenceRefs: [feedbackRef, ...(contractReceiptRef ? [contractReceiptRef] : [])], resultRef: feedbackRef };
              this.options.db.prepare('UPDATE graph_public_delivery SET result_json=? WHERE graph_run_id=?').run(JSON.stringify(output), run.graphRunId);
            },
            qualityInterrupt: proof => {
              const current = this.runs.get(run.scope, run.graphRunId)!;
              this.options.db.transaction(() => {
                synchronous(registration.authorize());
                this.options.db.prepare('UPDATE graph_public_delivery SET interrupt_json=? WHERE graph_run_id=?').run(JSON.stringify(proof), run.graphRunId);
                this.runs.project({ ...fence(current), status: 'waiting_input' });
              }).immediate();
            } });
          await entry.driver.run();
          const stopped = this.runs.get(run.scope, run.graphRunId)!;
          if (stopped.generation === 0 && stopped.status === 'cancel_pending') this.runs.project({ ...fence(stopped), status: 'cancelled' });
        } catch {
          const current = this.runs.get(run.scope, run.graphRunId)!;
          // Before a worker exists, failure has no ambiguous external effects.
          if (current.generation === 0 && current.status === 'queued') this.runs.project({ ...fence(current), status: 'failed' });
        }
      });
    }
    return this.status(run, registration);
  }
  close(): Promise<void> {
    this.closed = true;
    this.closing ??= this.drain();
    return this.closing;
  }
  private async drain(): Promise<void> {
    const draining = [...this.live.entries()].map(async ([id, entry]) => {
      try {
        const descriptor = JSON.parse(this.row(id).descriptor_json) as Descriptor;
        const run = this.runs.get(descriptor.request.scope, id)!;
        if (!['ready', 'failed', 'cancelled', 'cancel_pending'].includes(run.status)) this.runs.project({ ...fence(run), status: 'cancel_pending' });
        if (run.generation === 0) {
          const current = this.runs.get(run.scope, id)!;
          if (current.status === 'cancel_pending') this.runs.project({ ...fence(current), status: 'cancelled' });
        }
        await entry.driver?.cancel();
      } finally {
        // run() may still be preparing the module, and a bounded worker stop
        // may leave action persistence/release pending. Neither is DB quiescence.
        try { await entry.done; } finally { await entry.driver?.drainDatabaseUsers(); }
      }
    });
    // A deadline cannot prove database users are gone. Forced process recovery
    // keeps its existing durable uncertainty; graceful shutdown waits for I/O.
    await Promise.allSettled([...draining, ...this.pendingRequests]);
  }
}

// Explicit local owner trust: owner-only config plus a private Unix socket, never
// a scope/principal taken from RPC JSON. Remote identities require resolveCaller.
const ref = z.string().min(1).max(256);
const hostConfig = z.object({ schemaVersion: z.literal(1), admissionEnabled: z.boolean().optional(), registration: z.object({ id: ref, scope: GraphScope,
  repositoryRef: ref, repositoryPath: z.string().min(1), taskRef: ref, task: z.string().min(1).max(65536), inputArtifactRef: ref,
  deliveryContract: DeliveryContractV2.optional(),
  executionContextPath: z.string().min(1).max(4096).optional(),
  writer: z.object({ agentId: ref, capabilityId: ref }).strict(), reviewer: z.object({ agentId: ref, capabilityId: ref }).strict(),
  boundedMessages: z.object({ provider: z.enum(['anthropic', 'deepseek']).optional(), baseUrl: z.string().min(1).max(2048).optional(),
    model: ref, files: z.array(ref).min(1).max(16), acceptance: z.string().min(1).max(16000) }).strict().optional(),
  acceptanceRegistry: z.array(z.object({ ref, command: z.string().min(1), passExitCodes: z.array(z.number().int()), failExitCodes: z.array(z.number().int()) }).strict()).min(1),
  grant: z.object({ scope: GraphScope, expiresAt: z.number().int().positive(), operations: z.array(z.enum(['read', 'execute', 'cancel', 'resume'])).min(1) }).strict(),
}).strict() }).strict();
function trustedConfig(path: string) {
  const fd = openSync(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
  try { const stat = fstatSync(fd);
    if (!stat.isFile() || stat.size > 131072 || (stat.mode & 0o077) !== 0 || stat.uid !== process.getuid?.()) throw Error('GRAPH_CONFIG_OWNER_REQUIRED');
    const bytes = Buffer.alloc(131073); let offset = 0;
    for (;;) { const count = readSync(fd, bytes, offset, bytes.length - offset, offset); if (!count) break; offset += count;
      if (offset === bytes.length) throw Error('GRAPH_CONFIG_LIMIT'); }
    return hostConfig.parse(JSON.parse(bytes.subarray(0, offset).toString('utf8')));
  } finally { closeSync(fd); }
}
export interface LocalGraphRegistrationOptions {
  resolveDeliveryContractPolicy?: (input: { repositoryPath: string; scope: GraphScope; contract: DeliveryContractV2 }) => Promise<{
    policy: GraphContractHostPolicy;
    /** Synchronous live policy binding; not an async best-effort recheck. */
    assertCurrent: () => void;
  }>;
  resolveExecutionContext?: (input: { configPath: string; repositoryPath: string; scope: GraphScope;
    frozen: NonNullable<GraphDeliveryOptions['deliveryContract']>['frozen'] }) => Promise<GraphExecutionContextRegistration>;
}
export async function loadLocalGraphRegistration(path: string, adapters: ReadonlyMap<string, AgentAdapter>, options: LocalGraphRegistrationOptions = {}): Promise<GraphHostRegistration> {
  const config = trustedConfig(path).registration;
  const bindingDigest = (value: typeof config) => { const { grant: _grant, ...binding } = value; return createHash('sha256').update(workerCanonical(binding)).digest('hex'); };
  const digest = bindingDigest(config);
  let policyAuthority: (() => void) | undefined;
  let executionContext: GraphExecutionContextRegistration | undefined;
  const authorize = (operation: 'read' | 'execute' | 'cancel' | 'resume' = 'execute') => {
    const fresh = trustedConfig(path).registration;
    if (bindingDigest(fresh) !== digest || !same(fresh.grant.scope, config.scope)
      || !fresh.grant.operations.includes(operation) || ((operation === 'execute' || operation === 'resume') && fresh.grant.expiresAt <= Date.now()))
      throw new GraphContractError('GRANT_EXPIRED');
    if (operation === 'execute' || operation === 'resume') synchronous(policyAuthority?.());
    if (operation === 'execute' || operation === 'resume') synchronous(executionContext?.assertCurrent());
  };
  authorize('read');
  let deliveryContract: GraphDeliveryOptions['deliveryContract'];
  if (config.deliveryContract) {
    if (!options.resolveDeliveryContractPolicy) throw Error('GRAPH_CONTRACT_POLICY_REQUIRED');
    const resolved = await options.resolveDeliveryContractPolicy({ repositoryPath: config.repositoryPath, scope: config.scope, contract: config.deliveryContract });
    if (typeof resolved.assertCurrent !== 'function') throw Error('GRAPH_CONTRACT_POLICY_REQUIRED');
    policyAuthority = resolved.assertCurrent;
    deliveryContract = parseGraphContractRegistration({ frozen: freezeDeliveryContract(config.deliveryContract), policy: resolved.policy });
    authorize('execute');
  }
  if (config.executionContextPath) {
    if (!deliveryContract || !options.resolveExecutionContext) throw Error('GRAPH_CONTEXT_REGISTRATION_REQUIRED');
    if (config.scope.tenantId !== 'default') throw Error('GRAPH_CONTEXT_SCOPE_UNSUPPORTED');
    executionContext = snapshotGraphExecutionContext(await options.resolveExecutionContext({ configPath: config.executionContextPath,
      repositoryPath: config.repositoryPath, scope: config.scope, frozen: deliveryContract.frozen }));
    authorize('execute');
  }
  let writer = adapters.get(config.writer.agentId); let reviewer = adapters.get(config.reviewer.agentId);
  if (config.boundedMessages) {
    const provider = config.boundedMessages.provider ?? 'anthropic';

/** Raw bytes/modes/links, including ignored and untracked files. Git status alone
 * misses ignored files, assume-unchanged and timestamp-preserving modifications. */
function workspaceFingerprint(path: string): string {
  const hash = createHash('sha256'); let count = 0;
  const visit = (directory: string, prefix: string) => {
    for (const name of readdirSync(directory).sort()) {
      if (!prefix && name === '.git') continue;
      if (++count > 100000) throw new Error('workspace inventory limit');
      const relative = `${prefix}${name}`; const full = join(directory, name); const stat = lstatSync(full);
      hash.update(JSON.stringify([relative, stat.mode & 0o777, stat.isDirectory() ? 'directory' : stat.isSymbolicLink() ? 'link' : 'file']));
      if (stat.isSymbolicLink()) {
        const target = resolve(dirname(full), readlinkSync(full));
        if (!target.startsWith(`${path}/`) || (existsSync(target) && !realpathSync(target).startsWith(`${path}/`))) throw new Error('shared external symlink in candidate workspace');
        hash.update(readlinkSync(full));
      }
      else if (stat.isDirectory()) visit(full, `${relative}/`);
      else if (stat.isFile() && stat.size <= 256 * 1024 * 1024 && stat.nlink === 1) hash.update(readFileSync(full));
      else throw new Error('unsupported workspace entry/shared writable hardlink');
    }
  };
  visit(path, ''); return hash.digest('hex');
}
export function assertExactCandidateFiles(path: string, revision: string): void {
  if (resolve(path) !== path || realpathSync(path) !== path) throw new Error('candidate root must be canonical');
  const expected = new Set<string>();

    expected.add(name);
    let parent = dirname(name); while (parent !== '.') { expected.add(parent); parent = dirname(parent); }
    const file = join(path, name);
    parent = dirname(file);
    while (parent !== path) {
      if (!lstatSync(parent).isDirectory() || lstatSync(parent).isSymbolicLink()) throw new Error('candidate tracked parent changed');
      parent = dirname(parent);
    }
    const stat = lstatSync(file);
    if (stat.nlink !== 1 || stat.size > 256 * 1024 * 1024) throw new Error('unsupported tracked entry');
    const bytes = match[1] === '120000' && stat.isSymbolicLink() ? Buffer.from(readlinkSync(file))
      : match[1] !== '120000' && stat.isFile() ? readFileSync(file) : undefined;
    if (!bytes || (match[1] !== '120000' && ((stat.mode & 0o111) !== 0) !== (match[1] === '100755'))

  }

}
function repositoryStamp(path: string): string {
  const gitDir = join(path, '.git');
  for (const p of [path, gitDir, join(gitDir, 'objects')]) if (!lstatSync(p).isDirectory() || realpathSync(p) !== p) throw new Error('workspace repository is not independent');
  if (git(path, 'rev-parse', '--absolute-git-dir') !== gitDir || git(path, 'rev-parse', '--git-common-dir') !== '.git'
    || existsSync(join(gitDir, 'objects', 'info', 'alternates')) || existsSync(join(gitDir, 'objects', 'info', 'http-alternates'))) throw new Error('shared repository storage forbidden');
  // Every restored repository starts without caches, symlinked metadata or shared objects.
  const inspect = (directory: string): void => { for (const name of readdirSync(directory)) { const p = join(directory, name); const stat = lstatSync(p); if (stat.isSymbolicLink() || (!stat.isDirectory() && (!stat.isFile() || stat.nlink !== 1))) throw new Error('shared repository metadata'); if (stat.isDirectory()) inspect(p); } };
  inspect(gitDir);
  return verificationCanonical([path, gitDir, join(gitDir, 'objects')].map(p => { const stat = lstatSync(p); return [p, stat.dev, stat.ino]; }));
}

        const scope = { repo_path: workspace, tenant: "default" as const };
        const mandatory: ExecutionContextEntry[] = [hostEntry("instruction", invocation.prompt, scope, sampledAt), hostEntry("safety", HOST_SAFETY, scope, sampledAt)];
        if (contractDigest !== undefined) mandatory.push(hostEntry("contract", contractBrief!, scope, sampledAt, contractDigest));
        const assembly = assembleExecutionContext({ mandatory, optional: entries, scope, nowMs: sampledAt, maxPromptBytes, maxKnowledgeBytes });
        const receipt: ExecutionContextReceipt = {
          schema_version: 1, phase: "prepared", evidence_boundary: "agent_task_prompt_not_provider_wire",
          run_id: invocation.run_id, task_id: invocation.task_id, session_id: invocation.session_id, capability_id: invocation.capability_id, adapter_id: adapter.id,
          repo_path: workspace, execution_repo_path: executionRepo, tenant: "default", contract_digest: contractDigest ?? null,
          sampled_at_ms: sampledAt, prompt_sha256: assembly.prompt_sha256, prompt_bytes: assembly.prompt_bytes, knowledge_bytes: assembly.knowledge_bytes,
          max_prompt_bytes: maxPromptBytes, max_knowledge_bytes: maxKnowledgeBytes, token_window_qualification: "unqualified",
          mandatory: assembly.mandatory, selected: assembly.selected, omitted: assembly.omitted,
        };
        const persist = async (value: ExecutionContextReceipt, afterInvoke: boolean) => {
          try {
            if (Buffer.byteLength(JSON.stringify(value)) > 65536) throw new Error();
            const ref = await recordReceipt(value);
            if (!ref || typeof ref.path !== "string" || !isAbsolute(ref.path) || ref.path.length > 4096 || !sha256.test(ref.digest)) throw new Error();
            if (closed || controller.signal.aborted) throw new Error();
            refs.push({ ref: { path: ref.path, digest: ref.digest }, run_id: invocation.run_id, task_id: invocation.task_id, session_id: invocation.session_id });
          } catch { throw new ExecutionContextError(afterInvoke ? "execution_context_receipt_failed_after_invoke" : "execution_context_receipt_failed_before_invoke"); }
        };
        await abortable(persist(receipt, false), controller.signal);
        checkAbort(controller.signal);
        const dispatchAt = now();
        if (!Number.isSafeInteger(dispatchAt) || dispatchAt < sampledAt || [...assembly.mandatory, ...assembly.selected].some(row =>
          row.verified_at_ms > dispatchAt || (row.expires_at_ms !== undefined && row.expires_at_ms < dispatchAt)))
          throw new ExecutionContextError("execution_context_expired_before_invoke");
        active = adapter.invoke({ ...invocation, prompt: assembly.prompt }, controller.signal)[Symbol.asyncIterator]();
        let observed = false;
        while (true) {
          checkAbort(controller.signal);
          const step = await abortable(Promise.resolve(active.next()), controller.signal);
          if (step.done) { completed = true; break; }
          if (!observed) {
            const observedAt = now();
            if (!Number.isSafeInteger(observedAt) || observedAt < sampledAt) throw new ExecutionContextError("execution_context_invalid_clock");
            await abortable(persist({ ...receipt, phase: "adapter_observed", observed_at_ms: observedAt }, true), controller.signal);
            observed = true;
          }
          yield step.value;
        }
      } finally {
        closed = true;
        outerSignal.removeEventListener("abort", cancel);
        if (!completed) {
          controller.abort();
          if (active?.return) await boundedCleanup(() => active!.return!());
        }
      }
    };
    const methods = new Map<PropertyKey, unknown>();
    return new Proxy(adapter, { get(target, property) {
      if (property === "invoke") return invoke;
      const value: unknown = Reflect.get(target, property, target);
      if (typeof value !== "function") return value;
      if (!methods.has(property)) methods.set(property, value.bind(target));
      return methods.get(property);
    } });
  });
  return { adapters, receiptRefs: filter => refs.filter(row =>
    (!filter || (filter.run_id === undefined || row.run_id === filter.run_id) &&
      (filter.task_id === undefined || row.task_id === filter.task_id) && (filter.session_id === undefined || row.session_id === filter.session_id)))
    .map(row => ({ ...row.ref })) };
}

function hostEntry(kind: "instruction" | "safety" | "contract", text: string, scope: ExecutionContextEntry["scope"], nowMs: number, sourceDigest?: string): ExecutionContextEntry {
  return { id: `host:${kind}`, source_kind: kind, source_ref: `osstack:execution-context:${kind}:v1`, source_digest: sourceDigest ?? executionContextDigest(text),
    text_sha256: executionContextDigest(text), text, scope, tier: "mandatory", evidence_kind: "host", verified_at_ms: nowMs };
}
function checkAbort(signal: AbortSignal): void { if (signal.aborted) throw new ExecutionContextError("execution_context_aborted"); }
function abortable<T>(promise: Promise<T>, signal: AbortSignal): Promise<T> {
  return new Promise<T>((resolvePromise, reject) => {
    const abort = () => reject(new ExecutionContextError("execution_context_aborted"));
    if (signal.aborted) abort(); else signal.addEventListener("abort", abort, { once: true });
    promise.then(resolvePromise, reject).finally(() => signal.removeEventListener("abort", abort));
  });
}
async function boundedCleanup(cleanup: () => Promise<unknown>): Promise<void> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { await Promise.race([Promise.resolve().then(cleanup).catch(() => {}), new Promise<void>(done => { timer = setTimeout(done, 100); })]); }
  finally { if (timer) clearTimeout(timer); }
}

export function createPinnedLocalContextLoader(input: ExecutionContextConfig): ExecutionContextLoader {

        throw new ExecutionContextError("execution_context_pack_scope_or_version_mismatch");

      for (const chunk of manifest.chunks) {
        if (chunk.chunk_id.length > 96 || chunks.has(chunk.chunk_id)) throw new ExecutionContextError("invalid_execution_context_pack");
        chunks.add(chunk.chunk_id);
      }
      packs.push({ manifest, path: pin.path });
    }
    checkAbort(request.signal);
    const results = searchKnowledgePacks(packs, task.prompt.slice(0, 4096), { limit: 20 });
    return results.map(result => {
      const pin = config.packs.find(row => row.pack_id === result.pack_id && row.version === result.pack_version)!;
      return { id: `knowledge:${executionContextDigest(JSON.stringify([result.pack_id, result.pack_version, result.chunk_id]))}`,
        source_kind: "knowledge" as const, source_ref: JSON.stringify({ path: pin.path, pack_id: pin.pack_id, version: pin.version, chunk_id: result.chunk_id }),

async function canonicalDirectory(path: string): Promise<string> {
  try { const root = await realpath(path); if (!isAbsolute(path) || !(await stat(root)).isDirectory()) throw new Error(); return root; }
  catch { throw new ExecutionContextError("execution_context_repository_unavailable"); }
}
function inside(root: string, path: string): boolean { const rel = relative(root, path); return rel !== ".." && !rel.startsWith(`..${sep}`) && !isAbsolute(rel); }
async function confinedRead(root: string, input: string, maximum: number): Promise<Buffer> {
  let file: Awaited<ReturnType<typeof open>> | undefined;
  try {
    const path = await realpath(resolve(root, input));
    if (!inside(root, path)) throw new ExecutionContextError("execution_context_source_escape");
    file = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
    const before = await file.stat();
    if (!before.isFile()) throw new ExecutionContextError("execution_context_source_unavailable");
    if (before.size > maximum) throw new ExecutionContextError("execution_context_source_too_large");
    const buffer = Buffer.alloc(maximum + 1); let length = 0;
    while (length < buffer.length) { const read = await file.read(buffer, length, buffer.length - length, null); if (!read.bytesRead) break; length += read.bytesRead; }
    const after = await file.stat();
    if (length > maximum || before.size !== after.size || before.mtimeMs !== after.mtimeMs) throw new ExecutionContextError("execution_context_source_changed");
    const current = await realpath(path); const identity = await stat(current);
    if (!inside(root, current) || before.ino !== identity.ino || before.dev !== identity.dev) throw new ExecutionContextError("execution_context_source_escape");
    return buffer.subarray(0, length);
  } catch (error) { if (error instanceof ExecutionContextError) throw error; throw new ExecutionContextError("execution_context_source_unavailable"); }
  finally { await file?.close(); }
}

import { createHash } from 'node:crypto';
import type Database from 'better-sqlite3';
import { GraphJsonValue, GraphResult, freezeWorkerData, workerCanonical } from '@osstack/protocol';
import type { GraphAction, GraphActionBudget } from './graph-actions.js';
import { readGraphActionState } from './graph-actions.js';
export { readGraphActionState } from './graph-actions.js';
import { fencedGraphTransaction } from './graph-storage.js';
import { GraphRunStore, checkGraphRunFence, type GraphRun, type GraphRunFence } from './graph-runs.js';
export interface GraphVerificationSnapshot {
  run: GraphRun; actions: GraphAction[]; budget: GraphActionBudget;
  pendingAdmissions: number; unresolvedActions: number; digest: string;
}
export interface GraphVerificationRecord {
  receiptRef: string; graphRunId: string; epoch: number; candidateRevision: string;
  generation: number; owner: string; bundle: GraphJsonValue; createdAt: number;
}
const hash = (value: unknown) => createHash('sha256').update(workerCanonical(value)).digest('hex');
function fence(run: GraphRun, now: number): GraphRunFence {
  return { scope: run.scope, graphRunId: run.graphRunId, identity: run.identity, expectedGeneration: run.generation,
    expectedOwner: run.leaseOwner, expectedRevision: run.revision, now };
}
/** A durable immutable evidence publication, not an epoch scheduler or node FSM.
 * The trusted host verifier must bind the bundle to current ledger attempts and
 * independent artifact/executor evidence. No worker JSON port calls this store. */
export class GraphVerificationStore {
  constructor(private readonly db: Database.Database,
    private readonly verify: (snapshot: GraphVerificationSnapshot, bundle: GraphJsonValue) => void) {}
  private capture(run: GraphRun): GraphVerificationSnapshot {
    const state = readGraphActionState(this.db, run.graphRunId);
    const admissions = this.db.prepare('SELECT effect_key,released_at FROM graph_action_admissions WHERE graph_run_id=? ORDER BY effect_key').all(run.graphRunId) as Array<{ effect_key: string; released_at: number | null }>;
    const body = { run, ...state, pendingAdmissions: admissions.filter(a => a.released_at === null).length,
      unresolvedActions: state.actions.filter(a => ['intent','running','unknown','cancel_pending'].includes(a.state)).length };
    return freezeWorkerData({ ...body, digest: hash({ body, admissions }) });
  }
  snapshot(input: GraphRunFence): GraphVerificationSnapshot {
    return fencedGraphTransaction(this.db, input, run => ({ changed: false, value: this.capture(run) })).value;
  }
  /** Final synchronous publication check inside the caller's existing IMMEDIATE transaction.
   * No refreshed premise: includes admissions which can change without run revision. */
  assertCurrentInTransaction(snapshot: GraphVerificationSnapshot, now: number): void {
    if (!this.db.inTransaction) throw Error('GRAPH_PUBLICATION_TRANSACTION_REQUIRED');
    const input = fence(snapshot.run, now);
    const run = checkGraphRunFence(new GraphRunStore(this.db).get(input.scope, input.graphRunId), input, true);
    const current = this.capture(run);
    if (current.digest !== snapshot.digest || workerCanonical(current) !== workerCanonical(snapshot)) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
  }
  commit(snapshotValue: GraphVerificationSnapshot, bundleValue: unknown, now: number): { record: GraphVerificationRecord; created: boolean; run: GraphRun } {
    const snapshot = freezeWorkerData(structuredClone(snapshotValue)); const bundle = freezeWorkerData(GraphJsonValue.parse(bundleValue));
    const result = fencedGraphTransaction(this.db, fence(snapshot.run, now), run => {
      const current = this.capture(run);
      if (current.digest !== snapshot.digest || workerCanonical(current) !== workerCanonical(snapshot)) throw Error('VERIFICATION_SNAPSHOT_CHANGED');
      if (current.unresolvedActions) throw Error('UNRESOLVED_ACTION');
      if (current.pendingAdmissions) throw Error('ADMISSION_RELEASE_PENDING');
      if (!bundle || typeof bundle !== 'object' || Array.isArray(bundle)) throw Error('INVALID_JOIN');
      const object = bundle as Record<string, GraphJsonValue>; const binding = object.binding;
      if (!binding || typeof binding !== 'object' || Array.isArray(binding)) throw Error('INVALID_JOIN_BINDING');
      const b = binding as Record<string, GraphJsonValue>;
      const candidate = GraphResult.parse({ schemaVersion: 1, kind: 'candidate_ready', candidateRevision: b.candidateRevision, artifactDigest: b.artifactDigest, artifactRefs: ['join-validation'] });
      if (candidate.kind !== 'candidate_ready' || object.schemaVersion !== 1 || object.status !== 'complete'
        || b.graphRunId !== run.graphRunId || workerCanonical(b.scope) !== workerCanonical(run.scope) || b.specHash !== run.identity.specHash
        || !Number.isSafeInteger(b.epoch) || (b.epoch as number) < 0 || (b.epoch as number) >= (run.spec?.limits.maxEpochs ?? 0)
        || b.acceptanceRef !== run.spec?.input.acceptanceRef || typeof object.evidenceDigest !== 'string' || !/^[a-f0-9]{64}$/.test(object.evidenceDigest)) throw Error('INVALID_JOIN_BINDING');
      if (typeof this.verify !== 'function') throw Error('trusted verifier required');
      const verified: unknown = this.verify(current, bundle);
      if (verified !== undefined) { void Promise.resolve(verified).catch(() => {}); throw Error('join verifier must be synchronous'); }
      const receiptRef = `graph-join:${hash(binding)}`;
      const saved = this.db.prepare('SELECT bundle_json,generation,owner,created_at,receipt_ref FROM graph_verification_joins WHERE graph_run_id=? AND epoch=? AND candidate_revision=?')
        .get(run.graphRunId, b.epoch, candidate.candidateRevision) as { bundle_json: string; generation: number; owner: string; created_at: number; receipt_ref: string } | undefined;
      if (saved && (saved.bundle_json !== workerCanonical(bundle) || saved.receipt_ref !== receiptRef)) throw Error('JOIN_PUBLICATION_CONFLICT');
      if (!saved) this.db.prepare('INSERT INTO graph_verification_joins(receipt_ref,graph_run_id,epoch,candidate_revision,generation,owner,bundle_json,created_at) VALUES(?,?,?,?,?,?,?,?)')
        .run(receiptRef, run.graphRunId, b.epoch, candidate.candidateRevision, run.generation, run.leaseOwner, workerCanonical(bundle), now);
      const record: GraphVerificationRecord = { receiptRef, graphRunId: run.graphRunId, epoch: b.epoch as number, candidateRevision: candidate.candidateRevision,
        generation: saved?.generation ?? run.generation, owner: saved?.owner ?? run.leaseOwner!, bundle, createdAt: saved?.created_at ?? now };
      return { changed: !saved, value: { record, created: !saved } };
    });
    return freezeWorkerData({ ...result.value, run: result.run });
  }
}

/** Immutable lifecycle permits, not node state. Only the host can authorize a
 * subsequent epoch from an attested, published negative quality join. */
export class GraphDeliveryEpochStore {
  constructor(private readonly db: Database.Database) {}
  activate(input: GraphRunFence, epoch: number, cost: { actions: number; tokens: number }, joinRef?: string): GraphRun {
    return fencedGraphTransaction(this.db, input, run => {
      if (!run.spec || run.status !== 'running') throw Error('RUN_NOT_RUNNING');
      const state = readGraphActionState(this.db, run.graphRunId);
      if (state.actions.some(a => ['intent','running','unknown','cancel_pending'].includes(a.state))) throw Error('UNRESOLVED_ACTION');
      if (this.db.prepare('SELECT 1 FROM graph_action_admissions WHERE graph_run_id=? AND released_at IS NULL').get(run.graphRunId)) throw Error('ADMISSION_RELEASE_PENDING');
      const row = this.db.prepare('SELECT max(epoch) AS epoch FROM graph_delivery_epochs WHERE graph_run_id=?').get(run.graphRunId) as { epoch: number | null };
      if (row.epoch === epoch) {
        const prior = this.db.prepare('SELECT join_ref FROM graph_delivery_epochs WHERE graph_run_id=? AND epoch=?').get(run.graphRunId, epoch) as { join_ref: string | null };
        if (prior.join_ref !== (joinRef ?? null)) throw Error('EPOCH_PERMIT_CONFLICT');
        return { changed: false, value: undefined };
      }
      if (epoch !== (row.epoch ?? -1) + 1 || epoch >= run.spec.limits.maxEpochs) throw Error('EPOCH_MISMATCH');
      if (epoch === 0 && (joinRef || state.actions.length)) throw Error('INITIAL_EPOCH_ALREADY_CONSUMED');
      if (epoch > 0) {
        const saved = this.db.prepare('SELECT bundle_json,epoch FROM graph_verification_joins WHERE graph_run_id=? AND receipt_ref=?').get(run.graphRunId, joinRef) as { bundle_json: string; epoch: number } | undefined;
        if (!saved || saved.epoch !== epoch - 1) throw Error('IMMUTABLE_JOIN_REQUIRED');
        const bundle = JSON.parse(saved.bundle_json);
        if (bundle.status !== 'complete' || bundle.branches.length !== 2 || !bundle.branches.every((b: { state: string }) => b.state === 'confirmed')
          || !bundle.branches.some((b: { result: { verdict?: string; passed?: boolean } }) => b.result.verdict === 'changes_requested' || b.result.passed === false)) throw Error('NEGATIVE_JOIN_REQUIRED');
        for (const branch of bundle.branches) {
          const latest = state.actions.filter(a => a.effect.epoch === epoch - 1 && a.effect.nodeId === branch.nodeId && a.effect.effectSlot === branch.effectSlot).sort((a,b) => b.effect.logicalAttempt - a.effect.logicalAttempt)[0];
          if (!latest || latest.effectKey !== branch.effectKey || latest.state !== 'confirmed' || latest.outcome?.state !== 'confirmed' || workerCanonical(latest.outcome.result) !== workerCanonical(branch.result)) throw Error('STALE_JOIN');
        }
      }
      if (!Number.isSafeInteger(cost.actions) || cost.actions < 1 || !Number.isSafeInteger(cost.tokens) || cost.tokens < 0) throw Error('INVALID_COMPILED_COST');
      const budget = state.budget;
      if (input.now - run.createdAt >= run.spec.limits.deadlineSeconds * 1000 || budget.accountingBlocked
        || budget.actionStarts + cost.actions > run.spec.limits.maxActionStarts
        || BigInt(budget.chargedTokens) + BigInt(budget.reservedTokens) + BigInt(cost.tokens) > BigInt(run.spec.limits.tokenReservationLimit)) throw Error('BUDGET_EXHAUSTED');
      this.db.prepare('INSERT INTO graph_delivery_epochs(graph_run_id,epoch,join_ref,generation,created_at) VALUES(?,?,?,?,?)').run(run.graphRunId, epoch, joinRef ?? null, run.generation, input.now);
      return { changed: true, value: undefined };
    }).run;
  }
}
