import { realpathSync, statSync } from "node:fs";
import type Database from "better-sqlite3";

export type AttentionDashboardService = "running" | "disabled" | "unavailable";
export interface AttentionDashboardCounts {
  scopes: number;
  subjects: number;
  unmonitored_subjects: number;
  human_waits: number;
  system_waits: number;
  unknown_waits: number;
  open_items: number;
  pending_answers: number;
  uncertain_answers: number;
  missing_source_items: number;
  applied_answers: number;
  dead_letters: number;
}
export interface AttentionDashboardScope extends Omit<AttentionDashboardCounts, "scopes"> {
  scope_id: string;
  task_id: string;
  task_title: string;
  repo_path: string;
  sampled_at_ms: number;
  coverage: "covered" | "degraded" | "stale" | "unknown";
  reason: string | null;
}
export interface AttentionDashboard {
  status: "available" | "degraded" | "stale" | "disabled" | "unavailable" | "not_watched";
  reason: string | null;
  scope_filter: "workspace" | "local_overview";
  workspace_path: string | null;
  tenant: "default";
  sampled_at_ms: number;
  consistency: "attention_block_transaction";
  totals: AttentionDashboardCounts | null;
  scopes: AttentionDashboardScope[];
  returned: number;
  truncated: boolean;
  next_after: string | null;
  limitations: string[];
}
export interface AttentionDashboardOptions {
  workspacePath?: string;
  service: AttentionDashboardService;
  nowMs?: number;
  after?: string;
  limit?: number;
}

const countKeys = ["subjects", "unmonitored_subjects", "human_waits", "system_waits", "unknown_waits", "open_items", "pending_answers", "uncertain_answers", "missing_source_items", "applied_answers", "dead_letters"] as const;

/** An independent read transaction. Does not claim whole-dashboard consistency,
 * inspect processes, migrate/open a DB, mark seen, or acquire observation ownership. */
export function readAttentionDashboard(db: Database.Database, options: AttentionDashboardOptions): AttentionDashboard {
  const now = options.nowMs ?? Date.now();
  const limit = options.limit ?? 25;
  const after = options.after ?? "";
  if (!Number.isSafeInteger(limit) || limit < 1 || limit > 100 || !Number.isFinite(now)
    || (after !== "" && !/^[a-f0-9]{64}$/.test(after))) throw new Error("invalid_attention_page");
  const result: AttentionDashboard = {
    status: "unavailable", reason: null,
    scope_filter: options.workspacePath === undefined ? "local_overview" : "workspace",
    workspace_path: null, tenant: "default", sampled_at_ms: now,
    consistency: "attention_block_transaction", totals: null, scopes: [], returned: 0, truncated: false, next_after: null,
    limitations: ["Observation only; no execution or approval authority", "Counts are persisted facts, not proof of process liveness or task completion", "Unmonitored counts include only discovered subjects in registered scopes", "Only the Attention block shares a read transaction"],
  };
  if (options.workspacePath !== undefined) {
    try {
      result.workspace_path = realpathSync(options.workspacePath);
      if (!statSync(result.workspace_path).isDirectory()) throw new Error("not_directory");
    } catch { return { ...result, reason: "workspace_unavailable" }; }
  }
  if (options.service !== "running") return { ...result, status: options.service, reason: "observer_" + options.service };
  try {
    return db.transaction(() => {
      const tables = new Set((db.prepare("SELECT name FROM sqlite_master WHERE type='table'").all() as { name: string }[]).map(r => r.name));
      if (!["tasks", "daemon_owner", "attention_scopes", "attention_subjects", "attention_items", "attention_deliveries"].every(t => tables.has(t)))
        return { ...result, reason: "schema_unavailable" };
      const owner = db.prepare("SELECT generation FROM daemon_owner WHERE singleton=1").get() as { generation: number } | undefined;
      if (!owner) return { ...result, reason: "owner_unavailable" };
      // The SAME filtered relation feeds counts and paged rows. Missing task and
      // non-default tenant scopes are excluded; filesystem canonicalization precedes SQL.
      const query = `WITH filtered AS (
        SELECT s.*,t.title AS task_title FROM attention_scopes s JOIN tasks t ON t.task_id=s.task_id
        WHERE t.tenant_id IS NULL AND (@repo IS NULL OR s.repo_path=@repo)
      ), subject_counts AS (
        SELECT u.scope_id,COUNT(*) AS subjects,SUM(CASE WHEN admitted=1 THEN 0 ELSE 1 END) AS unmonitored_subjects,
          SUM(CASE WHEN json_valid(payload_json) THEN json_extract(payload_json,'$.wait.owner')='human' ELSE 0 END) AS human_waits,
          SUM(CASE WHEN json_valid(payload_json) THEN json_extract(payload_json,'$.wait.owner')='system' ELSE 0 END) AS system_waits,
          SUM(CASE WHEN json_valid(payload_json) THEN COALESCE(json_extract(payload_json,'$.wait.owner'),'unknown') NOT IN ('human','system','none') ELSE 1 END) AS unknown_waits,
          SUM(CASE WHEN u.sampled_at<@cutoff OR u.sampled_at>@now OR u.generation<>@generation OR NOT json_valid(payload_json) THEN 1
            WHEN COALESCE(json_extract(payload_json,'$.reason'),'snapshot_only')<>'snapshot_only' OR json_extract(payload_json,'$.gap')='history_gap' THEN 1 ELSE 0 END) AS faulty_subjects
        FROM attention_subjects u JOIN filtered f ON f.scope_id=u.scope_id GROUP BY u.scope_id
      ), item_counts AS (
        SELECT i.scope_id,SUM(state='open') AS open_items,SUM(state='answer_reserved') AS pending_answers,
          SUM(state='answer_uncertain') AS uncertain_answers,SUM(state='source_missing') AS missing_source_items,
          SUM(state='answer_applied') AS applied_answers
        FROM attention_items i JOIN filtered f ON f.scope_id=i.scope_id GROUP BY i.scope_id
      ), delivery_counts AS (
        SELECT d.scope_id,SUM(state='dead_letter') AS dead_letters FROM attention_deliveries d JOIN filtered f ON f.scope_id=d.scope_id GROUP BY d.scope_id
      ), stats AS (
        SELECT f.*,${countKeys.map(key => `COALESCE(${key},0) AS ${key}`).join(",")},
          CASE WHEN f.sampled_at>@now THEN 'unknown'
            WHEN f.generation<>@generation OR f.deadline<@now THEN 'stale'
            WHEN f.scan_complete<>1 OR f.error IS NOT NULL OR COALESCE(unmonitored_subjects,0)>0 OR COALESCE(faulty_subjects,0)>0 THEN 'degraded'
            ELSE 'covered' END AS coverage,
          CASE WHEN f.sampled_at>@now THEN 'clock_moved_backwards'
            WHEN f.generation<>@generation THEN 'owner_changed' WHEN f.deadline<@now THEN 'scan_expired'
            WHEN f.scan_complete<>1 THEN 'scan_incomplete' WHEN f.error IS NOT NULL THEN 'scan_failed'
            WHEN COALESCE(unmonitored_subjects,0)>0 THEN 'unmonitored_subjects'
            WHEN COALESCE(faulty_subjects,0)>0 THEN 'source_gap_or_error' ELSE NULL END AS reason
        FROM filtered f LEFT JOIN subject_counts USING(scope_id) LEFT JOIN item_counts USING(scope_id) LEFT JOIN delivery_counts USING(scope_id)
      ) `;
      const params = { repo: result.workspace_path, now, cutoff: now - 65_000, generation: owner.generation };
      const aggregate = db.prepare(query + `SELECT COUNT(*) AS scopes,${countKeys.map(k => `COALESCE(SUM(${k}),0) AS ${k}`).join(",")},
        COALESCE(SUM(coverage IN ('stale','unknown')),0) AS stale_scopes,COALESCE(SUM(coverage='degraded'),0) AS degraded_scopes FROM stats`).get(params) as AttentionDashboardCounts & { stale_scopes: number; degraded_scopes: number };
      const { stale_scopes, degraded_scopes, ...totals } = aggregate;
      const rows = db.prepare(query + `SELECT scope_id,task_id,task_title,repo_path,sampled_at AS sampled_at_ms,coverage,reason,${countKeys.join(",")} FROM stats WHERE scope_id>@after ORDER BY scope_id LIMIT @limit`).all({ ...params, after, limit: limit + 1 }) as AttentionDashboardScope[];
      const scopes = rows.slice(0, limit);
      return { ...result, status: totals.scopes === 0 ? "not_watched" : stale_scopes ? "stale" : degraded_scopes ? "degraded" : "available",
        reason: totals.scopes === 0 ? "no_registered_scope" : stale_scopes ? "stale_observation" : degraded_scopes ? "incomplete_coverage" : null,
        totals, scopes, returned: scopes.length, truncated: rows.length > limit,
        next_after: rows.length > limit ? scopes.at(-1)!.scope_id : null } as AttentionDashboard;
    })();
  } catch {
    // Bad/missing columns, malformed legacy data, busy/read failures all make
    // this block unavailable. Never turn failed observation into a healthy zero.
    return { ...result, reason: "source_unavailable" };
  }
}

/** Display-only: strip terminal escapes and control/bidi characters. Never use
 * display-cleaned strings as identities, paths, or executable command arguments. */
export function attentionDisplayText(value: unknown): string {
  // biome-ignore lint/suspicious/noControlCharactersInRegex: remove untrusted terminal control sequences
  return String(value ?? "").replace(/\u001b\[[0-?]*[ -/]*[@-~]/g, "")
    // biome-ignore lint/suspicious/noControlCharactersInRegex: transport controls must not reach a terminal
    .replace(/[\u0000-\u001f\u007f-\u009f\u202a-\u202e\u2066-\u2069]/g, " ");
}

export function createDashboardSnapshot(opts: DashboardSnapshotOptions): DashboardSnapshot {
  const generatedAt = opts.now ? opts.now() : Date.now() / 1000;
  const runCounts = opts.db.prepare(`
    SELECT
      COUNT(*) AS total_runs,
      SUM(CASE WHEN outcome = 'completed' THEN 1 ELSE 0 END) AS completed_runs,
      SUM(CASE WHEN outcome = 'failed' THEN 1 ELSE 0 END) AS failed_runs,
      SUM(CASE WHEN outcome = 'cancelled' THEN 1 ELSE 0 END) AS cancelled_runs,
      SUM(tokens_in) AS tokens_in,
      SUM(tokens_out) AS tokens_out,
      AVG(duration_ms) AS avg_duration_ms
    FROM runs
  `).get() as {
    total_runs: number;
    completed_runs: number | null;
    failed_runs: number | null;
    cancelled_runs: number | null;
    tokens_in: number | null;
    tokens_out: number | null;
    avg_duration_ms: number | null;
  };
  const totalEvents = countEvents(opts.db);
  const queuedTasks = countPendingTasks(opts.db);
  const boardOptions = opts.now
    ? { includeArchived: false, now: opts.now }
    : { includeArchived: false };
  const boardView = getBoard(opts.kanban, boardOptions);
  const boardColumns = boardView.changed ? boardView.columns : [];
  const byStatus = boardView.changed ? boardView.stats.by_status : {};
  const boardTotal = boardView.changed ? boardView.stats.total : 0;
  const doneCards = byStatus["done"] ?? 0;
  const progressBase = boardTotal > 0 ? boardTotal : runCounts.total_runs;
  const progressDone = boardTotal > 0 ? doneCards : (runCounts.completed_runs ?? 0);
  const progressPercent = progressBase > 0 ? Math.round((progressDone / progressBase) * 100) : 0;
  const completed = runCounts.completed_runs ?? 0;
  const failed = runCounts.failed_runs ?? 0;
  const cancelled = runCounts.cancelled_runs ?? 0;
  const terminalRuns = completed + failed + cancelled;
  const successRate = terminalRuns > 0 ? completed / terminalRuns : 1;
  const blockedCards = byStatus["blocked"] ?? 0;
  const grade: "A" | "B" | "C" = successRate >= 0.9 && blockedCards === 0
    ? "A"
    : successRate >= 0.5
      ? "B"
      : "C";

  const recentRuns = opts.db.prepare(`
    SELECT r.run_id, r.task_id, t.title, r.agent_id, r.capability_id, r.phase,
           r.outcome, r.started_at, r.duration_ms
    FROM runs r JOIN tasks t ON t.task_id = r.task_id

function dirnameOfDb(dbPath: string): string {
  return dbPath === ":memory:" ? "." : dirname(dbPath);
}

function countActivePitfallLessons(db: Database.Database): number {
  const row = db.prepare("SELECT COUNT(*) AS total FROM pitfall_lessons WHERE status = 'active'").get() as
    | { total?: number }
    | undefined;
  return row?.total ?? 0;
}

function countMarkdownFiles(dir: string, prefix: string): number {
  try {
    return readdirSync(dir).filter((name) => name.startsWith(prefix) && name.endsWith(".md")).length;
  } catch {
    return 0;
  }
}

function countProofPacks(dir: string): number {
  try {
    return readdirSync(dir).filter((name) => {
      if (!name.startsWith("osstack-proof-")) return false;
      try {
        return statSync(join(dir, name)).isDirectory();
      } catch {
        return false;
      }
    }).length;
  } catch {
    return 0;
  }
}

function latestMarkdownFile(dir: string, prefix: string): DashboardArtifact | null {
  return latestDirectoryEntry(dir, (name) => name.startsWith(prefix) && name.endsWith(".md"));
}

function latestProofPackArtifact(dir: string): DashboardProofPackArtifact | null {
  const artifact = latestDirectoryEntry(dir, (name, path) => {
    if (!name.startsWith("osstack-proof-")) return false;
    try {
      return statSync(path).isDirectory();
    } catch {
      return false;
    }
  });
  if (artifact === null) return null;
  const manifest = join(artifact.path, "manifest.json");
  try {
    statSync(manifest);
    return { ...artifact, manifest };
  } catch {
    return { ...artifact, manifest: null };
  }
}

function latestDirectoryEntry(
  dir: string,
  accept: (name: string, path: string) => boolean,
): DashboardArtifact | null {
  try {
    let latest: DashboardArtifact | null = null;
    for (const name of readdirSync(dir)) {
      const path = join(dir, name);
      if (!accept(name, path)) continue;
      let artifact: DashboardArtifact;
      try {
        const stats = statSync(path);
        artifact = {
          name,
          path,
          updated_at: stats.mtimeMs / 1000,
        };
      } catch {
        continue;
      }
      if (latest === null || artifact.updated_at > latest.updated_at) latest = artifact;
    }
    return latest;
  } catch {
    return null;
  }
}

function readReleaseReadinessBlockers(path: string): ReleaseReadinessBlocker[] {
  let markdown: string;
  try {
    markdown = readFileSync(path, "utf-8");
  } catch {
    return [];
  }
  const blockers: ReleaseReadinessBlocker[] = [];
  let inBlockers = false;
  for (const line of markdown.split(/\r?\n/)) {
    if (/^##\s+Blockers\s*$/i.test(line.trim())) {
      inBlockers = true;
      continue;
    }
    if (inBlockers && /^##\s+/.test(line.trim())) break;
    if (!inBlockers) continue;
    const match = line.match(/^-\s+\[([^\]]+)\]\s+(.+)$/);
    if (match) blockers.push({ id: match[1]!.trim(), detail: match[2]!.trim() });
  }
  return blockers;
}

function releaseReadinessNextSteps(
  blockers: ReleaseReadinessBlocker[],
  workspacePath: string,
): DashboardSnapshot["next_steps"] {
  if (blockers.length === 0) return [];
  const ids = blockers.map((blocker) => blocker.id).join(", ");
  return [{
    label: "Release Readiness",
    detail: `${blockers.length} blocker(s): ${ids}`,
    command: `osstack release doctor --target all --workspace ${shellArg(workspacePath)} --output release-readiness.md --json`,
    status: "blocked",
    blockers,
  }];
}

function shellArg(value: string): string {
  if (/^[A-Za-z0-9_./:-]+$/.test(value)) return value;
  return `'${value.replaceAll("'", "'\\''")}'`;
}

export async function startDashboardServer(
  snapshot: (params?: DashboardHttpSnapshotParams) => DashboardSnapshot,
  opts: DashboardUrlOptions = dashboardUrlFromEnv(),
): Promise<DashboardServerHandle> {
  let url = dashboardUrl(opts);
  const server = createServer((req, res) => {
    void handleDashboardRequest(req, res, snapshot, url);
  });
  await new Promise<void>((resolve, reject) => {
    server.once("error", reject);
    server.listen(opts.port, opts.host, () => {
      server.off("error", reject);
      resolve();
    });
  });
  const addr = server.address() as AddressInfo | null;
  const port = addr?.port ?? opts.port;
  const resolvedOpts = { ...opts, port };
  url = dashboardUrl(resolvedOpts);
  return {
    url,
    host: opts.host,
    port,
    close: () => new Promise<void>((resolve) => server.close(() => resolve())),
  };
}

async function handleDashboardRequest(
  req: IncomingMessage,
  res: ServerResponse,
  snapshot: (params?: DashboardHttpSnapshotParams) => DashboardSnapshot,
  url: string,
): Promise<void> {
  const requestUrl = new URL(req.url ?? "/", url);
  const path = requestUrl.pathname;
  if (path === "/" || path === "/dashboard") {
    send(res, 200, "text/html; charset=utf-8", renderDashboardHtml());
    return;
  }
  if (path === "/api/dashboard") {
    const workspacePath = requestUrl.searchParams.get("workspace") ?? undefined;
    const attentionAfter = requestUrl.searchParams.get("attention_after") ?? undefined;
    const rawLimit = requestUrl.searchParams.get("attention_limit");
    const attentionLimit = rawLimit === null ? undefined : Number(rawLimit);
    if ((attentionAfter !== undefined && !/^[a-f0-9]{64}$/.test(attentionAfter))
      || (attentionLimit !== undefined && (!Number.isSafeInteger(attentionLimit) || attentionLimit < 1 || attentionLimit > 100))) {
      send(res, 400, "application/json; charset=utf-8", JSON.stringify({ error: "invalid_attention_page" }));
      return;
    }
    send(res, 200, "application/json; charset=utf-8", JSON.stringify(snapshot({
      ...(workspacePath === undefined ? {} : { workspacePath }),
      ...(attentionAfter === undefined ? {} : { attentionAfter }),
      ...(attentionLimit === undefined ? {} : { attentionLimit }),
    })));
    return;
  }
  if (path === "/dashboard/url") {
    send(res, 200, "application/json; charset=utf-8", JSON.stringify({
      url: dashboardUrlWithWorkspace(url, requestUrl.searchParams.get("workspace") ?? undefined),
    }));
    return;
  }
  send(res, 404, "text/plain; charset=utf-8", "not found");
}

function dashboardUrlWithWorkspace(url: string, workspacePath: string | undefined): string {
  if (workspacePath === undefined) return url;
  const nextUrl = new URL(url);
  nextUrl.searchParams.set("workspace", workspacePath);
  return nextUrl.toString();
}

function send(res: ServerResponse, status: number, contentType: string, body: string): void {
  res.writeHead(status, { "Content-Type": contentType, "Cache-Control": "no-store" });
  res.end(body);
}

  }

  if (a.truncated) lines.push(label.more + ' (' + a.returned + '/' + a.totals.scopes + ')');
  lines.push(label.note);
  // Source text stays text, including angle brackets, quotes and backticks.
  el.textContent = lines.join('\\n');
  const current = new URL(window.location.href);
  const first = document.getElementById('attention-first');
  first.textContent = label.first;

export function initialCockpitTuiFullscreenState(model: CockpitTuiModel): CockpitTuiFullscreenState {
  return {
    active_section_index: model.sections.length === 0 ? -1 : 0,
    selected_action_index: model.actions.length === 0 ? 0 : 1,
  };
}

export function updateCockpitTuiFullscreenState(
  model: CockpitTuiModel,
  state: CockpitTuiFullscreenState,
  key: CockpitTuiKey,
): CockpitTuiKeyResult {
  const normalized = key.toLowerCase();
  if (normalized === "q" || normalized === "escape") return { type: "quit" };
  if (normalized === "enter") {
    const selection = selectCockpitTuiAction(model, state.selected_action_index);
    return selection === undefined ? { type: "none", state } : { type: "select", selection };
  }

  const nextState = { ...state };
  if (normalized === "up") {
    nextState.active_section_index = clamp(nextState.active_section_index - 1, 0, model.sections.length - 1);
    nextState.selected_action_index = clamp(nextState.selected_action_index - 1, 1, model.actions.length);
    return { type: "state", state: nextState };
  }
  if (normalized === "down") {
    nextState.active_section_index = clamp(nextState.active_section_index + 1, 0, model.sections.length - 1);
    nextState.selected_action_index = clamp(nextState.selected_action_index + 1, 1, model.actions.length);
    return { type: "state", state: nextState };
  }
  if (normalized === "left") {
    nextState.active_section_index = clamp(nextState.active_section_index - 1, 0, model.sections.length - 1);
    return { type: "state", state: nextState };
  }
  if (normalized === "right") {
    nextState.active_section_index = clamp(nextState.active_section_index + 1, 0, model.sections.length - 1);
    return { type: "state", state: nextState };
  }
  return { type: "none", state };
}

export function renderCockpitFullscreen(
  model: CockpitTuiModel,
  options: { width?: number; height?: number; state?: Partial<CockpitTuiFullscreenState> } = {},
): string {
  const width = Math.max(72, Math.min(options.width ?? process.stdout.columns ?? 100, 140));
  const height = Math.max(20, Math.min(options.height ?? process.stdout.rows ?? 34, 60));
  const baseState = initialCockpitTuiFullscreenState(model);
  const state = {
    active_section_index: options.state?.active_section_index ?? baseState.active_section_index,
    selected_action_index: options.state?.selected_action_index ?? baseState.selected_action_index,
  };
  const innerWidth = width - 4;
  const rule = "-".repeat(width - 2);
  const lines = [
    `+${rule}+`,
    frameLine("OSSTACK COCKPIT", width),
    frameLine(`Workspace: ${model.workspace}`, width),
    frameLine(`Daemon: ${model.daemon}`, width),
    frameLine("", width),
    ...model.sections.flatMap((section, index) => {
      const marker = index === state.active_section_index ? ">" : " ";
      return [
        frameLine(`${marker} ${section.title}`, width),
        ...section.lines.slice(0, 3).map((line) => frameLine(`  ${line}`, width)),
      ];
    }),
    frameLine("", width),
    frameLine("Actions", width),
    ...(model.actions.length === 0
      ? [frameLine("  none", width)]
      : model.actions.slice(0, 9).map((action, index) => {
        const selected = index + 1 === state.selected_action_index ? ">" : " ";
        return frameLine(`${selected} ${index + 1}. [${action.status}] ${action.label}: ${action.command}`, width);
      })),
    frameLine("", width),
    frameLine("Keys: up/down move | enter select | q quit", width),
    `+${rule}+`,
  ];
  const clipped = lines.slice(0, height);
  while (clipped.length < height) {
    clipped.splice(clipped.length - 1, 0, frameLine("", width));
  }
  return clipped.map((line) => trimToWidth(line, innerWidth + 4)).join("\n").trimEnd();
}

export function runCockpitTuiFullscreenKeys(model: CockpitTuiModel, keys: CockpitTuiKey[]): CockpitTuiFullscreenRunResult {
  let state = initialCockpitTuiFullscreenState(model);
  for (const key of keys) {
    const result = updateCockpitTuiFullscreenState(model, state, key);
    if (result.type === "state" || result.type === "none") {
      state = result.state;
      continue;
    }
    if (result.type === "select") {
      return { type: "select", state, selection: result.selection };
    }
    return { type: "quit", state };
  }
  return { type: "none", state };
}

export function cockpitTuiKeyFromInput(value: string): CockpitTuiKey {
  if (value === "\u0003") return "q";
  if (value === "\u001b") return "escape";
  if (value === "\r" || value === "\n") return "enter";
  if (value === "\u001b[A") return "up";
  if (value === "\u001b[B") return "down";
  if (value === "\u001b[C") return "right";
  if (value === "\u001b[D") return "left";
  return value;
}

export function selectCockpitTuiAction(model: CockpitTuiModel, selectedIndex: number): CockpitTuiActionSelection | undefined {
  const action = model.actions[selectedIndex - 1];
  if (action === undefined) return undefined;
  const command = concreteCockpitTuiCommand(action.command, model.workspace);
  const concreteAction = { ...action, command };
  return {
    mode: "terminal-cockpit-action",
    selected_index: selectedIndex,
    action: concreteAction,
    next_command: command,
  };
}

async function runCockpitTuiFullscreenRawMode(
  model: CockpitTuiModel,
  options: { json: boolean },
): Promise<CockpitTuiFullscreenRunResult> {
  const stdin = process.stdin as NodeJS.ReadStream & { setRawMode?: (mode: boolean) => void };
  if (typeof stdin.setRawMode !== "function") {
    console.error("osstack: cockpit tui --fullscreen --interactive raw-mode is unavailable in this terminal");
    return { type: "quit", state: initialCockpitTuiFullscreenState(model) };
  }
  let state = initialCockpitTuiFullscreenState(model);
  return await new Promise<CockpitTuiFullscreenRunResult>((resolve) => {
    const render = () => {
      if (options.json) return;
      process.stdout.write(`\u001b[?25l\u001b[2J\u001b[H${renderCockpitFullscreen(model, { state })}\n`);
    };
    const finish = (result: CockpitTuiFullscreenRunResult) => {
      stdin.off("data", onData);
      stdin.setRawMode?.(false);
      stdin.pause();
      if (!options.json) process.stdout.write("\u001b[?25h\n");
      resolve(result);
    };
    const onData = (chunk: Buffer) => {
      const key = cockpitTuiKeyFromInput(chunk.toString("utf-8"));
      const result = updateCockpitTuiFullscreenState(model, state, key);
      if (result.type === "state" || result.type === "none") {
        state = result.state;
        render();
        return;
      }
      if (result.type === "select") {
        finish({ type: "select", state, selection: result.selection });
        return;
      }
      finish({ type: "quit", state });
    };
    stdin.setRawMode(true);
    stdin.resume();
    stdin.on("data", onData);
    render();
  });
}

async function readCockpitTuiInteractiveInput(model: CockpitTuiModel, options: { json: boolean }): Promise<string | undefined> {
  const testInput = process.env["OSSTACK_TEST_COCKPIT_TUI_INPUT"];

function concreteCockpitTuiCommand(command: string, workspace: string): string {
  return command.replaceAll("<path>", shellQuote(workspace));
}

function frameLine(value: string, width: number): string {
  const innerWidth = width - 4;
  const text = trimToWidth(value, innerWidth);
  return `| ${text}${" ".repeat(Math.max(0, innerWidth - visibleLength(text)))} |`;
}

function trimToWidth(value: string, width: number): string {
  if (visibleLength(value) <= width) return value;
  if (width <= 1) return value.slice(0, width);
  return `${value.slice(0, Math.max(0, width - 1))}~`;
}

function visibleLength(value: string): number {
  return [...value].length;
}

function clamp(value: number, min: number, max: number): number {
  if (max < min) return 0;
  return Math.min(Math.max(value, min), max);
}

function shellQuote(value: string): string {
  return `'${value.replaceAll("'", "'\\''")}'`;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

/** Display only. These observations never become executable TUI actions. */
function attentionLines(value: Record<string, unknown>): string[] {
  const text = attentionDisplayText;
  const totals = isRecord(value["totals"]) ? value["totals"] : undefined;
  const lines = [
    `${text(value["status"])} ${text(value["reason"] ?? "read_only")} | ${text(value["scope_filter"])} default tenant`,
  ];
  if (totals === undefined) lines.push("counts=unknown");
  else {
    lines.push(`open_items=${text(totals["open_items"])} uncertain_answers=${text(totals["uncertain_answers"])} pending_answers=${text(totals["pending_answers"])}`);
    lines.push(`unmonitored_subjects=${text(totals["unmonitored_subjects"])} dead_letters=${text(totals["dead_letters"])} scopes=${text(totals["scopes"])}`);
    lines.push(`applied_answers=${text(totals["applied_answers"])} (not task completion); missing_source_items=${text(totals["missing_source_items"])}`);
    lines.push(`human_waits=${text(totals["human_waits"])} system_waits=${text(totals["system_waits"])} unknown_waits=${text(totals["unknown_waits"])}`);
  }
  const scopes = Array.isArray(value["scopes"]) ? value["scopes"].filter(isRecord) : [];
  for (const scope of scopes) lines.push(`${text(scope["task_title"])}: ${text(scope["coverage"])} ${text(scope["reason"])} scope=${text(scope["scope_id"])}`);
  if (value["truncated"] === true) lines.push(`More scopes: returned=${text(value["returned"])} next_after=${text(value["next_after"])}`);
  lines.push("Read-only observation; use attention inventory/inbox for an explicit scope.");
  return lines;
}

      selection: {
        selected_index: 2,
        next_command: "osstack cockpit proof --workspace '/workspace/project' --json",
      },
    });
    expect(updateCockpitTuiFullscreenState(model, movedDown.state, "q")).toEqual({ type: "quit" });
