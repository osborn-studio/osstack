import { createHash } from "node:crypto";
import { z } from "zod";
import type { Capability, CapabilityKind, RoutableAdapter } from "@osstack/core";

const id = z.string().trim().min(1).max(256);
const text = z.string().min(1).max(1000);
const capabilityOutcome = z.discriminatedUnion("kind", [
  z.object({ kind: z.literal("selected"), adapter_id: id, capability_id: id, reason: text }).strict(),
  z.object({ kind: z.literal("rejected"), code: z.enum(["agent_not_found", "agent_unavailable", "agent_capability_mismatch"]) }).strict(),
  z.object({ kind: z.literal("none") }).strict(),
]);
const profileOutcome = z.discriminatedUnion("kind", [
  z.object({ kind: z.literal("profile_selected"), builder: id, evaluator: id, independent: z.literal(true), overrides: z.record(id, id) }).strict(),
  z.object({ kind: z.literal("profile_rejected"), message: text }).strict(),
]);
const role = z.object({ primary: id, fallback: z.array(id).max(32) }).strict();
const roles = z.object({ coding: role, "test-verify": role, design: role }).strict();

// Capability kinds come from the real runtime schema without eagerly importing
// the core control-plane barrel into every supervisor consumer.
function schemas(kinds: typeof import("@osstack/core").CapabilityKind) {
  const adapter = z.object({
    id, cost_tier: z.enum(["low", "mid", "high"]),
    availability: z.enum(["available", "unavailable", "error"]),
    capabilities: z.array(z.object({ id, kind: kinds }).strict()).max(64),
    identity: z.object({ provider_id: id, model_id: id }).strict().optional(),
  }).strict().superRefine((a, ctx) => {
    if (new Set(a.capabilities.map(c => c.id)).size !== a.capabilities.length)
      ctx.addIssue({ code: "custom", message: "duplicate_capability_id" });
  });
  const common = { id, rationale: text, adapters: z.array(adapter).max(32) };
  const capability = z.object({ ...common, entrypoint: z.literal("capability"),
    request: z.object({ capability_id: id.optional(), kind: kinds.optional(), agent_id: id.optional() }).strict()
      .refine(r => Object.keys(r).length > 0, "routing_request_required"),
    expected: capabilityOutcome,
  }).strict();
  const profile = z.object({ ...common, entrypoint: z.literal("profile"), roles,
    explicit_builder: id.optional(), explicit_evaluators: z.array(id).max(1),
    role_overrides: z.object({ coding: id.optional(), "test-verify": id.optional(), design: id.optional() }).strict(),
    expected: profileOutcome,
  }).strict();
  return z.object({ version: z.literal(1), dataset_id: id,
    cases: z.array(z.discriminatedUnion("entrypoint", [capability, profile])).min(1).max(256),
  }).strict().superRefine((data, ctx) => {
    if (new Set(data.cases.map(c => c.id)).size !== data.cases.length)
      ctx.addIssue({ code: "custom", path: ["cases"], message: "duplicate_case_id" });
    for (const [index, item] of data.cases.entries()) {
      if (new Set(item.adapters.map(a => a.id)).size !== item.adapters.length)
        ctx.addIssue({ code: "custom", path: ["cases", index, "adapters"], message: "duplicate_adapter_id" });
    }
  });
}
export type RoutingGoldenDataset = z.infer<ReturnType<typeof schemas>>;
export type RoutingGoldenCase = RoutingGoldenDataset["cases"][number];
export type ProfileRoutingGoldenCase = Extract<RoutingGoldenCase, { entrypoint: "profile" }>;
export type ProfileRoutingGoldenOutcome = z.infer<typeof profileOutcome>;
type Outcome = z.infer<typeof capabilityOutcome> | ProfileRoutingGoldenOutcome;

/** Fixture adapters provide facts only. No credentials, provider clients, or
 * subprocess probes are created. Any future execution attempt fails visibly. */
export interface RoutingEvaluationAdapter extends RoutableAdapter {
  readonly display_name: string;
  isAvailable(): Promise<{ available: boolean; last_checked_at: number }>;
  runtimeIdentity(): Promise<{
    adapter_id: string; adapter_version: string; auth_mode: string; provider_id: string;
    model_id: string; evidence_source: "configured";
  }>;
  invoke(): never;
}
export type ProfileRoutingGoldenRunner = (
  item: ProfileRoutingGoldenCase, adapters: RoutingEvaluationAdapter[],
) => Promise<unknown>;
export interface RoutingGoldenCaseResult {
  id: string; entrypoint: "capability" | "profile";
  status: "passed" | "failed" | "incomplete";
  expected: Outcome; actual: Outcome | null;
  differences: string[]; error?: string;
}
export interface RoutingGoldenReport {
  mode: "router_golden"; dataset_hash: string | null; dataset_id: string | null;
  qualification: "deterministic_regression_only";
  status: "passed" | "failed" | "incomplete";
  total: number; evaluated: number; passed: number; failed: number; incomplete: number;
  provider_invocations: number; default_enabled: false;
  results: RoutingGoldenCaseResult[]; errors: string[];
}
function canonical(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  if (value !== null && typeof value === "object")
    return `{${Object.keys(value).sort().map(k => `${JSON.stringify(k)}:${canonical((value as Record<string, unknown>)[k])}`).join(",")}}`;
  return JSON.stringify(value) ?? "null";
}
function differences(expected: Outcome, actual: Outcome): string[] {
  const a = expected as Record<string, unknown>, b = actual as Record<string, unknown>;
  return [...new Set([...Object.keys(a), ...Object.keys(b)])].sort()
    .filter(key => canonical(a[key]) !== canonical(b[key]));
}

/** Execute one fixed entrypoint per case, never choose between predictions.
 * Profile wiring is supplied by CLI to avoid a supervisor -> CLI dependency. */
export async function evaluateRoutingGolden(raw: unknown, profileRunner?: ProfileRoutingGoldenRunner): Promise<RoutingGoldenReport> {
  let digest: string | null = null;
  try { digest = createHash("sha256").update(canonical(raw)).digest("hex"); } catch { /* Invalid cyclic input is incomplete. */ }
  const rawCases = raw && typeof raw === "object" && "cases" in raw ? raw.cases : undefined;
  const total = Array.isArray(rawCases) ? rawCases.length : 0;
  const report: RoutingGoldenReport = {
    mode: "router_golden", dataset_hash: digest, dataset_id: null,
    qualification: "deterministic_regression_only", status: "incomplete",
    total, evaluated: 0, passed: 0, failed: 0, incomplete: total,
    provider_invocations: 0, default_enabled: false, results: [], errors: [],
  };
  const { CapabilityKind: kinds, routeCapability, AgentOverrideError } = await import("@osstack/core");
  const parsed = schemas(kinds).safeParse(raw);
  if (!parsed.success || digest === null) {
    report.errors = parsed.success ? ["unhashable_dataset"] : parsed.error.issues.slice(0, 30).map(i => `${i.path.join(".")}: ${i.message}`);
    return report;
  }
  report.dataset_id = parsed.data.dataset_id;
  report.incomplete = 0;
  for (const item of parsed.data.cases) {
    const adapters: RoutingEvaluationAdapter[] = item.adapters.map(a => ({
      id: a.id, display_name: a.id, cost_tier: a.cost_tier,
      capabilities: (): Capability[] => a.capabilities.map(c => ({ ...c, kind: c.kind as CapabilityKind,
        agent_id: a.id, description: `Golden fixture ${a.id}`, cost_tier: a.cost_tier, hints: {}, version: "1" })),
      isAvailable: async () => {
        if (a.availability === "error") throw Error("fixture_probe_failed");
        return { available: a.availability === "available", last_checked_at: 1 };
      },
      runtimeIdentity: async () => ({ adapter_id: a.id, adapter_version: "fixture-v1", auth_mode: "fixture",
        provider_id: a.identity?.provider_id ?? `fixture:${a.id}`, model_id: a.identity?.model_id ?? "fixture-model",
        evidence_source: "configured" }),
      invoke: () => { report.provider_invocations++; throw Error("provider_invocation_forbidden"); },
    }));
    const result: RoutingGoldenCaseResult = { id: item.id, entrypoint: item.entrypoint,
      status: "incomplete", expected: item.expected, actual: null, differences: [] };
    const invokesBefore = report.provider_invocations;
    try {
      let actual: Outcome;
      if (item.entrypoint === "capability") {
        try {
          const picked = await routeCapability({
            ...(item.request.agent_id === undefined ? {} : { agent_id: item.request.agent_id }),
            ...(item.request.capability_id === undefined ? {} : { capability_id: item.request.capability_id }),
            ...(item.request.kind === undefined ? {} : { kind: item.request.kind }),
          }, adapters);
          actual = picked ? { kind: "selected", adapter_id: picked.adapter.id, capability_id: picked.capability.id, reason: picked.reason } : { kind: "none" };
        } catch (error) {
          if (!(error instanceof AgentOverrideError)) throw error;
          actual = { kind: "rejected", code: error.code };
        }
      } else {
        if (!profileRunner) throw Error("profile_runner_unavailable");
        const value = await profileRunner(item, adapters);
        const checked = profileOutcome.safeParse(value);
        if (!checked.success) throw Error("profile_result_missing_or_invalid");
        actual = checked.data;
      }
      if (report.provider_invocations !== invokesBefore) throw Error("provider_invocation_forbidden");
      result.actual = actual;
      result.differences = differences(item.expected, actual);
      result.status = result.differences.length ? "failed" : "passed";
      report.evaluated++;
      report[result.status]++;
    } catch (error) {
      result.error = error instanceof Error ? error.message : "routing_evaluation_failed";
      report.incomplete++;
    }
    report.results.push(result);
  }
  report.status = report.incomplete || report.provider_invocations ? "incomplete" : report.failed ? "failed" : "passed";
  return report;
}

import { selectProfile, type OsstackConfig } from "@osstack/config";
import type { ProfileRoutingGoldenRunner } from "@osstack/supervisor";
import { resolveLoopProfileRouting } from "./loop-profile-routing.js";

/** Use the production profile entrypoint with a complete fixture configuration.
 * Neither the user's config, machine profile, nor installed adapters are read. */
export const runProfileRoutingGolden: ProfileRoutingGoldenRunner = async (item, adapters) => {
  const config: OsstackConfig = {
    version: 1, active_profile: "golden", agent_runtime: {},
    defaults: { roles: item.roles },
    profiles: { golden: { roles: item.roles, match: { hostnames: [], model_hints: [] } } },
  };
  const args = ["--profile", "golden"];
  for (const [role, adapter] of Object.entries(item.role_overrides)) {
    if (adapter !== undefined) args.push("--role", `${role}=${adapter}`);
  }
  const result = await resolveLoopProfileRouting(args, item.explicit_builder, item.explicit_evaluators, {
    loadConfig: () => config,
    selectProfile: (value, options) => selectProfile(value, { ...options, hostname: "golden-fixture", model: "" }),
    adapters: () => adapters,
  });
  if (result instanceof Error) return { kind: "profile_rejected", message: result.message };
  if (!result) return undefined;
  return { kind: "profile_selected", builder: result.agentId, evaluator: result.verifierAgentIds[0],
    independent: result.summary.independent, overrides: result.summary.overrides };
};

import { open, writeFile, mkdir, realpath } from "node:fs/promises";
import { join } from "node:path";
import { randomUUID } from "node:crypto";
export async function handleAttentionPlanning(
  args: string[],
  out: (s: string) => void
): Promise<number> {
  const flags = new Map<string, string>();
  for (let i = 1; i < args.length; i++) {
    if (args[i] === "--json") continue;
    const key = args[i]!;
    const value = args[++i];
    if (
      !["--input", "--head", "--evidence", "--budget", "--repo", "--mode"].includes(
        key
      ) ||
      !value ||
      value.startsWith("--") ||
      flags.has(key)
    )
      throw new Error("invalid_planning_arguments");
    flags.set(key, value);
  }
  const read = async (path: string) => {
    const file = await open(path, "r");
    try {
      const data = Buffer.alloc(1048577);
      const result = await file.read(data, 0, data.length, 0);
      if (result.bytesRead > 1048576) throw new Error("oversized_input");
      return JSON.parse(data.subarray(0, result.bytesRead).toString());
    } finally {
      await file.close();
    }
  };
  const input = flags.get("--input");
  if (!input) throw new Error("input_required");
  const planning = await import("@osstack/supervisor");
  const mode = flags.get("--mode") ?? "supplied_predictions";
  if (flags.has("--mode") && args[0] !== "routing-eval") throw new Error("mode_only_valid_for_routing_eval");
  if (args[0] === "routing-eval") {
    if (!["supplied_predictions", "router_golden"].includes(mode)) throw new Error("invalid_routing_eval_mode");
    if ([...flags.keys()].some(k => !["--input", "--mode"].includes(k))) throw new Error("invalid_routing_eval_arguments");
    if (mode === "router_golden") {
      let raw: unknown;
      try { raw = await read(input); }
      catch {
        const report = await planning.evaluateRoutingGolden(null);
        report.errors = ["input_unreadable_or_invalid_json"];
        out(JSON.stringify(report, null, 2));
        return 1;
      }
      const { runProfileRoutingGolden } = await import("./attention-routing-evaluation.js");
      const report = await planning.evaluateRoutingGolden(raw, runProfileRoutingGolden);
      out(JSON.stringify(report, null, 2));
      return report.status === "passed" ? 0 : 1;
    }
  }
  const data = await read(input);
  let result: unknown;
  switch (args[0]) {
    case "contract":
      result = flags.has("--evidence")

        ? planning.promoteScout(data, flags.get("--head")!)

      result = { ...planning.evaluateAttentionRouting(data), mode: "supplied_predictions",
        qualification: "supplied_predictions_only", router_executed: false };
      break;
    case "triage":
      result = planning.acceptAttentionTriage(data, 1, {
        category: "source_uncertain",
        explanation:
          "Use deterministic source facts; model output unavailable or invalid.",
      });
      break;
    case "context":
      result = planning.budgetAttentionContext(
        data,
        Number(flags.get("--budget") ?? 8000),
        Date.now()
      );
      break;
    case "lesson": {
      if (!flags.has("--repo")) throw new Error("repo_required");

      const { renderLessonDraft } = await import("@osstack/loop");

      const directory = join(repo, "loop-lessons", "drafts");
      await mkdir(directory, { recursive: true });
      const path = join(directory, `attention-${randomUUID()}.md`);
      await writeFile(path, content, { flag: "wx" });
      result = { path, state: "draft", promoted: false };
      break;
    }
    default:
      throw new Error("unknown_planning_command");
  }
  out(JSON.stringify(result, null, 2));
  return 0;

# Deterministic routing regression evaluation

Run the production routing functions against an explicit, reviewable fixture matrix:

```sh
osstack attention routing-eval --mode router_golden --input packages/supervisor/src/fixtures/routing-golden-v1.json --json
```

When running a development checkout, set an explicit isolated `OSSTACK_HOME` as required by the repository guard. This command does not start a daemon, open the state DB, load the user's agent configuration, or create installed provider adapters. Availability, capabilities, cost tiers and runtime identities are fixture facts. A guarded fixture adapter records and rejects any attempt to invoke a provider.

The [version 1 matrix](../../packages/supervisor/src/fixtures/routing-golden-v1.json) contains 22 cases with label rationales. Each case fixes exactly one entrypoint:

- `capability` calls the actual `routeCapability` with capability/kind, optional explicit agent, fixture adapter availability and cost tiers. It compares the adapter, capability and reason, or the exact override rejection code/no candidate outcome.
- `profile` calls the actual `resolveLoopProfileRouting` with a complete fixture role configuration, a fixed profile and optional explicit builder/evaluator or role preferences. It compares builder/evaluator selection, adapter independence, recorded overrides or the exact rejection message.

The profile bridge is supplied by the CLI because supervisor cannot depend on CLI. Calling the supervisor evaluator directly without that bridge leaves profile cases `incomplete`; it does not substitute a fake profile prediction.

## Report and exit status

The report contains `mode: router_golden`, `qualification: deterministic_regression_only`, the canonical input SHA-256 `dataset_hash`, dataset ID, total case denominator, evaluated/passed/failed/incomplete counts, provider invocation count and every case's expected value, actual value, differences or incomplete reason. Object keys are canonicalized for hashing; case/adapter ordering and labels are retained in the hash. The hash identifies input data, not a signed qualification certificate or router build version.

Exit 0 requires all cases to produce valid matching results with zero provider invocations. Mismatches exit 1 and remain in the denominator. Missing results, invalid expectations, duplicate IDs/adapters, empty/invalid datasets and unexpected execution errors produce `incomplete` and exit 1. Dataset validation happens before any routing. For invalid datasets, no cases are executed; `total` is the supplied cases-array length, `incomplete` equals that length, and `errors` explains the rejection. A missing/non-array cases field has total 0 and an explicit incomplete status. Unreadable or invalid JSON inputs also produce an incomplete report.

Input is limited to 1 MiB; the schema caps datasets at 256 cases, 32 adapters per case and 64 capabilities per adapter. Unknown fields are rejected, including supplied predictions. Each profile case must explicitly provide all three role bindings, an `explicit_evaluators` array and a `role_overrides` object. The fixture is the versioned, fully worked format reference.

## What the current routers actually guarantee

Explicit capability `agent_id`, profile `explicit_builder` and profile `explicit_evaluators` are tested for failure-closed behavior. An unavailable pinned adapter must not silently pass by selecting a substitute.

The existing `--role` preference can fall back to configured alternatives; it is not the same contract as a hard evaluator pin. The matrix records that behavior rather than changing production policy. The capability router also has a legacy first-available fallback when no kind matches. Generic capability routing therefore does not establish review-role safety. The profile resolver enforces distinct builder/evaluator adapters at its current `adapter` independence level; it does not prove different providers/models, even when identities are configured.

This matrix is a deterministic regression suite, not a holdout study, model-intent classifier, production availability test, permission check, or token/cost savings experiment. It does not change the default router. A passing report means these reviewed expectations match this invocation of the routing code; it does not mean every existing routing behavior is desirable or safe for every caller. Real provider comparisons remain a separate qualification stage.

## Legacy supplied-prediction aggregation

Existing usage remains compatible:

```sh
osstack attention routing-eval --input routing-results.json --json
osstack attention routing-eval --mode supplied_predictions --input routing-results.json --json
```

Both aggregate the caller's existing prediction/label/usage rows. Reports retain those metrics and add `mode: supplied_predictions`, `qualification: supplied_predictions_only` and `router_executed: false`. They do not run the router or verify the caller's claimed predictions or token usage. Choose `router_golden` explicitly for actual deterministic router execution; unknown modes fail instead of falling back.

// Capability router (S2 basic version).
//
// Given a task's requested capability_id (or capability kind) and a set of
// available adapters, pick which adapter should run it. S3 will replace
// this with the data-driven policy engine; for now we keep it transparent
// and rule-based so users can reason about routing.
//
// Selection rules (in order, first match wins):
//   1. Exact capability_id match if the requested adapter is available.
//   2. Same kind, lowest cost_tier (low > mid > high preference).
//   3. Same kind, any available adapter.
//   4. None — return null and let the caller decide.

// @osstack/agents depends on @osstack/core for the Capability schema, so to
// avoid a cycle we re-declare the minimum routable-adapter surface here
// (just the fields the router actually reads).
import type { Capability, CapabilityKind, CostTier } from "../capability.js";

export interface RouterAvailability {
  available: boolean;
  reason?: string;
}

export interface RoutableAdapter {
  readonly id: string;
  readonly cost_tier: CostTier;
  capabilities(): readonly Capability[];
  isAvailable(): Promise<RouterAvailability>;
}

export interface RoutingRequest {
  capability_id?: string;
  kind?: CapabilityKind;
  /** Optional override: force this agent if it can serve the requested kind. */
  agent_id?: string;
}

export interface RoutingDecision {
  adapter: RoutableAdapter;
  capability: Capability;
  reason: string;
}

const TIER_ORDER: Record<CostTier, number> = { low: 0, mid: 1, high: 2 };

/** Explicit agent_id override could not be honored. FAIL-CLOSED: the caller
 *  asked for a specific agent, so silently substituting a different (often
 *  cheaper) agent is never acceptable — a review "performed" by the wrong
 *  engine is worse than an error (Phase 2 closure G1.1; incident 2026-07-18). */
export class AgentOverrideError extends Error {
  readonly code: "agent_not_found" | "agent_unavailable" | "agent_capability_mismatch";
  readonly agent_id: string;
  constructor(code: AgentOverrideError["code"], agentId: string, detail?: string) {
    super(`agent override failed (${code}): ${agentId}${detail ? ` — ${detail}` : ""}`);
    this.name = "AgentOverrideError";
    this.code = code;
    this.agent_id = agentId;
  }
}

export interface RouteOptions {
  /** Bound on a single adapter availability probe when honoring an explicit
   *  agent_id override. A hanging probe becomes agent_unavailable, never a
   *  silent fallback. */
  probeTimeoutMs?: number;
}

const DEFAULT_PROBE_TIMEOUT_MS = 10_000;

export async function routeCapability(
  request: RoutingRequest,
  adapters: ReadonlyArray<RoutableAdapter>,
  options: RouteOptions = {},
): Promise<RoutingDecision | null> {
  if (request.agent_id !== undefined && (typeof request.agent_id !== "string" || request.agent_id.length === 0)) {
    throw new AgentOverrideError("agent_not_found", "(invalid)", "agent_id must be a non-empty string");
  }
  if (adapters.length === 0) {
    if (request.agent_id !== undefined) throw new AgentOverrideError("agent_not_found", request.agent_id, "no adapters registered");
    return null;
  }

  // 1. Exact agent override — FAIL-CLOSED. When agent_id is set the ONLY
  //    outcomes are (a) that agent, or (b) AgentOverrideError. Never fall
  //    through to capability/kind routing.
  if (request.agent_id !== undefined) {
    const adapter = adapters.find((a) => a.id === request.agent_id);
    if (!adapter) throw new AgentOverrideError("agent_not_found", request.agent_id);
    const timeoutMs = options.probeTimeoutMs ?? DEFAULT_PROBE_TIMEOUT_MS;
    let avail: RouterAvailability;
    let timer: ReturnType<typeof setTimeout> | undefined;
    try {
      avail = await Promise.race([
        adapter.isAvailable(),
        new Promise<never>((_, reject) => {
          timer = setTimeout(() => reject(new Error(`availability probe timed out after ${timeoutMs}ms`)), timeoutMs);
          (timer as { unref?: () => void }).unref?.();
        }),
      ]);
    } catch (err) {
      throw new AgentOverrideError("agent_unavailable", request.agent_id, err instanceof Error ? err.message : String(err));
    } finally {
      if (timer !== undefined) clearTimeout(timer);
    }
    if (!avail.available) {
      throw new AgentOverrideError("agent_unavailable", request.agent_id, avail.reason ?? "adapter reports unavailable");
    }
    // STRICT capability selection under an override: a supplied capability_id
    // must match exactly; a supplied kind must match a capability of that kind;
    // only a fully unspecified request may take the adapter's first capability.
    let cap: Capability | undefined;
    if (request.capability_id !== undefined) {
      cap = adapter.capabilities().find((c) => c.id === request.capability_id);
    } else if (request.kind !== undefined) {
      cap = adapter.capabilities().find((c) => c.kind === request.kind);
    } else {
      cap = adapter.capabilities()[0];
    }
    if (!cap) {
      throw new AgentOverrideError("agent_capability_mismatch", request.agent_id, `no capability matching ${request.capability_id ?? request.kind ?? "(unspecified)"}`);
    }
    return { adapter, capability: cap, reason: "agent_id override" };
  }

  // 2. Exact capability_id match.
  if (request.capability_id) {
    for (const adapter of adapters) {
      const cap = adapter.capabilities().find((c) => c.id === request.capability_id);
      if (cap && await isAvailable(adapter)) {
        return { adapter, capability: cap, reason: "exact capability_id match" };
      }
    }
  }

  // 3. Kind-based: prefer lowest cost tier.
  const kind = request.kind ?? capabilityKindFromId(request.capability_id);
  if (kind) {
    const candidates: Array<{ adapter: RoutableAdapter; cap: Capability }> = [];
    for (const adapter of adapters) {
      const cap = adapter.capabilities().find((c) => c.kind === kind);
      if (cap && await isAvailable(adapter)) candidates.push({ adapter, cap });
    }
    candidates.sort((a, b) => TIER_ORDER[a.adapter.cost_tier] - TIER_ORDER[b.adapter.cost_tier]);
    if (candidates.length > 0) {
      const winner = candidates[0]!;
      return {
        adapter: winner.adapter,
        capability: winner.cap,
        reason: `kind=${kind} lowest-cost (${winner.adapter.cost_tier})`,
      };
    }
  }

  // 4. Fall back to the first available adapter's first capability.
  for (const adapter of adapters) {
    if (await isAvailable(adapter)) {
      const cap = adapter.capabilities()[0];
      if (cap) return { adapter, capability: cap, reason: "fallback first-available" };
    }
  }

  return null;
}

async function isAvailable(adapter: RoutableAdapter): Promise<boolean> {
  try {
    const a = await adapter.isAvailable();
    return a.available;
  } catch {
    return false;
  }
}


// Known CapabilityKind values, hand-mirrored from capability.ts. Kept here
// (rather than re-extracted from the Zod enum) so this module stays a leaf
// import — no circular dance with the Zod schema. If you add a kind there,
// add it here too. The router test exercises the round-trip.
const KIND_SET = new Set<string>([
  // Software lifecycle
  "code.write", "code.review", "review.code", "code.refactor", "code.test", "code.debug",
  "design.prd", "design.architecture", "design.review",
  "doc.write", "doc.summarize",
  "plan.decompose", "plan.estimate",
  "research.web", "research.code",
  "release.check", "release.verify",
  "ops.deploy", "ops.rollback",
  // Professional services (PartnerOS)
  "intent.draft", "intent.clarify", "intent.assign",
  "client.followup", "client.summarize",
  "matter.summarize", "matter.checklist",
  "risk.check", "verify.acceptance", "verify.alignment",
  "artifact.create", "artifact.revise", "approval.request",
  "memory.write", "memory.read",
  // Autonomous ML research (karpathy/autoresearch)
  "research.ml", "research.experiment",
]);

function capabilityKindFromId(id: string | undefined): CapabilityKind | undefined {
  if (!id) return undefined;
  // 1. If the whole id is a known kind ("code.write"), use it directly.
  if (KIND_SET.has(id)) return id as CapabilityKind;
  // 2. If it looks like "<agent_id>.<kind>" and the suffix is a known kind,
  //    extract that. Greedy: try the longest dotted suffix first so we don't
  //    confuse "ops.deploy" with "code.deploy".
  const parts = id.split(".");
  for (let i = 1; i < parts.length; i++) {
    const candidate = parts.slice(i).join(".");
    if (KIND_SET.has(candidate)) return candidate as CapabilityKind;
  }
  return undefined;
}

export function parseRoleOverrides(
  args: string[],
): Partial<Record<AgentRole, string>> | Error {
  const overrides: Partial<Record<AgentRole, string>> = {};
  for (const raw of flagValues(args, "--role")) {
    const [roleRaw, agentId] = raw.split("=", 2);
    if (
      roleRaw === undefined
      || agentId === undefined
      || agentId.trim() === ""
    ) {
      return new Error(`--role must use role=agent-id, got "${raw}"`);
    }
    const role = normalizeLoopRole(roleRaw);
    if (role === undefined) {
      return new Error(
        `unknown --role "${roleRaw}"; use coder, evaluator, or designer`,
      );
    }
    overrides[role] = agentId.trim();
  }
  return overrides;
}

export function normalizeLoopRole(
  value: string,
): AgentRole | undefined {
  const normalized = value.trim().toLowerCase();
  if (
    normalized === "coder"
    || normalized === "builder"
    || normalized === "coding"
  ) {
    return "coding";
  }
  if (
    normalized === "evaluator"
    || normalized === "judge"
    || normalized === "verifier"
    || normalized === "test-verify"
  ) {
    return "test-verify";
  }
  if (
    normalized === "designer"
    || normalized === "design"
  ) {
    return "design";
  }
  return undefined;
}

function flagValue(
  args: readonly string[],
  name: string,
): string | undefined {
  const index = args.indexOf(name);
  return index >= 0 && index + 1 < args.length
    ? args[index + 1]
    : undefined;
}

function flagValues(
  args: readonly string[],
  name: string,
): string[] {
  const values: string[] = [];
  for (let index = 0; index < args.length - 1; index += 1) {
    if (args[index] === name) {
      values.push(args[index + 1]!);
      index += 1;
    }
  }
  return values;

// Osstack role resolution: turn a RoleBinding (primary + fallback) plus the set
// of registered adapters into a concrete choice, probing isAvailable() so an
// agent that isn't installed on this machine is skipped in favour of the next.

import type {
  AgentAdapter,
  AgentRuntimeIdentity,
} from "@osstack/agents";
import type { RoutableAdapter } from "@osstack/core";
import { resolveRuntimeIdentity } from "./runtime-identity.js";
import {
  ALL_ROLES,
  ROLE_KINDS,
  type AgentRole,
  type AgentRuntimeIdentityOverride,
  type RoleBinding,
  type RoleMap,
} from "./types.js";

export interface RoleResolution {
  role: AgentRole;
  /** Chosen adapter id, or null if nothing usable is available. */
  chosen: string | null;
  reason: string;
  /** Every candidate we considered and why it was (not) picked. */
  tried: Array<{ id: string; ok: boolean; reason?: string }>;
  available: boolean;
}

function advertises(adapter: RoutableAdapter, role: AgentRole): boolean {
  const kinds = ROLE_KINDS[role];
  return adapter.capabilities().some((c) => kinds.includes(c.kind));
}

async function safeAvailable(
  adapter: RoutableAdapter,
): Promise<{ available: boolean; reason?: string }> {
  try {
    const a = await adapter.isAvailable();
    return a.reason !== undefined ? { available: a.available, reason: a.reason } : { available: a.available };
  } catch (err) {
    return { available: false, reason: err instanceof Error ? err.message : String(err) };
  }
}

export interface ResolveOpts {
  /** If no configured agent is available, try any other adapter for the role. */
  autoFallback?: boolean;
  /** Adapter ids that cannot serve this resolution because another role owns them. */
  excludeAgentIds?: ReadonlySet<string>;
}

export async function resolveRole(
  role: AgentRole,
  binding: RoleBinding,
  adapters: ReadonlyArray<RoutableAdapter>,
  opts: ResolveOpts = {},
): Promise<RoleResolution> {
  const byId = new Map(adapters.map((a) => [a.id, a]));
  const tried: RoleResolution["tried"] = [];
  const order = [binding.primary, ...binding.fallback];

  for (const id of order) {
    if (opts.excludeAgentIds?.has(id) === true) {
      tried.push({
        id,
        ok: false,
        reason: "excluded by role-separation policy",
      });
      continue;
    }
    const adapter = byId.get(id);
    if (!adapter) {
      tried.push({ id, ok: false, reason: "not registered" });
      continue;
    }
    if (!advertises(adapter, role)) {
      tried.push({ id, ok: false, reason: `does not serve role "${role}"` });
      continue;
    }
    const av = await safeAvailable(adapter);
    if (av.available) {
      const isPrimary = id === binding.primary;
      return {
        role,
        chosen: id,
        reason: isPrimary
          ? `primary "${id}" available`
          : `fell back to "${id}" (primary "${binding.primary}" unavailable)`,
        tried,
        available: true,
      };
    }
    tried.push({ id, ok: false, reason: av.reason ?? "unavailable" });
  }

  if (opts.autoFallback) {
    for (const adapter of adapters) {
      if (order.includes(adapter.id)) continue;
      if (opts.excludeAgentIds?.has(adapter.id) === true) {
        tried.push({
          id: adapter.id,
          ok: false,
          reason: "excluded by role-separation policy",
        });
        continue;
      }
      if (!advertises(adapter, role)) continue;
      const av = await safeAvailable(adapter);
      if (av.available) {
        return {
          role,
          chosen: adapter.id,
          reason: `auto-fallback to "${adapter.id}" (no configured agent available)`,
          tried,
          available: true,
        };
      }
      tried.push({ id: adapter.id, ok: false, reason: av.reason ?? "unavailable" });
    }
  }

  return {
    role,
    chosen: null,
    reason: `no available agent for role "${role}" (tried ${order.join(", ") || "none"})`,
    tried,
    available: false,
  };
}

export async function resolveAll(
  roles: RoleMap,
  adapters: ReadonlyArray<RoutableAdapter>,
  opts: ResolveOpts = {},
): Promise<Record<AgentRole, RoleResolution>> {
  const out = {} as Record<AgentRole, RoleResolution>;
  for (const role of ALL_ROLES) {
    out[role] = await resolveRole(role, roles[role], adapters, opts);
  }
  return out;
}

export type RuntimeIndependenceLevel = "adapter" | "provider-or-model";

export interface ResolveDeliveryRolesOpts extends ResolveOpts {
  identityOverrides?: Readonly<
    Record<string, AgentRuntimeIdentityOverride>
  >;
  independenceLevel?: RuntimeIndependenceLevel;
}

export interface DeliveryRoleResolution {
  builder: RoleResolution;
  evaluator: RoleResolution;
  designer: RoleResolution;
  identities: {
    builder?: AgentRuntimeIdentity;
    evaluator?: AgentRuntimeIdentity;
    designer?: AgentRuntimeIdentity;
  };
  independent: boolean;
  independence_reason: string;
}

export function runtimeIdentitiesIndependent(
  builder: AgentRuntimeIdentity,
  evaluator: AgentRuntimeIdentity,
  level: RuntimeIndependenceLevel,
): { pass: boolean; reason: string } {
  if (builder.adapter_id === evaluator.adapter_id) {
    return {
      pass: false,
      reason: "builder and evaluator use the same adapter",
    };
  }
  if (level === "adapter") {
    return {
      pass: true,
      reason: "builder and evaluator use distinct adapters",
    };
  }
  if (
    builder.evidence_source !== "reported"
    || evaluator.evidence_source !== "reported"
  ) {
    return {
      pass: false,
      reason: "configured/inferred identity cannot prove provider independence",
    };
  }
  if (
    builder.provider_id === "unknown"
    || builder.model_id === "unknown"
    || evaluator.provider_id === "unknown"
    || evaluator.model_id === "unknown"
  ) {
    return {
      pass: false,
      reason: "reported identity has unknown provider or model",
    };
  }
  if (
    builder.provider_id !== evaluator.provider_id
    || builder.model_id !== evaluator.model_id
  ) {
    return {
      pass: true,
      reason: "reported provider or model differs",
    };
  }
  return {
    pass: false,
    reason: "builder and evaluator share provider and model",
  };
}

export async function resolveDeliveryRoles(
  roles: RoleMap,
  adapters: ReadonlyArray<AgentAdapter>,
  opts: ResolveDeliveryRolesOpts = {},
): Promise<DeliveryRoleResolution> {
  const builder = await resolveRole(
    "coding",
    roles.coding,
    adapters,
    opts,
  );
  const evaluatorExclusions = new Set(opts.excludeAgentIds);
  if (builder.chosen !== null) {
    evaluatorExclusions.add(builder.chosen);
  }
  const evaluator = await resolveRole(
    "test-verify",
    roles["test-verify"],
    adapters,
    {
      ...opts,
      excludeAgentIds: evaluatorExclusions,
    },
  );
  const designer = await resolveRole(
    "design",
    roles.design,
    adapters,
    opts,
  );
  const byId = new Map(adapters.map((adapter) => [adapter.id, adapter]));
  const identities: DeliveryRoleResolution["identities"] = {};

  if (builder.chosen !== null) {
    const adapter = byId.get(builder.chosen);
    if (adapter !== undefined) {
      identities.builder = await resolveRuntimeIdentity(
        adapter,
        opts.identityOverrides?.[adapter.id],
      );
    }
  }
  if (evaluator.chosen !== null) {
    const adapter = byId.get(evaluator.chosen);
    if (adapter !== undefined) {
      identities.evaluator = await resolveRuntimeIdentity(
        adapter,
        opts.identityOverrides?.[adapter.id],
      );
    }
  }
  if (designer.chosen !== null) {
    const adapter = byId.get(designer.chosen);
    if (adapter !== undefined) {
      identities.designer = await resolveRuntimeIdentity(
        adapter,
        opts.identityOverrides?.[adapter.id],
      );
    }
  }

  const independence = identities.builder === undefined
    || identities.evaluator === undefined
    ? {
        pass: false,
        reason: "builder or evaluator unavailable",
      }
    : runtimeIdentitiesIndependent(
        identities.builder,
        identities.evaluator,
        opts.independenceLevel ?? "adapter",
      );

  return {
    builder,
    evaluator,
    designer,
    identities,
    independent: independence.pass,
    independence_reason: independence.reason,
  };
}

    version: 1,
    active_profile: "auto",
    profiles: {
      "mac-mini": {

        },

export function configExists(): boolean {
  return existsSync(configPath());
}

export function loadConfig(): OsstackConfigT {
  const path = configPath();
  if (!existsSync(path)) return defaultConfig();
  const raw = readFileSync(path, "utf8");
  return OsstackConfig.parse(parseYaml(raw) ?? {});
}

export function saveConfig(config: OsstackConfigT): string {
  const path = configPath();
  mkdirSync(dirname(path), { recursive: true });
  // Re-parse to normalise (apply defaults, drop unknown keys) before writing.
  writeFileSync(path, stringifyYaml(OsstackConfig.parse(config)), "utf8");
  return path;
}

export interface ResolvedProfile {
  name: string;
  roles: RoleMap;
  reason: string;
}

export interface SelectOpts {
  hostname?: string;
  model?: string;
  /** Explicit override (CLI --profile, $OSSTACK_AGENT_PROFILE, or $OSSTACK_PROFILE). */
  override?: string | undefined;
}

/**
 * Pick the active profile. Priority:
 *   1. explicit override (flag/env)
 *   2. active_profile, when not "auto"
 *   3. hostname match against each profile's match.hostnames
 *   4. defaults
 */
export function selectProfile(config: OsstackConfigT, opts: SelectOpts = {}): ResolvedProfile {
  const hostname = opts.hostname ?? osHostname();
  const model = opts.model ?? currentMacModel();
  const pin =
    opts.override ??
    (config.active_profile !== "auto" ? config.active_profile : undefined);

  if (pin) {
    const p = config.profiles[pin];
    if (p) return { name: pin, roles: p.roles, reason: `pinned profile "${pin}"` };
    return {
      name: "defaults",
      roles: config.defaults.roles,
      reason: `pinned profile "${pin}" not found; using defaults`,
    };
  }

  const target = hostname.toLowerCase();
  const modelTarget = model.toLowerCase();
  for (const [name, p] of Object.entries(config.profiles)) {
    if (p.match.hostnames.some((h) => h.toLowerCase() === target)) {
      return { name, roles: p.roles, reason: `hostname "${hostname}" matched profile "${name}"` };
    }
    if (modelTarget !== "" && p.match.model_hints.some((hint) => modelTarget.includes(hint.toLowerCase()))) {
      return { name, roles: p.roles, reason: `model "${model}" matched profile "${name}"` };
    }
  }

  return {
    name: "defaults",
    roles: config.defaults.roles,
    reason: `no profile matched hostname "${hostname}"; using defaults`,
  };
}

export function currentHostname(): string {
  return osHostname();
}

export function currentMacModel(): string {
  try {
    return execFileSync("sysctl", ["-n", "hw.model"], {
      encoding: "utf8",
      stdio: ["ignore", "pipe", "ignore"],
    }).trim();
  } catch {
    return "";
  }
}
