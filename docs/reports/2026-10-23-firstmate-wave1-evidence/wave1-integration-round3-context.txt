import { AttentionInbox } from "@osstack/protocol";
import { randomUUID } from "node:crypto";
import type Database from "better-sqlite3";
import {
  AttentionStore,
  AttentionObserver,
  AttentionExperience,
  AttentionActions,
  attentionDigest,
  type AttentionSender,
} from "@osstack/core";
type LoopReadInput = { repo: string; path: string; loop_id: string; signal: AbortSignal };
export interface SupervisionOptions {
  loopReader?: (input: LoopReadInput) => Promise<unknown>;
  readTimeoutMs?: number;
}
type ReadFailure = "timeout" | "aborted" | "invalid_payload" | "read_failed" | "read_capacity_exceeded";
type LoopFacts = Record<string, unknown> & { state: string; reason?: string };
interface ReadSlot {
  controller: AbortController;
  logical: Promise<LoopFacts>;
}
interface LoopBinding {
  binding_hash: string;
  admitted: number;
  generation: number;
  payload_json: string;
}
function loopPayload(json: string): Record<string, unknown> | null {
  try {
    const value: unknown = JSON.parse(json);
    return value && typeof value === "object" && !Array.isArray(value)
      ? value as Record<string, unknown> : null;
  } catch { return null; }
}
function validLoopFacts(value: unknown, id: string): value is LoopFacts {
  if (!value || typeof value !== "object") return false;
  const v = value as Record<string, unknown>;
  if (v.loop_id !== id || typeof v.path !== "string" || v.process_liveness !== "unknown") return false;
  if (v.state === "source_missing") return v.coverage === "uncovered" && v.gap === "source_missing";
  return v.state === "snapshot_available" && v.coverage === "snapshot_only"
    && v.gap === "best_effort_snapshot_no_event_replay" && typeof v.updated_at === "string"
    && Array.isArray(v.in_progress) && Array.isArray(v.pending)
    && Number.isSafeInteger(v.completed_count) && (v.completed_count as number) >= 0
    && typeof v.recovery_required === "boolean";
}

export class SupervisionService {
  readonly store: AttentionStore;
  readonly observer: AttentionObserver;
  readonly experience: AttentionExperience;
  private delivering = false;
  private deliveryRecoveryNeeded = false;
  private readonly deliveryAbort = new AbortController();
  private deliveryDone: Promise<void> | undefined;
  private ciReader: import("@osstack/github").AttentionCiReader | undefined;
  private timer: ReturnType<typeof setInterval> | undefined;
  private started = false;
  private stopped = false;
  private dirty = false;
  private scanDone: Promise<void> | undefined;
  private stopDone: Promise<void> | undefined;
  private epoch = 0;
  // A slot outlives its logical timeout until the underlying promise settles.
  // It holds no continuation that can access SQLite after stop.
  private readonly readers = new Map<string, ReadSlot>();
  private readonly logicalReads = new Set<Promise<LoopFacts>>();
  private readonly readTimeoutMs: number;
  constructor(
    db: Database.Database,
    path: string,
    readonly sender?: AttentionSender,
    private readonly options: SupervisionOptions = {}
  ) {
    this.readTimeoutMs = options.readTimeoutMs ?? 1000;
    if (!Number.isFinite(this.readTimeoutMs) || this.readTimeoutMs < 1 || this.readTimeoutMs > 1000)
      throw new Error("invalid_read_timeout");
    this.store = new AttentionStore(db, path);
    this.observer = new AttentionObserver(
      this.store,
      this.store.claimOwner(randomUUID())
    );
    this.experience = new AttentionExperience(this.store, this.observer.owner);
    this.experience.recoverDeliveries();
    new AttentionActions(this.store, this.observer.owner).recover();
  }
  private assertRunning(): void {
    if (this.stopped) throw new Error("supervision_stopped");
  }
  start(): void {
    this.assertRunning();
    if (this.started) return;
    this.started = true;
    void this.requestScan("start").catch(() => {});
    this.timer = setInterval(() => { void this.requestScan("timer").catch(() => {}); }, 30000);
    this.timer.unref();
  }
  /** Dirty signals are hints; each pass still reads the persisted source facts. */
  requestScan(_reason?: string): Promise<void> {
    if (this.stopped) return Promise.reject(new Error("supervision_stopped"));
    this.dirty = true;
    this.scanDone ??= Promise.resolve().then(async () => {
      while (this.dirty && !this.stopped) {
        this.dirty = false;
        try { await this.scan(); } catch { /* prior coverage expires */ }
      }
    }).finally(() => {
      this.scanDone = undefined;
      // A dirty signal can arrive after the loop exits but before this cleanup.
      if (this.dirty && !this.stopped) return this.requestScan("coalesced");
      return undefined;
    });
    return this.scanDone;
  }
  private async scan(): Promise<void> {
    this.assertRunning();
    this.observer.tick();
    const loops = this.store.db.prepare(
      "SELECT scope_id,subject_id,payload_json FROM attention_subjects WHERE kind='loop' AND admitted=1"
    ).all() as { scope_id: string; subject_id: string; payload_json: string }[];
    await Promise.allSettled(loops.map(async (row) => {
      let path: unknown;
      try { path = (JSON.parse(row.payload_json) as Record<string, unknown>).state_path; } catch { /* below */ }
      if (typeof path !== "string" || !path || path.length > 4096) {
        if (!this.stopped) this.store.write(this.observer.owner, () => {
          // Do not replace newer metadata or an explicit owner release.
          this.store.db.prepare("UPDATE attention_subjects SET sampled_at=?,payload_json=? WHERE scope_id=? AND kind='loop' AND subject_id=? AND payload_json=?")
            .run(Date.now(), JSON.stringify({ state: "unknown", reason: "invalid_payload", process_liveness: "unknown" }), row.scope_id, row.subject_id, row.payload_json);
        });
        return;
      }
      try {
        await this.handle("attention.loop", {
          scope_id: row.scope_id, loop_id: row.subject_id,
          state_path: path,
        });
      } catch { /* source errors never suppress other readers */ }
    }));
    if (this.stopped) return;
    for (const scope of this.store.listScopes()) {
      try { this.experience.enqueue(scope.scope_id); }
      catch { /* Keep observed items; one unavailable scope must not block other notices. */ }
    }
    if (this.sender && !this.delivering) {
      // Only this service owns the local idempotent sender. A failed receipt
      // write leaves `sending`; reconcile it before the next bounded batch.
      if (this.deliveryRecoveryNeeded) {
        this.experience.recoverDeliveries();
        this.deliveryRecoveryNeeded = false;
      }
      this.delivering = true;
      this.deliveryDone = Promise.resolve().then(() => this.experience.drainBatch(this.sender!, { signal: this.deliveryAbort.signal }))
        .then(() => {})
        .catch(() => { this.deliveryRecoveryNeeded = true; })
        .finally(() => { this.delivering = false; });
    }
  }
  stop(): Promise<void> {
    if (this.stopDone) return this.stopDone;
    this.stopped = true;
    this.deliveryAbort.abort();
    this.epoch++;
    this.dirty = false;
    if (this.timer) clearInterval(this.timer);
    this.timer = undefined;
    for (const slot of this.readers.values()) slot.controller.abort();
    this.stopDone = Promise.allSettled([this.scanDone, this.deliveryDone, ...this.logicalReads]).then(() => {});
    return this.stopDone;
  }
  private loopBinding(scope: string, id: string): LoopBinding | undefined {
    return this.store.db.prepare(
      "SELECT binding_hash,admitted,generation,payload_json FROM attention_subjects WHERE scope_id=? AND kind='loop' AND subject_id=?"
    ).get(scope, id) as LoopBinding | undefined;
  }
  private readLoop(scopeId: string, id: string, path: string): Promise<LoopFacts> {
    this.assertRunning();
    const epoch = this.epoch;
    const binding = attentionDigest([path, id]);
    const key = JSON.stringify([scopeId, "loop", id, binding]);
    const scope = this.store.getScope(scopeId)!;
    // Register the binding before IO, so release and competing RPCs see it.
    this.store.write(this.observer.owner, () => {
      this.assertRunning();
      const old = this.loopBinding(scopeId, id);
      if (old && old.binding_hash !== binding) throw new Error("loop_binding_conflict");
      if (old) {
        const payload = loopPayload(old.payload_json);
        if (!payload || payload.reason === "owner_released") return;
        const count = (this.store.db.prepare("SELECT COUNT(*) n FROM attention_subjects WHERE admitted=1").get() as { n: number }).n;
        if (old.generation !== this.observer.owner.generation || (!old.admitted && count < 100))
          this.store.db.prepare("UPDATE attention_subjects SET admitted=?,generation=?,sampled_at=0,payload_json=? WHERE scope_id=? AND kind='loop' AND subject_id=?")
            .run(old.admitted || count < 100 ? 1 : 0, this.observer.owner.generation,
              JSON.stringify({ ...payload, state: "unknown", reason: "read_pending" }), scopeId, id);
        return;
      }
      const count = (this.store.db.prepare("SELECT COUNT(*) n FROM attention_subjects WHERE admitted=1").get() as { n: number }).n;
      this.store.db.prepare(
        "INSERT INTO attention_subjects(scope_id,kind,subject_id,binding_hash,source_version,admitted,discovered_at,sampled_at,generation,payload_json) VALUES(?,'loop',?,?,0,?,?,?,?,?)"
      ).run(scopeId, id, binding, count < 100 ? 1 : 0, Date.now(), 0, this.observer.owner.generation,
        JSON.stringify({ state_path: path, state: "unknown", reason: count < 100 ? "read_pending" : "capacity_exceeded" }));
    });
    const old = this.loopBinding(scopeId, id)!;
    const failure = (reason: ReadFailure): LoopFacts => ({ loop_id: id, path, state: "unknown",
      process_liveness: "unknown", coverage: "uncovered", gap: "snapshot_unavailable", reason });
    const payload = loopPayload(old.payload_json);
    if (!payload) return Promise.resolve(failure("invalid_payload"));
    if (payload.reason === "owner_released")
      return Promise.resolve({ state: "unknown", reason: "owner_released" });
    const existing = this.readers.get(key);
    if (existing) return existing.logical;
    const commit = (facts: LoopFacts): LoopFacts => {
      // No await from the lifecycle check through the owner-fenced commit.
      if (this.stopped || epoch !== this.epoch) return facts;
      this.store.write(this.observer.owner, () => {
        const current = this.loopBinding(scopeId, id);
        const currentPayload = current ? loopPayload(current.payload_json) : null;
        if (!current || current.binding_hash !== binding || current.generation !== this.observer.owner.generation
          || current.admitted !== old.admitted || !currentPayload || currentPayload.reason === "owner_released") return;
        this.store.db.prepare(
          "UPDATE attention_subjects SET sampled_at=?,payload_json=? WHERE scope_id=? AND kind='loop' AND subject_id=?"
        ).run(Date.now(), JSON.stringify({ ...facts, state_path: path,
          reason: facts.reason ?? (!current.admitted ? "capacity_exceeded" : facts.state === "source_missing"
            ? "source_missing" : facts.recovery_required ? "recovery_required" : "snapshot_only") }), scopeId, id);
      });
      return facts;
    };
    if (!old.admitted || this.readers.size >= 100)
      return Promise.resolve(commit(failure("read_capacity_exceeded")));
    const controller = new AbortController();
    let reason: ReadFailure = "aborted";
    let timer: ReturnType<typeof setTimeout> | undefined;
    let abortListener: (() => void) | undefined;
    const aborted = new Promise<LoopFacts>((resolve) => {
      abortListener = () => resolve(failure(reason));
      controller.signal.addEventListener("abort", abortListener, { once: true });
      timer = setTimeout(() => { reason = "timeout"; controller.abort(); }, this.readTimeoutMs);
    });
    const physical = Promise.resolve().then(async () => {
      controller.signal.throwIfAborted();
      const reader = this.options.loopReader ?? (await import("./attention-loop-source.js")).readAttentionLoop;
      controller.signal.throwIfAborted();
      return await reader({ repo: scope.repo_path, path, loop_id: id, signal: controller.signal });
    }).then((value) => validLoopFacts(value, id) ? value : failure("invalid_payload"), (error: unknown) =>
      failure(controller.signal.aborted ? reason : error instanceof SyntaxError
        || (error instanceof Error && error.name === "ZodError") ? "invalid_payload" : "read_failed"));
    // Physical settlement has no DB access, even if the logical deadline passed.
    const settled = physical.finally(() => { this.readers.delete(key); });
    const logical = Promise.race([settled, aborted]).then(commit).finally(() => {
      if (timer) clearTimeout(timer);
      if (abortListener) controller.signal.removeEventListener("abort", abortListener);
      this.logicalReads.delete(logical);
    });
    this.readers.set(key, { controller, logical });
    this.logicalReads.add(logical);
    return logical;
  }
  async handle(method: string, params: unknown): Promise<unknown> {
    this.assertRunning();
    if (!params || typeof params !== "object")
      throw new Error("explicit_scope_required");
    const p = params as Record<string, unknown>;
    const str = (key: string) => {
      const value = p[key];
      if (typeof value !== "string" || !value || value.length > 4096)
        throw new Error(`invalid_${key}`);
      return value;
    };
    if (method === "attention.watch") {
      if (p["default_tenant"] !== true)
        throw new Error("explicit_default_tenant_required");
      const scope = this.store.registerScope(
        this.observer.owner,
        str("task_id"),
        str("repo_path")
      );
      await this.requestScan("watch");
      this.assertRunning();
      return this.observer.inventory(scope.scope_id);
    }
    const scopeId = str("scope_id");
    if (!this.store.getScope(scopeId)) throw new Error("not_found");
    if (method === "attention.release") {
      this.store.release(
        this.observer.owner,
        scopeId,
        str("kind"),
        str("subject_id"),
        Number(p["expected_version"])
      );
      return {
        released: true,
        execution_affected: false,
        coverage: "uncovered",
      };
    }
    if (method === "attention.loop")
      return await this.readLoop(scopeId, str("loop_id"), str("state_path"));
    if (method === "attention.ci") {
      const { AttentionCiReader } = await import("@osstack/github");
      this.assertRunning();
      this.ciReader ??= new AttentionCiReader();
      return await this.ciReader.read({
        repo: str("github_repo"),
        pr: p["pr"],
        head: str("head"),
        required: p["required"],
      });
    }
    if (method === "attention.brief")
      return this.experience.brief(scopeId, str("actor"));
    if (method === "attention.posture") {
      const posture = str("posture");
      if (!["quiet", "away", "attended"].includes(posture))
        throw new Error("invalid_posture");
      this.experience.posture(
        scopeId,
        str("actor"),
        posture as "quiet" | "away" | "attended"
      );
      return { posture };
    }
    if (method === "attention.seen") {
      if (!Array.isArray(p["revisions"])) throw new Error("invalid_revisions");
      this.experience.seen(scopeId, str("actor"), p["revisions"]);
      return { marked: true };
    }
    if (method === "attention.foreground") {
      if (!Array.isArray(p["subjects"])) throw new Error("invalid_subjects");
      return this.experience.foreground(scopeId, p["subjects"]);
    }
    if (method === "attention.handoff")
      return this.experience.handoff(scopeId, str("client_id"), str("token"));
    if (method === "attention.turn-end")
      return this.experience.turnEnd(
        scopeId,
        str("client_id"),
        str("token"),
        str("role")
      );
    if (method === "attention.deliveries")
      return this.store.db
        .prepare(
          "SELECT * FROM attention_deliveries WHERE scope_id=? AND delivery_id>? ORDER BY delivery_id LIMIT 100"
        )
        .all(scopeId, typeof p["after"] === "string" ? p["after"] : "");
    if (method === "attention.inventory")
      return this.observer.inventory(
        scopeId,
        Date.now(),
        typeof p["after"] === "string" ? p["after"] : ""
      );
    if (method === "attention.inbox") {
      const items = this.store.listItems(
        scopeId,
        typeof p["after"] === "string" ? p["after"] : ""
      );
      const counts = this.store.db
        .prepare(
          "SELECT state,COUNT(*) total FROM attention_items WHERE scope_id=? GROUP BY state"
        )
        .all(scopeId);
      return AttentionInbox.parse({
        schema_version: 1,
        execution_takeover: false,
        scope_id: scopeId,
        items,
        counts,
        next_after: items.length === 100 ? items.at(-1)!.item_id : null,
        coverage: this.observer.inventory(scopeId).coverage,
      });
    }
    if (method === "attention.get") {
      const item = this.store.getItem(scopeId, str("item_id"));
      if (!item) throw new Error("not_found");
      return item;
    }
    throw new Error("unsupported_attention_method");
  }
}

import { AttentionObserver } from "./attention-observer.js";
import { randomBytes, randomUUID, timingSafeEqual } from "node:crypto";
import {
  AttentionStore,
  attentionDigest,
  type AttentionOwner,
} from "./attention-store.js";
interface Delivery {
  delivery_id: string;
  scope_id: string;
  item_id: string;
  revision: number;
  state: string;
  attempts: number;
  next_at: number;
  error: string | null;
}
export type AttentionSender = (
  message: {
    delivery_id: string;
    scope_id: string;
    item_id: string;
    question: string;
    state: string;
  },
  signal: AbortSignal
) => Promise<void>;
/** Local owner UX state; posture never changes any execution authority. */
export class AttentionExperience {
  constructor(readonly store: AttentionStore, readonly owner: AttentionOwner) {}
  posture(
    scope: string,
    actor: string,
    posture: "attended" | "quiet" | "away"
  ): void {
    this.actor(actor);
    if (!["attended", "quiet", "away"].includes(posture))
      throw new Error("invalid_posture");
    this.store.write(this.owner, () => {
      this.scope(scope);
      this.store.db
        .prepare(
          "INSERT INTO attention_postures VALUES(?,?,?) ON CONFLICT(scope_id,actor) DO UPDATE SET posture=excluded.posture"
        )
        .run(scope, actor, posture);
    });
  }
  brief(scope: string, actor: string) {
    this.actor(actor);
    this.scope(scope);
    const db = this.store.db;
    const items = db
      .prepare(
        `SELECT i.item_id,i.revision,i.state,i.decision_json,i.created_at FROM attention_items i LEFT JOIN attention_seen s ON s.scope_id=i.scope_id AND s.item_id=i.item_id AND s.actor=?
   WHERE i.scope_id=? AND COALESCE(s.revision,0)<i.revision ORDER BY CASE WHEN i.state LIKE '%uncertain%' THEN 0 WHEN json_extract(i.decision_json,'$.continuation.gate_kind') IN ('risk_approval','task_authorization') THEN 1 WHEN i.state='open' THEN 2 ELSE 3 END,i.created_at,i.item_id LIMIT 3`
      )
      .all(actor, scope) as {
      item_id: string;
      revision: number;
      state: string;
      decision_json: string;
      created_at: number;
    }[];
    const total = (
      db
        .prepare(
          `SELECT COUNT(*) n FROM attention_items i LEFT JOIN attention_seen s ON s.scope_id=i.scope_id AND s.item_id=i.item_id AND s.actor=? WHERE i.scope_id=? AND COALESCE(s.revision,0)<i.revision`
        )
        .get(actor, scope) as { n: number }
    ).n;
    const dead = (
      db
        .prepare(
          "SELECT COUNT(*) n FROM attention_deliveries WHERE scope_id=? AND state='dead_letter'"
        )
        .get(scope) as { n: number }
    ).n;
    return {
      scope_id: scope,
      actor,
      total,
      dead_letters: dead,
      items: items.map(({ decision_json, ...r }) => ({
        ...r,
        question: JSON.parse(decision_json).question,
      })),
      read_is_not_seen: true,
    };
  }
  seen(
    scope: string,
    actor: string,
    revisions: { item_id: string; revision: number }[]
  ): void {
    this.actor(actor);
    if (revisions.length > 100) throw new Error("too_many_items");
    this.store.write(this.owner, () => {
      this.scope(scope);
      for (const r of revisions) {
        const item = this.store.getItem(scope, r.item_id);
        if (!item || item.revision !== r.revision)
          throw new Error("stale_item");
        this.store.db
          .prepare(
            "INSERT INTO attention_seen VALUES(?,?,?,?) ON CONFLICT(scope_id,actor,item_id) DO UPDATE SET revision=MAX(revision,excluded.revision)"
          )
          .run(scope, actor, r.item_id, r.revision);
      }
    });
  }
  enqueue(scope: string): void {
    this.store.write(this.owner, () => {
      this.scope(scope);
      this.store.db
        .prepare(
          `INSERT OR IGNORE INTO attention_deliveries(delivery_id,scope_id,item_id,revision)
    SELECT item_id||':'||revision,scope_id,item_id,revision FROM attention_items WHERE scope_id=? AND state IN ('open','answer_uncertain','source_missing')`
        )
        .run(scope);
    });
  }
  recoverDeliveries(): void {
    this.store.write(this.owner, () =>
      this.store.db
        .prepare(
          "UPDATE attention_deliveries SET state=CASE WHEN attempts>=5 THEN 'dead_letter' ELSE 'pending' END,error='interrupted_delivery' WHERE state='sending'"
        )
        .run()
    );
  }
  async drain(sender: AttentionSender, now = Date.now(), timeoutMs = 5000): Promise<void> {
    await this.drainOne(sender, now, timeoutMs);
  }
  /** Local idempotent transports only. A timeout may still have delivered the
   * same delivery_id; this does not authorize retrying public messages.
   * maxItems bounds all claimed work, including superseded projections.
   */
  async drainBatch(sender: AttentionSender, options: {
    maxItems?: number;
    maxDurationMs?: number;
    signal?: AbortSignal;
  } = {}) {
    const maxItems = options.maxItems ?? 20;
    const maxDurationMs = options.maxDurationMs ?? 1000;
    if (!Number.isInteger(maxItems) || maxItems < 1 || maxItems > 100 ||
        !Number.isInteger(maxDurationMs) || maxDurationMs < 1 || maxDurationMs > 10_000) {
      throw new Error("invalid_delivery_batch_bounds");
    }
    const started = performance.now();
    const result = { processed: 0, attempted: 0, delivered: 0, retried: 0,
      deadLetters: 0, superseded: 0, stale: 0, cancelled: 0,
      stopped: "limit" as "limit" | "empty" | "deadline" | "aborted" };
    while (result.processed < maxItems) {
      if (options.signal?.aborted) { result.stopped = "aborted"; break; }
      const remaining = maxDurationMs - (performance.now() - started);
      if (remaining <= 0) { result.stopped = "deadline"; break; }
      const outcome = await this.drainOne(sender, Date.now(), Math.min(5000, Math.ceil(remaining)), options.signal);
      if (outcome === "idle") { result.stopped = "empty"; break; }
      result.processed++;
      if (outcome === "superseded") result.superseded++;
      else {
        result.attempted++;
        if (outcome === "dead_letter") result.deadLetters++;
        else result[outcome]++;
      }
    }
    if (options.signal?.aborted) result.stopped = "aborted";
    // Counts use the same eligibility predicate as claim; quiet and delayed
    // retries remain persisted but are not currently deliverable.
    const due = this.store.db.prepare(`SELECT COUNT(*) n, MIN(d.next_at) oldest
      FROM attention_deliveries d WHERE d.state='pending' AND d.next_at<=?
      AND NOT EXISTS(SELECT 1 FROM attention_postures p WHERE p.scope_id=d.scope_id AND p.posture='quiet')`
    ).get(Date.now()) as { n: number; oldest: number | null };
    return { ...result, remainingDue: due.n, oldestDueAt: due.oldest };
  }
  private async drainOne(
    sender: AttentionSender,
    now = Date.now(),
    timeoutMs: number,
    signal?: AbortSignal
  ): Promise<"idle" | "superseded" | "delivered" | "retried" | "dead_letter" | "stale" | "cancelled"> {
    const db = this.store.db;
    const row = this.store.write(this.owner, () => {
      const d = db
        .prepare(
          `SELECT d.* FROM attention_deliveries d WHERE d.state='pending' AND d.next_at<=? AND NOT EXISTS(SELECT 1 FROM attention_postures p WHERE p.scope_id=d.scope_id AND p.posture='quiet') ORDER BY d.next_at,d.delivery_id LIMIT 1`
        )
        .get(now) as Delivery | undefined;
      if (d) {
        const item = this.store.getItem(d.scope_id, d.item_id);
        if (
          !item ||
          item.revision !== d.revision ||
          !["open", "answer_uncertain", "source_missing"].includes(item.state)
        ) {
          db.prepare(
            "UPDATE attention_deliveries SET state='superseded' WHERE delivery_id=?"
          ).run(d.delivery_id);
          return "superseded" as const;
        }
      }
      if (d)
        db.prepare(
          "UPDATE attention_deliveries SET state='sending',attempts=attempts+1 WHERE delivery_id=? AND state='pending'"
        ).run(d.delivery_id);
      return d;
    });
    if (row === "superseded") return "superseded";
    if (!row) return "idle";
    const item = this.store.getItem(row.scope_id, row.item_id);
    const controller = new AbortController();
    let timer: ReturnType<typeof setTimeout> | undefined;
    let error: string | null = null;
    let onAbort: (() => void) | undefined;
    let cancelled = false;
    try {
      if (!item) throw new Error("source_missing");
      await Promise.race([
        Promise.resolve().then(() => {
          if (controller.signal.aborted) throw new Error("delivery_aborted");
          return sender(
            {
              delivery_id: row.delivery_id,
              scope_id: row.scope_id,
              item_id: row.item_id,
              question: item.decision.question,
              state: item.state,
            },
            controller.signal
          );
        }),
        new Promise<never>((_, reject) => {
          onAbort = () => { cancelled = true; controller.abort(); reject(new Error("delivery_aborted")); };
          signal?.addEventListener("abort", onAbort, { once: true });
          if (signal?.aborted) onAbort();
        }),
        new Promise<never>((_, reject) => {
          timer = setTimeout(() => {
            controller.abort();
            reject(new Error("delivery_timeout"));
          }, timeoutMs);
        }),
      ]);
    } catch (e) {
      error =
        e instanceof Error && e.message === "delivery_timeout"
          ? "delivery_timeout"
          : "delivery_failed";
    } finally {
      if (timer) clearTimeout(timer);
      if (onAbort) signal?.removeEventListener("abort", onAbort);
    }
    let changed: number;
    try {
    const updated = this.store.write(this.owner, () =>
      db
        .prepare(
          "UPDATE attention_deliveries SET state=?,next_at=?,error=?,attempts=attempts-? WHERE delivery_id=? AND attempts=? AND state='sending'"
        )
        .run(
          cancelled ? "pending" : error
            ? row.attempts + 1 >= 5
              ? "dead_letter"
              : "pending"
            : "delivered",
          now + Math.min(300000, 1000 * 2 ** row.attempts),
          cancelled ? "delivery_cancelled_indeterminate" : error,
          cancelled ? 1 : 0,
          row.delivery_id,
          row.attempts + 1
        )
    );
    changed = updated.changes;
    } catch (e) {
      if (e instanceof Error && e.message === "stale_owner") return "stale";
      throw e;
    }
    if (changed !== 1) return "stale";
    if (cancelled) return "cancelled";
    return error ? row.attempts + 1 >= 5 ? "dead_letter" : "retried" : "delivered";
  }
  foreground(
    scope: string,
    subjects: { subject_id: string; source_version: number }[],
    now = Date.now()
  ) {
    if (
      !Array.isArray(subjects) ||
      subjects.length < 1 ||
      subjects.length > 100 ||
      new Set(subjects.map((s) => s.subject_id)).size !== subjects.length
    )
      throw new Error("invalid_subject_set");
    const token = randomBytes(32).toString("hex");
    const clientId = randomUUID();
    this.store.write(this.owner, () => {
      this.scope(scope);
      this.validateSubjects(scope, subjects);
      this.store.db
        .prepare(
          "INSERT INTO attention_foregrounds(client_id,token_hash,scope_id,generation,expires_at,subjects_json) VALUES(?,?,?,?,?,?)"
        )
        .run(
          clientId,
          attentionDigest(token),
          scope,
          this.owner.generation,
          now + 120000,
          JSON.stringify(subjects)
        );
    });
    return {
      client_id: clientId,
      token,
      expires_at: now + 120000,
      role: "foreground_owner",
    };
  }
  handoff(scope: string, client: string, token: string, now = Date.now()) {
    return this.store.write(this.owner, () => {
      const row = this.authenticate(scope, client, token, now, true);
      const prior = this.store.db
        .prepare("SELECT * FROM attention_handoffs WHERE client_id=?")
        .get(client) as
        | {
            receipt_id: string;
            client_id: string;
            scope_id: string;
            generation: number;
            subjects_json: string;
            created_at: number;
            valid_until: number;
          }
        | undefined;
      if (prior) {
        let currentlyValid =
          new AttentionObserver(this.store, this.owner).inventory(scope, now)
            .coverage === "covered";
        try {
          this.validateSubjects(scope, JSON.parse(row.subjects_json));
        } catch {
          currentlyValid = false;
        }
        return {
          ...prior,
          execution_takeover: false,
          process_liveness: "unknown",
          valid_until: (prior as { valid_until: number }).valid_until,
          expired: now > (prior as { valid_until: number }).valid_until,
          currently_valid:
            currentlyValid &&
            now <= (prior as { valid_until: number }).valid_until,
        };
      }
      if (row.expires_at < now) throw new Error("invalid_foreground");
      const s = this.scope(scope);
      if (
        new AttentionObserver(this.store, this.owner).inventory(scope, now)
          .coverage !== "covered"
      )
        throw new Error("coverage_not_ready");
      if (
        !s.scan_complete ||
        s.generation !== this.owner.generation ||
        s.deadline < now ||
        s.sampled_at > now ||
        s.error ||
        this.store.db
          .prepare(
            "SELECT 1 FROM attention_subjects WHERE scope_id=? AND (admitted=0 OR sampled_at<?) LIMIT 1"
          )
          .get(scope, now - 65000)
      )
        throw new Error("coverage_not_ready");
      const subjects = JSON.parse(row.subjects_json) as {
        subject_id: string;
        source_version: number;
      }[];
      this.validateSubjects(scope, subjects);
      const receipt = {
        receipt_id: randomUUID(),
        client_id: client,
        scope_id: scope,
        generation: this.owner.generation,
        subjects_json: row.subjects_json,
        created_at: now,
        valid_until: s.deadline,
      };
      this.store.db
        .prepare("INSERT INTO attention_handoffs VALUES(?,?,?,?,?,?,?)")
        .run(
          receipt.receipt_id,
          client,
          scope,
          this.owner.generation,
          row.subjects_json,
          now,
          s.deadline
        );
      this.store.db
        .prepare(
          "UPDATE attention_foregrounds SET handed_off=1 WHERE client_id=?"
        )
        .run(client);
      return {
        ...receipt,
        execution_takeover: false,
        process_liveness: "unknown",
        valid_until: s.deadline,
      };
    });
  }
  turnEnd(
    scope: string,
    client: string,
    token: string,
    role: string,
    now = Date.now()
  ) {
    return this.store.write(this.owner, () => {
      if (role !== "foreground_owner") throw new Error("wrong_role");
      const row = this.authenticate(scope, client, token, now, true);
      if (row.reminder_sent || row.handed_off) return { reminder: null };
      this.store.db
        .prepare(
          "UPDATE attention_foregrounds SET reminder_sent=1 WHERE client_id=?"
        )
        .run(client);
      return {
        reminder:
          "Foreground ended. Check attention coverage and explicitly hand off unresolved work.",
        continues: false,
        offline_detection: false,
        foreground_expired: row.expires_at < now,
      };
    });
  }
  private authenticate(
    scope: string,
    client: string,
    token: string,
    now: number,
    allowExpired = false
  ) {
    const row = this.store.db
      .prepare(
        "SELECT * FROM attention_foregrounds WHERE client_id=? AND scope_id=?"
      )
      .get(client, scope) as
      | {
          token_hash: string;
          generation: number;
          expires_at: number;
          subjects_json: string;
          handed_off: number;
          reminder_sent: number;
        }
      | undefined;
    if (
      typeof token !== "string" ||
      token.length !== 64 ||
      !row ||
      !timingSafeEqual(
        Buffer.from(row.token_hash),
        Buffer.from(attentionDigest(token))
      ) ||
      row.generation !== this.owner.generation ||
      (!allowExpired && row.expires_at < now)
    )
      throw new Error("invalid_foreground");
    return row;
  }
  private validateSubjects(
    scope: string,
    subjects: { subject_id: string; source_version: number }[]
  ): void {
    for (const s of subjects) {
      const row = this.store.db
        .prepare(
          "SELECT source_version,admitted,payload_json FROM attention_subjects WHERE scope_id=? AND kind='session' AND subject_id=?"
        )
        .get(scope, s.subject_id) as
        | { source_version: number; admitted: number; payload_json: string }
        | undefined;
      const live = this.store.db
        .prepare("SELECT version FROM agent_sessions WHERE session_id=?")
        .get(s.subject_id) as { version: number } | undefined;
      if (
        !row ||
        row.admitted !== 1 ||
        row.source_version !== s.source_version ||
        live?.version !== s.source_version ||
        JSON.parse(row.payload_json).reason ||
        JSON.parse(row.payload_json).gap === "history_gap" ||
        (JSON.parse(row.payload_json).unresolved_receipts ?? []).length > 0
      )
        throw new Error("subject_not_covered");
    }
  }
  private scope(scope: string) {
    const s = this.store.getScope(scope);
    if (!s) throw new Error("not_found");
    return s;
  }
  private actor(actor: string): void {
    if (typeof actor !== "string" || actor.length < 1 || actor.length > 256)
      throw new Error("invalid_actor");
  }
}

it("recovers a failed local delivery receipt on the next scan without restart", async () => {
  const { service, scope, db } = setup();
  const sent = new Set<string>();
  let calls = 0;
  Object.defineProperty(service, "sender", { value: async (message: {delivery_id:string}) => { calls++; sent.add(message.delivery_id); } });
  service.store.projectDecision(service.observer.owner, {scopeId:scope.scope_id,sessionId:"s",turnId:"turn",sourceVersion:1,decision:{decision_id:"d",question:"Choose",choices:[]},now:1});
  db.exec("CREATE TRIGGER fail_receipt BEFORE UPDATE ON attention_deliveries WHEN NEW.state='delivered' BEGIN SELECT RAISE(ABORT,'receipt disk failure'); END");
  await service.requestScan();
  await vi.waitFor(() => expect(calls).toBe(1));
  expect(db.prepare("SELECT state FROM attention_deliveries").get()).toEqual({state:"sending"});
  db.exec("DROP TRIGGER fail_receipt");
  await service.requestScan();
  await vi.waitFor(() => expect(db.prepare("SELECT state FROM attention_deliveries").get()).toEqual({state:"delivered"}));
  expect(calls).toBe(2);
  expect(sent.size).toBe(1);
});

it("aborts and joins a real hung notification batch before database close", async () => {
  const { service, scope, db } = setup();
  const late = deferred<void>();
  let called = false;
  let signal: AbortSignal | undefined;
  Object.defineProperty(service, "sender", { value: async (_message: unknown, abort: AbortSignal) => { called=true; signal=abort; await late.promise; } });
  service.store.projectDecision(service.observer.owner, {scopeId:scope.scope_id,sessionId:"s",turnId:"turn",sourceVersion:1,decision:{decision_id:"d",question:"Choose",choices:[]},now:1});
