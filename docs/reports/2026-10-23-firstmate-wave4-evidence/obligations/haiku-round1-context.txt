      case "follow-up.draft": return this.draft(input);

      }

}

import { open } from "node:fs/promises";
import { FollowUpCreateParams, FollowUpListParams, FollowUpDraftParams, FollowUpResolveParams, FollowUpMutationResult, FollowUpListResult } from "@osstack/protocol";
import { FOLLOW_UP_ERROR_CODES, FOLLOW_UP_MAX_BYTES, serializeFollowUp } from "@osstack/core";

export const FOLLOW_UP_HELP = `follow-up list --repo PATH --default-tenant [--limit 1..50] [--state STATE] [--cursor TOKEN] [--json]
follow-up create --repo PATH --default-tenant --input FILE [--json]
follow-up draft ID --repo PATH --default-tenant --revision N --actor LABEL [--json]
follow-up resolve ID --repo PATH --default-tenant --revision N --actor LABEL --mode waive|condition --reason TEXT [--json]
Local obligation ledger only; external delivery is unsupported. Drafts are UNSENT and do not resolve obligations.
Create input excludes repo_path/default_tenant; select scope explicitly with flags. Actor is a label, not an authorization credential.\n`;
export interface FollowUpCommandDependencies { rpc(method: string, params: unknown): Promise<unknown>; stdout?: (text: string) => void; stderr?: (text: string) => void }
export async function handleFollowUpCommand(args: string[], dependencies: FollowUpCommandDependencies): Promise<number> {
  const out = dependencies.stdout ?? (text => { process.stdout.write(text); }); const err = dependencies.stderr ?? (text => { process.stderr.write(text); });
  if (args.length === 0 || (args.length === 1 && ["help", "--help"].includes(args[0]!))) { out(FOLLOW_UP_HELP); return 0; }
  const action = args[0]; let params: unknown; let json = false;
  try {
    if (args.length > 22 || args.some(arg => arg.length > 4096) || !["list", "create", "draft", "resolve"].includes(action!)) throw new Error();
    const identity = action === "draft" || action === "resolve"; const id = identity ? args[1] : undefined;
    const flags = new Map<string, string | true>();
    const allowed = ["--repo", "--default-tenant", "--json", ...(action === "list" ? ["--limit", "--cursor", "--state"] : action === "create" ? ["--input"] : action === "draft" ? ["--revision", "--actor"] : ["--revision", "--actor", "--mode", "--reason"])];
    for (let i = identity ? 2 : 1; i < args.length; i++) {
      const key = args[i]!; if (!allowed.includes(key) || flags.has(key)) throw new Error();
      if (key === "--default-tenant" || key === "--json") flags.set(key, true);
      else { const value = args[++i]; if (!value || allowed.includes(value)) throw new Error(); flags.set(key, value); }
    }
    json = flags.has("--json");
    const scope = { repo_path: flags.get("--repo"), default_tenant: flags.get("--default-tenant") };
    const number = (name: string) => { const value = flags.get(name); if (typeof value !== "string" || !/^[1-9][0-9]*$/.test(value)) throw new Error(); return Number(value); };
    if (action === "list") params = FollowUpListParams.parse({ ...scope, ...(flags.has("--limit") ? { limit: number("--limit") } : {}), ...(flags.has("--cursor") ? { cursor: flags.get("--cursor") } : {}), ...(flags.has("--state") ? { state: flags.get("--state") } : {}) });
    else if (action === "create") {
      const path = flags.get("--input"); if (typeof path !== "string") throw new Error();
      const body = FollowUpCreateParams.omit({ repo_path: true, default_tenant: true }).parse(await readInput(path));
      params = FollowUpCreateParams.parse({ ...body, ...scope });
    } else {
      const base = { ...scope, obligation_id: id, expected_revision: number("--revision"), actor: flags.get("--actor") };
      params = action === "draft" ? FollowUpDraftParams.parse(base) : FollowUpResolveParams.parse({ ...base, mode: flags.get("--mode"), reason: flags.get("--reason") });
    }
  } catch { err("invalid_follow_up_request\n"); return 1; }
  let raw: unknown;
  try { raw = await dependencies.rpc(`follow-up.${action}`, params); }
  catch (error) { const code = error instanceof Error ? error.message : ""; err(`${(FOLLOW_UP_ERROR_CODES as readonly string[]).includes(code) ? code : "follow_up_request_failed"}\n`); return 1; }
  try {
    if (Buffer.byteLength(serializeFollowUp(raw)) + 1 > FOLLOW_UP_MAX_BYTES) throw new Error();
    const result = action === "list" ? FollowUpListResult.parse(raw) : FollowUpMutationResult.parse(raw);
    const text = json ? `${serializeFollowUp(result)}\n` : render(result);
    if (Buffer.byteLength(text) > FOLLOW_UP_MAX_BYTES) throw new Error(); out(text); return 0;
  } catch { err("invalid_follow_up_response\n"); return 1; }
}
async function readInput(path: string): Promise<unknown> {
  const file = await open(path, constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK);
  try {
    const stat = await file.stat(); if (!stat.isFile() || stat.size > 16384) throw new Error();
    const bytes = Buffer.alloc(16385); let length = 0;
    while (length < bytes.length) { const result = await file.read(bytes, length, bytes.length - length, null); if (!result.bytesRead) break; length += result.bytesRead; }
    if (length > 16384) throw new Error(); return JSON.parse(bytes.subarray(0, length).toString("utf8"));
  } finally { await file.close(); }
}
function render(result: FollowUpListResult | FollowUpMutationResult): string {
  const rows = "obligations" in result ? result.obligations : [result.obligation];
  const lines = ["Local obligations — external delivery unsupported; drafts are UNSENT."];
  if ("overdue_count" in result) lines.push(`Overdue: ${result.overdue_count}; page: ${rows.length}`);
  for (const row of rows) {
    lines.push(`${row.obligation_id} revision=${row.revision} state=${row.state}; source=${row.source_missing ? "source_missing" : row.source!.state}`);
    lines.push(`  promise=${serializeFollowUp("promise" in row ? row.promise : row.promise_preview)}`);
    if ("draft" in row && row.draft) lines.push(`  unsent_draft=${serializeFollowUp(row.draft.text)}`);
    if (row.state === "fulfilled_local") lines.push("  Local condition verified; no external delivery is claimed.");
  }
  if ("next_cursor" in result && result.next_cursor) lines.push(`next_cursor=${serializeFollowUp(result.next_cursor)}`);
  return `${lines.join("\n")}\n`;
}

# Local follow-up obligations

`follow-up` records a local operator's promise, builds an unsent draft, and records explicit local resolution. It does not send messages, recognize provider delivery receipts, or implement a public transport. `fulfilled_local` means only that the operator's selected local Session condition was verified; it does not mean anyone received a reply.

All commands require `--repo PATH --default-tenant`. The service canonicalizes an existing readable/searchable directory and joins an existing Session to its matching task and run in that exact repository, requiring both tenant IDs to be NULL. Graph and unbound run sources are unsupported. The daemon provides authentication and DB access; an `actor` is an audit label, never an authorization credential. This feature does not depend on the Attention enable flag.

```text
osstack follow-up create --repo /canonical/repository --default-tenant --input obligation.json --json
osstack follow-up list --repo /canonical/repository --default-tenant --limit 20 --json
osstack follow-up draft UUID --repo /canonical/repository --default-tenant --revision 1 --actor operator --json
osstack follow-up resolve UUID --repo /canonical/repository --default-tenant --revision 2 --actor operator --mode condition --reason "Verified local completion" --json
```

Create input is a JSON file of at most 16 KiB. Scope is supplied through flags and must not appear in the input file:

```json
{
  "source_session_id": "existing-local-session",
  "expected_source_version": 1,
  "idempotency_key": "review-local-result-1",
  "actor": "operator",
  "promise": "Review this Session's local result",
  "destination_ref": "local_operator",
  "resolution_condition": "session_completed",
  "due_at": null
}
```

`destination_ref` accepts only the exact built-in local operator reference. Arbitrary URLs, recipient strings and registered external channels are unsupported in this release. `resolution_condition` accepts `manual` or `session_completed`. Due dates are optional safe Unix epoch milliseconds; a past date is an overdue promise rather than an invalid request. Promise text is bounded to 4,096 UTF-8 bytes, actor labels to 128, and resolution reasons to 1,024. Whitespace-only values are rejected. Text is stored and displayed as data and is never sent to a model or shell.

Creation verifies the actual Session version and records an immutable source identity digest bound to the source session/task/run IDs, request digest, task creation time and run start time. Native session IDs, source prompts and raw model output are not read. Reusing a create key with exactly the same canonical payload returns the existing obligation, including after source cleanup or later resolution. Changing the content under an existing key fails. New keys require a currently valid matching source. The source snapshot and every transition remain in append-only local history.

`draft` requires the current obligation revision and records a deterministic text draft containing the operator's promise and current structured source facts. It always says `UNSENT`, changes only the local state to `drafted`, and never satisfies a condition or grants permission to send. A completed Session still requires a separate resolve operation. Missing/deleted/replaced sources remain visible as `source_missing`; the obligation and its original promise survive cleanup.

`resolve --mode condition` succeeds only when the original declared condition is `session_completed` and the same bound source Session is currently `completed`. A task's `done` status, a completed run, a model's assertion, a draft, a local spool file, or a URL cannot satisfy this condition. The resulting `fulfilled_local` record stores the verified source version/state/time and source identity binding. It records no external delivery. `resolve --mode waive` instead requires an explicit actor and reason; it works even when the original source is missing and retains the original promise. Terminal obligations are sealed. Both paths use expected-revision checks, so competing/stale callers cannot overwrite each other's decisions.

`list` is read-only, returns at most 50 records and less than 64 KiB including JSON escaping, and provides a keyset cursor for the next page. The cursor binds repository and state filter, but is a paging position, not an authority token. Each request must independently assert its scope. Listings are current views rather than immutable multi-page snapshots; new records or source changes can appear between pages. The overdue count covers all open/drafted obligations in the selected repository, regardless of the page's optional state filter. Source uncertainty never automatically removes or fulfills an obligation. List responses use previews; complete promise/draft values are available in mutation responses. Long paths/text may reduce page size or fail a mutation's response-size bound; no oversized evidence is silently truncated.

Migration 44 adds `follow_up_obligations` and `follow_up_obligation_events` after triage migration 43. No source table is changed. Constructors do not migrate or write. Row mutation, revision advancement and append-only event recording share one preflighted transaction; evidence failure rolls the mutation back. Existing `openDb` performs normal lineage checks and forward migration. Disable the command/RPC registration to roll back feature exposure while retaining history; do not lower schema versions or delete ledger rows.

The protocol schemas, `FollowUpObligationService(db, dbPath, {nowMs?})`, and `handleFollowUpCommand(args, {rpc, stdout?, stderr?})` support daemon integration. The handler expects raw validated RPC results; the root integration unwraps its existing transport response.
